function isValidColor(v) {
  return typeof v === "string" && /^#([0-9a-fA-F]{3}|[0-9a-fA-F]{6})$/.test(v);
}
function shapeValidCheckpoints(cps) {
  if (!Array.isArray(cps) || cps.length === 0) return false;
  for (const c of cps) {
    if (c === null || typeof c !== "object") return false;
    if (!("timeMs" in c) || !("count" in c)) return false;
  }
  return true;
}
function validateCandidateCheckpoints(cps, totalLines) {
  if (!shapeValidCheckpoints(cps) || cps.length < 2) return false;
  if (cps[0].timeMs !== 0 || cps[0].count !== 0) return false;
  for (let i = 0; i < cps.length; i++) {
    const c = cps[i];
    if (!Number.isFinite(c.timeMs) || !Number.isFinite(c.count)) return false;
    if (c.timeMs < 0 || c.count < 0 || c.count > totalLines) return false;
    if (!Number.isInteger(c.count)) return false;
    if (i > 0 && (c.timeMs <= cps[i - 1].timeMs || c.count <= cps[i - 1].count)) return false;
  }
  return cps[cps.length - 1].count === totalLines;
}
function validateDerivedCheckpoints(cps, totalLines) {
  if (!shapeValidCheckpoints(cps) || cps.length < 2) return false;
  if (cps[0].timeMs !== 0 || cps[0].count !== 0) return false;
  for (let i = 0; i < cps.length; i++) {
    const c = cps[i];
    if (!Number.isFinite(c.timeMs) || !Number.isFinite(c.count)) return false;
    if (c.timeMs < 0 || c.count < 0 || c.count > totalLines) return false;
    if (!Number.isInteger(c.count)) return false;
    if (i > 0 && c.timeMs <= cps[i - 1].timeMs) return false;
    if (i > 0 && c.count < cps[i - 1].count) return false;
    if (i < cps.length - 1 && totalLines > 0 && c.count >= totalLines) return false;
  }
  return cps[cps.length - 1].count === totalLines;
}
function validateCheckpoints(cps, totalLines) {
  return validateCandidateCheckpoints(cps, totalLines);
}
function isCoherentBasis(cps) {
  if (!shapeValidCheckpoints(cps) || cps.length < 2) return false;
  if (cps[0].timeMs !== 0 || cps[0].count !== 0) return false;
  for (let i = 0; i < cps.length; i++) {
    const c = cps[i];
    if (!Number.isFinite(c.timeMs) || !Number.isFinite(c.count)) return false;
    if (c.timeMs < 0 || c.count < 0 || !Number.isInteger(c.count)) return false;
    if (i > 0 && (c.timeMs <= cps[i - 1].timeMs || c.count < cps[i - 1].count)) return false;
  }
  return cps[cps.length - 1].count > 0;
}

function deriveFromBasis(basis, totalLines) {
  const specTotal = basis[basis.length - 1].count;
  return basis.map((c, i) => {
    let count;
    if (i === 0) count = 0;
    else if (i === basis.length - 1) count = totalLines;
    else count = Math.min(totalLines - 1, Math.round((c.count / specTotal) * totalLines));
    return { timeMs: c.timeMs, count };
  });
}
function normalizeCheckpoints(basis, totalLines) {
  return deriveFromBasis(basis, totalLines);
}


function strictParseNumber(v) {
  if (typeof v !== "string") return { ok: false, value: NaN };
  const s = v.trim();
  if (!/^[+-]?(\d+(\.\d*)?|\.\d+)([eE][+-]?\d+)?$/.test(s)) return { ok: false, value: NaN };
  const n = Number(s);
  if (!Number.isFinite(n)) return { ok: false, value: NaN };
  return { ok: true, value: n };
}
function resolveSchedule(candidate, totalLines, trustedBasis, durationMs, captionAtMs, defaultDurationMs, defaultCaptionAtMs) {
  const warnings = [];
  let table = null, kind = "";
  const finalTime = trustedBasis[trustedBasis.length - 1].timeMs;
  if (!Number.isInteger(totalLines) || totalLines < 0) totalLines = 0;
  if (totalLines === 0) {
    table = [{ timeMs: 0, count: 0 }, { timeMs: finalTime, count: 0 }]; kind = "empty";
    warnings.push("line list is empty; the effective schedule carries no lines");
  } else if (validateCandidateCheckpoints(candidate, totalLines)) {
    table = candidate.map(c => ({ timeMs: c.timeMs, count: c.count })); kind = "candidate";
  } else {
    const basis = isCoherentBasis(candidate) ? candidate : trustedBasis;
    const derived = deriveFromBasis(basis, totalLines);
    if (validateDerivedCheckpoints(derived, totalLines)) {
      table = derived; kind = basis === candidate ? "derived-candidate" : "derived-trusted";
      warnings.push("checkpoints failed validation; a count-derived schedule was produced and re-validated");
    } else if (validateCandidateCheckpoints(trustedBasis, totalLines)) {
      table = trustedBasis.map(c => ({ timeMs: c.timeMs, count: c.count })); kind = "trusted";
      warnings.push("checkpoints rejected; the validated default schedule was restored");
    } else {
      table = [{ timeMs: 0, count: 0 }, { timeMs: finalTime, count: totalLines }];
      kind = "fallback";
      warnings.push("checkpoints rejected; a two-anchor fallback schedule was produced");
    }
  }
  let dur = durationMs, cap = captionAtMs;
  if (!Number.isFinite(dur) || dur <= 0) { dur = defaultDurationMs; warnings.push("duration rejected: not a positive finite value; the default duration was restored"); }
  if (!Number.isFinite(cap)) { cap = defaultCaptionAtMs; warnings.push("caption point rejected: not a finite value; the default caption point was restored"); }
  if (table[table.length - 1].timeMs > dur) {
    table = validateCandidateCheckpoints(trustedBasis, totalLines)
      ? trustedBasis.map(c => ({ timeMs: c.timeMs, count: c.count }))
      : deriveFromBasis(trustedBasis, totalLines);
    kind = "default-restored";
    warnings.push("schedule rejected: its final anchor exceeds the total duration; the default schedule was restored");
    if (table[table.length - 1].timeMs > dur) {
      dur = defaultDurationMs;
      warnings.push("duration rejected: shorter than the default schedule; the default duration was restored");
    }
  }
  if (cap < 0 || cap > dur) { cap = Math.min(Math.max(0, cap), dur); warnings.push("caption point aligned inside the total duration"); }
  return { table, durationMs: dur, captionAtMs: cap, warnings, kind };
}

function lineTimeMs(index, checkpoints) {
  for (let s = 0; s < checkpoints.length - 1; s++) {
    const a = checkpoints[s], b = checkpoints[s + 1];
    if (index < b.count) {
      const span = b.count - a.count, dt = b.timeMs - a.timeMs;
      if (span <= 0) continue;
      return a.timeMs + (index - a.count + 1) * (dt / span);
    }
  }
  return Infinity;
}
function countAtMs(t, checkpoints) {
  const total = checkpoints[checkpoints.length - 1].count;
  let c = 0;
  for (let i = 0; i < total; i++) { if (lineTimeMs(i, checkpoints) <= t) c = i + 1; else break; }
  return c;
}
function pulseScaleAt(t, checkpoints, pulseScale, pulseMs) {
  if (pulseScale === 1) return 1;
  const shown = countAtMs(t, checkpoints);
  if (shown === 0) return 1;
  const lastReveal = lineTimeMs(shown - 1, checkpoints);
  const age = t - lastReveal;
  if (age < 0 || age >= pulseMs) return 1;
  return 1 + (pulseScale - 1) * (1 - age / pulseMs);
}

}
