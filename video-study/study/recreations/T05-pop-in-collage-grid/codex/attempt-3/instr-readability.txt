/* ---- fit models: one model, consumed by validation and by the layout ----
   Label room = panel width − both borders − left padding − the check
   reservation (check offset + check glyph advance). Lyric room = line width
   − prompt advance − prompt margin − cursor footprint. Both are model-only
   inside the supported envelope: monospace families whose advance matches
   the configured advance ratio. */
function scaleForRef(W){ return Math.min(1, Math.max(CONFIG.responsiveScaleFloor, W/960)); }
function labelAvailAt(W){
  const sc=scaleForRef(W);
  const pw=(W*CONFIG.gridWidthPct/100 - CONFIG.gridGapPx*(CONFIG.gridCols-1))/CONFIG.gridCols;
  const checkPx=Math.max(CONFIG.checkMinPx, CONFIG.checkFontPxAt960*sc);
  return pw - 2*CONFIG.panelBorderWidthPx - CONFIG.labelPadLeftPx - (CONFIG.checkRightPx + CONFIG.modelAdvanceEm*checkPx);
}
function lyricAvailWith(W, px){
  return W*CONFIG.lyricWidthPct/100
    - CONFIG.promptText.length*CONFIG.modelAdvanceEm*px
    - CONFIG.promptMarginEm*px
    - (CONFIG.cursorWidthEm*px + CONFIG.cursorMarginPx);
}
function lyricAvailAt(W){ return lyricAvailWith(W, Math.max(CONFIG.lyricMinPx, CONFIG.lyricFontPxAt960*scaleForRef(W))); }
const LABEL_FIT=[];
const LYRIC_FIT={ px:0, rejected:[] };
function correctFit(w){
  const adv=CONFIG.modelAdvanceEm, sc360=scaleForRef(360);
  const baseLabelPx=Math.max(CONFIG.labelMinPx, CONFIG.labelFontPxAt960*sc360);
  const availL=labelAvailAt(360);
  CONFIG.panels.forEach(function(p,i){
    let px=baseLabelPx, text=p.label, shortened=false, hard=false;
    while(px>CONFIG.labelMinPx && text.length*adv*px>availL) px=Math.max(CONFIG.labelMinPx, px-CONFIG.fitStepPx);
    if(text.length*adv*px>availL){
      /* Fit wins: trailing words are dropped first (the identifying head word
         survives longest); if even that word alone cannot fit at the floor
         size, the word itself is truncated — a fitting partial word renders,
         an overflowing full word does not. */
      const words=text.split(/\s+/);
      while(words.length>1 && words.join(" ").length*adv*px>availL){ words.pop(); shortened=true; }
      text=words.join(" ");
      while(text.length>0 && text.length*adv*px>availL){ text=text.slice(0,-1); hard=true; }
    }
    if(px!==baseLabelPx) w.push("LABEL_CORRECTED:"+i+":"+baseLabelPx+"->"+px);
    if(hard) w.push("LABEL_TRUNCATED_HARD:"+i);
    else if(shortened) w.push("LABEL_SHORTENED:"+i);
    LABEL_FIT[i]={ px:px, text:text };
  });
  const baseLyricPx=Math.max(CONFIG.lyricMinPx, CONFIG.lyricFontPxAt960*sc360);
  const rejected=[];
  CONFIG.lyricLines.forEach(function(l,i){
    if(l.startMs + l.text.length*CONFIG.lyricCharMs > CONFIG.durationMs){ rejected.push(i); w.push("LYRIC_REJECTED_COMPLETION:"+i); }
  });
  function lineFits(l,px){ return l.text.length*adv*px <= lyricAvailWith(360,px); }
  let lpx=baseLyricPx, guardN=0;
  while(lpx>CONFIG.lyricMinPx && guardN++<400){
    const bad=CONFIG.lyricLines.some(function(l,i){ return rejected.indexOf(i)<0 && !lineFits(l,lpx); });
    if(!bad) break;
    lpx=Math.max(CONFIG.lyricMinPx, lpx-CONFIG.fitStepPx);
  }
  CONFIG.lyricLines.forEach(function(l,i){
    if(rejected.indexOf(i)<0 && !lineFits(l,lpx)){ rejected.push(i); w.push("LYRIC_REJECTED:"+i); }
  });
  if(lpx!==baseLyricPx) w.push("LYRIC_CORRECTED:"+baseLyricPx+"->"+lpx);
  LYRIC_FIT.px=lpx; LYRIC_FIT.rejected=rejected;
}

    labelPxAt:function(W){ return Math.max(CONFIG.labelMinPx, CONFIG.labelFontPxAt960*scaleForRef(W)); } };
}

function render(){
  const t=elapsed()%CONFIG.durationMs;
  if(prevT!==null && t<prevT) suppressTransitions();
  prevT=t;
  const inReset = CONFIG.resetFadeMs>0 && t > CONFIG.durationMs-CONFIG.resetFadeMs;
  const rf = inReset ? Math.max(0,(CONFIG.durationMs-t)/CONFIG.resetFadeMs) : 1;
  activePanels.forEach(function(p,i){
    panelEls[i].classList.toggle("on", p.initial===true || t>=p.popAtMs);
    panelEls[i].style.setProperty("--fade", String(p.initial===true ? 1 : rf));
  });
  lyricsEl.style.opacity=String(rf);
  let active=-1;
  CONFIG.lyricLines.forEach(function(l,i){
    const row=lyricRows[i], txt=row.querySelector(".lt"), cur=row.querySelector(".cursor");
    if(LYRIC_FIT.rejected.indexOf(i)>=0){ row.style.display="none"; txt.textContent=""; return; }
    if(t<l.startMs){ row.style.display="none"; txt.textContent=""; return; }
    row.style.display="block";
    const n=Math.min(l.text.length, Math.floor((t-l.startMs)/CONFIG.lyricCharMs));
    txt.textContent=l.text.slice(0,n);
    const isActive=(n<l.text.length);
    cur.style.background=isActive?CONFIG.accentGreen:"transparent";
    if(isActive) active=i;
  });
  lyricRows.forEach(function(r,i){ r.querySelector(".cursor").style.background=(i===active)?CONFIG.accentGreen:"transparent"; });
}

function play(){ if(playing)return; playing=true; baseStamp=performance.now(); tick(); }

function restart(){ baseElapsed=0; baseStamp=performance.now(); if(!playing){playing=true; tick();} render(); }
function seek(ms){ baseElapsed=Math.max(0,ms); baseStamp=performance.now(); render(); }
