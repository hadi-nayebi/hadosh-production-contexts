  const s = String(raw).trim();

  return v;
}

function buildTimeline(cfg){
  const events = []; let t = 0;
  for (let i = 0; i < cfg.content.requests.length; i++){
    const w = cfg.timing.modalWindowsMs[i], land = cfg.timing.landDelaysMs[i];
    const flash = cfg.timing.flashModalIndices.indexOf(i) >= 0 ? cfg.timing.arrivalFlashMs : 0;
    events.push({ at: t, modal: i, phase: flash > 0 ? "flash" : "unselected", sel: -1 });
    if (flash > 0) events.push({ at: t + flash, modal: i, phase: "unselected", sel: -1 });
    events.push({ at: t + land, modal: i, phase: "selected", sel: cfg.content.finalSelections[i] });
    events.push({ at: t + w, modal: i, phase: "end", sel: cfg.content.finalSelections[i], holdMs: w - land });
    t += w;
  }
  return { events: events, cycleMs: t, holdsMs: cfg.content.requests.map(function(_, i){ return cfg.timing.modalWindowsMs[i] - cfg.timing.landDelaysMs[i]; }) };
}

function computeLayout(cfg, stageW, stageH, measure){

    let reqPx = T.refPx, optPx = T.refPx;
    /* shrink-to-fit from measured widths at ref size (bold title; bold selected rows incl marker+prefix) */
    let longestReq = 0, longestOpt = 0;

    cfg.content.options.forEach(function(o, i){

    });
    reqPx = Math.floor(T.refPx * (inner / Math.max(1, longestReq)));
    optPx = Math.floor(T.refPx * (inner / Math.max(1, longestOpt)));
    if (phone) optPx = Math.max(optPx, L.phoneBreakpointPx ? T.optFloorPhonePx : optPx);
    reqPx = Math.max(reqPx, Math.ceil(optPx * T.titleRatioMin), optPx + T.titleMinGapPx);
    reqPx = Math.min(reqPx, Math.floor(stageH * T.heightCapRatio * cfg.modalScale) + 14);
    const lines = [];

    cfg.content.requests.forEach(function(r){
      const wdt = lineWidth(r, reqPx, true);
      const wrapped = Math.max(1, Math.ceil(wdt / Math.max(1, inner)));
      if (wrapped > cfg.behavior.maxWrapLines) overflow = true;
      lines.push({ kind: "req", text: r, px: reqPx, width: wdt, inner: inner, wrappedLines: wrapped, outcome: wrapped > 1 ? "wrapped" : "fits" });
    });
    cfg.content.options.forEach(function(o, i){
      const full = "\u276F " + (i + 1) + ". " + o;

    const modalH = borderTotal + 2 * pad * 0.9 + reqLines * reqPx * T.reqLineHeight + reqPx * L.reqMarginBottomEm
      + optLinesTotal * (optPx * T.optLineHeight + 2 * Math.max(2, optPx * L.optPadYRatio));
    return { reqPx: reqPx, optPx: optPx, lines: lines, modalH: modalH, overflow: overflow, inner: inner, refLongestReq: longestReq, refLongestOpt: longestOpt };
  }

  if (ev.overflow || ev.lines.some(function(l){ return l.outcome === "wrapped"; })){
    if (cfg.behavior.longTextPolicy === "reject" && (ev.overflow || ev.lines.some(function(l){ return l.wrappedLines > 1; }))){

    }
  }
  return {

    refLongestReq: ev.refLongestReq, refLongestOpt: ev.refLongestOpt
  };
}

/* ===== message validation: source relationship AND origin AND schema ===== */
function validateMessage(cfg, msg){
  if (!msg || typeof msg !== "object") return { ok: false, reason: "shape" };
  if (msg.sourceIsParent !== true) return { ok: false, reason: "source" };
  const origins = cfg.container.allowedOrigins;
  if (origins.indexOf(msg.origin) < 0 && origins.indexOf("*") < 0) return { ok: false, reason: "origin" };
  if (["play", "pause", "restart"].indexOf(msg.cmd) < 0) return { ok: false, reason: "schema" };
  return { ok: true, reason: "ok" };
}

    const target = origin + timeline.events[evIdx].at;
    const delay = Math.max(0, target - performance.now());
    timer = setTimeout(function(){
      /* catch-up: apply every event whose scheduled boundary has passed, in order */
      const nowAbs = performance.now() - origin;
      while (evIdx < timeline.events.length && timeline.events[evIdx].at <= nowAbs){ applyEvent(timeline.events[evIdx]); evIdx++; }
      scheduleNext();
    }, delay);
  }
  function play(){
    if (playing) return; playing = true;
    if (pausedAt !== null && origin !== null){ origin += performance.now() - pausedAt; pausedAt = null; }
    else if (origin === null){ origin = performance.now(); evIdx = 0; }
    scheduleNext();
  }
  function pause(){ if (!playing) return; playing = false; pausedAt = performance.now(); if (timer) clearTimeout(timer); timer = null; }
  function restart(){ pause(); origin = null; evIdx = 0; cycles = 0; pausedAt = null; current = { modal: 0, phase: "unselected", sel: -1 }; play(); }
  window.addEventListener("resize", render);
  window.addEventListener("message", function(e){
    const verdict = validateMessage(cfg, { sourceIsParent: e.source === window.parent, origin: e.origin, cmd: e.data && e.data.cmd, data: e.data });
    if (!verdict.ok){ record("reject-message", verdict.reason); document.documentElement.dataset.t07Validation = String(VALIDATION.length); return; }
    if (e.data.cmd === "play") play(); else if (e.data.cmd === "pause") pause(); else if (e.data.cmd === "restart") restart();
  });
  render(); emit("ready", { cycleMs: timeline.cycleMs }); play();
})();
