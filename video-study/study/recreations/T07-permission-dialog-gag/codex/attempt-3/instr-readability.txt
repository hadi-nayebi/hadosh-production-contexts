function processElapsed(timeline, runs, totalElapsedMs){
  const cycleMs = timeline.cycleMs;
  let cyclesCompleted = Math.floor(totalElapsedMs / cycleMs);
  let finalCompleteCount = 0, exhausted = false;
  if (isFinite(runs) && cyclesCompleted >= runs){ cyclesCompleted = runs; exhausted = true; finalCompleteCount = 1; }
  const within = exhausted ? cycleMs : totalElapsedMs - Math.floor(totalElapsedMs / cycleMs) * cycleMs;

    const ev = timeline.events[i];
    if (exhausted || ev.at <= within){ visible = ev; processed++; }

function wrapCount(text, inner, px, bold, cfg, measure){
  if (!(inner > 0)) return { lines: 1, oversizedToken: true };
  const words = String(text).split(/\s+/).filter(function(w){ return w.length; });
  if (!words.length) return { lines: 1, oversizedToken: false };
  let lines = 1, cur = "";
  let oversized = false;
  words.forEach(function(w){
    if (measure(w, px, bold, cfg.type.family) > inner) oversized = true;
    const cand = cur ? cur + " " + w : w;
    if (measure(cand, px, bold, cfg.type.family) <= inner || !cur){ cur = cand; }
    else { lines++; cur = w; }
  });
  return { lines: lines, oversizedToken: oversized };
}
