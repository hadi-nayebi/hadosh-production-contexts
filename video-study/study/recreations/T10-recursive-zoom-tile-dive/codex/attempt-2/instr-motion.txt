  if (eff.tiles.length !== eff.gridCols * eff.gridRows) return fail("tiles.length != gridCols*gridRows");
  if (!(eff.targetIndex >= 0 && eff.targetIndex < eff.tiles.length)) return fail("targetIndex out of range");
  return { effective: eff, rejected: rejected, scaleFactor: scaleFactor, minSepMs: minSep };
}

/* Pure state from time — the normalized checkpoint table decides every phase. */
function computeState(tMs, cfg){
  const t = ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs;
  const cp = cfg.checkpoints;
  const ease = EASINGS[cfg.easing] || EASINGS.easeInOutCubic;
  const diveRaw = clamp01((t - cp.diveStart) / Math.max(1, cp.diveEnd - cp.diveStart));
  const dive = ease(diveRaw);
  const scale = 1 + (cfg.zoomScale - 1) * dive;
  let pill = clamp01((t - cp.pillAt) / cfg.pillFadeMs);
  if (t >= cp.pillOutAt) pill = pill * (1 - clamp01((t - cp.pillOutAt) / cfg.pillOutFadeMs));
  const status = clamp01((t - cp.statusAt) / cfg.statusFadeMs);
  const col = cfg.targetIndex % cfg.gridCols;
  const row = Math.floor(cfg.targetIndex / cfg.gridCols);
  const originX = (col + 0.5) / cfg.gridCols * 100;
  const originY = (row + 0.5) / cfg.gridRows * 100;
  return { t: t, diveRaw: diveRaw, dive: dive, scale: scale, pill: pill, status: status,
           originX: originX, originY: originY, targetCol: col, targetRow: row };
}

function resolveColor(role, tile, cfg){
  if (role === "accent") return tile.accent;
  if (role === "detail") return tile.detail;
  if (role === "bg") return tile.bg;
  if (typeof role === "string" && role.charAt(0) === "#") return role;
  if (typeof role === "string" && typeof cfg[role] === "string") return cfg[role];
  throw namedError("T10DescriptorError", "unknown descriptor color role: " + role);
}
function renderDescriptor(parent, d, tile, cfg){
  const nums = ["left","top","width","height"];
  for (const k of nums){ if (typeof d[k] !== "number" || !isFinite(d[k])) throw namedError("T10DescriptorError", "descriptor missing numeric " + k); }
  const s = document.createElement("div");
  s.className = "shape";
  s.style.left = d.left + "%"; s.style.top = d.top + "%";
  s.style.width = d.width + "%"; s.style.height = d.height + "%";
  if (typeof d.borderRadius === "number") s.style.borderRadius = d.borderRadius + "%";
  s.style.background = resolveColor(d.color, tile, cfg);
  parent.appendChild(s); return s;
}
function buildTiles(wallEl, cfg){
  wallEl.innerHTML = "";
  const tileEls = [];
  for (let i = 0; i < cfg.tiles.length; i++){
    const tile = cfg.tiles[i];
    if (!tile || !Array.isArray(tile.art)) throw namedError("T10DescriptorError", "tile " + i + " has no art descriptor array");
    const d = document.createElement("div");
    d.className = "tile";
    d.style.background = tile.bg;
    d.style.borderColor = (i === cfg.targetIndex) ? cfg.highlightColor : cfg.tileBorderColor;
    d.style.borderWidth = ((i === cfg.targetIndex) ? cfg.highlightBorderPx : cfg.tileBorderPx) + "px";
    const art = document.createElement("div");
    art.className = "tileArt";
    for (const desc of tile.art) renderDescriptor(art, desc, tile, cfg);
    d.appendChild(art);
    const lab = document.createElement("div");
    lab.className = "tileLabel";
    lab.textContent = tile.label;
    lab.style.left = cfg.labelLeftPct + "%";
    lab.style.bottom = cfg.labelBottomPct + "%";
    lab.style.fontSize = cfg.tileLabelFontPx + "px";
    lab.style.color = cfg.labelColor;
    lab.style.textShadow = "0 1px 2px " + cfg.labelShadowColor;
    d.appendChild(lab);
    wallEl.appendChild(d); tileEls.push(d);
  }
  return tileEls;
}

function render(tMs){
  const s = computeState(tMs, EFF);
  wall.style.transformOrigin = s.originX + "% " + s.originY + "%";
  wall.style.transform = "scale(" + s.scale.toFixed(4) + ")";
  pill.style.opacity = s.pill.toFixed(3);
  /* transformed-bounds clamp: price the pill's DISPLAYED rect, slide offset included */
  const sw = stage.clientWidth || 0, pw = pill.offsetWidth || 0;
  const slidePx = (1 - s.pill) * (EFF.pillSlidePct / 100) * pw;
  pill.style.transform = "translateX(" + slidePx.toFixed(2) + "px)";
  if (sw && pw){
    const desired = sw * (50 + EFF.wallWidthPct / 2 + EFF.pillOffsetXPct) / 100;
    const maxLeft = Math.max(0, sw - pw - slidePx - sw * EFF.pillEdgeMarginPct / 100);
    pill.style.left = Math.min(desired, maxLeft).toFixed(1) + "px";
  } else {
    pill.style.left = (50 + EFF.wallWidthPct / 2 + EFF.pillOffsetXPct) + "%";
  }
  statusPill.style.display = s.status > 0 ? "flex" : "none";
  statusPill.style.opacity = s.status.toFixed(3);
  return s;
}

let playing = false, startStamp = 0, pausedAt = 0, rafId = null;
function nowMs(){ return (typeof performance !== "undefined" && performance.now) ? performance.now() : Date.now(); }
function tick(){ if (!playing) return; render(nowMs() - startStamp); rafId = requestAnimationFrame(tick); }
function play(){ if (playing) return; playing = true; startStamp = nowMs() - pausedAt; tick(); }
function pause(){ if (!playing) return; pausedAt = nowMs() - startStamp; playing = false; if (rafId) cancelAnimationFrame(rafId); rafId = null; }
function restart(){ pausedAt = 0; if (playing){ startStamp = nowMs(); } else { render(0); } }

window.addEventListener("message", function(e){
  if (!e.data || typeof e.data !== "object") return;
  if (e.data.cmd === "play") play();
  if (e.data.cmd === "pause") pause();
  if (e.data.cmd === "restart") restart();
});
