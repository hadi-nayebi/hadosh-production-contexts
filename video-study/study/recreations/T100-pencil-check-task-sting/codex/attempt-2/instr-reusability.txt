- Phases (exclusive, in order): ENTER (pencil slides from its configured corner to the tick start) → DRAW (pencil tip travels the curved tick path as the stroke reveals; the checkbox exits at a configured fraction of draw progress, 0.85; the tick completes, then clears at the end of DRAW) → punctuation beat 1 (red family: a curved arrow reveals by stroke draw-on while scattered confetti rectangles displace outward from a configured start fraction to their full offsets) → punctuation beat 2 (green family, same construction) → HOLD (empty indigo field; all transient marks cleared) → loop.
- Tick-clear timing specification: the tick is visible during ENTER (undrawn) and DRAW (revealing); its opacity is set to 0 in every phase after DRAW, so no tick is displayed during either punctuation beat or the hold.

  return base;
}

function setBeat(b, p){
  if (p <= 0){ b.group.setAttribute("opacity", "0"); return; }
  b.group.setAttribute("opacity", "1");
  const e = easeOut(clamp01(p));
  b.arrow.style.strokeDashoffset = b.arrowLen * (1 - e);
  b.head.setAttribute("opacity", e > G.headAt ? "1" : "0");
  const s = b.cfg.confettiStartScale + (1 - b.cfg.confettiStartScale) * e;
  const travel = b.cfg.confettiFrom + (1 - b.cfg.confettiFrom) * e;
  for (const r of b.confetti){
    const x = C.checkbox.cx + parseFloat(r.dataset.dx) * travel;
    const y = C.checkbox.cy + parseFloat(r.dataset.dy) * travel;
    r.setAttribute("transform", "translate(" + x + " " + y + ") rotate(" + r.dataset.deg + ") scale(" + s + ")");
    r.setAttribute("opacity", "1");
  }
}
function frame(){
  const el = elapsed();
  const t = el % C.cycleMs;
  const cyc = Math.floor(el / C.cycleMs);
  if (cyc !== lastCycle){ lastCycle = cyc; buildBeats(); }
  const P = phases();
  const startPt = tick.getPointAtLength(0);
  if (t < C.enterMs){
    const sx = C.pencil.enterFrom.indexOf("right") >= 0 ? 1 : -1;
    const sy = C.pencil.enterFrom.indexOf("bottom") >= 0 ? 1 : -1;
    const p = easeOut(clamp01(t / C.enterMs));
    const dx = sx * Math.abs(C.pencil.enterDx) * (1 - p), dy = sy * Math.abs(C.pencil.enterDy) * (1 - p);
    pencil.setAttribute("transform", "translate(" + (startPt.x + dx) + " " + (startPt.y + dy) + ")");
    pencil.setAttribute("opacity", String(clamp01(p * G.enterFadeGain)));
    tick.style.strokeDashoffset = tickLen;
    tick.setAttribute("opacity", "1");
    box.setAttribute("opacity", "1");
    setBeat(beats[0], 0); setBeat(beats[1], 0);
