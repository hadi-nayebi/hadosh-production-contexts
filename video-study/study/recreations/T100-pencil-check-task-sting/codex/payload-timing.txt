function applyOverride(base, over){

  }

}

/* Burst marks: independently allocated per cycle; rebuilt each loop so a
   changed burst.count can never leave frozen remnants in hold or across

const burstLayer = document.getElementById("burstLayer");
let burstMarks = [];
function buildBurst(){
  burstLayer.innerHTML = "";
  burstMarks = [];
  for (let i = 0; i < C.burst.count; i++){
    const a = (2 * Math.PI * i) / C.burst.count;

/* Phase machine — phase exclusivity (Owner Cure 2): enter, then draw, then
   burst, then hold. Fixed-total invariant: cycle is exactly cycleMs; the
   hold is the residual (cycleMs - enter - draw - burst), so a changed
   tick.drawMs re-derives the hold and never stretches the cycle. */
function phases(){
  const drawEnd = C.enterMs + C.tick.drawMs;
  const burstEnd = drawEnd + C.burstMs;
  return {drawEnd, burstEnd};
}
function clamp01(v){ return v < 0 ? 0 : v > 1 ? 1 : v; }
function easeOut(v){ return 1 - Math.pow(1 - v, G.easePow); }

/* Clock with comparison-page sync control (play / pause / restart). */
let base = performance.now(), acc = 0, paused = false;
function elapsed(){ return paused ? acc : acc + (performance.now() - base); }
window.addEventListener("message", function(ev){
  const m = ev.data && ev.data.t100;
  if (m === "pause" && !paused){ acc = elapsed(); paused = true; }
  else if (m === "play" && paused){ base = performance.now(); paused = false; }
  else if (m === "restart"){ acc = 0; base = performance.now(); lastCycle = -1; }
});
let lastCycle = -1;
function frame(){
  const el = elapsed();
  const t = el % C.cycleMs;
  const cyc = Math.floor(el / C.cycleMs);
  if (cyc !== lastCycle){ lastCycle = cyc; buildBurst(); }
  const P = phases();
  if (t < C.enterMs){
    /* ENTER: pencil slides in from its CONFIG enter corner to the tick end.
       The corner is read from pencil.enterFrom; geom offsets give magnitude. */
    const sx = C.pencil.enterFrom.indexOf("right") >= 0 ? 1 : -1;
    const sy = C.pencil.enterFrom.indexOf("bottom") >= 0 ? 1 : -1;
    const p = easeOut(clamp01(t / C.enterMs));
    const dx = sx * Math.abs(G.pencilEnterDx) * (1 - p), dy = sy * Math.abs(G.pencilEnterDy) * (1 - p);
    pencil.setAttribute("transform", "translate(" + (G.pencilTipX + dx) + " " + (G.pencilTipY + dy) + ")");
    pencil.setAttribute("opacity", String(clamp01(p * G.enterFadeGain)));
    tick.style.strokeDashoffset = tickLen;
    setBurst(0);
  } else if (t < P.drawEnd){
    /* DRAW: the pencil tip travels the tick path as the tick draws on. */
    const p = clamp01((t - C.enterMs) / C.tick.drawMs);
    const pt = tick.getPointAtLength(p * tickLen);
    pencil.setAttribute("transform", "translate(" + pt.x + " " + pt.y + ")");
    pencil.setAttribute("opacity", "1");
    tick.style.strokeDashoffset = tickLen * (1 - p);
    setBurst(0);
  } else if (t < P.burstEnd){
    /* BURST: pencil gone; burst marks punctuate around the checked box. */
    const p = clamp01((t - P.drawEnd) / C.burstMs);
    pencil.setAttribute("opacity", "0");
    tick.style.strokeDashoffset = 0;
    setBurst(p);
  } else {
    /* HOLD: checked state only — box + full tick, no transient marks. */
    pencil.setAttribute("opacity", "0");
    tick.style.strokeDashoffset = 0;
    setBurst(0);
  }
  requestAnimationFrame(frame);
}
function setBurst(p){
  const s = G.burstStartScale + (1 - G.burstStartScale) * easeOut(p);
  const o = p <= 0 ? 0 : 1;
  for (const r of burstMarks){
    r.setAttribute("transform", "translate(" + r.dataset.x + " " + r.dataset.y + ") rotate(" + r.dataset.deg + ") scale(" + s + ")");
    r.setAttribute("opacity", String(o));
  }
}
requestAnimationFrame(frame);
