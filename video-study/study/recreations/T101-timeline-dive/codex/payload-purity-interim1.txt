function pose(t){
  if (t < CONFIG.travelMs) {           /* TRAVEL: lateral move at scale 1 */
    const u = ease(t / CONFIG.travelMs);
    return {x: travelPx*u, y:0, s:1};
  }
  if (t < CONFIG.travelMs + CONFIG.diveMs) {  /* DIVE: push in until card fills frame */
    const u = ease((t - CONFIG.travelMs) / CONFIG.diveMs);
    const s = 1 + (CONFIG.dive.endScale - 1)*u;
    const endX = stageCx - cardCx*CONFIG.dive.endScale;
    const endY = stageCy - cardCy*CONFIG.dive.endScale;

  const s = CONFIG.dive.endScale;      /* HOLD: card fills frame */
  return {x: stageCx - cardCx*s, y: stageCy - cardCy*s, s: s};
}
function render(tms){
  const t = ((tms % CYCLE) + CYCLE) % CYCLE;
  const p = pose(t);
  world.style.transform = "translate("+p.x+"px,"+p.y+"px) scale("+p.s+")";
  const prog = Math.min(t / (CONFIG.travelMs + CONFIG.diveMs), 1);
  playhead.style.left = (prog * CONFIG.layout.worldW * CONFIG.graph.span) + "px";
}

  if (start === null) start = ts;
