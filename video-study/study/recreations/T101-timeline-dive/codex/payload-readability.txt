   global progress drives all three, so camera velocity never falls to zero at
   the travel/dive boundary — the mechanism accumulates in one continuous move. */

const POSE_END = {x: stageCx - cardCx*CONFIG.dive.endScale, y: stageCy - cardCy*CONFIG.dive.endScale, s: CONFIG.dive.endScale};
function lerpPose(a, b, u){ return {x: a.x+(b.x-a.x)*u, y: a.y+(b.y-a.y)*u, s: a.s+(b.s-a.s)*u}; }
function pose(t){
  const active = CONFIG.travelMs + CONFIG.diveMs;
  if (t >= active) return POSE_END;                       /* HOLD */
  const u = ease(t / active);                             /* one continuous ease */
  const split = ease(CONFIG.travelMs / active);           /* travel pose completes exactly at travelMs, velocity nonzero */
  if (u < split) return lerpPose({x:0,y:0,s:1}, POSE_TRAVEL_END, u/split);
  return lerpPose(POSE_TRAVEL_END, POSE_END, (u-split)/(1-split));
}
function render(tms){

  const p = pose(t);
  world.style.transform = "translate("+p.x+"px,"+p.y+"px) scale("+p.s+")";
  /* designed loop reset: fade out/in across the seam (CONFIG loopFadeMs) */
  const fade = CONFIG.layout.loopFadeMs;
  const edge = Math.min(t, CYCLE - t);
  world.style.opacity = edge < fade ? String(0.15 + 0.85*(edge/fade)) : "1";

  playhead.style.left = (prog * CONFIG.layout.worldW * CONFIG.graph.span) + "px";
}
