/* ============ override (?cfg= JSON) ============ */
function deepMerge(base, over){
  if (Array.isArray(over)){
    if (!Array.isArray(base)) return over.slice();
    const merged = base.map(function(b, i){ return i < over.length ? deepMerge(b, over[i]) : b; });
    for (let i = base.length; i < over.length; i++) merged.push(over[i]);
    return merged;
  }
  if (over && typeof over === "object"){
    const out = Array.isArray(base) ? base.slice() : Object.assign({}, base);
    for (const k of Object.keys(over)) out[k] = deepMerge(base ? base[k] : undefined, over[k]);
    return out;
  }
  return over;
}
function clone(o){ return JSON.parse(JSON.stringify(o)); }
function resolveConfig(raw){
  if (raw === null || raw === undefined || raw === "") return {cfg: clone(CONFIG), errors: []};
  let over;
  try { over = JSON.parse(raw); } catch(e){ return {cfg: null, errors: ["cfg: malformed JSON"]}; }
  if (!over || typeof over !== "object" || Array.isArray(over)) return {cfg: null, errors: ["cfg: expected object"]};
  return {cfg: deepMerge(clone(CONFIG), over), errors: []};
}

/* ============ pure logic (node-testable) ============ */
function cycleMs(cfg){ return cfg.landMs + cfg.holdMs; }
function easeOutCubic(p){ return 1 - Math.pow(1 - p, 3); }
function phaseAt(t, cfg){
  const cyc = cycleMs(cfg);
  const u = ((t % cyc) + cyc) % cyc;
  if (u < cfg.landMs){
    const p = easeOutCubic(u / cfg.landMs);
    return {phase: "LAND", u: u, p: p,
      scale: cfg.motion.fromScale + (1 - cfg.motion.fromScale) * p,
      opacity: cfg.motion.fromOpacity + (1 - cfg.motion.fromOpacity) * p,
      risePx: cfg.motion.fromRisePx * (1 - p)};
  }
  return {phase: "HOLD", u: u, p: 1, scale: 1, opacity: 1, risePx: 0};
}
function gradientCSS(cfg){

function hexRGB(h){
  return [parseInt(h.slice(1,3),16), parseInt(h.slice(3,5),16), parseInt(h.slice(5,7),16)];
}
function relLum(rgb){
  function f(c){ c = c/255; return c <= 0.03928 ? c/12.92 : Math.pow((c+0.055)/1.055, 2.4); }
  return 0.2126*f(rgb[0]) + 0.7152*f(rgb[1]) + 0.0722*f(rgb[2]);
}
function contrastRatio(fgHex, bgRGB){
  const l1 = relLum(hexRGB(fgHex)), l2 = relLum(bgRGB);
  const hi = Math.max(l1,l2), lo = Math.min(l1,l2);
  return (hi + 0.05) / (lo + 0.05);
}
function blendOver(topHex, opacity, botRGB){
  const t = hexRGB(topHex);
  return [0,1,2].map(function(i){ return opacity*t[i] + (1-opacity)*botRGB[i]; });
}
/* Worst-case ink contrast: ink over the text backing composited on every
   gradient stop and every interpolated midpoint between consecutive stops. */
function worstCaseContrast(cfg){
  const stops = cfg.gradient.stops.map(hexRGB);
  const samples = stops.slice();
  for (let i = 0; i < stops.length - 1; i++)
    for (const f of [0.25, 0.5, 0.75])
      samples.push([0,1,2].map(function(k){ return stops[i][k] + f*(stops[i+1][k]-stops[i][k]); }));
  let worst = Infinity;
  for (const s of samples){
    const bg = blendOver(cfg.textBack.color, cfg.textBack.opacity, s);
    worst = Math.min(worst, contrastRatio(cfg.palette.ink, bg));
  }
  return worst;
}
/* Character width model (Arial advance widths, em fractions) for the fit harness. */
const CHAR_W = {upper: 0.68, lower: 0.52, space: 0.278, digit: 0.556, punct: 0.30};
function textWidthPx(text, fontPx, weight){
  let em = 0;
  for (const ch of text){
    if (ch === " ") em += CHAR_W.space;
    else if (ch >= "A" && ch <= "Z") em += CHAR_W.upper;
    else if (ch >= "0" && ch <= "9") em += CHAR_W.digit;
    else if (".,;:!?'’-—".indexOf(ch) >= 0) em += CHAR_W.punct;
    else em += CHAR_W.lower;
  }
  if (weight >= 700) em *= 1.06;
  return em * fontPx;
}
function wrapLines(text, fontPx, weight, maxW){
  const words = text.split(" ");
  const lines = []; let cur = "";
  for (const w of words){
    const trial = cur === "" ? w : cur + " " + w;
    if (textWidthPx(trial, fontPx, weight) <= maxW || cur === "") cur = trial;
    else { lines.push(cur); cur = w; }
  }
  if (cur !== "") lines.push(cur);
  return lines;
}
function layoutModel(cfg){
  const W = cfg.stage.width;
  const H = Math.round(W * cfg.geom.stageAspectH / cfg.geom.stageAspectW);
  const cardW = Math.round(W * cfg.geom.cardWFrac), cardH = Math.round(H * cfg.geom.cardHFrac);
  const cardX = Math.round((W - cardW) / 2), cardY = Math.round((H - cardH) / 2);
  const bodyMaxW = cardW * cfg.geom.bodyWFrac;
  const bodyLines = wrapLines(cfg.card.body, cfg.type.bodyPx, cfg.type.bodyWeight, bodyMaxW);
  const titleW = textWidthPx(cfg.card.title, cfg.type.titlePx, cfg.type.titleWeight) + cfg.type.titleLetterSpacingPx * Math.max(0, cfg.card.title.length - 1);
  return {stageW: W, stageH: H, cardX: cardX, cardY: cardY, cardW: cardW, cardH: cardH,

  const path = "M " + ax0 + " " + archBase + " L " + ax0 + " " + (archTop + (ax1-ax0)/2) +
    " A " + ((ax1-ax0)/2) + " " + ((ax1-ax0)/2) + " 0 0 1 " + ax1 + " " + (archTop + (ax1-ax0)/2) +
    " L " + ax1 + " " + archBase;
  svg.innerHTML =
    '<defs><linearGradient id="capGrad" x1="0" y1="0" x2="0" y2="1">' +
    '<stop offset="0" stop-color="' + fg.capsuleFrom + '"/><stop offset="1" stop-color="' + fg.capsuleTo + '"/></linearGradient></defs>' +
    '<path d="' + path + '" fill="none" stroke="' + fg.pathColor + '" stroke-width="' + (sw * fg.pathWidthFrac) +
    '" stroke-dasharray="' + (sw * fg.pathDashFrac) + ' ' + (sw * fg.pathDashFrac) + '" stroke-linecap="round"/>' +
    '<rect x="' + capX + '" y="' + capY + '" width="' + capW + '" height="' + capH + '" rx="' + (capW/2) + '" fill="url(#capGrad)"/>';
  return L;
}
