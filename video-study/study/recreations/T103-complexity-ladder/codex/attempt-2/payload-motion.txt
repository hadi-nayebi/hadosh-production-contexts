## Construction
- DOM three-row menu card; one persistent marker node repositioned from derived state; render(t) is a pure function of elapsed time (seek, throttle and wrap cannot strand state).
- Two acts selected by configuration. Act "segment" (default): a phase table in configuration (total / assembly start / names / items / departure boundaries) drives a five-phase machine — absent, assembly, names-only, marked, departed. Act "stepping": per tier, a names-only span, a marked span (that row full ink + its items), and a return gap with the device absent; after the last tier a terminal hold on the final marked state, then wrap. Traversal order derives from the configured start row.
- Items visibility is a phase output: items render only in marked phases.
- Assembly: row offsets and marker position interpolate from entry positions to settled positions as a function of the assembly phase's progress. Dimming is colour-based at every state; no opacity is used anywhere.
- Row heights are derived per row from content (name line + gap + per-item lines + padding, lines derived from configured type sizes); row tops are cumulative; stage height derives from the rows.
- Configuration validation runs before any override is applied: structural, range, finiteness and ordering checks; an invalid override is rejected as a whole (the prior valid configuration stays in force) and the rejection is recorded in an observable list.
- Clock ownership: standalone, the file starts its own animation loop at load and renders elapsed time from its own first frame; an external driver may call the exposed render function at any moment without corrupting the loop (render holds no accumulated state).

function rowHeight(c, i){
  const nameLine = Math.ceil(c.type.namePx * c.type.lineScale);
  const itemLine = Math.ceil(c.type.itemPx * c.type.lineScale);
  return nameLine + c.geom.nameItemGap + c.rows[i].items.length * (itemLine + c.geom.itemGap) + c.geom.rowPad;
}
function rowTops(c){
  const tops = []; let acc = c.geom.rowTop;
  for (let i = 0; i < c.rows.length; i++){ tops.push(acc); acc += rowHeight(c, i); }
  return tops;
}
function stageHeight(c){
  const tops = rowTops(c);
  return tops[tops.length - 1] + rowHeight(c, c.rows.length - 1) + c.geom.rowTop;
}
function totalMs(c){
  if (c.act === "stepping"){
    const per = c.allocNamesMs + c.stepMs + c.allocReturnMs;
    return c.rows.length * per - c.allocReturnMs + c.holdMs;
  }
  return c.seg.totalMs;
}
function stateAt(c, t){
  const total = totalMs(c);
  const tt = ((t % total) + total) % total;
  if (c.act === "stepping"){
    const n = c.rows.length;
    const per = c.allocNamesMs + c.stepMs + c.allocReturnMs;
    for (let k = 0; k < n; k++){
      const rowIndex = (c.marker.startRow + k) % n;
      const base = k * per;
      const last = k === n - 1;
      const markedEnd = base + c.allocNamesMs + c.stepMs + (last ? c.holdMs : 0);
      if (tt < base + c.allocNamesMs) return { t: tt, totalMs: total, act: c.act, phase: "names", rowIndex: rowIndex, itemsVisible: false, present: true, settle: 1 };
      if (tt < markedEnd) return { t: tt, totalMs: total, act: c.act, phase: "marked", rowIndex: rowIndex, itemsVisible: true, present: true, settle: 1 };
      if (!last && tt < base + per) return { t: tt, totalMs: total, act: c.act, phase: "return", rowIndex: rowIndex, itemsVisible: false, present: false, settle: 1 };
    }
    return { t: tt, totalMs: total, act: c.act, phase: "marked", rowIndex: (c.marker.startRow + n - 1) % n, itemsVisible: true, present: true, settle: 1 };
  }
  const s = c.seg;
  const startRow = ((c.marker.startRow % c.rows.length) + c.rows.length) % c.rows.length;
  if (tt < s.assemblyStartMs) return { t: tt, totalMs: total, act: c.act, phase: "absent", rowIndex: startRow, itemsVisible: false, present: false, settle: 0 };
  if (tt < s.namesAtMs) return { t: tt, totalMs: total, act: c.act, phase: "assembly", rowIndex: startRow, itemsVisible: false, present: true, settle: (tt - s.assemblyStartMs) / (s.namesAtMs - s.assemblyStartMs) };
  if (tt < s.itemsAtMs) return { t: tt, totalMs: total, act: c.act, phase: "names", rowIndex: startRow, itemsVisible: false, present: true, settle: 1 };
  if (tt < s.departAtMs) return { t: tt, totalMs: total, act: c.act, phase: "marked", rowIndex: startRow, itemsVisible: true, present: true, settle: 1 };
  return { t: tt, totalMs: total, act: c.act, phase: "departed", rowIndex: startRow, itemsVisible: false, present: false, settle: 1 };
}
function validateConfig(c){
  const errs = [];
  if (!c || typeof c !== "object") return ["config not an object"];
  if (!Array.isArray(c.rows) || c.rows.length === 0) errs.push("rows must be a non-empty array");
  else c.rows.forEach(function(r, i){
    if (!r || typeof r.name !== "string" || r.name.length === 0) errs.push("rows[" + i + "].name must be a non-empty string");
    if (!r || !Array.isArray(r.items)) errs.push("rows[" + i + "].items must be an array");
  });
  if (!c.marker || typeof c.marker !== "object") errs.push("marker group missing");
  else {
    if (!Number.isInteger(c.marker.startRow)) errs.push("marker.startRow must be an integer");
    else if (Array.isArray(c.rows) && c.rows.length && (c.marker.startRow < 0 || c.marker.startRow >= c.rows.length)) errs.push("marker.startRow out of range");
  }
  if (!c.seg || typeof c.seg !== "object") errs.push("seg group missing");
  else {
    const keys = ["totalMs", "assemblyStartMs", "namesAtMs", "itemsAtMs", "departAtMs"];
    keys.forEach(function(k){ if (!Number.isFinite(c.seg[k]) || c.seg[k] < 0) errs.push("seg." + k + " must be finite and >= 0"); });
    if (errs.length === 0 && !(c.seg.assemblyStartMs < c.seg.namesAtMs && c.seg.namesAtMs < c.seg.itemsAtMs && c.seg.itemsAtMs < c.seg.departAtMs && c.seg.departAtMs <= c.seg.totalMs)) errs.push("seg ordering violated");
  }
  [["stepMs", 1], ["holdMs", 0], ["allocNamesMs", 0], ["allocReturnMs", 0]].forEach(function(pair){
    if (!Number.isFinite(c[pair[0]]) || c[pair[0]] < pair[1]) errs.push(pair[0] + " out of range");
  });
  return errs;
}
function merge(base, over){
  if (Array.isArray(over)) return over.slice();
  if (over && typeof over === "object"){
    const out = {};
    for (const k in base) out[k] = merge(base[k], over[k]);
    for (const k in over) if (!(k in base)) out[k] = over[k];
    return out;
  }
  return over === undefined ? base : over;
}
