/* ---------- ONE clock: A = 1-p, B = p ---------- */
function easeFn(name, x){
  if (name === "smoothstep") return x * x * (3 - 2 * x);
  return x;
}

const D1 = D0 + CONFIG.dissolveMs;
const D2 = D1 + CONFIG.holdBMs;
const D3 = D2 + CONFIG.loopFadeMs;
const CYCLE = CONFIG.cycleMs;
function progressAt(t){
  t = ((t % CYCLE) + CYCLE) % CYCLE;
  if (t < D0) return 0;
  if (t < D1) return easeFn(CONFIG.ease, (t - D0) / CONFIG.dissolveMs);
  if (t < D2) return 1;
  if (t < D3) return 1 - easeFn(CONFIG.ease, (t - D2) / CONFIG.loopFadeMs);
  return 0;
}
function render(t){

  return window.__state;
}
