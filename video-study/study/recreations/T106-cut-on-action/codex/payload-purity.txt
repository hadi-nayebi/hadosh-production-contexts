function easeInOutCubic(p){ return p < 0.5 ? 4*p*p*p : 1 - Math.pow(-2*p + 2, 3) / 2; }

function thetaAt(t){
  if (t < CONFIG.holdBeforeMs) return CONFIG.lid.fromDeg;
  if (t < ACTION_END){
    const p = (t - CONFIG.holdBeforeMs) / CONFIG.lid.actionMs;
    return CONFIG.lid.fromDeg + (CONFIG.lid.toDeg - CONFIG.lid.fromDeg) * easeInOutCubic(p);

  return CONFIG.lid.toDeg;
}
function screenOpacityAt(theta){
  const f = (theta - CONFIG.lid.screenOnDeg) / (CONFIG.lid.screenFullDeg - CONFIG.lid.screenOnDeg);

/* ---------- render: phase predicates partition the cycle; cutMs belongs to scene B ---------- */
function render(t){
  const tt = ((t % CYCLE) + CYCLE) % CYCLE;
  const theta = thetaAt(tt);
  lidEl.style.transform = "rotate(" + (-theta) + "deg)";
  const so = screenOpacityAt(theta);
  lidScreen.style.opacity = so;
  beam.style.opacity = CONFIG.laptop.beamOpacity * so;
  const inA = tt < CUT_MS;
  sceneA.style.display = inA ? "block" : "none";
  sceneB.style.display = inA ? "none" : "block";
  const fadeStart = CYCLE - CONFIG.loopFadeMs;
  fadeWrap.style.opacity = tt > fadeStart ? Math.max(0, (CYCLE - tt) / CONFIG.loopFadeMs) : 1;
}
window.__render = render;
window.__theta = thetaAt;
window.__state = function(t){
  const tt = ((t % CYCLE) + CYCLE) % CYCLE;
  return {theta: thetaAt(tt), scene: tt < CUT_MS ? "A" : "B", cutMs: CUT_MS, cycle: CYCLE, thetaCut: THETA_CUT};
};

let startStamp = null;
let manualUntil = -1;
function loop(now){
  if (startStamp === null) startStamp = now;
  if (now > manualUntil) render(now - startStamp);
  requestAnimationFrame(loop);
}
const origRender = window.__render;
window.__render = function(t){
  manualUntil = performance.now() + 600;
  origRender(t);
};
if (typeof requestAnimationFrame === "function") requestAnimationFrame(loop);
