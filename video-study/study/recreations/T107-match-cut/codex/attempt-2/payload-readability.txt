  const growthE = 1 - Math.pow(1 - settleU, 3);
  const markScaleB = 1 + (CONFIG.settleScale - 1) * growthE;
  let wordmarkU = 0;
  if (t >= CUT_MS) wordmarkU = Math.min((t - CUT_MS) / CONFIG.wordmarkMs, 1);
  const wordmarkE = 1 - Math.pow(1 - wordmarkU, 3);

  const fade = CONFIG.loopFadeMs;
  const edge = Math.min(t, CONFIG.cycleMs - t);
  const stageOpacity = edge < fade ? edge / fade : 1;
