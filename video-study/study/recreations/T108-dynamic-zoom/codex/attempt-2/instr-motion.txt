function cursorPosAt(t){
  const P = CONFIG.cursor.path;
  if (t <= P[0].t) return {x:P[0].x, y:P[0].y};
  for (let i=0;i<P.length-1;i++){
    if (t < P[i+1].t){
      const q = smooth((t-P[i].t)/(P[i+1].t-P[i].t));
      return {x:P[i].x+(P[i+1].x-P[i].x)*q, y:P[i].y+(P[i+1].y-P[i].y)*q};
    }
  }
  return {x:P[P.length-1].x, y:P[P.length-1].y};
}
/* entry event: first cycle time the cursor hotspot (its path point, the glyph tip)
   lies inside the screen rectangle, both in world coordinates */
let entryT = null;
for (let t=0;t<=CONFIG.holdBeforeMs+CONFIG.zoomMs;t++){
  const p = cursorPosAt(t);
  if (p.x>=CONFIG.screenRect.x && p.x<=CONFIG.screenRect.x+CONFIG.screenRect.w && p.y>=CONFIG.screenRect.y && p.y<=CONFIG.screenRect.y+CONFIG.screenRect.h){ entryT = t; break; }
}
function pose(tms){
  const t = ((tms % CONFIG.cycleMs) + CONFIG.cycleMs) % CONFIG.cycleMs;
  const zoomStart = CONFIG.holdBeforeMs, zoomEnd = zoomStart + CONFIG.zoomMs;
  const arrivedEnd = zoomEnd + CONFIG.holdAfterMs;
  let raw = 0, cam = 0, phase;
  if (t < zoomStart){ raw = 0; cam = 0; phase = "distant"; }
  else if (t < zoomEnd){ raw = (t-zoomStart)/CONFIG.zoomMs; cam = EASES[CONFIG.ease](Math.min(raw/CONFIG.cameraFraction,1)); phase = "zoom"; }
  else if (t < arrivedEnd){ raw = 1; cam = 1; phase = "arrived"; }
  else { raw = 1; cam = 1; phase = "extended"; }
  /* settle: inside the zoom window, after the camera sub-span */
  let settleScale = 1, settleDy = 0;
  if (phase === "zoom" && raw > CONFIG.cameraFraction){
    const se = EASES[CONFIG.ease]((raw-CONFIG.cameraFraction)/(1-CONFIG.cameraFraction));
    settleScale = CONFIG.settle.fromScale + (1-CONFIG.settle.fromScale)*se;
    settleDy = CONFIG.settle.fromDy*(1-se);
  } else if (phase === "zoom"){ settleScale = CONFIG.settle.fromScale; settleDy = CONFIG.settle.fromDy; }
  const cp = cursorPosAt(t);
  const txTarget = cp.x + (screenCx-cp.x)*cam;
  const tyTarget = cp.y + (screenCy-cp.y)*cam;
  const s = CONFIG.startScale + (endScale - CONFIG.startScale)*cam;
  const tx = stageCx - txTarget*s;
  const ty = stageCy - tyTarget*s;
  const edge = Math.min(t, CONFIG.cycleMs - t);
  const opacity = edge < CONFIG.loopFadeMs ? edge/CONFIG.loopFadeMs : 1;
  let cursorOpacity;
  if (entryT === null || t < entryT) cursorOpacity = 1;
  else if (t >= entryT + CONFIG.cursor.fadeMs) cursorOpacity = 0;
  else cursorOpacity = 1 - (t-entryT)/CONFIG.cursor.fadeMs;
  return {s:s, tx:tx, ty:ty, u:cam, raw:raw, opacity:opacity, cursorOpacity:cursorOpacity,
          cursorX:cp.x, cursorY:cp.y, targetX:txTarget, targetY:tyTarget,
          settleScale:settleScale, settleDy:settleDy, phase:phase, entryT:entryT};
}
function render(tms){
  const p = pose(tms);
  world.style.transform = "translate("+p.tx+"px,"+p.ty+"px) scale("+p.s+")";
  world.style.opacity = String(p.opacity);
  world.dataset.phase = p.phase;
  cursor.style.opacity = String(p.cursorOpacity);
  cursor.style.left = p.cursorX+"px";
  cursor.style.top = p.cursorY+"px";
  const eff = fitScale * p.settleScale;
  content.style.transform = "scale("+eff+")";
  content.style.left = ((CONFIG.screenRect.w - CONFIG.screenRect.w*eff)/2)+"px";
  content.style.top = ((visTop + visibleH/2) - (CONFIG.screenRect.h*eff)/2 + p.settleDy)+"px";
}
window.__render = render;
window.__pose = pose;
window.__config = CONFIG;
window.__layout = {endScale:endScale, visibleW:visibleW, visibleH:visibleH, visTop:visTop, fitScale:fitScale, naturalH:naturalH, naturalW:naturalW, rowCount:rowCount, titleW:titleW, effPillW:effPillW, effRowW:effRowW, aspectValid:aspectValid, errors:validationErrors, phaseTable:phaseTable, entryT:entryT};
let start = null;
function frame(ts){
  if (start === null) start = ts;
  render(ts - start);
  requestAnimationFrame(frame);
}
requestAnimationFrame(frame);
