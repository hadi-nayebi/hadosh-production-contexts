function easeInOutCubic(u){ return u<0.5 ? 4*u*u*u : 1-Math.pow(-2*u+2,3)/2; }
const EASES = {"easeInOutCubic-single": easeInOutCubic};
function pose(tms){
  const t = ((tms % CONFIG.cycleMs) + CONFIG.cycleMs) % CONFIG.cycleMs;
  /* half-open phase intervals: [0,holdBefore) distant hold · [holdBefore,+zoom) zoom · [+zoom,+holdAfter) arrived hold · [+holdAfter,cycle) extended arrived hold (same end pose; the seam reset lives in the loop fade, opacity 0) */
  const zoomEnd = CONFIG.holdBeforeMs + CONFIG.zoomMs;
  const arrivedEnd = zoomEnd + CONFIG.holdAfterMs;
  let u;
  if (t < CONFIG.holdBeforeMs) u = 0;
  else if (t < zoomEnd) u = EASES[CONFIG.ease]((t - CONFIG.holdBeforeMs)/CONFIG.zoomMs);
  else if (t < arrivedEnd) u = 1;
  else u = 1;
  const s = CONFIG.startScale + (endScale - CONFIG.startScale)*u;
  const tx = stageCx - screenCx*s;
  const ty = stageCy - screenCy*s;
  const edge = Math.min(t, CONFIG.cycleMs - t);
  const opacity = edge < CONFIG.loopFadeMs ? edge/CONFIG.loopFadeMs : 1;
  let cursorOpacity;
  if (u <= CONFIG.cursor.fadeFrom) cursorOpacity = 1;
  else if (u >= CONFIG.cursor.fadeTo) cursorOpacity = 0;
  else cursorOpacity = 1 - (u - CONFIG.cursor.fadeFrom)/(CONFIG.cursor.fadeTo - CONFIG.cursor.fadeFrom);
  return {s:s, tx:tx, ty:ty, u:u, opacity:opacity, cursorOpacity:cursorOpacity};
}
function render(tms){
  const p = pose(tms);
  world.style.transform = "translate("+p.tx+"px,"+p.ty+"px) scale("+p.s+")";
  world.style.opacity = String(p.opacity);
  cursor.style.opacity = String(p.cursorOpacity);
}
window.__render = render;
window.__pose = pose;
window.__config = CONFIG;
window.__layout = {endScale:endScale, visibleW:visibleW, visibleH:visibleH, visTop:visTop, fitScale:fitScale, naturalH:naturalH};
let start = null;
function frame(ts){
  if (start === null) start = ts;
  render(ts - start);
  requestAnimationFrame(frame);
}
