/* Deterministic state reconstruction (CA-2): state is a pure function of phase.
   A frame that skips several checkpoints lands on exactly the state those
   checkpoints would have produced, in event-table order. */
function stateAt(c, tl, phase){
  const n = c.prs.length;
  let cardIdx = 0, tx = 0, blur = 0, streak = 0, whipping = false, whipProgress = 0;
  for (let i = 1; i < n; i++){
    const s = tl.starts[i], w0 = s - c.whipMs/2, w1 = s + c.whipMs/2;
    if (phase >= w0 && phase < w1){
      whipping = true; whipProgress = (phase - w0) / c.whipMs;
      const p = whipProgress;
      blur = c.whipBlurPx * Math.sin(Math.PI * p);
      streak = c.whipStreakPx * Math.sin(Math.PI * p);
      if (p < 0.5){ cardIdx = i - 1; tx = -c.whipDistancePx * (p * 2); }
      else { cardIdx = i; tx = c.whipDistancePx * (1 - (p - 0.5) * 2); }
    } else if (phase >= w1){ cardIdx = i; }
  }
  const cs = tl.counterStarts[cardIdx], cp = Math.max(0, Math.min(1, (phase - cs) / tl.counterSpanMs));
  const pr = c.prs[cardIdx];
  const approvals = Math.round(pr.approvalsFrom + (pr.approvalsTo - pr.approvalsFrom) * cp);
  const counterBlur = cp < c.counterBlurFraction ? c.counterBlurPx : 0;
  const checksIdx = phase >= tl.checksReadyAt ? c.mergeChecksIndex : cardIdx;
  let stampState = "idle", stampScale = c.stampStartScale, stampOpacity = 0;
  if (phase >= tl.stampAt && phase < tl.stampAt + c.slamMs){ stampState = "slamming"; const k = (phase - tl.stampAt) / c.slamMs; stampScale = c.stampStartScale + (1 - c.stampStartScale) * k; stampOpacity = 1; }
  else if (phase >= tl.stampAt + c.slamMs && phase < tl.stampHideAt){ stampState = "held"; stampScale = 1; stampOpacity = 1; }
  else if (phase >= tl.stampHideAt){ stampState = "hiding"; stampScale = 1; stampOpacity = Math.max(0, 1 - (phase - tl.stampHideAt) / c.stampHideMs); }
  return { cardIdx, tx, blur, streak, whipping, whipProgress, counterProgress: cp, approvals,
           counterBlur, checksIdx, stampState, stampScale, stampOpacity, opacity: 1 };
}

/* Absolute timeline scheduler (CA-2): t0 origin; pause stores the phase and
   resume shifts the origin by exactly the paused span; loop rollover is
   phase modulo, so a resumed tab reconstructs state from phase alone. */
let playing=false, t0=0, pausedPhase=0, rafId=null;
function frame(now){
  if(!playing) return;
  const phase = (now - t0) % CONFIG.loopMs;
  render(stateAt(CONFIG, TL, phase));
  rafId = requestAnimationFrame(frame);
}
function play(){ if(playing) return; playing=true; t0 = performance.now() - pausedPhase; rafId = requestAnimationFrame(frame); }
function pause(){ if(!playing) return; pausedPhase = (performance.now() - t0) % CONFIG.loopMs; playing=false; if(rafId) cancelAnimationFrame(rafId); rafId=null; }
function restart(){ playing=false; if(rafId) cancelAnimationFrame(rafId); pausedPhase=0; renderedCard=-1; renderedChecks=-1; play(); }
window.addEventListener("message", function(e){
  if(!e.data||typeof e.data!=="object") return;
  if(e.data.cmd==="play") play(); if(e.data.cmd==="pause") pause(); if(e.data.cmd==="restart") restart();
});
play();
