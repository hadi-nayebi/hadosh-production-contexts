  if (!inside || legGap) return { filled: false, cls: "", glyph: " " };
  const band = Math.sin(ny * A.bandFreq + x * A.bandPhase) > A.bandThreshold;
  const dark = (x * A.darkMulX + y * A.darkMulY) % A.darkMod < A.darkRem;
  if (band) return { filled: true, cls: "g-band", glyph: A.glyphBand };
  if (dark) return { filled: true, cls: "g-dark", glyph: A.glyphDark };
  return { filled: true, cls: "g-base", glyph: A.glyphBase };
}
function stateAt(c, tl, phase){
  const n = c.prs.length;
  let cardIdx = 0, tx = 0, blur = 0, streak = 0, skew = 0, whipping = false, whipProgress = 0;
  for (let i = 1; i < n; i++){
    const s = tl.starts[i], w0 = s - c.whipMs/2, w1 = s + c.whipMs/2;
    if (phase >= w0 && phase < w1){
      whipping = true; whipProgress = (phase - w0) / c.whipMs;
      const p = whipProgress;
      blur = c.whipBlurPx * Math.sin(Math.PI * p);
      streak = c.whipStreakPx * Math.sin(Math.PI * p);
      skew = -c.whipSkewDeg * Math.sin(Math.PI * p);
      if (p < 0.5){ cardIdx = i - 1; tx = -c.whipDistancePx * (p * 2); }
      else { cardIdx = i; tx = c.whipDistancePx * (1 - (p - 0.5) * 2); }
    } else if (phase >= w1){ cardIdx = i; }
  }
  const cs = tl.counterStarts[cardIdx], cp = Math.max(0, Math.min(1, (phase - cs) / tl.counterSpanMs));
  const pr = c.prs[cardIdx];
  const approvalsFloat = pr.approvalsFrom + (pr.approvalsTo - pr.approvalsFrom) * cp;
  const approvals = Math.round(approvalsFloat);
  const digits = digitsOf(c, approvals);
  const roll = rollPositions(c, approvalsFloat);
  let stampState = "idle", stampScale = c.stampStartScale, stampOpacity = 0, stamped = false;
  const st = tl.stamps.find(s=> s.cardIdx === cardIdx);
  if (st){
    if (phase >= st.at && phase < st.slamEnd){ stampState = "slamming"; const k = (phase - st.at) / c.slamMs; stampScale = c.stampStartScale + (1 - c.stampStartScale) * k; stampOpacity = 1; stamped = true; }
    else if (phase >= st.slamEnd && phase < st.holdEnd){ stampState = "held"; stampScale = 1; stampOpacity = 1; stamped = true; }
    else if (phase >= st.holdEnd && phase < st.hideEnd){ stampState = "hiding"; stampScale = 1; stampOpacity = Math.max(0, 1 - (phase - st.holdEnd) / c.stampHideMs); stamped = true; }
  }
  return { cardIdx, tx, blur, streak, skew, whipping, whipProgress, counterProgress: cp, approvals, approvalsFloat,
           digits, roll, checksIdx: cardIdx, stamped, stampState, stampScale, stampOpacity, opacity: 1 };
}

function frame(now){
  if(!playing) return;
  const phase = (now - t0) % CONFIG.loopMs;
  render(stateAt(CONFIG, TL, phase));
  rafId = requestAnimationFrame(frame);
}
function play(){ if(playing) return; playing=true; t0 = performance.now() - pausedPhase; rafId = requestAnimationFrame(frame); }
function pause(){ if(!playing) return; pausedPhase = (performance.now() - t0) % CONFIG.loopMs; playing=false; if(rafId) cancelAnimationFrame(rafId); rafId=null; }
function restart(){ playing=false; if(rafId) cancelAnimationFrame(rafId); pausedPhase=0; renderedCard=-1; renderedStamped=null; play(); }

  if(!e.data||typeof e.data!=="object") return;

});
