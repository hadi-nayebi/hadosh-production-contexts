function stateAt(t){
  t = ((t % CYCLE) + CYCLE) % CYCLE;
  const cutAt = CONFIG.demoTrigger === "on-count-complete-same-frame" ? CUT : CUT + CONFIG.label.stepMs;
  if (t < COUNT_START) return {scene:"out", step:-1, phase:"pre"};
  if (t < cutAt){
    let idx = Math.floor((t - COUNT_START) / CONFIG.label.stepMs);
    if (idx > CONFIG.label.steps.length - 1) idx = CONFIG.label.steps.length - 1;
    return {scene:"out", step:idx, phase:"count"};

function buildDemo(){

    hero.style.borderRadius = g.radius + "px";
    const tile = el(hero, g.tileX, g.tileY, g.tileW, g.tileW, p.dark);

      });
    });

  stage.appendChild(layer);
  return layer;

const layerOut = buildOutgoing();

function render(t){
  const s = stateAt(t);
  layerOut.style.display = s.scene === "out" ? "block" : "none";
  layerDemo.style.display = s.scene === "demo" ? "block" : "none";
  if (s.phase === "count"){
    band.layer.style.display = "block";
    band.text.textContent = labelText(s.step);
  } else {
    band.layer.style.display = "none";
  }
  layerSeam.style.opacity = String(seamOpacity(t));
}

window.__seamOpacity = seamOpacity;
window.__labelText = labelText;
window.__render = render;
render(0);
let startStamp = null;
function frame(ts){
  if (startStamp === null) startStamp = ts;
  render(ts - startStamp);
  requestAnimationFrame(frame);
}
