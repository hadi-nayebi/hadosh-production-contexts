const EASINGS = {
  easeOutCubic: function (x) { return 1 - Math.pow(1 - x, 3); },
  easeOutQuad: function (x) { return 1 - (1 - x) * (1 - x); },
  linear: function (x) { return x; }
};
function clamp01(x) { return x < 0 ? 0 : (x > 1 ? 1 : x); }
function staggerEff() {
  const n = CONFIG.triangles.length;
  if (n <= 1) return 0;
  return Math.min(CONFIG.staggerMs, Math.max(0, (CONFIG.assemblyMs - 1) / (n - 1)));
}
function pieceDuration() {
  return Math.max(1, CONFIG.assemblyMs - (CONFIG.triangles.length - 1) * staggerEff());
}
function resolveStart() { return CONFIG.assemblyMs; }
function holdStart() { return CONFIG.assemblyMs + CONFIG.wordmark.resolveMs; }
function fadeStart() { return holdStart() + CONFIG.holdMs; }
function fadeEnd() { return fadeStart() + CONFIG.loopFadeMs; }
function cycleEff() { return Math.max(CONFIG.cycleMs, fadeEnd()); }
function phaseAt(t) {
  t = ((t % cycleEff()) + cycleEff()) % cycleEff();
  if (t < resolveStart()) return "ASSEMBLE";
  if (t < holdStart()) return "RESOLVE";
  if (t < fadeStart()) return "HOLD";
  if (t < fadeEnd()) return "LOOP-FADE";
  return "REST";
}
function render(t) {
  t = ((t % cycleEff()) + cycleEff()) % cycleEff();
  const ease = EASINGS[CONFIG.ease] || EASINGS.linear;
  const dur = pieceDuration();
  const stag = staggerEff();
  CONFIG.triangles.forEach(function (tri, i) {
    const local = clamp01((t - i * stag) / dur);
    const e = ease(local);
    const dx = tri.from.x * (1 - e);
    const dy = tri.from.y * (1 - e);
    pieceEls[i].setAttribute("transform", "translate(" + dx + "," + dy + ")");
  });
  let fade = 0;
  if (t >= fadeEnd()) { fade = 1; }
  else if (t >= fadeStart()) { fade = clamp01((t - fadeStart()) / Math.max(1, CONFIG.loopFadeMs)); }
  let wop = 0;
  if (t >= holdStart()) { wop = 1; }
  else if (t >= resolveStart()) { wop = clamp01((t - resolveStart()) / Math.max(1, CONFIG.wordmark.resolveMs)); }
  wordEl.style.opacity = String(wop * (1 - fade));
  wordEl.style.transform = "translateY(" + (CONFIG.wordmark.risePx * (1 - wop)) + "px)";
  seamEl.setAttribute("opacity", String(fade));
}
