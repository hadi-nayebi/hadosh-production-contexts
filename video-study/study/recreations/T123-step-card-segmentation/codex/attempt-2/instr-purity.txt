function deepMerge(base, over){
  if (Array.isArray(over)){
    /* Array overrides are ATOMIC: an override array replaces the base
       array whole — no trailing base entries survive. */
    return JSON.parse(JSON.stringify(over));
  }
  if (over && typeof over === "object"){
    const out = Array.isArray(base) ? base.slice() : Object.assign({}, base);
    for (const k of Object.keys(over)) out[k] = deepMerge(base ? base[k] : undefined, over[k]);
    return out;
  }
  return over;
}
function clone(o){ return JSON.parse(JSON.stringify(o)); }
/* Validation: anchors finite, >= 0, < cycleMs; per-type anchor sequences

function easeOutCubic(p){ return 1 - Math.pow(1 - p, 3); }
function enterP(u, atMs, durationMs, cfg){
  if (u < atMs) return 0;
  const raw = Math.min(1, (u - atMs) / durationMs);
  const easing = (cfg && cfg.geom && cfg.geom.enterEasing) || "cubic-out";
  if (easing === "linear") return raw;
  return easeOutCubic(raw);
}
/* Split-anchor defaulting: a step that supplies only the legacy atMs derives
   both split anchors from it (shared-anchor default preserved). */
function stepAnchors(s){
  const legacy = (s && typeof s.atMs === "number") ? s.atMs : undefined;
  return {
    headlineAtMs: (s && typeof s.headlineAtMs === "number") ? s.headlineAtMs : legacy,
    tileAtMs: (s && typeof s.tileAtMs === "number") ? s.tileAtMs : legacy
  };
}
/* State at cycle time t. Exactly one active headline: the step with the
   GREATEST eligible headlineAtMs (not array order), or none before the
   first. Tiles are MONOTONIC within a cycle: tile i is present iff
   u >= steps[i].tileAtMs; only the cycle wrap (u returns to 0) clears
   them. Accumulation is a construction property. Headline and tile
   entrances use separate durations (headlineEnterMs / tileEnterMs);
   onset is the anchor itself, fully-visible is anchor + duration. */
function slotLeft(rank, count, cfg){
  const g = cfg.geom;
  return (cfg.stage.width - (count * g.tileW + Math.max(0, count - 1) * g.tileGap)) / 2 + rank * (g.tileW + g.tileGap);
}
/* Tile x positions at cycle time u. Slots are COMPUTED for the currently
   present set (centred row), never reflowed: when a tile joins, every
   already-present tile interpolates from its position at the join instant
   to its new slot over the joining tile's entrance window, eased by the
   same entrance progress. If another tile joins mid-shift, the new
   segment starts from the positions interpolated at that instant, so x
   stays continuous under overlapping entrances. A joining tile starts
   at its own new slot (its entrance is scale/rise/opacity). */
function tilePositions(u, cfg){
  const joins = cfg.steps.map(function(s, i){ return {i: i, at: stepAnchors(s).tileAtMs}; })
    .filter(function(j){ return typeof j.at === "number"; })
    .sort(function(a, b){ return a.at - b.at; });
  let segStart = null, fromPos = {}, targetPos = {}, presentList = [];
  for (const j of joins){
    if (u < j.at) break;
    const atPos = {};
    if (segStart !== null){
      const p = enterP(j.at, segStart, cfg.tileEnterMs, cfg);
      for (const k in targetPos) atPos[k] = fromPos[k] + (targetPos[k] - fromPos[k]) * p;
    }
    presentList = presentList.concat([j.i]).sort(function(a, b){ return a - b; });
    const newTarget = {};
    presentList.forEach(function(stepIdx, rank){ newTarget[stepIdx] = slotLeft(rank, presentList.length, cfg); });
    if (!(j.i in atPos)) atPos[j.i] = newTarget[j.i];
    fromPos = atPos; targetPos = newTarget; segStart = j.at;
  }
  if (segStart === null) return {};
  const p = enterP(u, segStart, cfg.tileEnterMs, cfg);
  const out = {};
  for (const k in targetPos) out[k] = fromPos[k] + (targetPos[k] - fromPos[k]) * p;
  return out;
}
function stateAt(t, cfg){
  const u = ((t % cfg.cycleMs) + cfg.cycleMs) % cfg.cycleMs;
  let active = -1, activeAt = -Infinity;
  cfg.steps.forEach(function(s, i){
    const a = stepAnchors(s).headlineAtMs;
    if (typeof a === "number" && u >= a && a > activeAt){ active = i; activeAt = a; }
  });
  const headlineP = active < 0 ? 0 : enterP(u, activeAt, cfg.headlineEnterMs, cfg);
  /* Crossfade: during the incoming headline's entrance window the outgoing
     headline (the predecessor by anchor time) stays rendered at opacity
     1 - headlineP. If a further headline arrives before the fade completes,
     the predecessor is the then-active headline and earlier layers drop —
     exactly one incoming + one outgoing layer ever render. Logical
     exclusivity (one ACTIVE headline) is unchanged. */
  let prevHeadline = -1, prevOpacity = 0;
  if (active >= 0 && u < activeAt + cfg.headlineEnterMs){
    let predAt = -Infinity;
    cfg.steps.forEach(function(s, i){
      const a = stepAnchors(s).headlineAtMs;
      if (i !== active && typeof a === "number" && a < activeAt && a > predAt){ prevHeadline = i; predAt = a; }
    });
    if (prevHeadline >= 0) prevOpacity = 1 - headlineP;
    else prevHeadline = -1;
  }
  const positions = tilePositions(u, cfg);
  const tiles = cfg.steps.map(function(s, i){
    const a = stepAnchors(s).tileAtMs;
    const p = enterP(u, a, cfg.tileEnterMs, cfg);
    return {index: i, present: u >= a, p: p, x: (i in positions) ? positions[i] : null,
      opacity: cfg.geom.enterFromOpacity + (1 - cfg.geom.enterFromOpacity) * p,
      scale: cfg.geom.enterFromScale + (1 - cfg.geom.enterFromScale) * p,
      risePx: cfg.geom.enterRisePx * (1 - p)};
  });
  return {u: u, active: active, headlineP: headlineP,
    prevHeadline: prevHeadline, prevOpacity: prevOpacity,
    tiles: tiles};
}
function gridCSS(cfg){
  const w = cfg.grid.linePx + "px";
  return "linear-gradient(" + cfg.palette.grid + " " + w + ", transparent " + w + ")," +
         "linear-gradient(90deg, " + cfg.palette.grid + " " + w + ", transparent " + w + ")";
}
/* Character width model (Arial advance widths, em fractions) for the fit harness. */
const CHAR_W = {upper: 0.68, lower: 0.52, space: 0.278, digit: 0.556, punct: 0.30};
function textWidthPx(text, fontPx, weight){
  let em = 0;
  for (const ch of text){
    if (ch === " ") em += CHAR_W.space;
    else if (ch >= "A" && ch <= "Z") em += CHAR_W.upper;
    else if (ch >= "0" && ch <= "9") em += CHAR_W.digit;
    else if (".,;:!?'’-—".indexOf(ch) >= 0) em += CHAR_W.punct;
    else em += CHAR_W.lower;
  }
  if (weight >= 700) em *= 1.06;
  return em * fontPx;
}
function wrapLines(text, fontPx, weight, maxW){
  const words = text.split(" ");
  const lines = []; let cur = "";
  for (const w of words){
    const trial = cur === "" ? w : cur + " " + w;
    if (textWidthPx(trial, fontPx, weight) <= maxW || cur === "") cur = trial;
    else { lines.push(cur); cur = w; }
  }
  if (cur !== "") lines.push(cur);
  return lines;
}
function headlineText(step){ return step.headline.map(function(s){ return s.t; }).join(""); }
function layoutModel(cfg){
  const g = cfg.geom, ty = cfg.type;
  const tileInnerW = g.tileW - 2 * g.tilePadX - 2 * g.tileBorderPx;
  const rows = cfg.steps.map(function(s, i){
    return {index: i,
      headlineW: textWidthPx(headlineText(s), ty.headlinePx, ty.headlineWeight),
      numeralW: textWidthPx(s.n, ty.numeralPx, ty.numeralWeight),
      titleW: textWidthPx(s.title, ty.titlePx, ty.titleWeight),
      subLines: wrapLines(s.sub, ty.subPx, ty.subWeight, tileInnerW),
      subLineW: wrapLines(s.sub, ty.subPx, ty.subWeight, tileInnerW).map(function(l){ return textWidthPx(l, ty.subPx, ty.subWeight); })};
  });
  const ink = 0.716; /* cap/ascender ratio — every subject line carries ascenders or caps */
  return {stageW: cfg.stage.width, stageH: cfg.stage.height, rows: rows, tileInnerW: tileInnerW,
    headlineInkH: ty.headlinePx * ink, numeralInkH: ty.numeralPx * ink,
    titleInkH: ty.titlePx * ink, subInkH: ty.subPx * ink,
    tilesTotalW: cfg.steps.length * g.tileW + Math.max(0, cfg.steps.length - 1) * g.tileGap,
    slotsFullSet: cfg.steps.map(function(s, i){ return slotLeft(i, cfg.steps.length, cfg); }),
    tileBottom: g.tileY + g.tileH, cycle: cfg.cycleMs};
}

let renderedHeadline = -2, renderedPrev = -2;
function renderHeadlineInto(cfg, el, stepIdx){
  el.innerHTML = "";

  const g = cfg.geom, pal = cfg.palette;
  cfg.steps[stepIdx].headline.forEach(function(span){

    sp.textContent = span.t;
    if (span.style === "blue"){ sp.style.color = pal.blue; }

      sp.style.padding = g.chipPadY + "px " + g.chipPadX + "px";
      sp.style.borderRadius = g.chipRadius + "px";
    }
    el.appendChild(sp);
  });
}
function renderHeadline(cfg, active){
  if (active === renderedHeadline) return;
  renderedHeadline = active;
  renderHeadlineInto(cfg, document.getElementById("headline"), active);
}
function renderPrev(cfg, prev){
  if (prev === renderedPrev) return;
  renderedPrev = prev;
  renderHeadlineInto(cfg, document.getElementById("headlinePrev"), prev);
}
function applyFrame(cfg, t){
  const st = stateAt(t, cfg);
  renderHeadline(cfg, st.active);
  renderPrev(cfg, st.prevHeadline);
  const hl = document.getElementById("headline");
  hl.style.opacity = String(cfg.geom.enterFromOpacity + (1 - cfg.geom.enterFromOpacity) * st.headlineP);

  hp.style.opacity = String(st.prevOpacity);
  const row = document.getElementById("tiles");
  st.tiles.forEach(function(tile, i){
    const el = row.children[i];
    el.style.display = tile.present ? "block" : "none";
    if (tile.x !== null && tile.x !== undefined) el.style.left = tile.x + "px";
    el.style.opacity = String(tile.opacity);
    el.style.transform = "scale(" + tile.scale + ") translateY(" + tile.risePx + "px)";
  });
  return st;
}
