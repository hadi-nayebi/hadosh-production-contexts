function deepMerge(base, over){
  if (Array.isArray(over)){
    if (!Array.isArray(base)) return over.slice();
    const merged = base.map(function(b, i){ return i < over.length ? deepMerge(b, over[i]) : b; });
    for (let i = base.length; i < over.length; i++) merged.push(over[i]);
    return merged;
  }
  if (over && typeof over === "object"){
    const out = Array.isArray(base) ? base.slice() : Object.assign({}, base);
    for (const k of Object.keys(over)) out[k] = deepMerge(base ? base[k] : undefined, over[k]);
    return out;
  }
  return over;
}
function clone(o){ return JSON.parse(JSON.stringify(o)); }

/* Character width model (Arial advance widths, em fractions) — the fit
   harness's measurement model; the strike endpoints derive from it. */
const CHAR_W = {upper: 0.68, lower: 0.52, space: 0.278, digit: 0.556, punct: 0.30};
function textWidthPx(text, fontPx, weight){
  let em = 0;
  for (const ch of text){
    if (ch === " ") em += CHAR_W.space;
    else if (ch >= "A" && ch <= "Z") em += CHAR_W.upper;
    else if (ch >= "0" && ch <= "9") em += CHAR_W.digit;
    else if (".,;:!?'’-—…".indexOf(ch) >= 0) em += CHAR_W.punct;
    else em += CHAR_W.lower;
  }
  if (weight >= 700) em *= 1.06;
  return em * fontPx;
}

/* ONE elapsed-time schedule derives every state (phase exclusivity):
   headline: hidden → entering → shown → flipped at flipAtMs.
   per card: waiting → entering (enterMs) → landed → striking
   (strike starts strikeDelayMs after the card's atMs; enterMs <
   strikeDelayMs by construction, so the strike never starts before
   the landing completes) → struck; at flipAtMs the list fades out
   over flipFadeMs (the measured flip clears the list). */
function stateAt(t, cfg){
  const u = ((t % cfg.cycleMs) + cfg.cycleMs) % cfg.cycleMs;
  const flipped = u >= cfg.flipAtMs;
  let headline;
  if (u < cfg.headlineAtMs) headline = {mode: "hidden", opacity: 0, risePx: cfg.layout.enterRisePx, flipped: false};
  else if (u < cfg.headlineAtMs + cfg.enterMs){
    const p = easeOutCubic((u - cfg.headlineAtMs) / cfg.enterMs);
    headline = {mode: "entering", opacity: p, risePx: cfg.layout.enterRisePx * (1 - p), flipped: false};
  } else headline = {mode: flipped ? "flipped" : "shown", opacity: 1, risePx: 0, flipped: flipped};
  const listFade = flipped ? 1 - clamp01((u - cfg.flipAtMs) / cfg.flipFadeMs) : 1;
  const cards = cfg.tasks.map(function(task, i){
    const at = cfg.atMs[i];
    const strikeStart = at + cfg.strikeDelayMs;
    let mode, opacity, risePx;
    if (u < at){ mode = "waiting"; opacity = 0; risePx = cfg.layout.enterRisePx; }
    else if (u < at + cfg.enterMs){
      const p = easeOutCubic((u - at) / cfg.enterMs);
      mode = "entering"; opacity = p; risePx = cfg.layout.enterRisePx * (1 - p);
    } else if (u < strikeStart){ mode = "landed"; opacity = 1; risePx = 0; }
    else if (u < strikeStart + cfg.strikeMs){ mode = "striking"; opacity = 1; risePx = 0; }
    else { mode = "struck"; opacity = 1; risePx = 0; }
    const strikeP = clamp01((u - strikeStart) / cfg.strikeMs);
    return {mode: mode, opacity: opacity * listFade, risePx: risePx, strikeP: strikeP};
  });
  return {u: u, headline: headline, cards: cards, listFade: listFade};
}

    const strike = document.createElement("div");
    strike.className = "strike";
    strike.style.left = (-cfg.strike.overshootPx) + "px";
    strike.style.top = (cfg.type.cardPx * cfg.type.lineHFrac * cfg.strike.yFracOfLine - cfg.strike.thicknessPx / 2) + "px";
    strike.style.height = cfg.strike.thicknessPx + "px";
    strike.style.width = "0px";
    strike.style.background = cfg.strikeColor;
    strike.style.borderRadius = (cfg.strike.thicknessPx / 2) + "px";
    wrap.appendChild(strike);
    card.appendChild(wrap);
    list.appendChild(card);
    cardEls.push({card: card, strike: strike, fullW: lay.strikeFullW});
  });
  return {layout: L, cardEls: cardEls, headlineFlipped: null};
}
function applyFrame(cfg, view, t){
  const st = stateAt(t, cfg);
  const headline = document.getElementById("headline");
  if (view.headlineFlipped !== st.headline.flipped){
    renderSpans(headline, st.headline.flipped ? cfg.flipHeadline : cfg.headline, cfg);
    view.headlineFlipped = st.headline.flipped;
  }
  headline.style.opacity = String(st.headline.opacity);
  headline.style.transform = "translateY(" + st.headline.risePx + "px)";
  view.cardEls.forEach(function(ce, i){
    const cs = st.cards[i];
    ce.card.style.opacity = String(cs.opacity);
    ce.card.style.transform = "translateY(" + cs.risePx + "px)";
    ce.card.style.visibility = cs.mode === "waiting" ? "hidden" : "visible";
    ce.strike.style.width = (cs.strikeP * ce.fullW) + "px";
  });
  return st;
}
