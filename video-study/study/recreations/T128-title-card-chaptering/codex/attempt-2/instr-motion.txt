function measureText(text, px, trackingPx, mono){
  let adv = 0;
  for(const ch of String(text)){
    if(mono) adv += 0.60;
    else if(ch===" ") adv += 0.30;
    else if(ch==="."||ch===","||ch==="-"||ch==="…") adv += 0.28;
    else if(ch===ch.toUpperCase() && ch!==ch.toLowerCase()) adv += 0.62;
    else adv += 0.52;
  }
  return adv*px + trackingPx*([...String(text)].length);
}

function lum(hex){
  const c = [1,3,5].map(i=>parseInt(hex.slice(i,i+2),16)/255).map(v=> v<=0.03928 ? v/12.92 : Math.pow((v+0.055)/1.055, 2.4));

/* Phase machine — exactly one phase per t, pure function of cycle-relative t. */
function stateAt(cfg, t){
  const cyc = cycleMs(cfg); const tt = ((t % cyc) + cyc) % cyc;
  const ex = cfg.exhibit;
  const exitEnd = ex.exitAtMs + ex.exitMs;
  const ebC = eyebrowCompleteMs(cfg), mC = mainCompleteMs(cfg);
  const holdEnd = mC + cfg.holdMs;
  const exhibitP = clamp01((tt - ex.exitAtMs)/ex.exitMs);
  const exhibitO = tt < exitEnd ? 1 - exhibitP : 0;
  let mapO = 0;
  if (tt >= ex.exitAtMs && tt < exitEnd) mapO = exhibitP;
  else if (tt >= exitEnd) mapO = 1 - (1 - cfg.map.fadeTo)*clamp01((tt - exitEnd)/(cfg.mainAtMs - exitEnd));
  const eyebrowChars = charsAt(cfg.eyebrowAtMs, cfg.eyebrowCharMs, glyphCount(cfg.eyebrow), tt);
  const strikeP = clamp01((tt - cfg.strikeAtMs)/cfg.strikeMs);
  const mainChars = charsAt(cfg.mainAtMs, cfg.mainCharMs, glyphCount(cfg.mainText), tt);
  const exitP = tt < holdEnd ? 0 : clamp01((tt - holdEnd)/cfg.exitMs);
  const darken = tt < exitEnd ? 0 : (1 - mapO)/(1 - cfg.map.fadeTo);
  const phase = tt < ex.exitAtMs ? "EXHIBIT" : (tt < exitEnd ? "EXIT" : (tt < cfg.eyebrowAtMs ? "MAP" :
    (tt < ebC ? "EYEBROW-TYPE" : (tt < cfg.strikeAtMs ? "EYEBROW-DWELL" : (tt < cfg.strikeAtMs + cfg.strikeMs ? "STRIKE" :
    (tt < cfg.mainAtMs ? "STRUCK-HOLD" : (tt < mC ? "MAIN-TYPE" : (tt < holdEnd ? "HOLD" : "EXIT-OUT"))))))));
  return { t: tt, phase, exhibitO, exhibitP, mapO, darken, eyebrowChars, strikeP, mainChars, exitP,
           cardO: 1 - exitP, strikeO: (1 - Math.min(1, exitP*cfg.exitStrikeLead)) * (strikeP>0?1:0) };
}

  function check(kind, v, path){
    if (kind==="num" && !num(v)) errors.push(path+": not a finite number");
    if (kind==="pos" && !(num(v) && v>0)) errors.push(path+": not a positive finite number");
    if (kind==="ms" && !(num(v) && v>=0)) errors.push(path+": not a non-negative finite number");
    if (kind==="unit" && !(num(v) && v>=0 && v<=1)) errors.push(path+": not within 0..1");
    if (kind==="str" && !(typeof v==="string" && v.length>0)) errors.push(path+": not a non-empty string");
    if (kind==="text" && !(typeof v==="string" && v.trim().length>0)) errors.push(path+": empty text");
    if (kind==="hex" && !hex(v)) errors.push(path+": not a #rrggbb colour");
    if (kind==="bool" && typeof v!=="boolean") errors.push(path+": not a boolean");
    if (kind==="arrUnit" && !(Array.isArray(v) && v.length>0 && v.every(x=>num(x) && x>=0 && x<=1))) errors.push(path+": not an array of numbers within 0..1");
    if (kind==="arrStr" && !(Array.isArray(v) && v.length>0 && v.every(x=>typeof x==="string" && x.length>0))) errors.push(path+": not an array of non-empty strings");
  }
  for (const k of Object.keys(o)){
    if (Object.prototype.hasOwnProperty.call(groups, k)){
      if (o[k]===null || typeof o[k]!=="object" || Array.isArray(o[k])){ errors.push(k+": group is not an object"); continue; }
      for (const sub of Object.keys(o[k])){
        if (!Object.prototype.hasOwnProperty.call(groups[k], sub)){ errors.push(k+"."+sub+": unknown key"); continue; }
        check(groups[k][sub], o[k][sub], k+"."+sub); cfg[k][sub] = o[k][sub];
      }
    } else if (Object.prototype.hasOwnProperty.call(top, k)){ check(top[k], o[k], k); cfg[k] = o[k]; }
    else errors.push(k+": unknown key");
  }
  if (!errors.length){
    const seq = [cfg.exhibit.atMs, cfg.exhibit.exitAtMs, cfg.eyebrowAtMs, cfg.strikeAtMs, cfg.mainAtMs, mainCompleteMs(cfg), cycleMs(cfg)];
    for (let i=1;i<seq.length;i++) if (!(seq[i-1] < seq[i])) errors.push("ordering: schedule is not strictly increasing at position "+i);

  }
  return errors.length ? { ok: false, errors, cfg: base } : { ok: true, errors: [], cfg };
}
