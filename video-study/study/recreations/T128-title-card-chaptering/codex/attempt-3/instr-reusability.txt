function stateAt(cfg, t){
  const cyc = cycleMs(cfg); const tt = ((t % cyc) + cyc) % cyc;
  const ex = cfg.exhibit;
  const exitEnd = ex.exitAtMs + ex.exitMs;
  const ebC = eyebrowCompleteMs(cfg), mC = mainCompleteMs(cfg);
  const holdEnd = mC + cfg.holdMs;
  const exhibitP = clamp01((tt - ex.exitAtMs)/ex.exitMs);
  const exhibitO = tt < exitEnd ? 1 - exhibitP : 0;
  const gate = clamp01((tt - exitEnd)/cfg.map.revealMs);
  const fadeP = clamp01((tt - exitEnd)/(cfg.mainAtMs - exitEnd));
  const mapFade = tt < exitEnd ? 0 : 1 - (1 - cfg.map.fadeTo)*fadeP;
  const mapO = gate * mapFade;
  const eyebrowChars = charsAt(cfg.eyebrowAtMs, cfg.eyebrowCharMs, glyphCount(cfg.eyebrow), tt);
  const strikeP = clamp01((tt - cfg.strikeAtMs)/cfg.strikeMs);
  const mainChars = charsAt(cfg.mainAtMs, cfg.mainCharMs, glyphCount(cfg.mainText), tt);
  const exitP = tt < holdEnd ? 0 : clamp01((tt - holdEnd)/cfg.exitMs);
  const darken = tt < exitEnd ? 0 : (1 - mapFade)/(1 - cfg.map.fadeTo);
  const phase = tt < ex.exitAtMs ? "EXHIBIT" : (tt < exitEnd ? "EXIT" : (tt < cfg.eyebrowAtMs ? "MAP" :
    (tt < ebC ? "EYEBROW-TYPE" : (tt < cfg.strikeAtMs ? "EYEBROW-DWELL" : (tt < cfg.strikeAtMs + cfg.strikeMs ? "STRIKE" :
    (tt < cfg.mainAtMs ? "STRUCK-HOLD" : (tt < mC ? "MAIN-TYPE" : (tt < holdEnd ? "HOLD" : "EXIT-OUT"))))))));
  return { t: tt, phase, exhibitO, exhibitP, mapO, darken, eyebrowChars, strikeP, mainChars, exitP, gate, mapFade,
           labelO: gate * cfg.map.labelO, vignette: gate * cfg.map.vignetteStrength,
           contourO: cfg.map.contourO.map((o)=> gate * o * mapFade),
           cardO: 1 - exitP, strikeO: (1 - Math.min(1, exitP*cfg.exitStrikeLead)) * (strikeP>0?1:0) };
}

function validateOverride(base, o){
  const errors = [];
  if (o === null || typeof o !== "object" || Array.isArray(o)) return { ok: false, errors: ["override is not an object"], cfg: base };
  const cfg = deepClone(base);

        kindErrors(SCHEMA.groups[k][sub], o[k][sub], k+"."+sub, errors); cfg[k][sub] = o[k][sub];
      }
    } else if (Object.prototype.hasOwnProperty.call(SCHEMA.top, k)){ kindErrors(SCHEMA.top[k], o[k], k, errors); cfg[k] = o[k]; }
    else errors.push(k+": unknown key");
  }

}
