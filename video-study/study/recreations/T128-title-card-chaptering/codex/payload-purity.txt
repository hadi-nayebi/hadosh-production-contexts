/* ?cfg= JSON override (swap-test route) */
function applyOverride(cfg){
  try{
    const q = new URLSearchParams(location.search).get("cfg");
    if(q){ const o = JSON.parse(q);
      for(const k of Object.keys(o)){ if(k==="palette"||k==="type"||k==="geom"||k==="exhibit"||k==="stage"){ Object.assign(cfg[k], o[k]); } else { cfg[k]=o[k]; } } }
  }catch(e){}
  return cfg;
}
if (typeof location !== "undefined") applyOverride(CONFIG);

function cycleMs(cfg){ return cfg.mainAtMs + (cfg.mainWords.length-1)*cfg.wordMs + cfg.holdMs; }

/* Deterministic text-width model (Georgia caps advance ~0.62em, mixed ~0.52em; LOGIC level). */
function measureText(text, px, trackingPx){
  let adv = 0;
  for(const ch of text){ adv += (ch===" " ? 0.30 : (ch==="…" ? 0.78 : (ch===ch.toUpperCase() && ch!==ch.toLowerCase() ? 0.62 : 0.52))); }
  return adv*px + trackingPx*(text.length);
}
function fitRows(cfg){
  const ew = measureText(cfg.eyebrow, cfg.type.eyebrowPx, cfg.type.eyebrowTrackingPx);
  let mw = 0; cfg.mainWords.forEach((w,i)=>{ mw += measureText(w, cfg.type.mainPx, 0) + (i? cfg.type.mainGapPx:0); });
  const lw = measureText(cfg.exhibit.label, cfg.exhibit.labelPx, 0);
  const sw = measureText(cfg.exhibit.sub, cfg.exhibit.subPx, 0);
  return {
    eyebrow: { text: cfg.eyebrow, inkW: ew, inkH: cfg.type.eyebrowPx*0.72, boxW: cfg.stage.w },
    main: { text: cfg.mainWords.join(" "), inkW: mw, inkH: cfg.type.mainPx*0.72, boxW: cfg.stage.w },
    exhibitLabel: { text: cfg.exhibit.label, inkW: lw, inkH: cfg.exhibit.labelPx*0.72, boxW: cfg.exhibit.wPx - 2*cfg.exhibit.padPx },
    exhibitSub: { text: cfg.exhibit.sub, inkW: sw, inkH: cfg.exhibit.subPx*0.72, boxW: cfg.exhibit.wPx - 2*cfg.exhibit.padPx }
  };
}
/* Phase machine — exclusive states, pure function of t (ms, cycle-relative). */
function stateAt(cfg, t){
  const cyc = cycleMs(cfg); const tt = ((t % cyc) + cyc) % cyc;
  const ex = cfg.exhibit;
  let exhibitO = 0;
  if (tt >= ex.atMs && tt < ex.untilMs - ex.fadeMs) exhibitO = 1;
  else if (tt >= ex.untilMs - ex.fadeMs && tt < ex.untilMs) exhibitO = 1 - (tt-(ex.untilMs-ex.fadeMs))/ex.fadeMs;
  const darken = tt < ex.untilMs ? 0 : Math.min(1, (tt-ex.untilMs)/cfg.darkenMs);
  const eyebrowO = tt < cfg.eyebrowAtMs ? 0 : Math.min(1, (tt-cfg.eyebrowAtMs)/cfg.eyebrowFadeMs);
  const words = cfg.mainWords.map((w,i)=>{
    const at = cfg.mainAtMs + i*cfg.wordMs;
    return tt < at ? 0 : Math.min(1, (tt-at)/cfg.wordFadeMs);
  });
  const phase = tt < ex.untilMs ? "EXHIBIT" : (tt < cfg.eyebrowAtMs ? "DARKEN" : (tt < cfg.mainAtMs ? "EYEBROW" : (words.some(o=>o<1) ? "MAIN" : "HOLD")));
  return { t: tt, phase, exhibitO, darken, eyebrowO, words };
}
