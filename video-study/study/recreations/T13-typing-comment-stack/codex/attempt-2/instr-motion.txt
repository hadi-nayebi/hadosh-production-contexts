function segmentGraphemes(text){
  if (typeof Intl !== "undefined" && typeof Intl.Segmenter === "function") {
    const seg = new Intl.Segmenter(undefined, { granularity: "grapheme" });
    return Array.from(seg.segment(text), function(s){ return s.segment; });
  }
  return Array.from(text);
}
function graphemeRangeOf(haystack, needle){
  if (!needle) return null;
  const h = segmentGraphemes(haystack), n = segmentGraphemes(needle);
  for (let i = 0; i + n.length <= h.length; i++) {
    let ok = true;
    for (let j = 0; j < n.length; j++) if (h[i + j] !== n[j]) { ok = false; break; }
    if (ok) return { start: i, end: i + n.length };
  }
  return null;
}

const GLYPH_FACTOR = 0.58; /* conservative worst-case glyph width, in em */
function cqwOf(token){ return parseFloat(token); }
function modelFor(mode, stageWidthPx){
  const L = CONFIG.layout[mode];
  const cqwPx = stageWidthPx / 100;
  const titleFontPx = cqwOf(L.titleFont) * cqwPx;
  const commentsWidthCqw = mode === "wide" ? (100 - cqwOf(L.commentsLeft) - cqwOf(L.commentsRight)) : 94;
  const innerCqw = commentsWidthCqw - 2 * cqwOf(L.cardPadX) - L.cardBorderCqw;
  const charsPerLine = Math.max(1, Math.floor(innerCqw / (cqwOf(L.titleFont) * GLYPH_FACTOR)));
  function linesFor(text){ return Math.max(1, Math.ceil(segmentGraphemes(text).length / charsPerLine)); }
  function cardHeightCqw(text){
    return 2 * cqwOf(L.cardPadY) + cqwOf(L.avatarSize) + cqwOf(L.titleMargin)
      + linesFor(text) * cqwOf(L.titleFont) * 1.25 + cqwOf(L.footMargin) + cqwOf(L.cardFootFont) * 1.2;
  }
  const headCqw = cqwOf(L.commentsHeadFont) * 1.2 + cqwOf(L.commentsHeadMargin);
  let stackCqw = headCqw;
  CONFIG.cards.forEach(function(c, i){ stackCqw += cardHeightCqw(c.text) + (i ? CONFIG.layout[mode].stackGapCqw : 0); });
  const stageHeightCqw = mode === "wide"
    ? 100 * CONFIG.wideStageAspect[1] / CONFIG.wideStageAspect[0]

    footFontPx: cqwOf(L.cardFootFont) * cqwPx, charsPerLine: charsPerLine,
    linesFor: linesFor, stackCqw: stackCqw, availCqw: availCqw };
}
function validateConfig(){
  const errors = OVERRIDE_WARNINGS.slice();

  if (CONFIG.beats.length < CONFIG.cards.length) errors.push("T13_BEAT_TABLE_MISMATCH");
  CONFIG.beats.forEach(function(b){ if (!(b.charMs > 0) || !(b.appearMs >= 0) || !(b.holdAfterMs >= 0)) errors.push("T13_BEAT_INVALID"); });

  if (CONFIG.cards.length > 1) {
    const g = segmentGraphemes(CONFIG.cards[1].text), p = segmentGraphemes(prefix);
    let ok = p.length <= g.length;
    for (let i = 0; ok && i < p.length; i++) if (g[i] !== p[i]) ok = false;

    CONFIG.cards.forEach(function(c){ if (m.linesFor(c.text) > CONFIG.maxTitleLines) errors.push("T13_TITLE_UNSUPPORTED:" + mode); });
    if (m.stackCqw > m.availCqw) errors.push("T13_STACK_OVERFLOW:" + mode);
  });

}

/* --- Sequential schedule derivation (strict sequencing by construction) --- */
function buildSchedule(){
  const sched = [];
  let t = 0;
  CONFIG.cards.forEach(function(c, i){
    const b = CONFIG.beats[i];
    const n = segmentGraphemes(c.text).length;
    if (i === 0 && CONFIG.opening.firstCardLanded) {
      sched.push({ appearStart: -CONFIG.opening.offsetMs, typeStart: -CONFIG.opening.offsetMs, typeEnd: -CONFIG.opening.offsetMs, count: n, preLanded: true });
      return;
    }
    if (i === 1 && CONFIG.opening.secondCardPrefix) {
      const pre = segmentGraphemes(CONFIG.opening.secondCardPrefix).length;
      const typeEnd = (n - pre) * b.charMs;
      sched.push({ appearStart: -CONFIG.opening.offsetMs, typeStart: -pre * b.charMs, typeEnd: typeEnd, count: n, preTyped: pre });
      t = typeEnd + b.holdAfterMs;
      return;
    }
    const appearStart = t;
    const typeStart = appearStart + b.appearMs;
    const typeEnd = typeStart + n * b.charMs;
    sched.push({ appearStart: appearStart, typeStart: typeStart, typeEnd: typeEnd, count: n });
    t = typeEnd + b.holdAfterMs;
  });
  const lastEnd = sched.length ? sched[sched.length - 1].typeEnd : 0;
  return { cards: sched, endMs: lastEnd + CONFIG.finalHoldMs };
}

    else typed = 0;
    const landed = t >= s.typeEnd;
