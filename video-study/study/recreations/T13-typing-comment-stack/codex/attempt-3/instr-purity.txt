function segmentClustersFallback(text){

function buildScheduleFor(cards, beats, opening, finalHoldMs){
  const sched = [];
  let t = 0;
  const openingApplies = cards.length >= 2 && opening.firstCardLanded && opening.secondCardPrefix;
