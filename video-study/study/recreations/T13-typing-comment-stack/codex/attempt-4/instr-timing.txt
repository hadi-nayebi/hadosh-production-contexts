function segmentClustersFallback(text){

function graphemeRangeOf(haystack, needle){

}

    _ctx.font = fontStr;
    w = _ctx.measureText(text).width;
  } else {
    w = estimateWidthPx(text, sizePx);
  }
  const g = segmentGraphemes(text).length;
  if (letterSpacingPx && g > 1) w += letterSpacingPx * (g - 1);
  return Math.ceil(w * 100) / 100;
}
/* Wrap simulation: CSS white-space normal collapsing + overflow-wrap
   break-word long-word breaking. Mark padding charged per occupied fragment. */
function fitLines(text, availPx, weight, sizePx, markRange, markPadPx){
  const norm = String(text).replace(/\s+/g, " ").replace(/^ | $/g, "");
  const words = norm === "" ? [] : norm.split(" ");
  const graphemes = segmentGraphemes(norm);
  let gi = 0;
  const wordInfos = words.map(function(wd){
    const g = segmentGraphemes(wd);
    const info = { text: wd, width: measureWidthPx(wd, weight, sizePx, 0), start: gi, end: gi + g.length, gcount: g.length };
    gi += g.length + 1;
    return info;
  });
  const spaceW = measureWidthPx(" ", weight, sizePx, 0);
  const effAvail = availPx - 1;
  let lines = 1, curW = 0, curMark = false;
  const lineWidths = [];
  function markOn(a, b){ return !!(markRange && b > markRange.start && a < markRange.end); }
  function markCharge(){ return curMark ? 2 * markPadPx : 0; }
  function closeLine(){ lineWidths.push(curW + markCharge() + 1); lines = lineWidths.length; curW = 0; curMark = false; }
  wordInfos.forEach(function(wd, wi){
    const wdMark = markOn(wd.start, wd.end);
    const sep = (curW > 0) ? spaceW : 0;
    const prospective = curW + sep + wd.width + ((curMark || wdMark) ? 2 * markPadPx : 0) - markCharge();
    if (curW > 0 && prospective > effAvail && wd.width + (wdMark ? 2 * markPadPx : 0) <= effAvail) { closeLine(); }
    if (wd.width > effAvail) {
      if (curW > 0) closeLine();
      const gs = segmentGraphemes(wd.text);
      gs.forEach(function(gch, k){
        const gw = measureWidthPx(gch, weight, sizePx, 0);
        const gMark = markOn(wd.start + k, wd.start + k + 1);
        if (curW > 0 && curW + gw + ((curMark || gMark) ? 2 * markPadPx : 0) - markCharge() > effAvail) closeLine();
        curW += gw;
        if (gMark) curMark = true;
      });
    } else {
      curW += ((curW > 0) ? spaceW : 0) + wd.width;
      if (wdMark) curMark = true;
    }
    if (wi === wordInfos.length - 1) lineWidths.push(curW + markCharge() + 1);
  });
  if (!wordInfos.length) lineWidths.push(0);
  const maxW = Math.max.apply(null, lineWidths);
  const lastW = lineWidths.length ? lineWidths[lineWidths.length - 1] - 1 : 0;
  const uncertain = (FIT_PATH === "estimate" || FONT_UNCERTAIN) && maxW > 0.9 * availPx;
  return { lines: lineWidths.length, maxWidthPx: maxW, lastLineWidthPx: lastW, uncertain: uncertain, path: FIT_PATH };
}
function normalizeWs(s){ return String(s).replace(/\s+/g, " ").replace(/^ | $/g, ""); }
function markRangeFor(text, highlight){
  if (!highlight) return null;
  return graphemeRangeOf(normalizeWs(text), normalizeWs(highlight));
}
function parseUnitToken(token){
  if (typeof token === "number") return { value: token, unit: "cqw" };
  const parts = String(token).trim().split(/\s+/).map(function(p){
    const mm = p.match(/^(-?[\d.]+)([a-z%]*)$/);
    return mm ? { value: parseFloat(mm[1]), unit: mm[2] || "" } : { value: 0, unit: "" };
  });
  return parts.length === 1 ? parts[0] : { components: parts };
}
function lenPx(token, cqwPx, refPx){
  const t = parseUnitToken(token);
  const one = function(c){
    if (c.unit === "cqw") return c.value * cqwPx;
    if (c.unit === "px" || c.unit === "") return c.value;
    if (c.unit === "%") return c.value * (refPx || 0) / 100;
    return c.value;
  };
  return t.components ? one(t.components[0]) : one(t);
}
function cqwOf(token){ const t = parseUnitToken(token); return t.components ? t.components[0].value : t.value; }
function scaledToken(token, factor){
  const t = parseUnitToken(token);
  if (t.components) return t.components.map(function(c){ return (c.value * factor) + c.unit; }).join(" ");
  return (t.value * factor) + t.unit;
}

    let pLines = 0, pWorst = 0, pUnc = false, pOverflow = false;
    for (let k = 0; k <= gs.length; k++) {
      const pre = gs.slice(0, k).join("");
      const pmr = mr ? { start: mr.start, end: Math.min(mr.end, k) } : null;
      const pf = fitLines(pre, innerPx, L.titleWeight, titleFontPx, (pmr && pmr.end > pmr.start) ? pmr : null, markPadPx);
      if (pf.lines > pLines) pLines = pf.lines;

      if (caretRight > innerPx) {
        if (pf.lines >= CONFIG.maxTitleLines) pOverflow = true;
        if (caretRight > pWorst) pWorst = caretRight;
      } else {
        if (caretRight > pWorst) pWorst = caretRight;
        if ((FIT_PATH === "estimate" || FONT_UNCERTAIN) && pf.lines >= CONFIG.maxTitleLines && caretRight > 0.9 * innerPx) pUnc = true;
      }
    }
    return { lines: full.lines, maxWidthPx: full.maxWidthPx, uncertain: full.uncertain, path: full.path,

  const postInnerPx = postWidthPx - 2 * lenPx(L.postPadX, cqwPx, ref);

  const postMark = markRangeFor(postFull, CONFIG.post.bodyHighlight);
  const postBodyFit = fitLines(postFull, postInnerPx, L.bodyWeight, lenPx(L.postBodyFont, cqwPx, ref), postMark, markPadPx);

function validateConfig(){
  const errors = [];
  if (!CONFIG || !Array.isArray(CONFIG.cards) || !Array.isArray(CONFIG.beats) || !CONFIG.layout || !CONFIG.layout.wide || !CONFIG.layout.narrow || !CONFIG.palette || !CONFIG.post) {
    errors.push("T13_CONFIG_INVALID");
    return errors;
  }
  if (CONFIG.cards.length > CONFIG.maxCards) errors.push("T13_CARD_COUNT_UNSUPPORTED");
  if (CONFIG.beats.length < CONFIG.cards.length) errors.push("T13_BEAT_TABLE_MISMATCH");
  CONFIG.beats.forEach(function(b){ if (!b || !(b.charMs > 0) || !isFinite(b.charMs) || !(b.appearMs >= 0) || !isFinite(b.appearMs) || !(b.holdAfterMs >= 0) || !isFinite(b.holdAfterMs)) errors.push("T13_BEAT_INVALID"); });
  ["landMs", "caretBlinkMs", "finalHoldMs"].forEach(function(k){
    const v = CONFIG[k];
    if (!(typeof v === "number" && isFinite(v) && v > 0 && v <= CONFIG.catalogueWindowMs)) errors.push("T13_DURATION_INVALID:" + k);
  });
  CONFIG.beats.forEach(function(b, i){
    if (b && i < CONFIG.cards.length - 1 && isFinite(b.holdAfterMs) && b.holdAfterMs < CONFIG.landMs) errors.push("T13_HOLD_LT_LAND:" + i);
  });
  if (CONFIG.cards.length >= 2 && CONFIG.opening.firstCardLanded) {
    const g = segmentGraphemes(CONFIG.cards[1].text), p = segmentGraphemes(CONFIG.opening.secondCardPrefix);
    let ok = p.length <= g.length;
    for (let i = 0; ok && i < p.length; i++) if (g[i] !== p[i]) ok = false;
    if (!ok) errors.push("T13_PREFIX_MISMATCH");
  }
  ["wide", "narrow"].forEach(function(mode){
    const m = modelFor(mode, mode === "wide" ? CONFIG.wideStageAspect[0] : CONFIG.narrowStageAspect[0]);
    m.titles.forEach(function(t){
      if (t.lines > CONFIG.maxTitleLines) errors.push("T13_TITLE_UNSUPPORTED:" + mode);
      else if (t.uncertain) errors.push("T13_FIT_UNCERTAIN:" + mode);
      if (t.maxWidthPx > m.innerPx) errors.push("T13_TITLE_TOO_WIDE:" + mode);
      if (t.prefixLines > CONFIG.maxTitleLines) errors.push("T13_TITLE_UNSUPPORTED:caret:" + mode);
      if (t.prefixOverflow) errors.push("T13_TITLE_TOO_WIDE:caret:" + mode);
      else if (t.prefixUncertain) errors.push("T13_FIT_UNCERTAIN:caret:" + mode);
    });
    m.headers.forEach(function(h){
      if (!h.fits) errors.push("T13_CHROME_OVERFLOW:" + mode);
      else if (h.uncertain) errors.push("T13_FIT_UNCERTAIN:chrome:" + mode);
    });
    ["postHead", "commentsHead", "cardFoot", "postFoot", "stamp"].forEach(function(s){
      const c = m.chrome[s];
      if (!c.fits) errors.push("T13_CHROME_OVERFLOW:" + s + ":" + mode);
      else if (c.uncertain) errors.push("T13_FIT_UNCERTAIN:" + s + ":" + mode);
    });
    m.chrome.avatarInitials.forEach(function(a){
      if (!a.fits) errors.push("T13_CHROME_OVERFLOW:avatarInitials:" + mode);
      else if (a.uncertain) errors.push("T13_FIT_UNCERTAIN:avatarInitials:" + mode);
    });
    if (m.postBody.maxWidthPx > m.postInnerPx + 1) errors.push("T13_CHROME_OVERFLOW:postBody:" + mode);
    if (m.postBody.pessimisticLines > m.postBody.lines + 1) errors.push("T13_FIT_UNCERTAIN:postBody:" + mode);
    if (mode === "narrow" && m.supportingPx < CONFIG.phoneMinSupportingPx) errors.push("T13_FONT_FLOOR:" + mode);
    if (m.stackCqw > m.availCqw) errors.push("T13_STACK_OVERFLOW:" + mode);
  });
  return errors;
}

/* --- Sequential schedule derivation (strict sequencing by construction) --- */
function buildScheduleFor(cards, beats, opening, finalHoldMs){
  const sched = [];
  let t = 0;
  const openingApplies = cards.length >= 2 && opening.firstCardLanded && opening.secondCardPrefix;
  cards.forEach(function(c, i){
    const b = beats[i];
    const n = segmentGraphemes(c.text).length;
    if (i === 0 && openingApplies) {
      sched.push({ appearStart: -opening.offsetMs, typeStart: -opening.offsetMs, typeEnd: -opening.offsetMs, count: n, preLanded: true });
      return;
    }
    if (i === 1 && openingApplies) {
      const pre = segmentGraphemes(opening.secondCardPrefix).length;
      const typeEnd = (n - pre) * b.charMs;
      sched.push({ appearStart: -opening.offsetMs, typeStart: -pre * b.charMs, typeEnd: typeEnd, count: n, preTyped: pre });
      t = typeEnd + b.holdAfterMs;
      return;
    }
    const appearStart = t;
    const typeStart = appearStart + b.appearMs;
    const typeEnd = typeStart + n * b.charMs;
    sched.push({ appearStart: appearStart, typeStart: typeStart, typeEnd: typeEnd, count: n });
    t = typeEnd + b.holdAfterMs;
  });
  const lastEnd = sched.length ? sched[sched.length - 1].typeEnd : 0;
  return { cards: sched, endMs: lastEnd + finalHoldMs };
}

function hexRgb(hex){
  if (!isHexColor(hex)) return [0, 0, 0];
  const n = parseInt(hex.slice(1), 16);
  return [n >> 16 & 255, n >> 8 & 255, n & 255];
}
function colorLerp(a, b, p){
  const A = hexRgb(a), B = hexRgb(b);
  const c = A.map(function(v, i){ return Math.round(v + (B[i] - v) * p); });
  return "rgb(" + c[0] + "," + c[1] + "," + c[2] + ")";
}
