function parseStrictNum(raw){
  if (raw === null || raw === undefined) return null;
  const s = String(raw).trim();
  if (s === "") return null;
  const v = Number(s);
  return isFinite(v) ? v : null;
}
function segmentGraphemes(text){
  if (typeof Intl !== "undefined" && typeof Intl.Segmenter === "function") {
    const seg = new Intl.Segmenter(undefined, { granularity: "grapheme" });
    return Array.from(seg.segment(text), function(s){ return s.segment; });

  return segmentClustersFallback(text);
}
function segmentClustersFallback(text){
  const cps = Array.from(text);
  const clusters = [];
  let cur = null;
  function isAttach(cp){
    return /[\p{Mn}\p{Me}\p{Mc}]|[\uFE00-\uFE0F]|[\u{1F3FB}-\u{1F3FF}]|\u20E3/u.test(cp);
  }
  function isRI(cp){ const n = cp.codePointAt(0); return n >= 0x1F1E6 && n <= 0x1F1FF; }
  for (let i = 0; i < cps.length; i++) {
    const cp = cps[i];
    if (cur === null) { cur = cp; continue; }
    if (isAttach(cp)) { cur += cp; continue; }
    if (cp === "\u200D") { cur += cp; if (i + 1 < cps.length) { cur += cps[i + 1]; i++; } continue; }
    if (isRI(cp)) {
      const prev = Array.from(cur);
      if (prev.length === 1 && isRI(prev[0])) { cur += cp; continue; }
    }
    clusters.push(cur); cur = cp;
  }
  if (cur !== null) clusters.push(cur);
  return clusters;
}
function graphemeRangeOf(haystack, needle){
  if (!needle) return null;
  const h = segmentGraphemes(haystack), n = segmentGraphemes(needle);
  for (let i = 0; i + n.length <= h.length; i++) {
    let ok = true;
    for (let j = 0; j < n.length; j++) if (h[i + j] !== n[j]) { ok = false; break; }
    if (ok) return { start: i, end: i + n.length };
  }
  return null;
}

}

let FONT_UNCERTAIN = false;
let _ctx = null;

    if (cv && typeof cv.getContext === "function") { _ctx = cv.getContext("2d"); if (_ctx) FIT_PATH = "canvas"; }

function estimateCharEm(ch){
  const E = CONFIG.estimateEm;
  if ("iljtf.,:;'!| ".indexOf(ch) >= 0) return E.narrow;
  if ("mwMW@#%".indexOf(ch) >= 0) return E.wide;
  const n = ch.codePointAt(0);
  if (n >= 0x2E80) return E.cjk;
  if (/[A-Z0-9]/.test(ch)) return E.upperDigit;
  return E.base;
}
function estimateWidthPx(text, sizePx){
  let em = 0;
  Array.from(text).forEach(function(ch){ em += estimateCharEm(ch); });
  return em * sizePx;
}
function fontLoaded(fontStr){
  try {
    if (typeof document !== "undefined" && document.fonts && typeof document.fonts.check === "function") {
      return document.fonts.check(fontStr);
    }
  } catch (e) { return true; }
  return true;
}
function measureWidthPx(text, weight, sizePx, letterSpacingPx){
  const fontStr = weight + " " + sizePx + "px " + CONFIG.fontFamily;
  let w;
  if (FIT_PATH === "canvas" && _ctx) {
    if (!fontLoaded(fontStr)) FONT_UNCERTAIN = true;
    _ctx.font = fontStr;
    w = _ctx.measureText(text).width;
  } else {
    w = estimateWidthPx(text, sizePx);
  }
  const g = segmentGraphemes(text).length;
  if (letterSpacingPx && g > 1) w += letterSpacingPx * (g - 1);
  return Math.ceil(w * 100) / 100;
}
/* Wrap simulation: CSS white-space normal collapsing + overflow-wrap
   break-word long-word breaking. Mark padding charged per occupied fragment. */
function fitLines(text, availPx, weight, sizePx, markRange, markPadPx){
  const norm = canonicalText(text);
  const words = norm === "" ? [] : norm.split(" ");
  const graphemes = segmentGraphemes(norm);
  let gi = 0;
  const wordInfos = words.map(function(wd){
    const g = segmentGraphemes(wd);
    const info = { text: wd, width: measureWidthPx(wd, weight, sizePx, 0), start: gi, end: gi + g.length, gcount: g.length };
    gi += g.length + 1;
    return info;
  });
  const spaceW = measureWidthPx(" ", weight, sizePx, 0);
  const effAvail = availPx - 1;
  let lines = 1, curW = 0, curMark = false;
  const lineWidths = [];
  const markRanges = !markRange ? [] : (Array.isArray(markRange) ? markRange : [markRange]);
  function markOn(a, b){ return markRanges.some(function(r){ return b > r.start && a < r.end; }); }
  function markCharge(){ return curMark ? 2 * markPadPx : 0; }
  function closeLine(){ lineWidths.push(curW + markCharge() + 1); lines = lineWidths.length; curW = 0; curMark = false; }
  wordInfos.forEach(function(wd, wi){
    const wdMark = markOn(wd.start, wd.end);
    const sep = (curW > 0) ? spaceW : 0;
    const prospective = curW + sep + wd.width + ((curMark || wdMark) ? 2 * markPadPx : 0) - markCharge();
    if (curW > 0 && prospective > effAvail && wd.width + (wdMark ? 2 * markPadPx : 0) <= effAvail) { closeLine(); }
    if (wd.width > effAvail) {
      if (curW > 0) closeLine();
      const gs = segmentGraphemes(wd.text);
      gs.forEach(function(gch, k){
        const gw = measureWidthPx(gch, weight, sizePx, 0);
        const gMark = markOn(wd.start + k, wd.start + k + 1);
        if (curW > 0 && curW + gw + ((curMark || gMark) ? 2 * markPadPx : 0) - markCharge() > effAvail) closeLine();
        curW += gw;
        if (gMark) curMark = true;
      });
    } else {
      curW += ((curW > 0) ? spaceW : 0) + wd.width;
      if (wdMark) curMark = true;
    }
    if (wi === wordInfos.length - 1) lineWidths.push(curW + markCharge() + 1);
  });
  if (!wordInfos.length) lineWidths.push(0);
  const maxW = Math.max.apply(null, lineWidths);
  const lastW = lineWidths.length ? lineWidths[lineWidths.length - 1] - 1 : 0;
  const uncertain = (FIT_PATH === "estimate" || FONT_UNCERTAIN) && maxW > 0.9 * availPx;
  return { lines: lineWidths.length, maxWidthPx: maxW, lastLineWidthPx: lastW, lineWidthsPx: lineWidths.map(function(w){ return Math.max(0, w - 1); }), uncertain: uncertain, path: FIT_PATH };
}
/* The canonical sequence: the one character sequence fitting, counting, marking
   and playback all consume. CSS white-space collapsing folds runs of tab,
   newline, form feed, carriage return and space to one space and trims the
   ends; a nonbreaking space is never collapsed and never trimmed. */
function canonicalText(s){ return String(s).replace(/[\t\n\f\r ]+/g, " ").replace(/^[\t\n\f\r ]+|[\t\n\f\r ]+$/g, ""); }
function normalizeWs(s){ return canonicalText(s); }
function markRangeFor(text, highlight){
  if (!highlight) return null;
  return graphemeRangeOf(canonicalText(text), canonicalText(highlight));
}
function markRangesFor(text, highlight){
  if (!highlight) return [];
  const hs = Array.isArray(highlight) ? highlight : [highlight];
  const ct = canonicalText(text);
  return hs.map(function(h){ return graphemeRangeOf(ct, canonicalText(h)); }).filter(Boolean);
}
function parseUnitToken(token){
  if (typeof token === "number") return { value: token, unit: "cqw" };
  const parts = String(token).trim().split(/\s+/).map(function(p){
    const mm = p.match(/^(-?[\d.]+)([a-z%]*)$/);
    return mm ? { value: parseFloat(mm[1]), unit: mm[2] || "" } : { value: 0, unit: "" };
  });
  return parts.length === 1 ? parts[0] : { components: parts };
}
function lenPx(token, cqwPx, refPx){
  const t = parseUnitToken(token);
  const one = function(c){
    if (c.unit === "cqw") return c.value * cqwPx;
    if (c.unit === "px") return c.value;
    if (c.unit === "%") return c.value * (refPx || 0) / 100;
    throw new Error("T13_UNIT_UNSUPPORTED:" + token);
  };
  return t.components ? one(t.components[0]) : one(t);
}
function cqwOf(token){ const t = parseUnitToken(token); return t.components ? t.components[0].value : t.value; }
function scaledToken(token, factor){
  const t = parseUnitToken(token);
  if (t.components) return t.components.map(function(c){ return (c.value * factor) + c.unit; }).join(" ");
  return (t.value * factor) + t.unit;
}
function modelFor(mode, stageWidthPx){
  const L = CONFIG.layout[mode];
  /* Unit basis: the stage is the query container. Its own padding token resolves in
     the stage's own unit context (no ancestor query container exists, so the basis is
     the full stage width); every descendant unit resolves against the stage's
     content box. The model carries both bases explicitly. */
  const selfCqwPx = stageWidthPx / 100;
  const padPx = lenPx(L.stagePad, selfCqwPx, stageWidthPx);
  const contentPx = stageWidthPx - 2 * padPx;
  const cqwPx = contentPx / 100;
  const ref = stageWidthPx;
  const titleFontPx = lenPx(L.titleFont, cqwPx, ref);
  const headFontPx = lenPx(L.cardHeadFont, cqwPx, ref);
  const commentsWidthPx = mode === "wide"
    ? stageWidthPx - lenPx(L.commentsLeft, cqwPx, ref) - lenPx(L.commentsRight, cqwPx, ref)
    : contentPx;
  const innerPx = commentsWidthPx - 2 * lenPx(L.cardPadX, cqwPx, ref) - L.cardBorderCqw * cqwPx;
  const gapPx = lenPx(L.headGap, cqwPx, ref);
  const markPadPx = lenPx(L.markPadX, cqwPx, ref);
  const caretPx = lenPx(L.caretWidth, cqwPx, ref) + lenPx(L.caretMarginLeft, cqwPx, ref);
  const titles = CONFIG.cards.map(function(c){
    const mr = markRangesFor(c.text, c.highlight);
    const full = fitLines(c.text, innerPx, L.titleWeight, titleFontPx, mr, markPadPx);
    const gs = segmentGraphemes(canonicalText(c.text));
    let pLines = 0, pWorst = 0, pUnc = false, pOverflow = false;
    for (let k = 0; k <= gs.length; k++) {
      const pre = gs.slice(0, k).join("");
      const pmr = mr.map(function(r){ return { start: r.start, end: Math.min(r.end, k) }; }).filter(function(r){ return r.end > r.start; });
      const pf = fitLines(pre, innerPx, L.titleWeight, titleFontPx, pmr.length ? pmr : null, markPadPx);
      if (pf.lines > pLines) pLines = pf.lines;
      const caretRight = pf.lastLineWidthPx + caretPx;
      if (caretRight > innerPx) {
        if (pf.lines >= CONFIG.maxTitleLines) pOverflow = true;
        if (caretRight > pWorst) pWorst = caretRight;
      } else {
        if (caretRight > pWorst) pWorst = caretRight;
        if ((FIT_PATH === "estimate" || FONT_UNCERTAIN) && pf.lines >= CONFIG.maxTitleLines && caretRight > 0.9 * innerPx) pUnc = true;
      }
    }
    return { lines: full.lines, maxWidthPx: full.maxWidthPx, uncertain: full.uncertain, path: full.path,
      prefixLines: pLines, prefixWorstPx: pWorst, prefixUncertain: pUnc, prefixOverflow: pOverflow, caretPx: caretPx };
  });
  const headers = CONFIG.cards.map(function(c){
    const hw = lenPx(L.avatarSize, cqwPx, ref) + gapPx * 3
      + measureWidthPx(c.handle, L.headWeight, headFontPx, 0)
      + measureWidthPx(c.badge, L.badgeWeight, lenPx(L.badgeFont, cqwPx, ref), 0) + 2 * lenPx(L.badgePadX, cqwPx, ref)
      + measureWidthPx(c.when, L.timeWeight, headFontPx, 0);
    return { widthPx: hw, fits: hw <= innerPx, uncertain: (FIT_PATH === "estimate" || FONT_UNCERTAIN) && hw > 0.9 * innerPx };
  });
  const headerHeightPx = Math.max(lenPx(L.avatarSize, cqwPx, ref),
    headFontPx * L.headLineHeight,
    lenPx(L.badgeFont, cqwPx, ref) * L.headLineHeight + 2 * lenPx(L.badgePadY, cqwPx, ref));
  function cardHeightCqw(ti){
    return (2 * lenPx(L.cardPadY, cqwPx, ref) + headerHeightPx + lenPx(L.titleMargin, cqwPx, ref)
      + titles[ti].lines * titleFontPx * L.titleLineHeight + lenPx(L.footMargin, cqwPx, ref) + lenPx(L.cardFootFont, cqwPx, ref) * L.footLineHeight) / cqwPx;
  }
  const postWidthPx = mode === "wide" ? lenPx(L.postWidth, cqwPx, ref) : contentPx;
  const postInnerPx = postWidthPx - 2 * lenPx(L.postPadX, cqwPx, ref);
  const postFull = CONFIG.post.bodyBefore + CONFIG.post.bodyHighlight + CONFIG.post.bodyAfter;
  const postMark = markRangeFor(postFull, CONFIG.post.bodyHighlight);
  const postBodyFit = fitLines(postFull, postInnerPx, L.bodyWeight, lenPx(L.postBodyFont, cqwPx, ref), postMark, markPadPx);
  const postBodyPess = (FIT_PATH === "estimate" || FONT_UNCERTAIN)
    ? fitLines(postFull, postInnerPx * 0.9, L.bodyWeight, lenPx(L.postBodyFont, cqwPx, ref), postMark, markPadPx) : postBodyFit;
  const postBody = { lines: postBodyFit.lines, maxWidthPx: postBodyFit.maxWidthPx, lineWidthsPx: postBodyFit.lineWidthsPx, uncertain: false, pessimisticLines: postBodyPess.lines };
  const postHeightCqw = (2 * lenPx(L.postPadY, cqwPx, ref) + lenPx(L.postHeadFont, cqwPx, ref) * L.headLineHeight + lenPx(L.bodyMargin, cqwPx, ref)
    + postBody.lines * lenPx(L.postBodyFont, cqwPx, ref) * L.postBodyLineHeight + lenPx(L.postFootMargin, cqwPx, ref) + lenPx(L.postFootFont, cqwPx, ref) * L.footLineHeight) / cqwPx;
  const headCqw = (lenPx(L.commentsHeadFont, cqwPx, ref) * L.headLineHeight + lenPx(L.commentsHeadMargin, cqwPx, ref)) / cqwPx;
  let stackCqw = headCqw;
  CONFIG.cards.forEach(function(c, i){ stackCqw += cardHeightCqw(i) + (i ? L.stackGapCqw : 0); });
  const aspect = mode === "wide" ? CONFIG.wideStageAspect : CONFIG.narrowStageAspect;
  const stageHeightPx = stageWidthPx * aspect[1] / aspect[0];
  const postHeightPx = postHeightCqw * cqwPx;
  const availPx = mode === "wide"
    ? stageHeightPx - lenPx(L.commentsTop, cqwPx, stageHeightPx) - lenPx(L.commentsBottom, cqwPx, stageHeightPx)
    : (stageHeightPx - 2 * padPx) - postHeightPx - lenPx(L.postGap, cqwPx, ref);
  const availCqw = availPx / cqwPx;
  function surf(widthPx, boundPx){ return { widthPx: widthPx, fits: widthPx <= boundPx, uncertain: (FIT_PATH === "estimate" || FONT_UNCERTAIN) && widthPx > 0.9 * boundPx }; }
  const postHeadFontPx = lenPx(L.postHeadFont, cqwPx, ref);
  const chrome = {};
  chrome.postHead = surf(measureWidthPx(CONFIG.post.author, L.headWeight, postHeadFontPx, 0) + gapPx
    + measureWidthPx(CONFIG.post.handle, L.handleWeight, postHeadFontPx, 0) + gapPx
    + measureWidthPx(CONFIG.post.closeMark, L.closeWeight, postHeadFontPx, 0), postInnerPx);
  chrome.commentsHead = surf(measureWidthPx(CONFIG.commentsTitle + " " + (CONFIG.commentsBaseCount + CONFIG.cards.length), L.headWeight, lenPx(L.commentsHeadFont, cqwPx, ref), 0)
    + measureWidthPx(CONFIG.sortLabel, L.sortWeight, lenPx(L.sortFont, cqwPx, ref), 0), commentsWidthPx);
  chrome.cardFoot = surf(measureWidthPx(CONFIG.cardFooter.mark, L.footWeight, lenPx(L.cardFootFont, cqwPx, ref), 0)
    + lenPx(L.cardFootGap, cqwPx, ref) + measureWidthPx(CONFIG.cardFooter.label, L.footWeight, lenPx(L.cardFootFont, cqwPx, ref), 0), innerPx);
  let pfW = 0;
  CONFIG.post.actions.forEach(function(a, ai){ pfW += measureWidthPx(a, L.footWeight, lenPx(L.postFootFont, cqwPx, ref), 0) + (ai ? lenPx(L.footGap, cqwPx, ref) : 0); });
  chrome.postFoot = surf(pfW, postInnerPx);
  const stampFontPx = lenPx(L.stampFont, cqwPx, ref);
  const stampW = measureWidthPx(CONFIG.post.stampText, L.stampWeight, stampFontPx, lenPx(L.stampLetterSpacing, cqwPx, ref))
    + 2 * lenPx(L.stampPadX, cqwPx, ref) + 2 * lenPx(L.stampBorder, cqwPx, ref);
  const stampH = stampFontPx * L.headLineHeight + 2 * lenPx(L.stampPadY, cqwPx, ref) + 2 * lenPx(L.stampBorder, cqwPx, ref);
  const rotRad = Math.abs(cqwOf(L.stampRotate)) * Math.PI / 180;
  const stampRW = stampW * Math.cos(rotRad) + stampH * Math.sin(rotRad);
  const stampRH = stampW * Math.sin(rotRad) + stampH * Math.cos(rotRad);
  chrome.stamp = surf(stampRW, postInnerPx);
  chrome.stamp.rotatedWidthPx = stampRW;
  const bodyLineHPx = lenPx(L.postBodyFont, cqwPx, ref) * L.postBodyLineHeight;

  const stampCx = postWidthPx - lenPx(L.stampRight, cqwPx, postWidthPx) - stampW / 2;
  const stampCy = lenPx(L.stampTop, cqwPx, postHeightPx) + stampH / 2;

  function rectsHit(a, b){ return a.left < b.right && a.right > b.left && a.top < b.bottom && a.bottom > b.top; }
  chrome.stamp.rect = stampRect;
  chrome.stamp.contained = stampRect.left >= 0 && stampRect.top >= 0 && stampRect.right <= postWidthPx && stampRect.bottom <= postHeightPx;
  chrome.stamp.overlapsBodyInk = postBody.lineWidthsPx.some(function(w, li){
    return rectsHit(stampRect, { left: lenPx(L.postPadX, cqwPx, ref), right: lenPx(L.postPadX, cqwPx, ref) + w, top: bodyTopPx + li * bodyLineHPx, bottom: bodyTopPx + (li + 1) * bodyLineHPx });
  });
  const footTopPx = bodyTopPx + postBody.lines * bodyLineHPx + lenPx(L.postFootMargin, cqwPx, ref);
  chrome.stamp.overlapsFoot = rectsHit(stampRect, { left: lenPx(L.postPadX, cqwPx, ref), right: postWidthPx - lenPx(L.postPadX, cqwPx, ref), top: footTopPx, bottom: footTopPx + lenPx(L.postFootFont, cqwPx, ref) * L.footLineHeight });
  chrome.avatarInitials = CONFIG.cards.map(function(c){
    return surf(measureWidthPx(c.initials, L.avatarWeight, lenPx(L.avatarFont, cqwPx, ref), 0), lenPx(L.avatarSize, cqwPx, ref) * 0.92);
  });
  const supportingPx = Math.min(headFontPx, lenPx(L.badgeFont, cqwPx, ref), lenPx(L.avatarFont, cqwPx, ref),
    lenPx(L.cardFootFont, cqwPx, ref), lenPx(L.sortFont, cqwPx, ref), lenPx(L.postFootFont, cqwPx, ref), postHeadFontPx);
  return { titleFontPx: titleFontPx, headFontPx: headFontPx,
    footFontPx: lenPx(L.cardFootFont, cqwPx, ref), innerPx: innerPx, postInnerPx: postInnerPx,
    titles: titles, headers: headers, chrome: chrome, postBody: postBody, supportingPx: supportingPx,
    postBodyLines: postBody.lines,
    basis: { cqwPx: cqwPx, contentPx: contentPx, padPx: padPx, selfCqwPx: selfCqwPx },
    headCqw: headCqw, postHeightPx: postHeightPx, headerHeightPx: headerHeightPx,
    postBodyTopPx: bodyTopPx, bodyLineHPx: bodyLineHPx, stampHPx: stampH,
    postHeightCqw: postHeightCqw, stackCqw: stackCqw, availCqw: availCqw, fitPath: FIT_PATH };
}
function validateConfig(){
  const errors = [];
  if (!CONFIG || !Array.isArray(CONFIG.cards) || !Array.isArray(CONFIG.beats) || !CONFIG.layout || !CONFIG.layout.wide || !CONFIG.layout.narrow || !CONFIG.palette || !CONFIG.post) {
    errors.push("T13_CONFIG_INVALID");
    return errors;
  }
  if (CONFIG.cards.length > CONFIG.maxCards) errors.push("T13_CARD_COUNT_UNSUPPORTED");
  if (CONFIG.beats.length < CONFIG.cards.length) errors.push("T13_BEAT_TABLE_MISMATCH");
  CONFIG.beats.forEach(function(b){ if (!b || !(b.charMs > 0) || !isFinite(b.charMs) || !(b.appearMs >= 0) || !isFinite(b.appearMs) || !(b.holdAfterMs >= 0) || !isFinite(b.holdAfterMs)) errors.push("T13_BEAT_INVALID"); });
  ["landMs", "caretBlinkMs", "finalHoldMs"].forEach(function(k){
    const v = CONFIG[k];
    if (!(typeof v === "number" && isFinite(v) && v > 0 && v <= CONFIG.catalogueWindowMs)) errors.push("T13_DURATION_INVALID:" + k);
  });
  CONFIG.beats.forEach(function(b, i){
    if (b && i < CONFIG.cards.length - 1 && isFinite(b.holdAfterMs) && b.holdAfterMs < CONFIG.landMs) errors.push("T13_HOLD_LT_LAND:" + i);
  });
  if (CONFIG.cards.length >= 2 && CONFIG.opening.firstCardLanded) {
    const g = segmentGraphemes(canonicalText(CONFIG.cards[1].text)), p = segmentGraphemes(canonicalText(CONFIG.opening.secondCardPrefix));
    let ok = p.length <= g.length;
    for (let i = 0; ok && i < p.length; i++) if (g[i] !== p[i]) ok = false;
    if (!ok) errors.push("T13_PREFIX_MISMATCH");
  }
  const widthRuns = [{ mode: "wide", w: CONFIG.wideStageAspect[0] }, { mode: "narrow", w: CONFIG.narrowStageAspect[0] }, { mode: "narrow", w: 390 }];
  try {
    if (typeof document !== "undefined" && document.getElementById) {
      const stageEl = document.getElementById("stage");
      const rw = stageEl && stageEl.clientWidth;
      if (rw > 0 && !widthRuns.some(function(r){ return r.w === rw; })) widthRuns.push({ mode: rw <= CONFIG.narrowBreakpointPx ? "narrow" : "wide", w: rw });
    }
  } catch (e) { /* the runtime width is additive only */ }
  widthRuns.forEach(function(run){
    const mode = run.mode;
    let m;
    try { m = modelFor(mode, run.w); } catch (e) {
      if (e && String(e.message).indexOf("T13_UNIT_UNSUPPORTED") === 0) { errors.push(e.message); return; }
      throw e;
    }
    m.titles.forEach(function(t){
      if (t.lines > CONFIG.maxTitleLines) errors.push("T13_TITLE_UNSUPPORTED:" + mode);
      else if (t.uncertain) errors.push("T13_FIT_UNCERTAIN:" + mode);
      if (t.maxWidthPx > m.innerPx) errors.push("T13_TITLE_TOO_WIDE:" + mode);

function freshRangeFor(ce, s, typed, t){
  if (!typed) return null;
  if (t >= s.typeEnd && landingProgressAt(t, s.typeEnd) >= 1) return null;
  const g = ce.graphemes;
  let start = typed - 1;
  while (start > 0 && g[start - 1] !== " ") start--;
  return { start: start, end: typed };
}
function renderTitle(ce, c, typedCount, freshRange){
  ce.title.textContent = "";
  const shown = ce.graphemes.slice(0, typedCount);
  function kindAt(idx){
    let k = null;
    ce.hlRanges.forEach(function(r){ if (idx >= r.start && idx < r.end && !k) k = "mark"; });
    if (freshRange && idx >= freshRange.start && idx < freshRange.end) k = "fresh";
    return k;
  }
  let runStart = 0;
  function flush(end){
    if (end <= runStart) return;
    const text = shown.slice(runStart, end).join("");
    const k = kindAt(runStart);
    if (k === "mark") {
      const m = document.createElement("mark"); m.textContent = text;
      m.style.background = P.highlightBg; m.style.color = P.highlightInk;
      ce.title.appendChild(m);
    } else if (k === "fresh") {
      const sp = document.createElement("span"); sp.className = "fresh"; sp.textContent = text;
      ce.title.appendChild(sp);
    } else {
      ce.title.appendChild(document.createTextNode(text));
    }
    runStart = end;
  }
  for (let idx = 1; idx <= shown.length; idx++) {
    if (idx === shown.length || kindAt(idx) !== kindAt(runStart)) flush(idx);
  }
  ce.title.appendChild(ce.caret);
}
