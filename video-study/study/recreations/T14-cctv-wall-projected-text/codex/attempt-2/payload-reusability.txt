function parseBaseMs(s){

/* Wall answer wraps into lines of at most maxChars; a word longer than the limit is
   hard-split at the limit so it enters the same wrap path instead of shrinking alone. */
function splitLines(text, maxChars){
  const raw = text.split(" "); const words = [];
  for (let i = 0; i < raw.length; i++){
    let w = raw[i];
    while (w.length > maxChars){ words.push(w.slice(0, maxChars)); w = w.slice(maxChars); }
    if (w !== "") words.push(w);
  }
  const lines = []; let cur = "";
  for (let i = 0; i < words.length; i++){
    const w = words[i];
    const next = cur === "" ? w : cur + " " + w;
    if (next.length <= maxChars || cur === "") cur = next;
    else { lines.push(cur); cur = w; }
  }
  if (cur !== "") lines.push(cur);
  return lines;
}

/* Two-dimensional fit: the font is the smallest of the width fit (measured line widths
   against the box width, less the fit margin), the height fit (box height divided by the
   line count times the line height) and the configured maximum. If that size falls below
   the readable floor, the floor is used and an explicit diagnostic is reported. */
function fitWallFont(stageW, stageH, cfg){
  const lines = splitLines(cfg.wallText, cfg.wallLineMaxChars);
  const boxW = stageW * cfg.wallBoxWidthFraction, boxH = stageH * cfg.wallBoxHeightFraction;
  const usableW = boxW * (1 - cfg.wallFitMarginFraction);
  let widthFit = cfg.wallMaxFontPx;
  for (let i = 0; i < lines.length; i++){
    const atRef = measureLineWidthPx(lines[i], 100, cfg);
    if (atRef > 0) widthFit = Math.min(widthFit, usableW / (atRef / 100));
  }
  const heightFit = lines.length > 0 ? boxH / (lines.length * cfg.wallLineHeight) : cfg.wallMaxFontPx;
  let font = Math.min(cfg.wallMaxFontPx, widthFit, heightFit);
  let diagnostic = null;
  if (font < cfg.wallMinFontPx){ diagnostic = "wall font floor: two-dimensional fit " + font.toFixed(1) + "px is below the readable floor " + cfg.wallMinFontPx + "px"; font = cfg.wallMinFontPx; }
  return { fontPx: font, lines: lines, boxW: boxW, boxH: boxH, widthFit: widthFit, heightFit: heightFit, basis: measureBasis(), diagnostic: diagnostic };
}

  let wallLinesCache = splitLines(CONFIG.wallText, CONFIG.wallLineMaxChars);
  function fitWall(){
    const w = stage.clientWidth || 0, h = stage.clientHeight || 0;
    if (w > 0 && h > 0){ const f = fitWallFont(w, h, CONFIG); wallEl.style.fontSize = f.fontPx.toFixed(1) + "px"; if (f.diagnostic) wallEl.setAttribute("data-fit-diagnostic", f.diagnostic); }
  }
  fitWall();
  window.addEventListener("resize", fitWall);
  let lastKey = "";
  function render(tMs){
    const s = computeState(tMs, CONFIG);
    const key = s.tv1Chars + "|" + s.wallChars + "|" + s.tv2Chars + "|" + s.timestamp;
    if (key === lastKey) return s;
    lastKey = key;
    let tvStr = CONFIG.tvPrompt + CONFIG.tvLine1.slice(0, s.tv1Chars);
    if (s.tv2Chars > 0) tvStr += "\n" + CONFIG.tvLine2.slice(0, s.tv2Chars);
    tvText.textContent = tvStr;
    let remaining = s.wallChars; const out = [];
    for (let i = 0; i < wallLinesCache.length; i++){
      const take = Math.max(0, Math.min(wallLinesCache[i].length, remaining));
      out.push(wallLinesCache[i].slice(0, take));
      remaining -= wallLinesCache[i].length + 1;
      if (remaining < 0) remaining = 0;
    }
    wallEl.textContent = out.join("\n");
    tsEl.textContent = s.timestamp;
    return s;
  }
  let playing = false, startStamp = 0, pausedAt = 0, rafId = null;
  function nowMs(){ return (typeof performance !== "undefined" && performance.now) ? performance.now() : Date.now(); }
  function tick(){ if (!playing) return; render(nowMs() - startStamp); rafId = requestAnimationFrame(tick); }
  function play(){ if (playing) return; playing = true; startStamp = nowMs() - pausedAt; tick(); }
  function pause(){ if (!playing) return; pausedAt = nowMs() - startStamp; playing = false; if (rafId) cancelAnimationFrame(rafId); rafId = null; }
  function restart(){ pausedAt = 0; lastKey = ""; if (playing){ startStamp = nowMs(); } else { render(0); } }
  window.addEventListener("message", function(e){
    if (!e.data || typeof e.data !== "object") return;
    if (e.data.cmd === "play") play();
    if (e.data.cmd === "pause") pause();
    if (e.data.cmd === "restart") restart();
  });
  render(0);
  play();
}
