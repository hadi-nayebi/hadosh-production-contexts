function layoutPhrase(text){
  const widthBound = W * CONFIG.maxWidthRatio, heightBound = H * CONFIG.maxHeightRatio;
  const words = text.split(/\s+/).filter(Boolean);
  const linesFor = (px) => {
    pctx.font = CONFIG.fontWeight + " " + px + "px " + CONFIG.fontFamily;
    if (CONFIG.wrap !== "multiline" || words.length === 0) return [text];
    const lines = []; let line = "";
    for (const w of words){
      const cand = line ? line + " " + w : w;
      if (line && pctx.measureText(cand).width > widthBound){ lines.push(line); line = w; }
      else line = cand;
    }
    if (line) lines.push(line);
    while (lines.length > CONFIG.maxLines){
      const extra = lines.pop();
      lines[lines.length - 1] = lines[lines.length - 1] + " " + extra;
    }
    return lines;
  };
  let px = CONFIG.maxFontPx, lines = linesFor(px), maxLineWidth = 0, blockHeight = 0;
  while (true){
    pctx.font = CONFIG.fontWeight + " " + px + "px " + CONFIG.fontFamily;
    maxLineWidth = 0;
    for (const ln of lines) maxLineWidth = Math.max(maxLineWidth, pctx.measureText(ln).width);
    blockHeight = lines.length * px * CONFIG.lineHeightRatio;
    if ((maxLineWidth <= widthBound && blockHeight <= heightBound) || px <= CONFIG.fontFloorPx) break;
    px = Math.max(CONFIG.fontFloorPx, px - 4);
    lines = linesFor(px);
  }
  const fits = maxLineWidth <= widthBound && blockHeight <= heightBound;
  return { text: text, lines: lines, fontPx: px, fits: fits, shortfall: !fits,
           maxLineWidth: maxLineWidth, blockHeight: blockHeight,
           widthBound: widthBound, heightBound: heightBound };
}

function particleColor(){
  const t = rand();
  const r = Math.round(cB[0] + (cA[0]-cB[0])*t), g = Math.round(cB[1] + (cA[1]-cB[1])*t), b = Math.round(cB[2] + (cA[2]-cB[2])*t);

/* --- particle pool: one persistent pool, two fixed roles ---
   Every particle is created either BORROWABLE or RESERVE; the role
   never changes. Reserve particles are never given a target.
   Every particle carries its own stream velocity (svx, svy): the
   shared wind plus a small per-particle drift, fixed at creation. */
let particles = [];
function newParticle(isReserve){
  const driftX = (rand() - 0.5) * CONFIG.driftXRange, driftY = (rand() - 0.5) * CONFIG.driftYRange;
  return {
    x: rand() * W, y: rand() * H,
    svx: CONFIG.windX + driftX, svy: CONFIG.windY + driftY,
    rvx: 0, rvy: 0,               // release component (dissolve only)
    c: particleColor(),
    delay: rand() * CONFIG.staggerFraction, // per-particle form stagger
    phase: rand() * Math.PI * 2,  // jitter phase while borrowed
    isReserve: !!isReserve,
    borrowed: false,
    released: false,              // dissolve release state (render/targeting separated)
    releaseAt: 0,                 // ms into dissolve at which this particle's release begins
    q: 0,                         // release progress 0..1 during dissolve
    tx: 0, ty: 0, sx: 0, sy: 0,
    renderAlpha: CONFIG.reserveAlpha
  };
}
function reserveCount(){ let n = 0; for (const p of particles) if (p.isReserve) n++; return n; }
function borrowableCount(){ return particles.length - reserveCount(); }
/* Grow the pool until the borrowable share can cover targetCount
   targets AND the reserve share is at least reserveFraction of the
   whole pool. The pool never shrinks, so the reserve share can only
   grow toward the configured fraction, never drop below the count
   already established. */
function ensureCapacity(targetCount){
  while (borrowableCount() < targetCount) particles.push(newParticle(false));
  while (particles.length > 0 && reserveCount() / particles.length < CONFIG.reserveFraction)
    particles.push(newParticle(true));
  if (particles.length === 0) particles.push(newParticle(true));
}

/* --- scheduler: strictly one lifecycle at a time ---
   FORM -> HOLD -> DISSOLVE -> GAP -> FORM (next phrase).
   buildPhrase for phrase N+1 runs only in the GAP -> FORM
   transition, i.e. after phrase N's dissolve has completed.
   schedulerTrace records the transitions for inspection. */
let current = 0, phase = "gap", phaseT = 0, lastTs = null, paused = false;
let clockMs = 0;
const schedulerTrace = [];
function trace(event, phraseIdx){ schedulerTrace.push({ event: event, phrase: phraseIdx, t: Math.round(clockMs) }); }

function buildPhrase(idx){
  const targets = samplePhrase(CONFIG.phrases[idx]);
  ensureCapacity(targets.length);
  let ti = 0;
  for (const p of particles){
    p.released = false; p.q = 0; p.rvx = 0; p.rvy = 0;
    if (p.isReserve){ p.borrowed = false; continue; }   // reserve: never borrowed
    if (ti < targets.length){
      p.borrowed = true;
      p.tx = targets[ti].x; p.ty = targets[ti].y; ti++;
      p.sx = p.x; p.sy = p.y;   // form starts wherever the stream currently is
    } else {
      p.borrowed = false;       // surplus: governed by surplusPolicy at render
    }
  }
  trace("formStart", idx);
}

const DUR = () => ({ form: CONFIG.formMs, hold: CONFIG.holdMs, dissolve: CONFIG.dissolveMs, gap: CONFIG.gapMs })[phase];
const easeOut = t => 1 - Math.pow(1 - t, 3);
function wrapCoords(o){ if (o.x < -CONFIG.wrapMargin) o.x = W + CONFIG.wrapMargin; if (o.x > W + CONFIG.wrapMargin) o.x = -CONFIG.wrapMargin; if (o.y < -CONFIG.wrapMargin) o.y = H + CONFIG.wrapMargin; if (o.y > H + CONFIG.wrapMargin) o.y = -CONFIG.wrapMargin; }

function releaseBorrowed(){
  for (const p of particles){
    if (!p.borrowed) continue;
    /* the particle enters the RELEASED state: targeting ends, but
       its render alpha and release velocity follow the release
       progress q — nothing flips in one frame */
    p.released = true;
    p.releaseAt = rand() * CONFIG.releaseStaggerFraction * CONFIG.dissolveMs;
    p.rvx = (rand() - 0.5) * CONFIG.releaseRangeX;
    p.rvy = (rand() - CONFIG.releaseBiasY) * CONFIG.releaseRangeY;
    p.q = 0;
  }
}

function step(ts){
  requestAnimationFrame(step);
  if (lastTs === null) lastTs = ts;
  let dt = Math.min(64, ts - lastTs); lastTs = ts;
  if (paused) return;
  phaseT += dt; clockMs += dt;
  const dur = DUR();
  let t = dur > 0 ? Math.min(1, phaseT / dur) : 1;
  if (t >= 1){
    phaseT = 0;
    if (phase === "gap"){ phase = "form"; buildPhrase(current); }
    else if (phase === "form"){ phase = "hold"; trace("holdStart", current); }
    else if (phase === "hold"){ phase = "dissolve"; trace("dissolveStart", current); releaseBorrowed(); }
    else if (phase === "dissolve"){
      trace("dissolveEnd", current);
      /* finalize every release at the boundary: a released particle
         leaves dissolve as ordinary surplus (q = 1, alpha already at
         the surplus end of its ramp) — it never re-enters a borrowed
         branch during the gap */
      for (const p of particles) if (p.released){ p.released = false; p.borrowed = false; p.q = 1; }
      phase = "gap";
      current = (current + 1) % CONFIG.phrases.length;
    }
    t = 0;
  }
  /* background: near-black with a breath of the previous frame (stream trails) */
  ctx.fillStyle = CONFIG.background; ctx.globalAlpha = CONFIG.trailFade; ctx.fillRect(0,0,W,H); ctx.globalAlpha = 1;
  const r = CONFIG.particleRadius;
  const dtS = dt / 1000;
  let surplusDrawn = 0;
  /* draw reserve first (background), then surplus/released, then borrowed (foreground) */
  const ordered = [];
  for (const p of particles) if (p.isReserve) ordered.push(p);
  for (const p of particles) if (!p.isReserve && !p.borrowed) ordered.push(p);
  for (const p of particles) if (p.borrowed) ordered.push(p);
  for (const p of ordered){
    if (p.borrowed && p.released && phase === "dissolve"){
      /* RELEASED: release progress q over its own staggered window;
         motion = stream + release component growing with q;
         alpha ramps foreground -> surplus across the same q */
      const win = Math.max(1, CONFIG.dissolveMs - p.releaseAt);
      p.q = Math.max(0, Math.min(1, (phaseT - p.releaseAt) / win));
      p.x += (p.svx + p.rvx * p.q) * dtS;
      p.y += (p.svy + p.rvy * p.q) * dtS;
      wrapCoords(p);
      p.renderAlpha = CONFIG.foregroundAlpha + (CONFIG.surplusAlpha - CONFIG.foregroundAlpha) * p.q;
      if (p.q >= 1){ p.released = false; p.borrowed = false; }
    } else if (p.borrowed && phase === "form"){
      /* anchor advects with the stream every frame (during the
         stagger delay the particle is its anchor: it streams) */
      p.sx += p.svx * dtS; p.sy += p.svy * dtS;
      const anchor = { x: p.sx, y: p.sy }; wrapCoords(anchor); p.sx = anchor.x; p.sy = anchor.y;
      const lt = Math.max(0, Math.min(1, (t - p.delay) / (1 - p.delay || 1)));
      const e = easeOut(lt);
      const jx = Math.sin(clockMs * CONFIG.jitterFreqX + p.phase) * CONFIG.jitter;
      const jy = Math.cos(clockMs * CONFIG.jitterFreqY + p.phase) * CONFIG.jitter;
      p.x = p.sx + (p.tx - p.sx) * e + jx * e;
      p.y = p.sy + (p.ty - p.sy) * e + jy * e;
      p.renderAlpha = CONFIG.foregroundAlpha;
    } else if (p.borrowed){ /* HOLD: on target, low-amplitude jitter, never still */
      p.x = p.tx + Math.sin(clockMs * CONFIG.jitterFreqX + p.phase) * CONFIG.jitter;
      p.y = p.ty + Math.cos(clockMs * CONFIG.jitterFreqY + p.phase) * CONFIG.jitter;
      p.renderAlpha = CONFIG.foregroundAlpha;
    } else {
      /* streaming: reserve, surplus, and released particles all ride
         the flow field; a decaying release component rides on top
         during and after dissolve — stream velocity never resets */
      const decay = phase === "dissolve" ? 1 : Math.max(0, 1 - phaseT / (CONFIG.releaseDecayMs || 1));
      p.x += (p.svx + p.rvx * decay) * dtS;
      p.y += (p.svy + p.rvy * decay) * dtS;
      if (phase !== "dissolve" && decay === 0){ p.rvx = 0; p.rvy = 0; }
      wrapCoords(p);
      if (p.isReserve){
        p.renderAlpha = CONFIG.reserveAlpha;
      } else if (CONFIG.surplusPolicy === "capped-ambient"){
        surplusDrawn++;
        if (surplusDrawn > CONFIG.ambientCap) continue; // surplus only; reserve always drawn
        p.renderAlpha = CONFIG.reserveAlpha;
      } else { /* drift-fade */
        p.renderAlpha = CONFIG.surplusAlpha;
      }
    }
    ctx.globalAlpha = p.renderAlpha;
    ctx.fillStyle = p.c;
    ctx.fillRect(p.x - r/2, p.y - r/2, r, r);
  }
  ctx.globalAlpha = 1;
}
trace("gapStart", 0);
requestAnimationFrame(step);

## Motion model (one constant flow field)

- Every particle is created with a fixed role — borrowable or reserve — and a persistent stream
  velocity: the shared wind plus a per-particle drift drawn from the configured drift ranges.
- FORM: a borrowed particle's trajectory is `position = a(t) + (target − a(t))·e + jitterOffset(t)·e`,
  where `a(t)` is its form-start anchor advected by its own stream velocity every frame,
  `e = easeOutCubic(clamp((phaseProgress − delay)/(1 − delay)))`, and `jitterOffset` is the same
  sinusoidal offset used in HOLD. During the stagger delay (e = 0) the particle sits exactly on
  its advecting anchor, i.e. it streams. At e = 1 the position equals `target + jitterOffset(t)`
  (the HOLD expression) and, since de/du = 0 at u = 1, the velocity equals the HOLD jitter
  velocity; both were measured continuous across the boundary (own-check results below). HOLD
  deviation from target is bounded by the jitter amplitude.
- HOLD: borrowed particles sit at target + jitterOffset; every non-borrowed particle streams.
- DISSOLVE: each borrowed particle enters a released state with a staggered release time
  (`releaseAt`, spread over `releaseStaggerFraction` of the dissolve). Its release progress q runs
  0→1 over its own release window; its velocity is stream + release component × q; its render
  alpha is `foregroundAlpha + (surplusAlpha − foregroundAlpha)·q`. At the dissolve→gap boundary
  every release is finalized (q = 1, borrowed/released flags cleared), so a released particle
  enters the gap as ordinary surplus. After dissolve, the release component decays over
  `releaseDecayMs`.
- GAP: all particles stream; the next phrase's targets are sampled and assigned only in the
  GAP → FORM transition.
- Render order is reserve first, surplus/released second, borrowed last; each particle's render
  alpha is assigned from its current role/state every frame.
- All motion integrates in seconds from the frame dt (clamped at 64 ms); jitter is a function of
  the absolute logical clock.

## Scheduler

States: `gap → form → hold → dissolve → gap`, one phrase lifecycle at a time. A
`schedulerTrace` array records `{event, phrase, t}` for `gapStart`, `formStart`, `holdStart`,
`dissolveStart`, `dissolveEnd` against an accumulated logical clock. Default trace (own-check
run): gapStart(0)@0 · formStart(0)@250 · holdStart(0)@3250 · dissolveStart(0)@5750 ·
dissolveEnd(0)@7750 · formStart(1)@8000 — the next phrase's formation begins strictly after the
previous dissolve ends. `window.T15_API` exposes CONFIG, the sampler, the particle array, the
current phase/phrase, the trace, the last layout record, and the normalization notes; it has no
effect on rendering.

## Layout

`samplePhrase` lays the phrase out before sampling: in single mode the whole phrase is one line;
in multiline mode words wrap greedily to at most `maxLines` lines. The font shrinks from
`maxFontPx` until the widest line fits `maxWidthRatio` of the canvas AND the block height
(lines × fontPx × `lineHeightRatio`) fits `maxHeightRatio`, or `fontFloorPx` is reached. The
layout record (`T15_API.getLayout()`) reports lines, fontPx, measured widths/heights, bounds, and
a `shortfall` flag when the floor is reached without fitting.

## Restart

The `recreation:restart` handler re-seeds the RNG from the effective `CONFIG.seed`, clears the
particle pool (it rebuilds at the next formation), and resets current/phase/phaseT/clock/trace.
Restart reproduces the fresh-load state exactly under fixed-step driving (own-check result below).

## Surplus disposition

Surplus rendering is policy-governed: under `drift-fade` every surplus particle is drawn at
`surplusAlpha`; under `capped-ambient` at most `ambientCap` surplus particles are drawn in a
frame (reserve and borrowed always drawn). The pool itself never shrinks: its size is set by the
largest phrase sampled so far, and the reserve count is non-decreasing across phrases. Bounded
pool shrinkage is not implemented in this attempt.
