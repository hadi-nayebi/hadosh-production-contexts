/* ---------- checkpoint model ----------
   Derivation: three anchors — first page at 0, a middle page at
   the configured page/time fractions, the last page at the
   configured end-time fraction — so any resolved pageCount /
   flipMs pair produces a coherent schedule whose last page is
   reached before the flip phase ends. An explicitly supplied list
   is used only if EVERY entry is strictly valid: object with
   number-typed finite fields, integer page within range, pages
   and times strictly increasing, first entry at time 0, last time
   before the flip phase ends. Any violation derives the schedule
   instead, with a recorded warning. */
function deriveCheckpoints(){
  const pc = CONFIG.pageCount, fm = CONFIG.flipMs;
  const midPage = Math.min(pc, Math.max(2, 1 + Math.round((pc - 1) * CONFIG.checkpointMidPageFraction)));
  const cps = [ {page:1, atMs:0},
                {page:midPage, atMs: Math.round(fm * CONFIG.checkpointMidTimeFraction)},
                {page:pc, atMs: Math.round(fm * CONFIG.checkpointEndTimeFraction)} ];
  return cps.filter((c, i) => i === 0 || (c.page > cps[i-1].page && c.atMs > cps[i-1].atMs));
}
function strictCheckpoints(cps){
  if (!Array.isArray(cps) || cps.length < 1) return null;
  for (const c of cps) {
    if (!c || typeof c !== "object") return null;
    if (!isNum(c.page) || !isNum(c.atMs)) return null;
    if (!Number.isInteger(c.page) || c.page < 1 || c.page > CONFIG.pageCount) return null;
    if (c.atMs < 0) return null;
  }
  if (cps[0].atMs !== 0) return null;
  for (let i = 1; i < cps.length; i++) {
    if (!(cps[i].atMs > cps[i-1].atMs) || !(cps[i].page > cps[i-1].page)) return null;
  }
  if (cps[cps.length-1].atMs >= CONFIG.flipMs) return null;
  return cps;
}
let RESOLVED_CHECKPOINTS = null;
function resolveCheckpoints(){
  if (CONFIG.flipCheckpoints === null || CONFIG.flipCheckpoints === undefined) {
    RESOLVED_CHECKPOINTS = deriveCheckpoints();
  } else {
    const ok = strictCheckpoints(CONFIG.flipCheckpoints);
    if (ok) RESOLVED_CHECKPOINTS = ok;
    else { RESOLVED_CHECKPOINTS = deriveCheckpoints(); warn("supplied checkpoint list failed strict validation; derived schedule used"); }
  }
  return RESOLVED_CHECKPOINTS;
}
function pageAtTime(t){
  const cps = RESOLVED_CHECKPOINTS || resolveCheckpoints();
  if (t <= cps[0].atMs) return Math.min(CONFIG.pageCount, Math.max(1, cps[0].page));
  for (let i = 1; i < cps.length; i++){
    if (t <= cps[i].atMs){
      const a = cps[i-1], b = cps[i];
      const f = (t - a.atMs) / Math.max(1, b.atMs - a.atMs);
      return Math.min(CONFIG.pageCount, Math.max(1, Math.floor(a.page + (b.page - a.page)*f)));
    }
  }
  return CONFIG.pageCount;
}
/* Schedule derivation: page-change instants, per-page intervals,
   the minimum interval, and the settled dwell of each page (its
   interval minus flip travel and maximum stagger). */
function computeSchedule(cfg){
  const saveCps = CONFIG.flipCheckpoints, savePc = CONFIG.pageCount, saveFm = CONFIG.flipMs, saveRes = RESOLVED_CHECKPOINTS;
  if (cfg) { if (cfg.flipCheckpoints !== undefined) CONFIG.flipCheckpoints = cfg.flipCheckpoints; if (cfg.pageCount) CONFIG.pageCount = cfg.pageCount; if (cfg.flipMs) CONFIG.flipMs = cfg.flipMs; }
  RESOLVED_CHECKPOINTS = null; resolveCheckpoints();
  const changes = [{page: pageAtTime(0), atMs: 0}];
  let prev = changes[0].page;
  const fm = cfg && cfg.flipMs ? cfg.flipMs : CONFIG.flipMs;
  for (let t = 0.5; t <= fm; t += 0.5){
    const pg = pageAtTime(t);
    if (pg !== prev){ changes.push({page: pg, atMs: t}); prev = pg; }
  }
  const c = cfg || CONFIG;
  const maxStagger = c.travelFlipMs * c.staggerFlipFraction;
  const dwellTable = changes.map((ch, i) => {
    const end = (i + 1 < changes.length) ? changes[i+1].atMs : fm;
    const intervalMs = end - ch.atMs;
    return { page: ch.page, atMs: ch.atMs, intervalMs: intervalMs,
             dwellMs: Math.max(0, intervalMs - c.travelFlipMs - maxStagger) };
  });
  let minInterval = Infinity;
  for (const d of dwellTable) if (d.intervalMs > 0 && d.intervalMs < minInterval) minInterval = d.intervalMs;
  const out = { changes: changes, dwellTable: dwellTable, minIntervalMs: minInterval,
           maxStaggerMs: maxStagger,
           budgetMs: c.travelFlipMs + maxStagger + c.minSettledHoldMs,
           budgetOk: c.travelFlipMs + maxStagger + c.minSettledHoldMs <= minInterval,
           checkpoints: (RESOLVED_CHECKPOINTS || []).map(x => ({page:x.page, atMs:x.atMs})) };
  CONFIG.flipCheckpoints = saveCps; CONFIG.pageCount = savePc; CONFIG.flipMs = saveFm; RESOLVED_CHECKPOINTS = saveRes;
  return out;
}
/* Budget enforcement: travel is reduced in steps until the budget
   holds against the resolved schedule's minimum interval. */
function enforceBudget(){
  let sched = computeSchedule();
  let guard = 0;
  while (!sched.budgetOk && CONFIG.travelFlipMs > 5 && guard < 40) {
    CONFIG.travelFlipMs = Math.max(5, CONFIG.travelFlipMs * 0.8);
    warn("travelFlipMs reduced to " + CONFIG.travelFlipMs + " to satisfy the schedule budget");
    sched = computeSchedule(); guard++;
  }
  return sched;
}

function sampleDrawn(ctx, w, h, gap){
  const img = ctx.getImageData(0,0,w,h).data, out = [];
  for (let y = 0; y < h; y += gap) for (let x = 0; x < w; x += gap)
    if (img[(y*w + x)*4 + 3] > CONFIG.sampleAlphaThreshold) out.push({x:x, y:y});
  return out;
}
function sampleDrawnRegion(ctx, rx, ry, rw, rh, W, H, gap){
  const x0 = Math.max(0, Math.floor(rx)-4), y0 = Math.max(0, Math.floor(ry)-4);
  const x1 = Math.min(W, Math.ceil(rx+rw)+4), y1 = Math.min(H, Math.ceil(ry+rh)+4);
  const img = ctx.getImageData(x0,y0,x1-x0,y1-y0).data, out = [];
  for (let y = y0; y < y1; y += gap) for (let x = x0; x < x1; x += gap)
    if (img[((y-y0)*(x1-x0) + (x-x0))*4 + 3] > CONFIG.sampleAlphaThreshold) out.push({x:x, y:y});
  return out;
}

/* The pool is sized once, at build: the largest simultaneous demand
   — the MAXIMUM over every page's target count, or title + ring
   together — grossed up by the reserve fraction. The pool never
   grows afterwards, and assignment completeness is asserted
   against this sizing. */
function buildPool(){

  let pageNeed = 0;
  for (let pg = 1; pg <= CONFIG.pageCount; pg++) pageNeed = Math.max(pageNeed, samplePageTargets(W, H, pg).length);
  const formNeed = sampleTitleTargets(W, H).length + sampleHaloTargets(W, H).length;
  const needTotal = Math.ceil(Math.max(pageNeed, formNeed) / (1 - CONFIG.reserveFraction));
  for (let i = 0; i < needTotal; i++) particles.push(newParticle(true));

  for (let i = 0; i < reserveNeed; i++) particles[i].borrowable = false;

function startTravel(p, t, now, travelMs, maxDelayMs){
  p.tx = t.x; p.ty = t.y; p.kind = t.kind; p.key = t.key; p.persistent = !!t.persistent;
  p.borrowed = true; p.t0 = now; p.delay = rng()*maxDelayMs; p.travel = travelMs;
  p.captured = false; p.cueT0 = -1e9;
}
function releaseParticle(p, now){ p.borrowed = false; p.captured = false; p.cueT0 = -1e9; p.relT = now; }

function borrowFresh(targets, now, travelMs, staggerFraction){
  const free = particles.filter(p => p.borrowable && !p.borrowed);
  const n = Math.min(free.length, targets.length);
  for (let i = 0; i < n; i++) startTravel(free[i], targets[i], now, travelMs, travelMs*staggerFraction);
  return n;
}
/* Retarget the borrowed set onto a new page's targets.
   Returns the transition record, including any target that could
   not be assigned (unassigned must be zero under the pool sizing). */
const transitions = [];
function retargetPage(targets, now, page){
  const current = particles.filter(p => p.borrowed);
  const byKey = new Map();
  for (const p of current) if (!byKey.has(p.key)) byKey.set(p.key, p);
  const used = new Set(), unmatched = [];
  let persisted = 0, reassigned = 0, fresh = 0, released = 0, unassigned = 0;
  for (const t of targets){
    const p = byKey.get(t.key);
    if (p && !used.has(p)){
      used.add(p); persisted++;
      if (p.tx === t.x && p.ty === t.y){ p.cueT0 = now; }
      else startTravel(p, t, now, CONFIG.travelFlipMs, CONFIG.travelFlipMs*CONFIG.staggerFlipFraction);
    } else unmatched.push(t);
  }
  const spare = current.filter(p => !used.has(p));
  for (const t of unmatched){
    let best = null, bestD = Infinity;
    for (const p of spare){
      if (used.has(p) || p.persistent) continue;
      const d = (p.tx - t.x)*(p.tx - t.x) + (p.ty - t.y)*(p.ty - t.y);
      if (d < bestD){ bestD = d; best = p; }
    }
    if (best){ used.add(best); reassigned++; startTravel(best, t, now, CONFIG.travelFlipMs, CONFIG.travelFlipMs*CONFIG.staggerFlipFraction); }
    else {
      const freeP = particles.find(p => p.borrowable && !p.borrowed);
      if (freeP){ used.add(freeP); fresh++; startTravel(freeP, t, now, CONFIG.travelFlipMs, CONFIG.travelFlipMs*CONFIG.staggerFlipFraction); }
      else unassigned++;
    }
  }
  for (const p of current) if (!used.has(p)){ releaseParticle(p, now); released++; }
  const rec = { page: page, atMs: Math.round(now), targets: targets.length,
    persisted: persisted, reassigned: reassigned, fresh: fresh, released: released,
    unassigned: unassigned, churn: reassigned + released };
  transitions.push(rec);
  return rec;
}

/* Dispersal measurement under the constant-field model: the spread
   of the most recently released silhouette relative to its spread
   at release, and its centroid displacement. The model prices the
   expected behaviour exactly: centroid displacement equals the
   stream velocity times elapsed time; spread growth is bounded by
   the drift ranges times elapsed time. */
function releaseSpreadRatio(){

  if (!s) return null;
  let x0=Infinity, y0=Infinity, x1=-Infinity, y1=-Infinity, n=0;
  for (const p of particles) if (!p.borrowed && p.relT > 0 && simT - p.relT <= CONFIG.releaseMs + 60){
    n++; if (p.x<x0)x0=p.x; if (p.x>x1)x1=p.x; if (p.y<y0)y0=p.y; if (p.y>y1)y1=p.y;
  }
  if (!n) return null;
  return Math.max((x1-x0)/s.w, (y1-y0)/s.h);
}
function releaseModelBound(elapsedMs){
  const s = diagnostics.lastSpread;
  if (!s) return null;

function isSettled(p, now){
  if (!p.borrowed) return false;
  if (now - p.cueT0 < p.travel) return false;
  if (!p.captured) return false;

/* Zero-end-derivative travel profile: s(f) with s(0)=0, s(1)=1 and
   s'(0)=s'(1)=0, so anchor velocity starts and ends at zero and the
   trajectory velocity (anchor + stream) is continuous at capture
   and settlement. */
function easeProfile(f){
  const e = CONFIG.easingExponent;
  const a = Math.pow(f, e), b = Math.pow(1-f, e);
  return a / (a + b || 1);
}
function updateParticles(dtMs){

  const pivotX = CONFIG.pageXR*W;
  for (const p of particles){

           its offset; the travel anchor origin is captured only once
           the delay has elapsed, from the rendered position then */

          p.ax0 = p.x - p.ox; p.ay0 = p.y - p.oy; p.captured = true;
          p.lastCapture = { t: now, x: p.x, y: p.y };

           during the flip window; backing-stack anchors never turn */

          (f < 1) || (now - p.cueT0 >= 0 && now - p.cueT0 < p.travel));
        if (turning){
          const tf = (f < 1) ? f : Math.min(1, Math.max(0, (now - p.cueT0) / p.travel));
          const prof = Math.sin(Math.PI*tf);
          const factor = 1 - (1 - CONFIG.turnCosMin) * prof * prof;
          ax = pivotX + (ax - pivotX) * factor;
          p.renderAlpha = CONFIG.travelAlpha;
        } else if (f < 1) {
          p.renderAlpha = CONFIG.travelAlpha;
        } else {
          p.renderAlpha = p.kind === "stack" ? CONFIG.stackAlpha : (p.kind === "halo" ? CONFIG.haloAlpha : CONFIG.foregroundAlpha);
        }
        p.ax = ax; p.ay = ay;
      }

      /* released / free: the anchor freezes once the jitter amplitude
         has faded out smoothly; the offset carries the particle */

        if (dec >= 1) p.jitAmp = 0;

      p.renderAlpha = p.borrowable ? CONFIG.surplusAlpha : CONFIG.reserveAlpha;

    p.x = p.ax + p.ox; p.y = p.ay + p.oy;
  }
}

### Construction decision recorded before building: schedule budget enforcement
The schedule budget (travel + maximum stagger + reserved settled hold, against the minimum page interval) is enforced during configuration validation: if the budget cannot hold, the flip travel time is reduced in steps and a warning is recorded, so execution never proceeds on a schedule whose own budget reports failure.

### Interval budget filing for the rapid pages (pages 6–13)
Preserved anchors: first checkpoint at 0 ms, middle checkpoint at 1500 ms, final page reached at 2200 ms, flip phase ends at 2600 ms, full cycle 6000 ms, title formation/completion and hold anchors unchanged.
- The rapid span (1500 ms → 2200 ms) holds 8 page intervals of 87.5 ms each.
- Scheduling criterion: at least one settled rendered frame per rapid page at multiple frame-phase offsets, at a 60 fps cadence and at permitted 50 ms steps (modelled).
- Reading criterion: settled dwell of at least 80 ms per rapid page (modelled).
- Chosen construction values: flip travel 28 ms, maximum stagger 8.4 ms (fraction 0.30), worst-case arrival 36.4 ms, modelled settled window 87.5 − 36.4 = 51.1 ms per rapid page.
- Scheduling arithmetic: the 51.1 ms settled window exceeds the 50 ms maximum step, so at least one step instant falls inside the window at every phase offset; at 60 fps (16.67 ms period) at least three frame instants fall inside it at every phase offset. The arrival completes inside the position update of the step in which travel elapses, and that step's draw shows the settled position, so the step instant at or after arrival counts as a settled rendered frame.
- Reading arithmetic: an 80 ms dwell inside an 87.5 ms interval leaves at most 7.5 ms for travel plus stagger combined, which is incompatible with any continuous travel motion (the arrival would be an instantaneous snap, not a travel). The 80 ms reading criterion is therefore UNREACHABLE under the preserved anchors. Filed before building: the reading dwell target is NOT carried by this construction; the construction carries the scheduling criterion above and the modelled dwell rises from 23.66 ms to 51.1 ms per rapid page. The dwell trade is priced against the preserved anchors, which are not moved.

## Construction facts
- Single self-contained HTML file; one canvas particle system; no external assets. Page, title and ring targets are produced at runtime by sampling content rendered from CONFIG values on offscreen canvases. No content value appears in markup or logic outside CONFIG: the counter and label elements start empty and are filled from CONFIG at initialization.
- Phase sequence inside one 6000 ms cycle: flip [0, 2600) · release [2600, 3100) · form [3100, 4600) · hold [4600, 5700) · seam [5700, 6000). At the seam's start the title and ring assignments release; no page target is borrowed until the next cycle begins. Title travel completes by ~4478 ms; ring travel by ~4552 ms; the settled hold begins at 4600 ms.
- Page schedule: derived from the resolved page count and flip duration through anchor fractions (middle page fraction 5/13 at time fraction 1500/2600; last page at time fraction 2200/2600). At the default configuration the resolved schedule is page 1 at 0 ms · page 6 at 1500 ms · page 14 at 2200 ms. An explicitly supplied checkpoint list replaces the derived schedule only when every entry passes strict validation (number-typed finite fields, integer pages within range, strictly increasing pages and times, first entry at time 0, last time before the flip phase ends); any violation derives the schedule and records a warning.
- Schedule budget enforcement: travel + maximum stagger + reserved settled hold is checked against the minimum page interval of the resolved schedule during validation; if it cannot hold, flip travel is reduced in steps (warning recorded) until it holds.
- Motion construction: every particle's rendered position is anchor + stream offset. The offset accumulates the particle's stream velocity every frame in every state. The anchor moves only through a travel profile whose derivative is zero at both ends; jitter amplitude fades in with the same profile and fades out smoothly after release before the anchor freezes. The settled reference frame is the target frame: a settled particle's anchor equals its target plus the faded-in jitter.
- Page turn: during a flip window the front page's anchors (outline, header, lines) compress horizontally toward the page's left edge by a zero-end-derivative profile reaching a minimum factor of 0.15 at mid-window; backing-stack anchors never take part. The turn changes no assignment.
- Release adds no field component: a released particle's anchor freezes and its offset carries it. Silhouette behaviour after release is priced against the constant-field model exactly (centroid translates at stream velocity; spread growth bounded by the drift ranges) and measured against that bound.
- Text fitting prices width AND height: header fits a width box (0.84 of page width) and a height box (0.17 of page height); the title fits a width box (0.86 of stage width) and a height box (0.30 of stage height). The named no-fit policy is CONDENSE-AT-FLOOR: at the minimum font size, a condensation transform brings the rendered bounds inside the box.
- Pool sizing: maximum target demand over EVERY page, and title + ring together, grossed up by the reserve fraction; the pool never grows afterwards. Transition records carry an unassigned count, asserted zero.
- Restart resets all run state: simulation time, phase, current page, trace, transition records, diagnostics, scheduler flags, the animation timestamp, and the re-seeded pool.
- Intensities are named and distinct: travelling 0.38 · settled stack 0.60 · settled page/title 1.0 · ring 0.95 · surplus 0.16 · reserve 0.28.
- Configuration validation runs before sampling: sampling gaps, radii, line pattern, colours, periods, intensities, easing exponent and turn factor are type- and range-checked with recorded warnings; URL numeric overrides are parsed strictly (a value that is not entirely a number is rejected with a warning).
- Outline and stack strokes are sampled at a denser gap (3 px) than filled content (4 px).
- CONFIG carries the technique's values, including the values that prescribe behaviour in code: stack vertical offset factor, line-width floor and thickness floor, fit floors and shrink steps, easing exponent, wrap margin, minimum stage dimensions, sampling alpha threshold. URL overrides: title, titleLabel, pageHeader, pageCount, flipMs, releaseMs, formMs, holdMs, seamReleaseMs, sampleGap, outlineGap, reserveFraction, palette colours, flipCheckpoints.
- comparison.html is unchanged: it embeds the recreation by reference and carries no configuration interface of its own; its ORIGINAL side is untouched (snapshot sha recorded in the preservation section above).
