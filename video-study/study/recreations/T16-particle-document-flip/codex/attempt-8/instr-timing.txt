/* ---------- checkpoint model ----------
   Derivation: three anchors — first page at 0, a middle page at
   the configured page/time fractions, the last page at the
   configured end-time fraction — so any resolved pageCount /
   flipMs pair produces a coherent schedule whose last page is
   reached before the flip phase ends. An explicitly supplied list
   is used only if EVERY entry is strictly valid AND the list is
   endpoint-complete: object with number-typed finite fields,
   integer page within range, pages and times strictly increasing,
   first entry = page 1 at time 0, last entry = the final page
   (a singleton list is complete only for a one-page stack).
   Any violation derives the schedule instead, with a warning. */
function deriveCheckpoints(){
  const pc = CONFIG.pageCount, fm = CONFIG.flipMs;
  const midPage = Math.min(pc, Math.max(2, 1 + Math.round((pc - 1) * CONFIG.checkpointMidPageFraction)));
  const cps = [ {page:1, atMs:0},
                {page:midPage, atMs: Math.round(fm * CONFIG.checkpointMidTimeFraction)},
                {page:pc, atMs: Math.round(fm * CONFIG.checkpointEndTimeFraction)} ];
  return cps.filter((c, i) => i === 0 || (c.page > cps[i-1].page && c.atMs > cps[i-1].atMs));
}
function strictCheckpoints(cps){
  if (!Array.isArray(cps) || cps.length < 1) return null;
  for (const c of cps) {
    if (!c || typeof c !== "object") return null;
    if (!isNum(c.page) || !isNum(c.atMs)) return null;
    if (!Number.isInteger(c.page) || c.page < 1 || c.page > CONFIG.pageCount) return null;
    if (c.atMs < 0) return null;
  }
  if (cps[0].atMs !== 0) return null;
  if (cps[0].page !== 1) return null;                                   // endpoint completeness: starts at page 1
  if (cps[cps.length-1].page !== CONFIG.pageCount) return null;        // endpoint completeness: ends at the final page
  for (let i = 1; i < cps.length; i++) {
    if (!(cps[i].atMs > cps[i-1].atMs) || !(cps[i].page > cps[i-1].page)) return null;
  }
  if (cps[cps.length-1].atMs >= CONFIG.flipMs) return null;
  return cps;
}
let RESOLVED_CHECKPOINTS = null;
function resolveCheckpoints(){
  if (CONFIG.flipCheckpoints === null || CONFIG.flipCheckpoints === undefined) {
    RESOLVED_CHECKPOINTS = deriveCheckpoints();
  } else {
    const ok = strictCheckpoints(CONFIG.flipCheckpoints);
    if (ok) RESOLVED_CHECKPOINTS = ok;
    else { RESOLVED_CHECKPOINTS = deriveCheckpoints(); warn("supplied checkpoint list failed strict validation; derived schedule used"); }
  }
  return RESOLVED_CHECKPOINTS;
}
function pageAtTime(t){
  const cps = RESOLVED_CHECKPOINTS || resolveCheckpoints();
  if (t <= cps[0].atMs) return Math.min(CONFIG.pageCount, Math.max(1, cps[0].page));
  for (let i = 1; i < cps.length; i++){
    if (t <= cps[i].atMs){
      const a = cps[i-1], b = cps[i];
      const f = (t - a.atMs) / Math.max(1, b.atMs - a.atMs);
      return Math.min(CONFIG.pageCount, Math.max(1, Math.floor(a.page + (b.page - a.page)*f)));
    }
  }
  return CONFIG.pageCount;
}
/* Schedule derivation: page-change instants, per-page intervals,
   the minimum interval, and the settled dwell of each page (its
   interval minus flip travel and maximum stagger). */
function turnEffForInterval(intervalMs, c){
  const cc = c || CONFIG;
  return Math.max(1, Math.min(cc.turnMs, intervalMs - cc.minSettledHoldMs));
}
function computeSchedule(cfg){
  const saveCps = CONFIG.flipCheckpoints, savePc = CONFIG.pageCount, saveFm = CONFIG.flipMs, saveRes = RESOLVED_CHECKPOINTS;
  if (cfg) { if (cfg.flipCheckpoints !== undefined) CONFIG.flipCheckpoints = cfg.flipCheckpoints; if (cfg.pageCount) CONFIG.pageCount = cfg.pageCount; if (cfg.flipMs) CONFIG.flipMs = cfg.flipMs; }
  RESOLVED_CHECKPOINTS = null; resolveCheckpoints();
  const changes = [{page: pageAtTime(0), atMs: 0}];
  let prev = changes[0].page;
  const fm = cfg && cfg.flipMs ? cfg.flipMs : CONFIG.flipMs;
  for (let t = 0.5; t <= fm; t += 0.5){
    const pg = pageAtTime(t);
    if (pg !== prev){ changes.push({page: pg, atMs: t}); prev = pg; }
  }
  const c = cfg || CONFIG;
  const maxStagger = c.travelFlipMs * c.staggerFlipFraction;
  const regMsC = isNum(c.registrationMs) ? c.registrationMs : CONFIG.registrationMs;
  const dwellTable = changes.map((ch, i) => {
    const end = (i + 1 < changes.length) ? changes[i+1].atMs : fm;
    const intervalMs = end - ch.atMs;
    const turnEffMs = turnEffForInterval(intervalMs, c);
    /* complete-lifecycle term for this interval: arrival plus the
       span-derived page registration the runtime applies (capped
       against the turn's readable onset), so a budget that passes
       also completes the lifecycle */
    const regPageMs = Math.max(1, Math.min(regMsC, turnEffMs * c.turnOnsetFraction - (c.travelFlipMs + c.travelFlipMs * c.staggerFlipFraction)));
    const lifecycleMs = c.travelFlipMs + maxStagger + regPageMs;
    return { page: ch.page, atMs: ch.atMs, intervalMs: intervalMs,
             turnEffMs: turnEffMs, regPageMs: regPageMs, lifecycleMs: lifecycleMs,
             dwellMs: Math.max(0, intervalMs - c.travelFlipMs - maxStagger),
             readableMs: Math.max(0, intervalMs - turnEffMs) };
  });
  let minInterval = Infinity;
  for (const d of dwellTable) if (d.intervalMs > 0 && d.intervalMs < minInterval) minInterval = d.intervalMs;
  const arrivalMs = c.travelFlipMs + maxStagger;
  let budgetMs = 0, budgetOk = true;
  for (const d of dwellTable) {
    const need = Math.max(arrivalMs + d.regPageMs, d.turnEffMs) + c.minSettledHoldMs;
    if (need > budgetMs) budgetMs = need;
    if (d.intervalMs > 0 && need > d.intervalMs) budgetOk = false;
  }
  const out = { changes: changes, dwellTable: dwellTable, minIntervalMs: minInterval,
           maxStaggerMs: maxStagger, arrivalMs: arrivalMs,
           budgetMs: budgetMs,
           budgetOk: budgetOk,
           checkpoints: (RESOLVED_CHECKPOINTS || []).map(x => ({page:x.page, atMs:x.atMs})) };
  CONFIG.flipCheckpoints = saveCps; CONFIG.pageCount = savePc; CONFIG.flipMs = saveFm; RESOLVED_CHECKPOINTS = saveRes;
  return out;
}

   holds against the resolved schedule's minimum interval. If it

   warning — instead of running on a schedule whose own budget
   reports failure. */
const diagnostics = { seamViolations: 0, seamChecks: 0, lastSpread: null, pageNeed: 0, formNeed: 0, titleShortfall: 0, haloShortfall: 0, titleShortfallMax: 0, haloShortfallMax: 0, resolvedHold: null, holdEvents: null, pagesProcessed: 0, pagesCoverage: 0, pagesAssigned: 0, pagesPresented: 0, pagePresented: {}, pageAssignments: [], droppedMs: 0, overlapNeed: 0, effectiveScatterMs: 0, frameStats: null, feasibility: null };
/* ---------- feasibility gate (the ONLY feasibility construction) ----------
   Validation produces exactly one of two outcomes for a candidate
   configuration: FEASIBLE — the resolved configuration is applied
   whole and execution starts from it; or REFUSED ATOMICALLY — the
   candidate is rejected as a unit, the previously applied (feasible)
   configuration remains in force untouched, execution does not
   proceed on the candidate, and the refusal is written to the
   observable REJECTED_CONFIGS record with its reasons. There is no
   state in which an infeasible resolved configuration executes.
   The gate decides on a conservative PRE-COMMIT lifecycle bound,
   computed before any execution: per-interval schedule lifecycle
   (travel + stagger + span-derived registration + the reserved
   hold), formation lifecycle (delay + travel + stagger + full
   registration + a detection allowance + the supported tick lag),
   complete page coverage of the executed schedule, positive
   intervals, the intensity hierarchy WITH its minimum separation,
   palette contrast floors, magnitude bounds on every numeric value,
   and non-empty target sets for every required form. A schedule
   fallback result is itself a candidate: it passes through this
   same gate, and a fallback that remains infeasible is a refusal,
   not a proceed. At a cold start there is no prior configuration:
   the shipped defaults are the standing prior, a refused startup
   candidate is recorded, and the defaults are restored whole. */

function assessFeasibility(){
  const reasons = [];
  for (const k of Object.keys(CONFIG)) { const v = CONFIG[k]; if (typeof v === "number" && (!isFinite(v) || Math.abs(v) > SCHEMA.maxAbsValue)) reasons.push(k + " outside the magnitude bound"); }
  const sched = computeSchedule();
  if (!sched.budgetOk) reasons.push("schedule lifecycle exceeds a page interval (pre-commit bound: travel + stagger + registration + reserved hold)");
  const seen = new Set(sched.changes.map(c => c.page));
  for (let pg = 1; pg <= CONFIG.pageCount; pg++) if (!seen.has(pg)) reasons.push("page " + pg + " absent from the executed schedule");
  for (const d of sched.dwellTable) if (!(d.intervalMs > 0)) reasons.push("non-positive interval at page " + d.page);
  const titleNeed = CONFIG.travelFormMs * (1 + CONFIG.staggerFormFraction);
  const haloNeed = CONFIG.haloDelayMs + CONFIG.travelFormMs * CONFIG.haloTravelMultiplier * (1 + CONFIG.staggerFormFraction) + CONFIG.registrationMs + CONFIG.frameCapMs + CONFIG.tickLagMs;
  if (CONFIG.travelFormMs < CONFIG.formationMinTravelMs) reasons.push("formation travel below the formation travel floor");
  if (titleNeed > CONFIG.formMs) reasons.push("formation lifecycle exceeds the form phase");
  if (haloNeed > CONFIG.formMs + CONFIG.holdMs) reasons.push("ring lifecycle exceeds form + hold (pre-commit bound incl. registration, detection and tick lag)");
  const chain = [["surplusAlpha","reserveAlpha"],["reserveAlpha","travelAlpha"],["travelAlpha","stackAlpha"],["stackAlpha","foregroundAlpha"]];
  for (const pair of chain) { const lo = pair[0], hi = pair[1]; if (!(CONFIG[hi] - CONFIG[lo] >= SCHEMA.minAlphaSeparation)) reasons.push("intensity separation below the minimum between " + lo + " and " + hi); }
  /* final-palette contrast, re-evaluated here on the resolved palette:
     the exact pairs sanitisation applies, plus the DOM colours, each
     against the resolved background */
  const lumOfRgb = (rgb) => { const f = (v) => { v /= 255; return v <= 0.03928 ? v / 12.92 : Math.pow((v + 0.055) / 1.055, 2.4); }; return 0.2126 * f(rgb[0]) + 0.7152 * f(rgb[1]) + 0.0722 * f(rgb[2]); };
  const contrastVsBg = (hex) => { if (!validColor(hex) || !validColor(CONFIG.background)) return 0; const la = lumOfRgb(hexRgb(hex)), lb = lumOfRgb(hexRgb(CONFIG.background)); return (Math.max(la, lb) + 0.05) / (Math.min(la, lb) + 0.05); };
  if (contrastVsBg(CONFIG.colorA) < SCHEMA.contrastColorA) reasons.push("final palette contrast below its floor: colorA against the resolved background");
  if (contrastVsBg(CONFIG.colorB) < SCHEMA.contrastColorB) reasons.push("final palette contrast below its floor: colorB against the resolved background");
  if (contrastVsBg(CONFIG.haloColor) < SCHEMA.contrastHalo) reasons.push("final palette contrast below its floor: haloColor against the resolved background");
  for (const dk of ["counterColor", "counterLabelColor", "labelColor"]) if (contrastVsBg(CONFIG[dk]) < SCHEMA.contrastDom) reasons.push("DOM colour contrast below its floor: " + dk + " against the resolved background");
  if (W > 0 && H > 0) {
    if (sampleTitleTargets(W, H).length === 0) reasons.push("empty title target set");
    if (sampleHaloTargets(W, H).length === 0) reasons.push("empty ring target set");
    if (samplePageTargets(W, H, 1).length === 0) reasons.push("empty page target set");
    /* fit and placement are one construction: the placed ink box the
       fit produces must lie inside its region, and a fit condensed
       below the sampling floor is refused, never silently thinned */
    if (lastHeaderFit && lastHeaderFit.box) { const b = lastHeaderFit.box, r = lastHeaderFit.region; if (b.x0 < r.x0 - 0.01 || b.x1 > r.x1 + 0.01 || b.y0 < r.y0 - 0.01 || b.y1 > r.y1 + 0.01) reasons.push("header placed ink box outside the page region"); if (lastHeaderFit.condensed && Math.min(lastHeaderFit.sx, lastHeaderFit.sy) < CONFIG.fitMinScaleRatio) reasons.push("header fit condensed below the sampling floor"); }
    if (lastTitleFit && lastTitleFit.box) { const b = lastTitleFit.box; if (b.x0 < -0.01 || b.x1 > W + 0.01 || b.y0 < -0.01 || b.y1 > H + 0.01) reasons.push("title placed ink box outside the stage"); if (lastTitleFit.condensed && Math.min(lastTitleFit.sx, lastTitleFit.sy) < CONFIG.fitMinScaleRatio) reasons.push("title fit condensed below the sampling floor"); }
  }
  return { ok: reasons.length === 0, reasons: reasons, budgetMs: Math.round(sched.budgetMs), titleNeedMs: Math.round(titleNeed), haloNeedMs: Math.round(haloNeed) };
}

function mulberry32(a){ return function(){ a|=0; a = a + 0x6D2B79F5 | 0; let t = Math.imul(a ^ a>>>15, 1|a); t = t + Math.imul(t ^ t>>>7, 61|t) ^ t; return ((t ^ t>>>14)>>>0)/4294967296; }; }

function sampleDrawn(ctx, w, h, gap){
  const img = ctx.getImageData(0,0,w,h).data, out = [];
  for (let y = 0; y < h; y += gap) for (let x = 0; x < w; x += gap)
    if (img[(y*w + x)*4 + 3] > CONFIG.sampleAlphaThreshold) out.push({x:x, y:y});
  return out;
}
function sampleDrawnRegion(ctx, rx, ry, rw, rh, W, H, gap){
  const pad = CONFIG.regionPaddingPx;
  const x0 = Math.max(0, Math.floor(rx)-pad), y0 = Math.max(0, Math.floor(ry)-pad);
  const x1 = Math.min(W, Math.ceil(rx+rw)+pad), y1 = Math.min(H, Math.ceil(ry+rh)+pad);
  const img = ctx.getImageData(x0,y0,x1-x0,y1-y0).data, out = [];
  for (let y = y0; y < y1; y += gap) for (let x = x0; x < x1; x += gap)
    if (img[((y-y0)*(x1-x0) + (x-x0))*4 + 3] > CONFIG.sampleAlphaThreshold) out.push({x:x, y:y});
  return out;
}
/* Text fitting against a width box AND a height box. Returns the
   fitted font size plus the condensation factors the no-fit policy
   (CONDENSE-AT-FLOOR) applies when the floor size still exceeds a
   box: sx brings the rendered width inside the box, sy the height.
   Termination is guaranteed by validation (shrink in (0, 0.98])
   and by an explicit iteration cap. */
function fitText(ctx, text, startFs, minFs, shrink, boxW, boxH){
  let fs = startFs;
  const setF = () => { ctx.font = CONFIG.fontWeight + " " + fs + "px " + CONFIG.fontFamily; };
  const measure = () => { const m = ctx.measureText(text);
    const inkW = (typeof m.actualBoundingBoxLeft === "number" && typeof m.actualBoundingBoxRight === "number") ? (m.actualBoundingBoxLeft + m.actualBoundingBoxRight) : m.width;
    const hgt = ((m.actualBoundingBoxAscent || fs*CONFIG.fitFallbackAscentRatio) + (m.actualBoundingBoxDescent || fs*CONFIG.fitFallbackDescentRatio));
    return { w: inkW, h: hgt }; };
  setF();
  let d = measure(), guard = 0;
  while ((d.w > boxW || d.h > boxH) && fs > minFs && guard < CONFIG.fitIterationCap){ fs = Math.max(minFs, fs * shrink); setF(); d = measure(); guard++; }
  const sx = d.w > boxW ? boxW / d.w : 1;
  const sy = d.h > boxH ? boxH / d.h : 1;
  return { fs: fs, sx: sx, sy: sy, w: d.w, h: d.h, condensed: (sx < 1 || sy < 1) };
}
function drawFittedText(ctx, text, cx, cy, fit, baseline){
  ctx.save();
  ctx.translate(cx, cy);
  if (fit.sx !== 1 || fit.sy !== 1) ctx.transform(fit.sx, 0, 0, fit.sy, 0, 0);
  ctx.textAlign = "center"; ctx.textBaseline = baseline;
  ctx.fillText(text, 0, 0);
  ctx.restore();
}
/* Page targets are sampled element by element so every target
   carries a stable identity key: outline, stack and header targets
   are identical on every page (the persistent set); line targets
   are keyed by line index, column and row, so a line point that survives
   a page change keeps its key while a point past a shortened line
   end vanishes. Outline and stack strokes are sampled at the
   denser outline gap. */
function samplePageTargets(W, H, page){
  const pw = CONFIG.pageWR*W, ph = CONFIG.pageHR*H, px = CONFIG.pageXR*W, py = CONFIG.pageYR*H;
  const off = makeCanvas(W,H), ctx = off.getContext("2d");
  ctx.clearRect(0,0,W,H); ctx.strokeStyle = "#fff"; ctx.fillStyle = "#fff"; ctx.lineWidth = CONFIG.pageStrokePx;
  const targets = [];
  for (let s = CONFIG.stackCount; s >= 1; s--){
    const ox = px + s*CONFIG.stackOffsetPx, oy = py - s*CONFIG.stackOffsetPx*CONFIG.stackOffsetYFactor;
    ctx.strokeRect(ox, oy, pw, ph);
    const pts = sampleDrawnRegion(ctx, ox, oy, pw, ph, W, H, CONFIG.outlineGap);
    for (const p of pts) targets.push({x:p.x, y:p.y, kind:"stack", key:"stack:" + p.x + ":" + p.y, persistent:true});
    ctx.clearRect(0,0,W,H);
  }
  ctx.strokeRect(px, py, pw, ph);
  const outline = sampleDrawn(ctx, W, H, CONFIG.outlineGap);
  for (const p of outline) targets.push({x:p.x, y:p.y, kind:"page", key:"outline:" + p.x + ":" + p.y, persistent:true});
  ctx.clearRect(0,0,W,H);
  const hFit = fitText(ctx, CONFIG.pageHeader, pw*CONFIG.pageHeaderFontRatio, CONFIG.headerMinFontPx, CONFIG.headerShrink, pw*CONFIG.pageHeaderWidthRatio, ph*CONFIG.pageHeaderHeightRatio);
  drawFittedText(ctx, CONFIG.pageHeader, px + pw/2, py + ph*CONFIG.pageHeaderTopRatio, hFit, "top");
  lastHeaderFit = { fs: hFit.fs, sx: hFit.sx, sy: hFit.sy, w: hFit.w * hFit.sx, h: hFit.h * hFit.sy,
                    boxW: pw*CONFIG.pageHeaderWidthRatio, boxH: ph*CONFIG.pageHeaderHeightRatio, condensed: hFit.condensed,
                    box: { x0: px + pw/2 - (hFit.w * hFit.sx)/2, x1: px + pw/2 + (hFit.w * hFit.sx)/2, y0: py + ph*CONFIG.pageHeaderTopRatio, y1: py + ph*CONFIG.pageHeaderTopRatio + hFit.h * hFit.sy },
                    region: { x0: px, y0: py, x1: px + pw, y1: py + ph } };
  const header = sampleDrawn(ctx, W, H, CONFIG.sampleGap);
  for (const p of header) targets.push({x:p.x, y:p.y, kind:"page", key:"header:" + p.x + ":" + p.y, persistent:true});
  ctx.clearRect(0,0,W,H);
  const rngL = mulberry32(CONFIG.seed + page*7919);
  const innerX = px + pw*CONFIG.pageInnerXRatio, innerW = pw*CONFIG.pageInnerWRatio;
  let ly = py + ph*CONFIG.pageLinesTopRatio;
  CONFIG.pageLinePattern.forEach((frac, li) => {
    const jittered = Math.max(CONFIG.lineMinFraction, Math.min(1, frac + (rngL()-0.5)*CONFIG.pageLineJitter));
    ctx.clearRect(0,0,W,H);
    ctx.fillRect(innerX, ly, innerW*jittered, Math.max(CONFIG.lineMinThicknessPx, ph*CONFIG.pageLineThicknessRatio));
    const pts = sampleDrawn(ctx, W, H, CONFIG.sampleGap);
    for (const p of pts) targets.push({x:p.x, y:p.y, kind:"line", key:"line:" + li + ":" + p.x + ":" + p.y, persistent:false});
    ly += ph*CONFIG.pageLineGapRatio;
  });
  return targets;
}
/* Title targets: the big words, fitted in width AND height against

                   boxW: W*CONFIG.maxWidthRatio, boxH: H*CONFIG.titleMaxHeightRatio, condensed: fit.condensed,
                   cx: W*CONFIG.titleCxRatio, cy: H*CONFIG.titleCyRatio,
                   box: { x0: W*CONFIG.titleCxRatio - (fit.w * fit.sx)/2, x1: W*CONFIG.titleCxRatio + (fit.w * fit.sx)/2, y0: H*CONFIG.titleCyRatio - (fit.h * fit.sy)/2, y1: H*CONFIG.titleCyRatio + (fit.h * fit.sy)/2 } };
  return sampleDrawn(ctx, W, H, CONFIG.sampleGap).map(p => ({x:p.x, y:p.y, kind:"title", key:"title:" + p.x + ":" + p.y, persistent:false}));

/* Ring targets: an ellipse rendered on its own offscreen canvas. */
function sampleHaloTargets(W, H){
  const off = makeCanvas(W,H), ctx = off.getContext("2d");
  ctx.clearRect(0,0,W,H); ctx.strokeStyle = "#fff"; ctx.lineWidth = CONFIG.haloStrokePx;
  ctx.beginPath(); ctx.ellipse(W*CONFIG.haloCxRatio, H*CONFIG.haloCyR, W*CONFIG.haloRxR, H*CONFIG.haloRyR, 0, 0, Math.PI*2); ctx.stroke();

}

/* ---------- particle pool: ONE constant flow field, registered ----------
   Particle state: anchor (ax, ay) + stream offset (ox, oy) +
   reference offset (refx, refy). Rendered position = anchor +
   (offset − reference). The reference is constant while free or
   travelling, eases to the live offset across the registration
   window once settled, and is re-based across the release window
   so the displayed offset re-emerges smoothly. Wrapping adjusts
   the anchor so the rendered position stays continuous apart from
   the wrap jump itself. */
const stage = document.getElementById("stage"), sctx = stage.getContext("2d");
let W = 0, H = 0, particles = [], rng = mulberry32(CONFIG.seed);
function hexRgb(h){ const n = parseInt(h.slice(1), 16); return [n>>16 & 255, n>>8 & 255, n & 255]; }
let cA = hexRgb(DEFAULTS.colorA), cB = hexRgb(DEFAULTS.colorB), cHalo = hexRgb(DEFAULTS.haloColor);
function refreshColors(){ cA = hexRgb(CONFIG.colorA); cB = hexRgb(CONFIG.colorB); cHalo = hexRgb(CONFIG.haloColor); }
function newParticle(borrowable){
  const mix = rng();
  const ax = rng()*W, ay = rng()*H;
  return { ax: ax, ay: ay, ox: 0, oy: 0, refx: 0, refy: 0, dox: 0, doy: 0, x: ax, y: ay,
    ax0: ax, ay0: ay, tx: 0, ty: 0, kind:"", key:"", persistent:false,
    borrowed:false, borrowable:borrowable,
    vx: CONFIG.windX + (rng()-0.5)*CONFIG.driftX, vy: CONFIG.windY + (rng()-0.5)*CONFIG.driftY,
    delay:0, travel:600, t0:0, captured:false, lastCapture:null,
    regT0:-1, regFromX:0, regFromY:0, regDur:300, regDoneAt:-1, px:0, py:0, hasPrev:false,
    relT:-1e9, relDispX:0, relDispY:0, relOx:0, relOy:0,
    scatActive:false, scatFromX:0, scatFromY:0, scatToX:0, scatToY:0,
    relDur:500, relDerivX:0, relDerivY:0, relAnchorVelX:0, relAnchorVelY:0,
    exiting:false, exitT0:0, exitDur:1, exitFromX:0, exitFromY:0, exitRelDur:500,
    velX:0, velY:0, doxVelX:0, doxVelY:0,
    cueT0:-1e9, jitAmp: 0,
    col: [Math.round(cA[0]+(cB[0]-cA[0])*mix), Math.round(cA[1]+(cB[1]-cA[1])*mix), Math.round(cA[2]+(cB[2]-cA[2])*mix)],
    ph: rng()*Math.PI*2, renderAlpha: CONFIG.reserveAlpha };
}
/* The pool is sized once, at build: the largest simultaneous demand
   — the MAXIMUM over every page's target count, or title + ring
   together — grossed up by the reserve fraction. The pool never
   grows afterwards, and assignment completeness is asserted
   against this sizing. */
function buildPool(){
  particles = [];
  let pageNeed = 0;
  let lineMax = 0;
  let edgeNeed = 0;
  for (let pg = 1; pg <= CONFIG.pageCount; pg++) { const ts = samplePageTargets(W, H, pg); pageNeed = Math.max(pageNeed, ts.length); let lc = 0, ec = 0; for (const t of ts) { if (t.kind === "line") lc++; if (t.key.indexOf("outline:") === 0) ec++; } if (lc > lineMax) lineMax = lc; if (ec > edgeNeed) edgeNeed = ec; }
  diagnostics.edgeNeed = edgeNeed;
  const formNeed = sampleTitleTargets(W, H).length + sampleHaloTargets(W, H).length;
  /* concurrent demand is priced, not assumed: if the effective
     release scatter (capped at the release window by the release
     site) could still be running when formation starts, the page
     set overlaps formation demand and the pool sizes for both */
  const effectiveScatterMs = CONFIG.releaseScatterMs;
  const overlapNeed = effectiveScatterMs > CONFIG.releaseMs ? pageNeed : 0;
  diagnostics.overlapNeed = overlapNeed; diagnostics.effectiveScatterMs = effectiveScatterMs;
  /* the outgoing edge set is concurrent demand at every handoff —
     priced here, not borrowed as an afterthought: at a rapid
     handoff the previous edge set is still exiting while the new
     one is borrowed, so two edge sets plus the exiting line set
     stand against the incoming page */
  /* scatter concurrency is priced too: a vanished set released at
     one change is still scattering when the next changes demand
     particles, so one full line set of scatter stands in the base */
  const scatterNeed = lineMax;
  diagnostics.scatterNeed = scatterNeed;
  const needTotal = Math.ceil(Math.max(pageNeed + lineMax + 2 * edgeNeed + scatterNeed, formNeed + overlapNeed) / (1 - CONFIG.reserveFraction));
  for (let i = 0; i < needTotal; i++) particles.push(newParticle(true));
  const reserveNeed = Math.floor(particles.length * CONFIG.reserveFraction);
  for (let i = 0; i < reserveNeed; i++) particles[i].borrowable = false;
  diagnostics.pageNeed = pageNeed; diagnostics.formNeed = formNeed;
}
function startTravel(p, t, now, travelMs, maxDelayMs){
  p.tx = t.x; p.ty = t.y; p.kind = t.kind; p.key = t.key; p.persistent = !!t.persistent;
  p.borrowed = true; p.t0 = now; p.delay = rng()*maxDelayMs; p.travel = travelMs;
  p.captured = false; p.cueT0 = -1e9; p.regT0 = -1; p.scatActive = false; p.exiting = false; p.regDoneAt = -1;
  /* registration duration for THIS borrow: a page borrow registers
     inside its own turn window (the duration is capped so capture
     + travel + registration completes with the turn, leaving the
     reserved tail readable); a form/hold borrow uses the full
     configured registration window */
  if (isFinite(borrowTurnEffMs)) {
    /* page borrow: registration is capped against the turn's
       readable onset (a fixed fraction of the effective turn
       window), so capture + travel + registration complete by the
       onset and the remainder of the window is readable geometry;
       the configured floor governs whenever the span allows it */
    const onsetMs = borrowTurnEffMs * CONFIG.turnOnsetFraction;
    const spanMs = onsetMs - (p.delay + p.travel);
    p.regDur = spanMs >= CONFIG.registrationFloorMs
      ? Math.max(CONFIG.registrationFloorMs, Math.min(CONFIG.registrationMs, spanMs))
      : Math.max(1, spanMs);
  } else if (isFinite(formRegDeadlineMs) && (t.kind === "title" || t.kind === "halo")) {
    /* formation borrow: registration is span-derived per particle
       against the hold entry — delay + travel + registration
       completes by the hold entry for every particle whose travel
       completes before it, so the hold is genuinely resolved */
    const spanMs = formRegDeadlineMs - (now + p.delay + p.travel);
    p.regDur = spanMs >= CONFIG.registrationFloorMs
      ? Math.max(CONFIG.registrationFloorMs, Math.min(CONFIG.registrationMs, spanMs))
      : Math.max(1, spanMs);
  } else {
    p.regDur = Math.max(CONFIG.registrationFloorMs, Math.min(CONFIG.registrationMs, borrowTurnEffMs - (p.delay + p.travel)));
  }
}
function releaseParticle(p, now, durMs){
  p.borrowed = false; p.captured = false; p.cueT0 = -1e9; p.regT0 = -1; p.regDoneAt = -1; p.relT = now;
  p.relDur = durMs || CONFIG.releaseScatterMs;
  /* release state: constructed from the ACTUAL RENDERED state - the
     scatter starts at the rendered position (anchor set so anchor +
     displayed offset equals it exactly, so no position jump), the
     displayed offset re-emerges from its current value, and both
     the anchor and the offset carry a derivative blend that starts
     at the captured rendered derivative and settles to the release
     profile's own slope over the first quarter of the window */
  p.relDispX = p.dox; p.relDispY = p.doy; p.relOx = p.ox; p.relOy = p.oy;
  p.relDerivX = p.doxVelX; p.relDerivY = p.doxVelY;
  p.relAnchorVelX = p.velX - p.doxVelX; p.relAnchorVelY = p.velY - p.doxVelY;
  p.scatActive = true; p.scatFromX = p.x - p.dox; p.scatFromY = p.y - p.doy;
  /* per-release randomness, stated exactly as constructed: TWO
     seeded rng() draws per release — the first displaces the scatter
     target's x (the +240 px coherent bias plus draw x the configured
     range), the second its y (the -200 px coherent bias minus draw
     x the range x the y-factor). The seed source is CONFIG.seed,
     through the run's mulberry32 stream. */
  p.scatToX = p.scatFromX + CONFIG.releaseScatterBiasX + rng() * CONFIG.releaseScatterPx;
  p.scatToY = p.scatFromY + CONFIG.releaseScatterBiasY - rng() * CONFIG.releaseScatterPx * CONFIG.releaseScatterYFactor;
}
function releaseAll(now, durMs){
  /* a phase release with no explicit duration runs on the phase's
     own window: the scatter duration is capped at the release
     window it must complete inside, so no released particle is
     still unavailable when the next phase borrows */
  const dur = durMs != null ? durMs : Math.min(CONFIG.releaseScatterMs, CONFIG.releaseMs);
  for (const p of particles) if (p.borrowed) releaseParticle(p, now, dur);
}
/* Endpoint commitment: a release whose scheduled end has passed is
   evaluated to its endpoint — scatter anchor, displayed offset and
   reference offset all committed — BEFORE any borrow decision may
   select the particle, so travel never begins from a stale sampled
   release position. */
function commitRelease(p){
  if (!p.scatActive) return;
  const se = easeProfile(1);
  const w = Math.min(1, 1 / CONFIG.releaseBlendFraction);
  const hb = w - w*w*w + 0.5*w*w*w*w;
  const corr = CONFIG.releaseBlendFraction * (p.relDur / 1000) * hb;
  p.ax = p.scatFromX + (p.scatToX - p.scatFromX) * se + p.relAnchorVelX * corr;
  p.ay = p.scatFromY + (p.scatToY - p.scatFromY) * se + p.relAnchorVelY * corr;
  p.dox = p.relDispX + (p.ox - p.relOx) * se + p.relDerivX * corr;
  p.doy = p.relDispY + (p.oy - p.relOy) * se + p.relDerivY * corr;
  p.x = p.ax + p.dox; p.y = p.ay + p.doy;
  p.scatActive = false; p.jitAmp = 0; p.refx = p.ox - p.dox; p.refy = p.oy - p.doy;
}
function borrowFresh(targets, now, travelMs, staggerFraction){
  /* a particle whose release is still in flight is never borrowed;
     a release complete at its SCHEDULED end is committed to its
     endpoint first (commitRelease), then becomes borrowable */
  const free = [];
  for (const p of particles){
    if (!p.borrowable || p.borrowed) continue;
    if (p.scatActive && now - p.relT >= p.relDur) commitRelease(p);
    if (!p.scatActive) free.push(p);
  }
  const n = Math.min(free.length, targets.length);
  for (let i = 0; i < n; i++) startTravel(free[i], targets[i], now, travelMs, travelMs*staggerFraction);
  return n;
}
/* Retarget the borrowed set onto a new page's targets; also starts
   the page-turn window (its own duration parameter).
   Returns the transition record, including any target that could
   not be assigned (unassigned must be zero under the pool sizing). */
const transitions = [];
let turnT0 = -1e9, turnEffCur = CONFIG.turnMs, lastTurnFactor = 1, borrowTurnEffMs = Infinity;
let turnFactorOut = 1, turnFactorIn = 1, turnFactorExit = 1;
let formRegDeadlineMs = Infinity; // absolute time by which formation registration completes (the hold entry), set by the scheduler
function retargetPage(targets, now, page, turnStartMs){
  /* EVERY page change starts its own turn: the turn window opens
     at the change's SCHEDULED instant (a stepped simulation may
     detect the change up to one step late; the turn belongs to
     the schedule, not to the detection) and its effective
     duration is derived from this change's own page interval */
  turnT0 = (turnStartMs != null ? turnStartMs : now);
  const anchor = turnT0; // travel and release start at the scheduled instant, not the detection tick
  const sched = computeSchedule();
  const entry = sched.dwellTable.find(d => d.page === page);
  turnEffCur = entry ? entry.turnEffMs : turnEffForInterval(CONFIG.flipMs, CONFIG);
  borrowTurnEffMs = turnEffCur;
  const current = particles.filter(p => p.borrowed);
  const byKey = new Map();
  for (const p of current) if (!byKey.has(p.key)) byKey.set(p.key, p);
  const used = new Set(), unmatched = [];
  let persisted = 0, reassigned = 0, fresh = 0, released = 0, unassigned = 0;
  for (const t of targets){
    const p = byKey.get(t.key);
    if (p && !used.has(p)){
      used.add(p); persisted++;
      if (p.tx === t.x && p.ty === t.y){ p.cueT0 = anchor; }
      else startTravel(p, t, anchor, CONFIG.travelFlipMs, CONFIG.travelFlipMs*CONFIG.staggerFlipFraction);
    } else unmatched.push(t);
  }
  const spare = current.filter(p => !used.has(p));
  for (const t of unmatched){
    let best = null, bestD = Infinity;
    for (const p of spare){
      if (used.has(p) || p.persistent) continue;
      const d = (p.tx - t.x)*(p.tx - t.x) + (p.ty - t.y)*(p.ty - t.y);
      if (d < bestD){ bestD = d; best = p; }
    }
    if (best){ used.add(best); reassigned++; startTravel(best, t, anchor, CONFIG.travelFlipMs, CONFIG.travelFlipMs*CONFIG.staggerFlipFraction); }
    else {
      let freeP = null;
      for (const cand of particles){ if (!cand.borrowable || cand.borrowed || used.has(cand)) continue; if (cand.scatActive && anchor - cand.relT >= cand.relDur) commitRelease(cand); if (!cand.scatActive){ freeP = cand; break; } }
      if (freeP){ used.add(freeP); fresh++; startTravel(freeP, t, anchor, CONFIG.travelFlipMs, CONFIG.travelFlipMs*CONFIG.staggerFlipFraction); }
      else unassigned++;
    }
  }
  /* vanished targets release on the interval of the page they
     leave: a mid-flip release disperses inside the flip tempo, so
     the in-flight release set stays within the pool margin */
  const relDur = entry ? entry.intervalMs : CONFIG.releaseScatterMs;
  let exitingN = 0;
  for (const p of current) if (!used.has(p)){
    if (!p.persistent) { /* the leaving page's own texture is the outgoing face: it stays borrowed on the fold path for the turn window, then releases */
      if (p.exiting && anchor - p.exitT0 >= p.exitDur) { /* its exit window has completed by this scheduled instant: release it, never re-exit it */
        p.exiting = false; releaseParticle(p, anchor, p.exitRelDur); released++; }
      else if (p.exiting) { exitingN++; released++; } /* already on its way out: its exit stands, its clock is not reset */
      else { p.exiting = true; p.exitT0 = anchor; p.exitDur = Math.max(1, turnEffCur); p.exitFromX = p.tx; p.exitFromY = p.ty; p.exitRelDur = relDur; exitingN++; released++; } }
    else { releaseParticle(p, anchor, relDur); released++; }
  }
  /* outgoing edge set: the leaving page's outline is borrowed as a
     dedicated exiting set, so through the handoff the outgoing face
     carries a coherent edge (outline + its line texture) while the
     incoming face carries its own outline + texture — neither face
     is texture-only or outline-only at any point in the turn */
  let edgeAssigned = 0, edgeUnassigned = 0;
  const edgeTargets = targets.filter(tg => tg.persistent && tg.key.indexOf("outline:") === 0);
  for (const tg of edgeTargets){
    let fp = null;
    for (const cand of particles){ if (!cand.borrowable || cand.borrowed || used.has(cand)) continue; if (cand.scatActive && anchor - cand.relT >= cand.relDur) commitRelease(cand); if (!cand.scatActive){ fp = cand; break; } }
    if (fp){
      used.add(fp);
      fp.borrowed = true; fp.kind = "edge"; fp.key = "edge:" + tg.key; fp.persistent = false;
      fp.tx = tg.x; fp.ty = tg.y; fp.captured = true; fp.ax0 = tg.x; fp.ay0 = tg.y; fp.ax = tg.x; fp.ay = tg.y;
      fp.t0 = anchor; fp.delay = 0; fp.travel = 1; fp.regT0 = anchor; fp.regDur = 1; fp.regDoneAt = anchor + 1;
      fp.exiting = true; fp.exitT0 = anchor; fp.exitDur = Math.max(1, turnEffCur); fp.exitFromX = tg.x; fp.exitFromY = tg.y; fp.exitRelDur = relDur;
      fp.renderAlpha = CONFIG.foregroundAlpha;
      edgeAssigned++;
    } else edgeUnassigned++;
  }
  if (edgeUnassigned > (diagnostics.edgeUnassignedMax || 0)) diagnostics.edgeUnassignedMax = edgeUnassigned;
  const rec = { page: page, atMs: Math.round(now), targets: targets.length,
    persisted: persisted, reassigned: reassigned, fresh: fresh, released: released, exiting: exitingN,
    edgeAssigned: edgeAssigned, edgeUnassigned: edgeUnassigned,
    unassigned: unassigned, churn: reassigned + released };
  transitions.push(rec);
  return rec;
}

/* ---------- scheduler: sequential phases, one lifecycle at a time ----------
   Cycle: flip [0, flipMs) · release · form · hold · seam. The seam is
   the final seamReleaseMs of the cycle, immediately after the
   hold window; the title and ring release at the seam's start and no
   page target is borrowed until the next cycle begins. */
const TOTAL = () => CONFIG.flipMs + CONFIG.releaseMs + CONFIG.formMs + CONFIG.holdMs + CONFIG.seamReleaseMs;
let simT = 0, phase = "", curPage = 0, trace = [], titleBorrowed = false, haloBorrowed = false;
let formTargets = null, titleAssigned = 0, haloAssigned = 0;
const counterEl = document.getElementById("counter"), counterNum = counterEl.querySelector(".num"),
      counterLab = counterEl.querySelector(".lab"), labelEl = document.getElementById("tlabel");
let lastCounterFit = null, counterMeasureCtx = null;
function fitCounter(){
  if (!counterMeasureCtx) counterMeasureCtx = makeCanvas(8, 8).getContext("2d");
  const ctx = counterMeasureCtx;
  const worst = CONFIG.pageCount + "/" + CONFIG.pageCount;
  const widthAt = (text, fs) => { ctx.font = CONFIG.fontWeight + " " + fs + "px " + CONFIG.fontFamily; return ctx.measureText(text).width; };
  const labelW = () => widthAt(CONFIG.counterLabelText, CONFIG.counterLabelPx) + Math.max(0, CONFIG.counterLabelText.length - 1) * CONFIG.counterLabelPx * CONFIG.counterLabelLetterSpacingEm;
  const groupW = (fs) => Math.max(labelW(), widthAt(worst, fs));
  const groupH = (fs) => CONFIG.counterLabelPx * 1.2 + fs * 1.05 + 4;
  const x1 = (1 - CONFIG.counterRightRatio) * W;
  const regionRight = (CONFIG.pageXR + CONFIG.pageWR) * W + CONFIG.stackCount * CONFIG.stackOffsetPx;
  const availW = x1 - regionRight - CONFIG.counterRegionMarginPx;
  let fs = Math.floor(CONFIG.counterNumPx);
  while (fs > CONFIG.counterNumMinPx && groupW(fs) > availW) fs--;
  if (groupW(fs) <= availW) {
    const y0 = CONFIG.counterTopRatio * H;
    lastCounterFit = { numPx: fs, mode: "inline", worst: worst, availW: availW,
      box: { x0: x1 - groupW(fs), x1: x1, y0: y0, y1: y0 + groupH(fs) }, groupW: groupW(fs) };
  } else {
    fs = Math.floor(CONFIG.counterNumMinPx);
    const y0 = CONFIG.counterFallbackTopRatio * H;
    const stackTop = CONFIG.pageYR * H - CONFIG.stackCount * CONFIG.stackOffsetPx * CONFIG.stackOffsetYFactor;
    lastCounterFit = { numPx: fs, mode: "above", worst: worst, availW: availW, clearancePx: Math.round((stackTop - (y0 + groupH(fs))) * 100) / 100,
      box: { x0: x1 - groupW(fs), x1: x1, y0: y0, y1: y0 + groupH(fs) }, groupW: groupW(fs) };
  }
  return lastCounterFit;
}

  counterEl.style.top = ((fit.mode === "above" ? CONFIG.counterFallbackTopRatio : CONFIG.counterTopRatio) * 100) + "%";
}

function setPhase(name, t){ phase = name; trace.push({event:name, t:Math.round(t)}); }
function spreadOfBorrowed(){
  let x0=Infinity, y0=Infinity, x1=-Infinity, y1=-Infinity, n=0;
  for (const p of particles) if (p.borrowed){ n++; if (p.x<x0)x0=p.x; if (p.x>x1)x1=p.x; if (p.y<y0)y0=p.y; if (p.y>y1)y1=p.y; }
  return n ? {w: x1-x0, h: y1-y0} : null;
}
let lastStepAbs = 0;
function segStarts(){ const F = CONFIG.flipMs, R = F + CONFIG.releaseMs, Fo = R + CONFIG.formMs, Ho = Fo + CONFIG.holdMs; return [0, F, R, Fo, Ho]; }
function segNameAt(tt){ const ss = segStarts(); const names = ["flip","release","form","hold","seam"]; for (let i = ss.length - 1; i >= 0; i--) if (tt >= ss[i]) return names[i]; return "flip"; }
function runSegmentEntry(name, segStartAbs, tt){
  const F = CONFIG.flipMs, R = F + CONFIG.releaseMs, Fo = R + CONFIG.formMs, Ho = Fo + CONFIG.holdMs;
  if (name === "flip"){ setPhase("flip", segStartAbs); diagnostics.seamChecks++; if (particles.some(p => p.borrowed && (p.kind === "title" || p.kind === "halo"))) diagnostics.seamViolations++; titleBorrowed = false; haloBorrowed = false; formTargets = null; curPage = 0; titleAssigned = 0; haloAssigned = 0; diagnostics.pagesCoverage = 0; diagnostics.pagesAssigned = 0; diagnostics.pageAssignments = []; diagnostics.pagePresented = {}; diagnostics.pagesPresented = 0; }
  else if (name === "release"){ setPhase("release", segStartAbs); diagnostics.lastSpread = spreadOfBorrowed(); releaseAll(segStartAbs); }
  else if (name === "form"){ setPhase("form", segStartAbs); formTargets = { title: sampleTitleTargets(W, H), halo: sampleHaloTargets(W, H) }; formRegDeadlineMs = segStartAbs + (Fo - R); }
  else if (name === "hold"){ setPhase("hold", segStartAbs); }
  else if (name === "seam"){ setPhase("seam", segStartAbs); diagnostics.lastSpread = spreadOfBorrowed(); releaseAll(segStartAbs, CONFIG.seamReleaseMs); titleBorrowed = false; haloBorrowed = false; titleAssigned = 0; haloAssigned = 0; }
}
function step(dtMs){
  const prevAbs = simT;
  simT += dtMs; const t = simT % TOTAL();
  /* crossed-boundary dispatch: every segment entered since the last
     step — including any segment this step crosses entirely — runs
     its entry in chronological order before the landing body */
  {
    const ss = segStarts(), names = ["flip","release","form","hold","seam"], TOT = TOTAL();
    /* one chronological event order across segment entries and page
       changes (ties: segment entry before page change before the
       particle updates of the landing body) */
    const events = [];
    const base0 = Math.floor(prevAbs / TOT) * TOT;
    for (let cyc = 0; cyc <= 1; cyc++) for (let i = 0; i < ss.length; i++) {
      const at = base0 + cyc * TOT + ss[i];
      if (at > prevAbs && at <= simT) events.push({ type: "seg", name: names[i], at: at });
    }
    const finalCycStart = Math.floor(simT / TOT) * TOT;
    const chsAll = computeSchedule().changes;
    for (let k = Math.floor(prevAbs / TOT); k <= Math.floor(simT / TOT); k++){
      const ws = k * TOT;
      if (ws === finalCycStart && t < CONFIG.flipMs) continue; // the landing flip window dispatches in its own body below
      for (const ch of chsAll){
        const at = ws + ch.atMs;
        if ((at > prevAbs && at <= simT) || (at === prevAbs && curPage === 0 && at === ws)){
          events.push({ type: "page", page: ch.page, atMs: ch.atMs, at: at, ws: ws });
        }
      }
    }
    events.sort((a, b) => (a.at - b.at) || ((a.type === "seg" ? 0 : 1) - (b.type === "seg" ? 0 : 1)));
    const landing = segNameAt(t);
    for (const e of events){
      if (e.type === "seg"){ if (phase !== e.name && e.name !== landing) runSegmentEntry(e.name, e.at, e.at % TOT); }
      else {
        const targets = samplePageTargets(W, H, e.page);
        if (curPage === 0) { turnT0 = -1e9; turnEffCur = CONFIG.turnMs; borrowTurnEffMs = CONFIG.turnMs; borrowFresh(targets, e.ws, CONFIG.travelFlipMs, CONFIG.staggerFlipFraction); }
        else retargetPage(targets, e.at, e.page, e.at);
        diagnostics.pagesAssigned++;
        diagnostics.pageAssignments.push({ page: e.page, atMs: e.atMs });
        let cov = 0; for (const c of chsAll) if (c.atMs <= e.atMs) cov++;
        if (cov > diagnostics.pagesCoverage) diagnostics.pagesCoverage = cov;
        diagnostics.pagesProcessed = diagnostics.pagesAssigned;
        curPage = e.page;
      }
    }
  }
  const F = CONFIG.flipMs, R = F + CONFIG.releaseMs, Fo = R + CONFIG.formMs, Ho = Fo + CONFIG.holdMs;
  if (t < F){
    if (phase !== "flip"){
      setPhase("flip", simT);
      diagnostics.seamChecks++;
      if (particles.some(p => p.borrowed && (p.kind === "title" || p.kind === "halo"))) diagnostics.seamViolations++;
      titleBorrowed = false; haloBorrowed = false; formTargets = null; curPage = 0; titleAssigned = 0; haloAssigned = 0; diagnostics.pagesCoverage = 0; diagnostics.pagesAssigned = 0; diagnostics.pageAssignments = []; diagnostics.pagePresented = {}; diagnostics.pagesPresented = 0;
    }
    const pg = pageAtTime(t);
    /* page-level dispatch: every page change crossed by this step is
       dispatched in chronological order at its scheduled instant —
       coverage (entries crossed), assignments (pages actually
       retargeted) and presented (pages drawn in the registered
       state) are separate counters and are never conflated */
    {
      const TOTd = TOTAL();
      const sameCycle = Math.floor(prevAbs / TOTd) === Math.floor(simT / TOTd);
      const fromT = sameCycle ? (prevAbs % TOTd) : -1;
      const chs = computeSchedule().changes;
      let cov = 0; for (const c of chs) if (c.atMs <= t) cov++;
      if (cov > diagnostics.pagesCoverage) diagnostics.pagesCoverage = cov;
      const crossed = chs.filter(c => c.atMs > fromT && c.atMs <= t);
      if (curPage === 0 && chs.length && chs[0].atMs <= t && !crossed.includes(chs[0])) crossed.unshift(chs[0]);
      for (const ch of crossed){
        const targets = samplePageTargets(W, H, ch.page);
        if (curPage === 0) { turnT0 = -1e9; turnEffCur = CONFIG.turnMs; borrowTurnEffMs = CONFIG.turnMs; borrowFresh(targets, simT - t, CONFIG.travelFlipMs, CONFIG.staggerFlipFraction); }
        else retargetPage(targets, simT, ch.page, simT - (t - ch.atMs));
        diagnostics.pagesAssigned++;
        diagnostics.pageAssignments.push({ page: ch.page, atMs: ch.atMs });
        curPage = ch.page;
      }
      if (pg !== curPage){
        const targets = samplePageTargets(W, H, pg);
        if (curPage === 0) { turnT0 = -1e9; turnEffCur = CONFIG.turnMs; borrowTurnEffMs = CONFIG.turnMs; borrowFresh(targets, simT - t, CONFIG.travelFlipMs, CONFIG.staggerFlipFraction); }
        else { const chEntry = computeSchedule().changes.find(c => c.page === pg); retargetPage(targets, simT, pg, chEntry ? simT - (t - chEntry.atMs) : simT); }
        diagnostics.pagesAssigned++;
        diagnostics.pageAssignments.push({ page: pg, atMs: Math.round(t) });
        curPage = pg;
      }
      diagnostics.pagesProcessed = diagnostics.pagesAssigned;
    }
    counterEl.style.opacity = 1; counterNum.textContent = curPage + "/" + CONFIG.pageCount; labelEl.style.opacity = 0;
  } else if (t < R){
    if (phase !== "release"){ setPhase("release", simT); diagnostics.lastSpread = spreadOfBorrowed(); releaseAll(simT - (t - F)); }
    counterEl.style.opacity = Math.max(0, 1 - (t-F)/CONFIG.releaseMs); labelEl.style.opacity = 0;
  } else if (t < Fo){
    if (phase !== "form"){
      setPhase("form", simT);
      formTargets = { title: sampleTitleTargets(W, H), halo: sampleHaloTargets(W, H) };
      formRegDeadlineMs = simT + (Fo - t); // the hold entry, in absolute time
    }
    counterEl.style.opacity = 0; labelEl.style.opacity = Math.min(1, (t-R)/CONFIG.labelFadeMs);
    /* formation flags are assignment-truthful: a flag is set only
       when its borrow assigned EVERY target; a short borrow records
       its shortfall and retries on later ticks while its phase
       window remains */
    if (!titleBorrowed){ borrowTurnEffMs = Infinity; titleAssigned += borrowFresh(formTargets.title.slice(titleAssigned), simT - (t - R), CONFIG.travelFormMs, CONFIG.staggerFormFraction); if (titleAssigned >= formTargets.title.length) { titleBorrowed = true; diagnostics.titleShortfall = 0; } else { diagnostics.titleShortfall = formTargets.title.length - titleAssigned; if (diagnostics.titleShortfall > diagnostics.titleShortfallMax) diagnostics.titleShortfallMax = diagnostics.titleShortfall; } }
    if (!haloBorrowed && t - R >= CONFIG.haloDelayMs){ borrowTurnEffMs = Infinity; haloAssigned += borrowFresh(formTargets.halo.slice(haloAssigned), simT - (t - R) + CONFIG.haloDelayMs, CONFIG.travelFormMs*CONFIG.haloTravelMultiplier, CONFIG.staggerFormFraction); if (haloAssigned >= formTargets.halo.length) { haloBorrowed = true; diagnostics.haloShortfall = 0; } else { diagnostics.haloShortfall = formTargets.halo.length - haloAssigned; if (diagnostics.haloShortfall > diagnostics.haloShortfallMax) diagnostics.haloShortfallMax = diagnostics.haloShortfall; } }
  } else if (t < Ho){
    if (phase !== "hold"){
      setPhase("hold", simT);
      /* resolved-hold accounting: the hold is measured from actual
         registration completion (per particle: start + delay +
         travel + registration), separately for title and ring; the
         trace distinguishes nominal hold entry from resolved hold */
      /* completion accounting: the resolved hold derives from the
         MEASURED registration-completion instant of every assigned
         particle (recorded by the runtime when its registration
         completed), never from scheduled instants. A set with any
         particle not yet complete at hold entry reports null for
         its completion — the hold is not resolved until it is. */
      const cycleStart = simT - t;
      let tBy = 0, hBy = 0, tAll = true, hAll = true, tCount = 0, hCount = 0;
      for (const p of particles) if (p.borrowed) {
        if (p.kind === "title") { tCount++; if (p.regDoneAt >= 0) tBy = Math.max(tBy, p.regDoneAt - cycleStart); else tAll = false; }
        if (p.kind === "halo") { hCount++; if (p.regDoneAt >= 0) hBy = Math.max(hBy, p.regDoneAt - cycleStart); else hAll = false; }
      }
      /* completion is reportable only when the assigned set equals the
         required set and every assigned particle has registered; an
         empty or partial assignment reports UNRESOLVED, never a hold */
      const tReq = formTargets ? formTargets.title.length : 0, hReq = formTargets ? formTargets.halo.length : 0;
      if (tCount !== tReq) tAll = false;
      if (hCount !== hReq) hAll = false;
      const jointBy = (tAll && hAll) ? Math.max(tBy, hBy) : null;
      diagnostics._holdCycleStart = cycleStart; diagnostics._holdHo = Ho; diagnostics._holdFo = Fo;
      diagnostics.resolvedHold = { basis: "measured", holdState: (tAll && hAll) ? "RESOLVED" : ("UNRESOLVED — " + ((tCount === 0 && hCount === 0) ? "empty assignment" : "partial or unregistered assignment")), titleAssignedCount: tCount, titleRequiredCount: tReq, ringAssignedCount: hCount, ringRequiredCount: hReq, titleResolvedByMs: tAll ? Math.round(tBy) : null, ringResolvedByMs: hAll ? Math.round(hBy) : null,
        jointResolvedByMs: jointBy === null ? null : Math.round(jointBy),
        titleMs: Math.round(Math.max(0, Ho - Math.max(Fo, tAll ? tBy : Ho))), ringMs: Math.round(Math.max(0, Ho - Math.max(Fo, hAll ? hBy : Ho))),
        observedHoldMs: jointBy === null ? null : Math.round(Math.max(0, Ho - Math.max(Fo, jointBy))) };
      trace.push({event:"resolved-hold", t:Math.round(simT), titleMs: diagnostics.resolvedHold.titleMs, ringMs: diagnostics.resolvedHold.ringMs});
    }
    if (!titleBorrowed && formTargets){ borrowTurnEffMs = Infinity; titleAssigned += borrowFresh(formTargets.title.slice(titleAssigned), simT - (t - R), CONFIG.travelFormMs, CONFIG.staggerFormFraction); if (titleAssigned >= formTargets.title.length) { titleBorrowed = true; diagnostics.titleShortfall = 0; } else { diagnostics.titleShortfall = formTargets.title.length - titleAssigned; if (diagnostics.titleShortfall > diagnostics.titleShortfallMax) diagnostics.titleShortfallMax = diagnostics.titleShortfall; } }
    if (!haloBorrowed && formTargets && t - R >= CONFIG.haloDelayMs){ borrowTurnEffMs = Infinity; haloAssigned += borrowFresh(formTargets.halo.slice(haloAssigned), simT - (t - R) + CONFIG.haloDelayMs, CONFIG.travelFormMs*CONFIG.haloTravelMultiplier, CONFIG.staggerFormFraction); if (haloAssigned >= formTargets.halo.length) { haloBorrowed = true; diagnostics.haloShortfall = 0; } else { diagnostics.haloShortfall = formTargets.halo.length - haloAssigned; if (diagnostics.haloShortfall > diagnostics.haloShortfallMax) diagnostics.haloShortfallMax = diagnostics.haloShortfall; } }
    if (diagnostics.resolvedHold && (diagnostics.resolvedHold.titleResolvedByMs === null || diagnostics.resolvedHold.ringResolvedByMs === null)) {
      /* finalize the entry snapshot from measured completions */
      const cs = diagnostics._holdCycleStart, Ho2 = diagnostics._holdHo, Fo2 = diagnostics._holdFo;
      let tBy2 = 0, hBy2 = 0, tAll2 = true, hAll2 = true, tCount2 = 0, hCount2 = 0;
      for (const p of particles) if (p.borrowed) {
        if (p.kind === "title") { tCount2++; if (p.regDoneAt >= 0) tBy2 = Math.max(tBy2, p.regDoneAt - cs); else tAll2 = false; }
        if (p.kind === "halo") { hCount2++; if (p.regDoneAt >= 0) hBy2 = Math.max(hBy2, p.regDoneAt - cs); else hAll2 = false; }
      }
      const tReq2 = formTargets ? formTargets.title.length : 0, hReq2 = formTargets ? formTargets.halo.length : 0;
      if (tCount2 !== tReq2) tAll2 = false;
      if (hCount2 !== hReq2) hAll2 = false;
      const joint2 = (tAll2 && hAll2) ? Math.max(tBy2, hBy2) : null;
      diagnostics.resolvedHold = { basis: "measured", holdState: (tAll2 && hAll2) ? "RESOLVED" : ("UNRESOLVED — " + ((tCount2 === 0 && hCount2 === 0) ? "empty assignment" : "partial or unregistered assignment")), titleAssignedCount: tCount2, titleRequiredCount: tReq2, ringAssignedCount: hCount2, ringRequiredCount: hReq2, titleResolvedByMs: tAll2 ? Math.round(tBy2) : null, ringResolvedByMs: hAll2 ? Math.round(hBy2) : null,
        jointResolvedByMs: joint2 === null ? null : Math.round(joint2),
        titleMs: Math.round(Math.max(0, Ho2 - Math.max(Fo2, tAll2 ? tBy2 : Ho2))), ringMs: Math.round(Math.max(0, Ho2 - Math.max(Fo2, hAll2 ? hBy2 : Ho2))),
        observedHoldMs: joint2 === null ? null : Math.round(Math.max(0, Ho2 - Math.max(Fo2, joint2))) };
      if (joint2 !== null) trace.push({event:"resolved-hold-final", t:Math.round(simT), observedHoldMs: diagnostics.resolvedHold.observedHoldMs});
    }
    labelEl.style.opacity = 1;
  } else {
    if (phase !== "seam"){ setPhase("seam", simT); diagnostics.lastSpread = spreadOfBorrowed(); releaseAll(simT - (t - Ho), CONFIG.seamReleaseMs); titleBorrowed = false; haloBorrowed = false; titleAssigned = 0; haloAssigned = 0; }
    counterEl.style.opacity = 0; labelEl.style.opacity = Math.max(0, 1 - (t-Ho)/CONFIG.seamReleaseMs);
  }
  updateParticles(dtMs);
}
/* Clearance measurement for the most recently released set: the
   fraction of released particles inside the fitted title box, and
   the released silhouette's bounding-box overlap with its own
   bounding box at the release instant. The metric prices geometric
   clearance only; whether the silhouette still READS as a page is
   a perceptual claim that stays unverified without rendering. */
let lastReleaseSnapshot = null;
function snapshotRelease(){
  const set = [];
  let x0=Infinity, y0=Infinity, x1=-Infinity, y1=-Infinity;
  for (let i = 0; i < particles.length; i++){ const p = particles[i];
    if (p.relT > 0 && simT - p.relT <= 1) { set.push(i); if (p.x<x0)x0=p.x; if (p.x>x1)x1=p.x; if (p.y<y0)y0=p.y; if (p.y>y1)y1=p.y; } }
  lastReleaseSnapshot = set.length ? { set: set, bbox: {x0:x0, y0:y0, x1:x1, y1:y1} } : null;
  return lastReleaseSnapshot;
}
function releaseClearance(){
  if (!lastReleaseSnapshot || !lastTitleFit) return null;
  const snap = lastReleaseSnapshot;
  const tb = { x0: lastTitleFit.cx - lastTitleFit.w/2, x1: lastTitleFit.cx + lastTitleFit.w/2,
               y0: lastTitleFit.cy - lastTitleFit.h/2, y1: lastTitleFit.cy + lastTitleFit.h/2 };
  let inside = 0, x0=Infinity, y0=Infinity, x1=-Infinity, y1=-Infinity;
  for (const i of snap.set){ const p = particles[i];
    if (p.x >= tb.x0 && p.x <= tb.x1 && p.y >= tb.y0 && p.y <= tb.y1) inside++;
    if (p.x<x0)x0=p.x; if (p.x>x1)x1=p.x; if (p.y<y0)y0=p.y; if (p.y>y1)y1=p.y; }
  const iw = Math.max(0, Math.min(x1, snap.bbox.x1) - Math.max(x0, snap.bbox.x0));
  const ih = Math.max(0, Math.min(y1, snap.bbox.y1) - Math.max(y0, snap.bbox.y0));
  const area0 = Math.max(1, (snap.bbox.x1-snap.bbox.x0) * (snap.bbox.y1-snap.bbox.y0));
  return { titleFraction: inside / snap.set.length, bboxOverlap: (iw*ih) / area0, n: snap.set.length };
}
function isSettled(p, now){
  if (!p.borrowed) return false;
  if (!p.captured) return false;
  return (now - p.t0 - p.delay) >= p.travel;
}
function isReadable(p, now){
  if (!isSettled(p, now)) return false;
  if (turnT0 > -1e8 && now - turnT0 < turnEffCur) return false;
  if (p.regT0 < 0) return false;
  return (now - p.regT0) >= p.regDur;
}
/* Geometric readability: settled and registered, with the turn's
   face factor for this particle's face at near-full width (the
   incoming face has unfolded, the persistent face has returned).
   This predicate prices geometry only; the strict predicate above
   additionally requires the turn window to be complete. */
function isReadableGeom(p, now){
  if (!isSettled(p, now)) return false;
  if (p.regT0 < 0) return false;
  if ((now - p.regT0) < p.regDur) return false;
  if (turnT0 > -1e8 && now - turnT0 < turnEffCur) {
    const f = (p.kind === "line") ? turnFactorIn : turnFactorOut;
    if (f < CONFIG.readableGeomFactor) return false;
  }
  return true;
}
/* Zero-end-derivative travel profile: s(f) with s(0)=0, s(1)=1 and
   s'(0)=s'(1)=0, so anchor velocity starts and ends at zero and the
   trajectory velocity (anchor + stream) is continuous at capture
   and settlement. */
function easeProfile(f){
  const e = CONFIG.easingExponent;
  const a = Math.pow(f, e), b = Math.pow(1-f, e);
  return a / (a + b || 1);
}
function updateParticles(dtMs){
  const dt = dtMs/1000, now = simT;
  const pivotX = CONFIG.pageXR*W;
  const turnActive = (now - turnT0 >= 0 && now - turnT0 < turnEffCur);
  const turnU = turnActive ? (now - turnT0) / turnEffCur : 0;
  lastTurnFactor = 1;
  for (const p of particles){
    const prevX = p.x, prevY = p.y, prevDox = p.dox, prevDoy = p.doy;
    /* the stream offset accumulates in EVERY state, every frame */
    p.ox += p.vx*dt; p.oy += p.vy*dt;
    if (p.borrowed){
      if (p.exiting){
        /* outgoing face: fold toward the pivot and past it on the
           fold profile (deepen over the ramp fraction, plateau,
           return by the readable onset), then release */
        const eu = Math.min(1, Math.max(0, (now - p.exitT0) / p.exitDur));
        const rampF2 = CONFIG.turnRampFraction;
        let efp;
        if (eu < rampF2) { const q = eu / rampF2; efp = q * q * (3 - 2 * q); }
        else if (eu < rampF2 + CONFIG.foldPlateauFraction) efp = 1;
        else if (eu < CONFIG.turnOnsetFraction) { const q = (eu - rampF2 - CONFIG.foldPlateauFraction) / (CONFIG.turnOnsetFraction - rampF2 - CONFIG.foldPlateauFraction); efp = 1 - q * q * (3 - 2 * q); }
        else efp = 0;
        const efactor = 1 - (1 - CONFIG.foldMinFactor) * efp;
        turnFactorExit = efactor;
        p.ax = pivotX + (p.exitFromX - pivotX) * efactor; p.ay = p.exitFromY;
        p.dox = p.ox - p.refx; p.doy = p.oy - p.refy;
        p.renderAlpha = CONFIG.travelAlpha;
        if (eu >= 1) { p.exiting = false; releaseParticle(p, now, p.exitRelDur); }
      }
      else if (!p.captured && now - p.t0 >= p.delay){
        /* capture: the travel anchor origin is taken from the
           rendered position actually occupied now, and travel
           progresses in THIS SAME update (no capture/update lag) */
        /* capture origin: the position occupied at THIS instant,
           including this tick's stream advance (anchor + the newly
           advanced offset), so the tick's displacement is carried
           into the travel curve instead of being cancelled by it */
        const capX = p.ax + (p.ox - p.refx), capY = p.ay + (p.oy - p.refy);
        p.dox = p.ox - p.refx; p.doy = p.oy - p.refy;
        p.ax0 = capX - p.dox; p.ay0 = capY - p.doy; p.captured = true;
        p.lastCapture = { t: now, x: capX, y: capY, vx: p.velX, vy: p.velY };
      }
      if (p.exiting){
        /* position set by the exit branch above */
      } else if (!p.captured){
        /* stagger window: the anchor holds; the displayed offset
           keeps accumulating the stream contribution */
        p.dox = p.ox - p.refx; p.doy = p.oy - p.refy;
        p.renderAlpha = CONFIG.travelAlpha;
      } else {
        const el = now - p.t0 - p.delay;
        const f = Math.min(1, Math.max(0, el / p.travel));
        const s = easeProfile(f);
        p.jitAmp = s;
        let ax = p.ax0 + (p.tx - p.ax0)*s + Math.sin(now/CONFIG.jitterPeriodXMs + p.ph)*CONFIG.jitter*p.jitAmp;
        let ay = p.ay0 + (p.ty - p.ay0)*s + Math.cos(now/CONFIG.jitterPeriodYMs + p.ph)*CONFIG.jitter*p.jitAmp;
        /* registration: once settled, the reference eases to the
           live stream offset, so the displayed offset decays to
           zero and the structure registers on its target */
        if (f >= 1) {
          /* registration starts at the arrival event instant (borrow
             start + delay + travel), not at the detecting tick; its
             completion is likewise recorded at its event instant */
          if (p.regT0 < 0) { p.regT0 = p.t0 + p.delay + p.travel; p.regFromX = p.refx; p.regFromY = p.refy; }
          const u = Math.min(1, Math.max(0, (now - p.regT0) / p.regDur));
          if (u >= 1 && p.regDoneAt < 0) p.regDoneAt = p.regT0 + p.regDur;
          if (u >= 1) { p.refx = p.ox; p.refy = p.oy; }
          else { const se = easeProfile(u); p.refx = p.regFromX + (p.ox - p.regFromX)*se; p.refy = p.regFromY + (p.oy - p.regFromY)*se; }
        }
        p.dox = p.ox - p.refx; p.doy = p.oy - p.refy;
        /* page turn: the turn is a two-face gesture inside its own
           window. The persistent face (page outline + header, the
           leaving page's surface) folds toward the pivot and past
           it — its horizontal factor runs below zero at mid-window,
           so the face's far edge crosses the pivot line as a page
           edge passing — while the incoming face (the line texture
           that actually changes between pages) unfolds FROM the
           pivot: its factor starts at the configured minimum and
           opens to full width by the window's readable onset. Both
           profiles have zero end derivatives and complete by the
           onset fraction of the window; backing stack anchors
           never turn */
        const turning = turnActive && (p.kind === "page" || p.kind === "line");
        if (turning){
          /* the incoming face — the new page's outline, header and
             line texture, sampled as its own target set — unfolds
             FROM the pivot: its factor starts at the configured
             minimum and opens to full width by the readable onset.
             The outgoing face is a different particle set entirely
             (the leaving page's texture, handled in the exit branch
             above), folding past the pivot on its own profile */
          const ou = Math.min(1, Math.max(0, turnU / CONFIG.turnOnsetFraction));
          const os = ou * ou * (3 - 2 * ou);
          const factor = CONFIG.turnCosMin + (1 - CONFIG.turnCosMin) * os;
          turnFactorIn = factor; turnFactorOut = factor;
          lastTurnFactor = factor;
          ax = pivotX + (ax - pivotX) * factor;
          /* intensity binds to this particle's own registration, not
             to the turn window's end: a particle that has arrived and
             registered renders at its settled intensity from that
             instant, even while the window is still running */
          if (f >= 1) {
            const regU2 = p.regT0 < 0 ? 0 : Math.min(1, Math.max(0, (now - p.regT0) / p.regDur));
            const settledA2 = p.kind === "stack" ? CONFIG.stackAlpha : CONFIG.foregroundAlpha;
            p.renderAlpha = CONFIG.travelAlpha + (settledA2 - CONFIG.travelAlpha) * easeProfile(regU2);
          } else p.renderAlpha = CONFIG.travelAlpha;
        } else if (f < 1) {
          p.renderAlpha = CONFIG.travelAlpha;
        } else {
          /* settled intensity is reached THROUGH registration: the
             intensity interpolates from the travelling intensity to
             the settled intensity by registration progress, so newly
             settled content is never at full intensity while still
             spatially displaced */
          const regU = p.regT0 < 0 ? 0 : Math.min(1, Math.max(0, (now - p.regT0) / p.regDur));
          const settledAlpha = p.kind === "stack" ? CONFIG.stackAlpha : (p.kind === "halo" ? CONFIG.haloAlpha : CONFIG.foregroundAlpha);
          p.renderAlpha = CONFIG.travelAlpha + (settledAlpha - CONFIG.travelAlpha) * easeProfile(regU);
        }
        p.ax = ax; p.ay = ay;
      }
    } else {
      if (p.scatActive){
        /* release window: the anchor scatters to its dispersed anchor
           over the release's OWN duration while the displayed offset
           re-emerges to the full stream contribution; the scatter
           profile has zero end derivatives, and a derivative blend
           over the first quarter of the window carries the captured
           rendered derivative into the profile's own slope. No
           positional jitter term is added in this branch: whatever
           jitter the rendered position held at release is already
           inside the captured start position. */
        const u = Math.min(1, Math.max(0, (now - p.relT) / p.relDur));
        const se = easeProfile(u);
        const w = Math.min(1, u / CONFIG.releaseBlendFraction);
        const hb = w - w*w*w + 0.5*w*w*w*w;
        const corr = CONFIG.releaseBlendFraction * (p.relDur / 1000) * hb;
        p.ax = p.scatFromX + (p.scatToX - p.scatFromX)*se + p.relAnchorVelX * corr;
        p.ay = p.scatFromY + (p.scatToY - p.scatFromY)*se + p.relAnchorVelY * corr;
        p.dox = p.relDispX + (p.ox - p.relOx)*se + p.relDerivX * corr;
        p.doy = p.relDispY + (p.oy - p.relOy)*se + p.relDerivY * corr;
        if (u >= 1) { p.scatActive = false; p.jitAmp = 0; p.refx = p.ox - p.dox; p.refy = p.oy - p.doy; }
      } else {
        p.dox = p.ox - p.refx; p.doy = p.oy - p.refy;
      }
      /* a released page/title particle holds its settled intensity
         until the clearance point of its scatter window — the old
         page silhouette clears geometrically before its intensity
         drops to surplus, never at release start */
      if (p.scatActive && (p.kind === "page" || p.kind === "line" || p.kind === "title" || p.kind === "halo" || p.kind === "stack") && ((now - p.relT) / Math.max(1, p.relDur)) < CONFIG.releaseClearanceHoldU) {
        p.renderAlpha = p.kind === "stack" ? CONFIG.stackAlpha : (p.kind === "halo" ? CONFIG.haloAlpha : CONFIG.foregroundAlpha);
      } else {
        p.renderAlpha = p.borrowable ? CONFIG.surplusAlpha : CONFIG.reserveAlpha;
      }
    }
    p.x = p.ax + p.dox; p.y = p.ay + p.doy;
    const m = CONFIG.wrapMarginPx;
    if (p.x < -m) p.ax += W + 2*m - CONFIG.wrapJumpInsetPx; if (p.x > W + m) p.ax -= W + 2*m - CONFIG.wrapJumpInsetPx;
    if (p.y < -m) p.ay += H + 2*m - CONFIG.wrapJumpInsetPx; if (p.y > H + m) p.ay -= H + 2*m - CONFIG.wrapJumpInsetPx;
    p.x = p.ax + p.dox; p.y = p.ay + p.doy;
    if (dt > 0) {
      const wrapped = Math.abs(p.x - prevX) > W / 2 || Math.abs(p.y - prevY) > H / 2;
      if (!wrapped) { p.velX = (p.x - prevX) / dt; p.velY = (p.y - prevY) / dt; p.doxVelX = (p.dox - prevDox) / dt; p.doxVelY = (p.doy - prevDoy) / dt; }
    }
  }
}

WHAT THE BUILD IS
A single self-contained HTML file. A stack of document pages, rendered as particles sampled from config-rendered text and shapes, flips through its page counter (pages 1 to 14 at counter checkpoints 1, 6 and 14, at 0 ms, 1500 ms and 2200 ms of a 6000 ms cycle); the finished stack disperses, a title condenses out of the particle stream with a ring above it, the title holds, then disperses inside a dedicated seam interval (5700–6000 ms) and the cycle restarts with no borrowed particles carried across the seam. CONSTRUCTION FACTS
- Two distinct faces per page turn, on separate particle sets. The OUTGOING face is the leaving page's own line texture PLUS a dedicated edge set: the leaving page's outline targets are borrowed as first-class exiting targets, so the outgoing face carries outline + texture through the whole handoff. Both faces fold around the page's left-edge pivot on the turn profiles (outgoing factor crosses below zero at mid-window; incoming factor rises from the configured minimum to 1.0 by the window's readable onset, 70% of the effective turn window). The INCOMING face is the new page's own sampled target set (outline + header + line texture) unfolding from the pivot. The pool is sized for the concurrent demand the handoff actually creates: the incoming page, the exiting line set, two edge sets (the previous handoff's edge may still be exiting), and one vanished set still scattering, grossed up by the reserve fraction.
- Page dispatch: every page change crossed by a simulation step is dispatched in chronological order at its scheduled instant (one event order across segment entries and page changes; ties resolve segment entry first). Diagnostics keep three separate counters that cannot be conflated: schedule coverage (entries crossed), assignments (pages whose targets were actually assigned), and presented (pages that reached a drawn frame). At 120 ms callbacks, coverage equals assignments for every page of the default schedule (14/14) and of a 40-page / 6000 ms schedule (40/40); a single step crossing many pages and the flip exit assigns the identical page sequence as fine-step execution.
