function figShapeChecks(f,errs){
  let bad=false;
  FIG_CONTAINER_FIELDS.forEach(k=>{ if(k in f && !Array.isArray(f[k])){ errs.push(f.id+": "+k+" must be an array"); bad=true; } });
  const req=FIG_REQUIRED_FIELDS[f.type]||[];
  req.forEach(k=>{ if(!(k in f)||f[k]===undefined||f[k]===null){ errs.push(f.id+": "+k+" is a required field and is absent"); bad=true; } });
  if(bad) return true;
  if(Array.isArray(f.bars)&&f.bars.length===0){ errs.push(f.id+": bars must name at least one bar (an empty array draws nothing)"); bad=true; }
  if(Array.isArray(f.bars)) f.bars.forEach((v,i)=>{ if(typeof v!=="number"||!isFinite(v)||v<0||v>1){ errs.push(f.id+": bars["+i+"] must be a finite number in 0..1"); bad=true; } });
  if(Array.isArray(f.points)&&f.points.length===0){ errs.push(f.id+": points must name at least one point (an empty array draws nothing)"); bad=true; }
  if(Array.isArray(f.points)) f.points.forEach((pt,i)=>{ if(!Array.isArray(pt)||pt.length<2||typeof pt[0]!=="number"||typeof pt[1]!=="number"||!isFinite(pt[0])||!isFinite(pt[1])||pt[0]<0||pt[0]>1||pt[1]<0||pt[1]>1){ errs.push(f.id+": points["+i+"] must be a [x,y] pair of finite numbers in 0..1"); bad=true; } });
  if(Array.isArray(f.yLabels)) f.yLabels.forEach((o,i)=>{ if(!o||typeof o!=="object"||typeof o.y!=="number"||!isFinite(o.y)||o.y<0||o.y>1){ errs.push(f.id+": yLabels["+i+"].y must be a finite number in 0..1"); bad=true; } });
  if("donutFraction" in f && (typeof f.donutFraction!=="number"||!isFinite(f.donutFraction)||f.donutFraction<0||f.donutFraction>1)){ errs.push(f.id+": donutFraction must be a finite number in 0..1"); bad=true; }
  if("cliffExponent" in f && (typeof f.cliffExponent!=="number"||!isFinite(f.cliffExponent)||!(f.cliffExponent>0))){ errs.push(f.id+": cliffExponent must be a finite number above 0"); bad=true; }
  if("cliffAt" in f && (typeof f.cliffAt!=="number"||!isFinite(f.cliffAt)||f.cliffAt<0||f.cliffAt>1)){ errs.push(f.id+": cliffAt must be a finite number in 0..1"); bad=true; }
  if("preCliffDrop" in f && (typeof f.preCliffDrop!=="number"||!isFinite(f.preCliffDrop)||f.preCliffDrop<0)){ errs.push(f.id+": preCliffDrop must be a finite number at or above 0"); bad=true; }
  if("counterStart" in f && (typeof f.counterStart!=="number"||!isFinite(f.counterStart))){ errs.push(f.id+": counterStart must be a finite number"); bad=true; }
  if("counterEnd" in f && (typeof f.counterEnd!=="number"||!isFinite(f.counterEnd))){ errs.push(f.id+": counterEnd must be a finite number"); bad=true; }
  return bad;
}

function schemaFieldChecks(f,errs){
  const sch=deriveSchema().perType[f.type]; if(!sch) return;
  sch.forEach(({field,kind,required})=>{
    if(!(field in f)||f[field]===undefined||f[field]===null){ if(required) errs.push(f.id+": "+field+" is read by its renderer and is absent"); return; }
    const v=f[field];
    if(kind==="fraction"&&(typeof v!=="number"||!isFinite(v)||v<0||v>1)) errs.push(f.id+": "+field+" must be a finite fraction within 0..1");
    if(kind==="positive"&&(typeof v!=="number"||!isFinite(v)||!(v>0))) errs.push(f.id+": "+field+" must be a positive finite number");
    if(kind==="positive-integer"&&(!Number.isInteger(v)||v<=0)) errs.push(f.id+": "+field+" must be a positive integer");
    if(kind==="array"&&!Array.isArray(v)) errs.push(f.id+": "+field+" must be an array");
    if(kind==="fractions"&&(!Array.isArray(v)||v.some(x=>typeof x!=="number"||!isFinite(x)||x<0||x>1))) errs.push(f.id+": "+field+" must be an array of finite fractions within 0..1");
  });
}

  const cfg=deps.config, P=deps.palette, dom=deps.dom, el=dom.el, stageEl=deps.stageEl;

  stageEl.style.background=P.paper; stageEl.style.color=P.ink;

  const dateS=el('span','',cfg.dateLine);
  const clockS=el('span','',cfg.clockStart); clockS.style.color=P.red;
  const priceS=el('span','',cfg.price);
  if(L.metaRows===2){ priceS.style.flexBasis='100%'; priceS.style.textAlign='right'; }
  meta.style.color=P.muted;
  meta.append(dateS, clockS, priceS); stageEl.appendChild(meta);
  const problems=validateConfig(cfg);
  const globalProblems=problems.filter(x=>!cfg.figures.some(f=>f&&x.indexOf(f.id+':')===0));
  if(globalProblems.length){ const gw=el('div','warn globalwarn','Validation: '+globalProblems.join('; ')); stageEl.appendChild(gw); }
  const mounted=[]; const appended=[];
  try{ cfg.figures.forEach(f=>{
    if(!f||typeof f!=="object") return;
    const root=el('section','fig'); root.id=f.id;
    const num=el('div','fignum',f.num); num.style.color=P.red; root.appendChild(num);
    const h=el('h2'); wordWrapLines(f.title,L.contentW,L.titlePx,cfg).forEach((ln,li,arr)=>{ h.appendChild(el('span','titleline',ln)); if(li<arr.length-1) h.appendChild(el('br')); });
    h.style.minHeight=(headingLines(f.title,L,cfg)*L.titlePx*cfg.layout.lineHeights.title)+'px'; root.appendChild(h);
    const body=el('div'); root.appendChild(body);
    const srcEl=el('div','src'); wordWrapLines(f.source,L.contentW,L.labelPx,cfg,"label").forEach((ln,li,arr)=>{ srcEl.appendChild(el('span','srcline',ln)); if(li<arr.length-1) srcEl.appendChild(el('br')); }); root.appendChild(srcEl);
    stageEl.appendChild(root); appended.push(root);
    const figProblems=problems.filter(x=>x.indexOf(f.id+':')===0);
    let draw=null;

    mounted.push({fig:f,root:root,draw:draw});
  });
  } catch(e){ appended.forEach(r=>r.remove()); throw e; }
  return mounted;

function fitFailureMessages(fit,W){
  const parts=[];
  (fit.globalErrors||[]).forEach(e=>parts.push(e));
  (fit.figures||[]).forEach(r=>{ if(r.rejected){ parts.push(r.id+': '+(r.errors||[]).join('; ')); return; }
    if(!r.fits) parts.push(r.id+': required '+r.required+'px exceeds allocated '+r.allocated+'px');
    if(!r.widthOk) parts.push(r.id+': occupied '+r.occupied+'px exceeds available '+r.available+'px or positioned text overflows'); });
  if(!fit.fontsOk) parts.push('type floors not met at width '+W);
  if(!fit.budgetOk) parts.push('stage height '+fit.stageH+'px exceeds the phone stage budget at width '+W);
  return parts.length?parts:["fit check failed at width "+W];
}
