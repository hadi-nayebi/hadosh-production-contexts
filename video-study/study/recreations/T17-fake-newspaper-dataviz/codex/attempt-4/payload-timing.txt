function nodeThresholds(fig,cfg){const T=(cfg||CONFIG).timing; const out=[]; for(let i=0;i<fig.nodes.length;i++) out.push(T.nodeStart+(i+1)*((T.nodeEnd-T.nodeStart)/fig.nodes.length)); return out;}
function orgState(fig,q,cfg){const T=(cfg||CONFIG).timing; const th=nodeThresholds(fig,cfg);
  const badgeQ=clamp01((q-T.badgeAt)/T.badgeDur); const badgeE=1-Math.pow(1-badgeQ,T.badgeEasePow); return {ceo:q>=T.ceoAt, nodes:th.map(t=>q>=t), badge:q>=T.badgeAt, badgeQ:badgeQ, badgeE:badgeE};}
function stampScaleAt(sq,T,cap){ const from=(cap!=null?cap:T.stampFrom);
  if(sq<=0) return from;
  if(sq<T.stampApproach){ const u=sq/T.stampApproach; const e=1-Math.pow(1-u,T.stampEasePowIn); return lerp(from,T.stampOvershoot,e); }
  if(sq<T.stampReboundEnd){ const u=(sq-T.stampApproach)/(T.stampReboundEnd-T.stampApproach); return lerp(T.stampOvershoot,T.stampCompress,u); }
  const u=(sq-T.stampReboundEnd)/(1-T.stampReboundEnd); return lerp(T.stampCompress,1,1-Math.pow(1-u,T.stampEasePowOut));
}

function stateAt(t,figures,duration){
  const tt=Math.max(0,Math.min(duration,t));
  for(let i=0;i<figures.length;i++){const f=figures[i]; const last=i===figures.length-1;
    if(tt>=f.start && (tt<f.start+f.dur || (last && tt<=duration))) return {index:i, p:clamp01((tt-f.start)/f.dur)};}
  let idx=0; figures.forEach((f,i)=>{if(tt>=f.start) idx=i;});
  return {index:idx, p:clamp01((tt-figures[idx].start)/figures[idx].dur)};
}
function computeLayout(w,cfg){const c=cfg||CONFIG; const F=c.typeFloors, LH=c.layout.lineHeights, TS=c.layout.typeScale, TC=c.layout.typeCaps, phone=w<=c.layout.phoneBreakpoint;
  const mastPx=Math.max(F.masthead, Math.min(TC.mast, w*TS.mast));
  const titlePx=Math.max(F.figureTitle, Math.min(TC.title, w*TS.title));
  const bodyPx=Math.max(F.body, Math.min(TC.body, w*TS.body));
  const labelPx=Math.max(F.label, Math.min(TC.label, w*TS.label));
  const bigPx=Math.max(F.bigCounter, Math.min(TC.big, w*TS.big));
  const handPx=Math.max(F.hand, Math.min(TC.hand, w*TS.hand));
  const padX=phone?c.layout.padXPhone:c.layout.padXDesktop, padTop=phone?c.layout.padTopPhone:c.layout.padTopDesktop;
  const metaTexts=[c.dateLine,c.clockStart,c.price];
  const metaWidths=metaTexts.map(t=>t.length*labelPx*c.layout.charWidthLabel);
  const metaAvail=w-2*c.layout.metaPadX;

  const metaH=metaRows*(labelPx*LH.label+10)+(metaRows-1)*2;
  const headerH=padTop + mastPx*LH.mast + 13 + metaH + 4;
  const contentW=w-2*padX, stageH=phone?null:w*9/16;
  const contentH=phone?null:stageH-headerH;
  const headSrcReserve=labelPx*LH.label + 2*titlePx*LH.title + (c.layout.headingMarginTopPx+c.layout.headingMarginBottomPx) + (labelPx*LH.label+c.layout.srcMarginTopPx);
  const availBlockH=phone?(c.layout.phoneStageBudgetPx-headerH-c.layout.bottomClearancePx):contentH;
  return {w, phone, mastPx, titlePx, bodyPx, labelPx, bigPx, handPx, padX, padTop, headerH, contentW, stageH, contentH, metaRows, metaWidths, metaAvail, headSrcReserve, availBlockH};

function figureGeom(fig,L,cfg){const c=cfg||CONFIG; const LH=c.layout.lineHeights; const cw=L.contentW;

    const titleH=L.labelPx*LH.label+4;
    return {chartW,chartH,cardW,cardH,titleH, blockH:L.phone?cardH+c.layout.rowGapPx+titleH+chartH:Math.max(cardH,titleH+chartH), maxW:cw};}
  if(fig.type==="waffle"){const gap=c.layout.waffleGapPx, dot=Math.max(c.layout.waffleDotMin,Math.min(c.layout.waffleDotMax,Math.floor((cw*c.layout.waffleGridFrac)/fig.waffleCols)-gap));

  if(fig.type==="stackBars"){const chartW=L.phone?cw:Math.min(c.layout.chartMaxW,cw*c.layout.chartWidthFrac.stack), chartH=L.phone?c.layout.chartHeights.stackPhone:c.layout.chartHeights.stackDesktop;

    return {chartW,chartH,unitH,unitHRaw,unitFloor,unitHOver:unitHRaw<unitFloor,stackH,counterH,stackBudget,arrivalReserve:arrival, blockH:L.phone?counterH+stackH+gap+chartH:Math.max(stackH+counterH,chartH), maxW:cw};}
  if(fig.type==="orgChart"){const longest=Math.max.apply(null,fig.nodes.map(n=>n.length));
    const nodeW=longest*L.labelPx*c.layout.charWidthLabel+c.layout.nodePadX; let perRow=Math.max(1,Math.floor(cw/nodeW)); while(perRow>1 && perRow*nodeW+(perRow-1)*c.layout.orgRowGapPx>cw) perRow--; const rows=Math.ceil(fig.nodes.length/perRow);

function validateConfig(cfg){
  const errs=[]; if(!cfg||typeof cfg!=="object"){errs.push("config absent"); return errs;}
  const B=cfg.bounds, T=cfg.timing;
  if(!isFinite(cfg.duration)||cfg.duration<=0) errs.push("duration must be a positive finite number");
  if(!T||!isFinite(T.minHoldSec)||!isFinite(T.readMsPerChar)||T.readMsPerChar<=0) errs.push("timing values must be finite with a positive reading rate");
  if(T&&(T.fillSec<T.minFillSec)) errs.push("timing: fill duration below its named minimum");
  if(!Array.isArray(cfg.figures)||cfg.figures.length<1||cfg.figures.length>B.maxFigures) errs.push("figures count outside supported boundary 1.."+B.maxFigures);
  (cfg.figures||[]).forEach(f=>{
    if(!f||typeof f!=="object"){errs.push("figure record absent"); return;}
    if(!isFinite(f.start)||!isFinite(f.dur)) errs.push(f.id+": figure window values must be finite");
    if(!(f.dur>cfg.timing.minHoldSec)) errs.push(f.id+": window must exceed the minimum hold");
    if(f.type==="waffle"){ if(f.waffleTotal>B.maxWaffle) errs.push(f.id+": waffleTotal "+f.waffleTotal+" exceeds supported boundary "+B.maxWaffle);
      if(typeof f.summaryText!=="string"||!f.summaryText.length) errs.push(f.id+": summary text absent");
      else { const st=summaryTiming(f,cfg); if(st.typeDurSec<T.minTypeSec) errs.push(f.id+": summary of "+f.summaryText.length+" chars cannot type in at least "+T.minTypeSec+"s and still hold for reading inside its window"); } }
    if(f.type==="stackBars"){ if(f.bars.length>B.maxBars) errs.push(f.id+": bars "+f.bars.length+" exceeds supported boundary "+B.maxBars);
      if(f.stackCount>B.maxStack) errs.push(f.id+": stackCount "+f.stackCount+" exceeds supported boundary "+B.maxStack);
      if(f.bars.length!==f.barLabels.length) errs.push(f.id+": barLabels length must match bars length");
      if(f.bars.some(v=>!isFinite(v))) errs.push(f.id+": bar values must be finite");
      const L360=computeLayout(cfg.layout.phoneProbeWidth,cfg); const g360=figureGeom(f,L360,cfg);
      if(g360.unitHOver) errs.push(f.id+": stackCount "+f.stackCount+" cannot hold the unit legibility floor at phone width"); }
    if(f.type==="orgChart"&&f.nodes.length>B.maxNodes) errs.push(f.id+": nodes "+f.nodes.length+" exceeds supported boundary "+B.maxNodes);
    if(f.type==="correction"&&f.notes.length>B.maxNotes) errs.push(f.id+": notes "+f.notes.length+" exceeds supported boundary "+B.maxNotes);
    if(f.type==="lineChart"){ if(f.points.length>B.maxPoints) errs.push(f.id+": points "+f.points.length+" exceeds supported boundary "+B.maxPoints);
      if(f.points.some(pt=>!pt||!isFinite(pt[0])||!isFinite(pt[1]))) errs.push(f.id+": point coordinates must be finite"); }
    if(f.type==="counterGraph"){ if(!(f.counterStart>f.counterEnd)) errs.push(f.id+": counterStart must exceed counterEnd");
      if(!isFinite(f.counterStart)||!isFinite(f.counterEnd)||!isFinite(f.cliffAt)) errs.push(f.id+": counter values must be finite");
      if(!(f.cliffAt>B.cliffMin)) errs.push(f.id+": cliffAt outside supported boundary "+B.cliffMin+"..animation fraction");
      else { const lim=animFraction(f,cfg)-T.cliffBudgetMargin; if(!(f.cliffAt<lim)) errs.push(f.id+": cliffAt "+f.cliffAt+" must sit below the animation fraction minus the named margin ("+lim.toFixed(3)+")"); }
      if(!(f.preCliffDrop>0&&f.preCliffDrop<(f.counterStart-f.counterEnd)*0.05)) errs.push(f.id+": preCliffDrop outside supported boundary (0, 5% of counter range)"); }
  });
  const knownTypes=["lineChart","waffle","stackBars","orgChart","counterGraph","correction"];
  (cfg.figures||[]).forEach(f=>{ if(f && knownTypes.indexOf(f.type)<0) errs.push(f.id+": unsupported figure type "+f.type); });
  const schedFigs=cfg.figures||[];
  for(let si=1;si<schedFigs.length;si++){ const prevF=schedFigs[si-1], curF=schedFigs[si]; if(Math.abs(curF.start-(prevF.start+prevF.dur))>1e-9) errs.push("schedule: figure "+curF.id+" start "+curF.start+" does not equal previous figure end "+(prevF.start+prevF.dur)+" (overlap or gap)"); }
  if(schedFigs.length){ const lastF=schedFigs[schedFigs.length-1]; if(Math.abs(lastF.start+lastF.dur-cfg.duration)>1e-9) errs.push("schedule: last figure end "+(lastF.start+lastF.dur)+" does not equal total duration "+cfg.duration); }
  return errs;
}

/* Scene assembly: builds the broadsheet frame and mounts every figure factory with
   the injected dependency set. Returns the mounted renderers. */
function createScene(deps){
  const cfg=deps.config, P=deps.palette, dom=deps.dom, el=dom.el, stageEl=deps.stageEl;
  const L=deps.layout;
  stageEl.style.background=P.paper; stageEl.style.color=P.ink;
  stageEl.style.setProperty('--warn',P.red); stageEl.style.setProperty('--muted',P.muted); stageEl.style.setProperty('--secondary',P.secondary); stageEl.style.setProperty('--srcmt',cfg.layout.srcMarginTopPx+'px');
  stageEl.style.setProperty('--card-tint',P.cardTint);
  stageEl.style.setProperty('--font-serif',deps.fonts.serif); stageEl.style.setProperty('--font-sans',deps.fonts.sans); stageEl.style.setProperty('--font-hand',deps.fonts.hand);
  stageEl.style.setProperty('--metapad',cfg.layout.metaPadX+'px');
  stageEl.style.setProperty('--lh-mast',cfg.layout.lineHeights.mast); stageEl.style.setProperty('--lh-title',cfg.layout.lineHeights.title);
  stageEl.style.setProperty('--fw-normal',cfg.layout.fontWeights.normal); stageEl.style.setProperty('--fw-black',cfg.layout.fontWeights.black);
  const mast=el('div','mast'); mast.appendChild(el('h1','',cfg.masthead)); stageEl.appendChild(mast);
  stageEl.appendChild(el('div','rule'));
  const meta=el('div','meta');
  const dateS=el('span','',cfg.dateLine);
  const clockS=el('span','',cfg.clockStart); clockS.style.color=P.red;
  const priceS=el('span','',cfg.price);
  if(L.metaRows===2){ priceS.style.flexBasis='100%'; priceS.style.textAlign='right'; }
  meta.append(dateS, clockS, priceS); stageEl.appendChild(meta);
  const problems=validateConfig(cfg);
  const globalProblems=problems.filter(x=>!cfg.figures.some(f=>x.indexOf(f.id+':')===0));
  if(globalProblems.length){ const gw=el('div','warn globalwarn','Validation: '+globalProblems.join('; ')); stageEl.appendChild(gw); }
  const mounted=[];
  cfg.figures.forEach(f=>{
    const root=el('section','fig'); root.id=f.id;
    const num=el('div','fignum',f.num); num.style.color=P.red; root.appendChild(num);
    const h=el('h2','',f.title); h.style.minHeight=(headingLines(f.title,L,cfg)*L.titlePx*cfg.layout.lineHeights.title)+'px'; root.appendChild(h);
    const body=el('div'); root.appendChild(body);
    root.appendChild(el('div','src',f.source));
    stageEl.appendChild(root);
    const figProblems=problems.filter(x=>x.indexOf(f.id+':')===0);
    let draw=null;
    if(figProblems.length){ const warn=el('div','warn','Unsupported data — '+figProblems.join('; ')); body.appendChild(warn); if(window.console) console.warn('T17 validation: '+figProblems.join('; ')); }
    else if(BUILDERS[f.type]) draw=BUILDERS[f.type]({root:root,body:body},f,deps);
    mounted.push({fig:f,root:root,draw:draw});
  });
  return mounted;
}
