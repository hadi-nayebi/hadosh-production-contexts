function barRects(fig,geom,q,cfg){const B=(cfg||CONFIG).layout.barGeom;
  const n=fig.bars.length, slot=(geom.w*B.slotFrac)/n, bw=slot*B.barFrac, base=geom.h-B.basePad, maxH=geom.h-B.topPad;
  return fig.bars.map((v,i)=>{const h=v*maxH*q; return {x:B.xPad+i*slot+(slot-bw)/2, y:base-h, w:bw, h, labelX:B.xPad+i*slot+slot/2};});
}
function stateAt(t,figures,duration){
  const tt=Math.max(0,Math.min(duration,t));
  for(let i=0;i<figures.length;i++){const f=figures[i]; const last=i===figures.length-1;
    if(tt>=f.start && (tt<f.start+f.dur || (last && tt<=duration))) return {index:i, p:clamp01((tt-f.start)/f.dur)};}
  let idx=0; figures.forEach((f,i)=>{if(tt>=f.start) idx=i;});
  return {index:idx, p:clamp01((tt-figures[idx].start)/figures[idx].dur)};
}
function smoothstep01(v){const x=clamp01(v); return x*x*(3-2*x);}
function transitionState(t,figures,cfg){const c=cfg||CONFIG; const T=c.timing.figureTransitionSec, rise=c.layout.transitionRisePx;
  const st=stateAt(t,figures,c.duration);
  if(T>0){ for(let i=1;i<figures.length;i++){ const b=figures[i].start;
    if(t>=b-T/2 && t<=b+T/2){ const s=(t-(b-T/2))/T; const outP=smoothstep01(s/c.timing.transitionOutSpan); const inP=smoothstep01((s-c.timing.transitionInStart)/c.timing.transitionInSpan);
      const rawOut=1-outP; const rel=c.timing.transitionFloorRelease; const ramp=1-smoothstep01((s-rel)/(1-rel)); const outOpacity=Math.max(rawOut, c.timing.transitionFloorOpacity*ramp);
      return {index:i, p:clamp01((t-b)/figures[i].dur), prevIndex:i-1, outOpacity:outOpacity, outTy:-rise*outP, outP:(t>=b?1:clamp01((t-figures[i-1].start)/figures[i-1].dur)), inOpacity:inP, inTy:rise*(1-inP), inP:clamp01((t-b)/figures[i].dur), s:s}; } } }
  return {index:st.index, p:st.p, prevIndex:-1, outOpacity:0, outTy:0, outP:st.p, inOpacity:1, inTy:0, inP:st.p, s:-1};
}
function computeLayout(w,cfg){const c=cfg||CONFIG; const F=c.typeFloors, LH=c.layout.lineHeights, TS=c.layout.typeScale, TC=c.layout.typeCaps, phone=w<=c.layout.phoneBreakpoint;
  const mastPx=Math.max(F.masthead, Math.min(TC.mast, w*TS.mast));
  const titlePx=Math.max(F.figureTitle, Math.min(TC.title, w*TS.title));
  const bodyPx=Math.max(F.body, Math.min(TC.body, w*TS.body));
  const labelPx=Math.max(F.label, Math.min(TC.label, w*TS.label));
  const bigPx=Math.max(F.bigCounter, Math.min(TC.big, w*TS.big));
  const handPx=Math.max(F.hand, Math.min(TC.hand, w*TS.hand));
  const padX=phone?c.layout.padXPhone:c.layout.padXDesktop, padTop=phone?c.layout.padTopPhone:c.layout.padTopDesktop;
  const contentW=w-2*padX, stageH=phone?null:w*c.layout.stageAspectH/c.layout.stageAspectW;
  const mastLines=wordWrapLines(c.masthead, contentW, mastPx, c, "body").length;
  const metaTexts=[c.dateLine,c.clockStart,c.price];
  const metaWidths=metaTexts.map(t=>textWidth(t,labelPx,c,"label"));
  const metaAvail=w-2*c.layout.metaPadX;
  let metaRows=1, rowW=0; metaWidths.forEach(mw=>{ const need=(rowW>0?c.layout.metaMinGapPx:0)+mw; if(rowW>0 && rowW+need>metaAvail){ metaRows++; rowW=mw; } else rowW+=need; });
  const metaH=metaRows*(labelPx*LH.label+10)+(metaRows-1)*2;
  const headerH=padTop + mastLines*mastPx*LH.mast + 13 + metaH + 4;
  const contentH=phone?null:stageH-headerH;
  const headSrcReserve=labelPx*LH.label + 2*titlePx*LH.title + (c.layout.headingMarginTopPx+c.layout.headingMarginBottomPx) + (labelPx*LH.label+c.layout.srcMarginTopPx);
  const availBlockH=phone?(c.layout.phoneStageBudgetPx-headerH-c.layout.bottomClearancePx):contentH;
  return {w, phone, mastPx, titlePx, bodyPx, labelPx, bigPx, handPx, padX, padTop, headerH, contentW, stageH, contentH, metaRows, mastLines, metaWidths, metaAvail, headSrcReserve, availBlockH};
}
function charFactor(ch,cfg,kind){const c=cfg||CONFIG; const W=c.layout.charWidths;
  if(ch===" ") return W.space;
  if("iljtf.,:;'!|()[]".indexOf(ch)>=0) return W.narrow;
  if("mwMW@#%".indexOf(ch)>=0) return W.wide;
  if(ch>="A"&&ch<="Z") return W.upper;
  if(ch>="0"&&ch<="9") return W.digit;
  return kind==="label"?c.layout.charWidthLabel:c.layout.charWidthBody;}
function textWidth(text,fontPx,cfg,kind){let s=0; for(const ch of String(text)) s+=charFactor(ch,cfg,kind); return s*fontPx;}
function wordWrapLines(text,widthPx,fontPx,cfg,kind){const words=String(text).split(" "); const lines=[]; let cur="";
  for(const wd of words){ if(cur===""){ cur=wd; continue; }
    if(textWidth(cur+" "+wd,fontPx,cfg,kind)<=widthPx) cur=cur+" "+wd; else { lines.push(cur); cur=wd; } }
  if(cur!==""||lines.length===0) lines.push(cur); return lines;}
function textLines(text,widthPx,fontPx,cfg,kind){return Math.max(1,wordWrapLines(text,widthPx,fontPx,cfg,kind).length);}
function headingLines(title,L,cfg){return textLines(title, L.contentW, L.titlePx, cfg);}
function stampMetrics(fig,L,cfg){const c=cfg||CONFIG; const T=c.timing, lay=c.layout;
  const w=fig.stamp.length*L.bodyPx*lay.stampCharWidth + lay.stampLetterSpacingEm*L.bodyPx*fig.stamp.length + lay.stampPadX + 2*lay.stampBorderPx;
  const h=L.bodyPx + lay.stampPadY + 2*lay.stampBorderPx;
  const rad=Math.abs(T.stampRotationDeg)*Math.PI/180, cos=Math.cos(rad), sin=Math.sin(rad);
  const aabbW1=w*cos+h*sin, aabbH1=w*sin+h*cos;
  return {w:w, h:h, aabbW1:aabbW1, aabbH1:aabbH1};
}
function stampCap(fig,L,cfg){const c=cfg||CONFIG; const m=stampMetrics(fig,L,c);
  const corrW=Math.min(c.layout.corrMaxW,L.contentW);
  return Math.min(c.timing.stampFrom, corrW/m.aabbW1, c.layout.stampAreaBudgetPx/m.aabbH1);
}
function stampAabbAt(fig,L,scale,cfg){const c=cfg||CONFIG; const m=stampMetrics(fig,L,c); const cap=stampCap(fig,L,c);
  const aabbW=m.aabbW1*scale, aabbH=m.aabbH1*scale; const offsetX=Math.max(0, m.aabbW1*cap/2 - m.w/2);
  const cx=offsetX + m.w/2; const corrW=Math.min(c.layout.corrMaxW,L.contentW);
  return {offsetX:offsetX, aabbW:aabbW, aabbH:aabbH, left:cx-aabbW/2, right:cx+aabbW/2, corrW:corrW};}
function badgeSegments(fig,cfg){const c=cfg||CONFIG; const F=c.formatting;
  return [ {text:fig.onlyNote, font:F.badgeHandFont}, {text:F.badgeStarSep+fig.badgeName+F.badgeDashSep+fig.badge, font:F.badgeSansFont} ];}
function badgeFontPx(seg,L,cfg){const c=cfg||CONFIG; return seg.font===(cfg||CONFIG).formatting.badgeHandFont? L.handPx : L.bodyPx;}
function badgeTextWidth(fig,L,cfg){const c=cfg||CONFIG; return badgeSegments(fig,c).reduce((s,seg)=>s+textWidth(seg.text, badgeFontPx(seg,L,c), c),0);}
function badgeMetrics(fig,L,cfg){const c=cfg||CONFIG;
  const text=fig.onlyNote+c.formatting.badgeStarSep+fig.badgeName+c.formatting.badgeDashSep+fig.badge;
  const clearedW=L.contentW-c.layout.badgeClearancePx;
  const lines=wordWrapLines(text, clearedW, L.bodyPx, c);
  const w=Math.max(badgeTextWidth(fig,L,c), Math.max.apply(null, lines.map(ln=>textWidth(ln, L.bodyPx, c))));
  const h=lines.length*L.bodyPx*c.layout.lineHeights.body+8;
  return {text:text, lines:lines.length, w:w, h:h, clearedW:clearedW};}
function badgeCap(fig,L,cfg){const m=badgeMetrics(fig,L,cfg); return Math.min((cfg||CONFIG).timing.badgeFrom, m.clearedW/Math.max(1,m.w));}
function badgeAabbAt(fig,L,scale,cfg){const m=badgeMetrics(fig,L,cfg); const cx=L.contentW/2;
  const aabbW=m.w*scale, aabbH=m.h*scale;
  return {left:cx-aabbW/2, right:cx+aabbW/2, aabbW:aabbW, aabbH:aabbH, w:m.w, h:m.h};}
function yAxisPad(fig,L,cfg){const c=cfg||CONFIG; const labs=fig.yLabels||[]; const maxLab=labs.reduce((m2,l2)=>Math.max(m2, textWidth(l2.text, L.labelPx, c, "label")),0); return Math.max(c.layout.chartPadLeft, Math.ceil(maxLab)+c.layout.yLabelGapPx+6);}
function lineLabelBounds(fig,L,cfg){const c=cfg||CONFIG; const g=figureGeom(fig,L,c);
  const geom={x0:yAxisPad(fig,L,c), w:g.chartW-yAxisPad(fig,L,c)-c.layout.chartPadRight};
  const n=fig.xLabels.length;
  return fig.xLabels.map((lab,i)=>{ const halfW=lab.length*L.labelPx*c.layout.charWidthLabel/2;
    const cx=geom.x0+geom.w*(n>1?i/(n-1):0);
    const anchor=i===0?'start':(i===n-1?'end':'middle');
    const left=anchor==='start'?cx:(anchor==='end'?cx-2*halfW:cx-halfW);
    const right=anchor==='start'?cx+2*halfW:(anchor==='end'?cx:cx+halfW);
    return {text:lab, anchor:anchor, cx:cx, halfW:halfW, left:left, right:right, chartW:g.chartW}; });
}
function cardInnerW(w,cfg){const c=cfg||CONFIG; return w-2*(c.layout.cardBorderPx+c.layout.cardPaddingPx);}
function cardPad2(cfg){const c=cfg||CONFIG; return 2*c.layout.cardPaddingPx;}
function figureGeom(fig,L,cfg){const c=cfg||CONFIG; const LH=c.layout.lineHeights; const cw=L.contentW;
  if(fig.type==="lineChart"){const chartW=L.phone?cw:Math.min(c.layout.chartMaxW,cw*c.layout.chartWidthFrac.line), chartH=L.phone?c.layout.chartHeights.linePhone:c.layout.chartHeights.lineDesktop;
    const cardW=L.phone?cw:cw-chartW-c.layout.rowGapPx, cardH=textLines(fig.cardText,cardInnerW(cardW,c),L.bodyPx,c)*L.bodyPx*LH.body+cardPad2(c);
    const titleH=L.labelPx*LH.label+4;
    return {chartW,chartH,cardW,cardH,titleH, blockH:L.phone?cardH+c.layout.rowGapPx+titleH+chartH:Math.max(cardH,titleH+chartH), maxW:cw};}
  if(fig.type==="waffle"){const gap=c.layout.waffleGapPx, dot=Math.max(c.layout.waffleDotMin,Math.min(c.layout.waffleDotMax,Math.floor((cw*c.layout.waffleGridFrac)/fig.waffleCols)-gap));
    const rows=Math.ceil(fig.waffleTotal/fig.waffleCols), gridW=fig.waffleCols*(dot+gap)-gap, gridH=rows*(dot+gap)-gap;
    const sumInnerW=cardInnerW(cw,c); const sumH=textLines(fig.summaryText,sumInnerW,L.bodyPx,c)*L.bodyPx*LH.body+cardPad2(c);
    return {dot,gap,rows,gridW,gridH,sumH,sumInnerW, blockH:gridH+6+L.bodyPx*c.layout.waffleLabelH+8+sumH, maxW:Math.max(cw,gridW)};}
  if(fig.type==="stackBars"){const chartW=L.phone?cw:Math.min(c.layout.chartMaxW,cw*c.layout.chartWidthFrac.stack), chartH=L.phone?c.layout.chartHeights.stackPhone:c.layout.chartHeights.stackDesktop;
    const gap=c.layout.fig22GapPx, unitGap=c.layout.stackUnitGapPx, arrival=c.layout.stackArrivalOffsetPx;
    const counterH=L.bodyPx*LH.stackCounter;
    const stackBudget=L.phone ? (L.availBlockH - L.headSrcReserve) - counterH - gap - chartH - arrival
                              : (L.availBlockH - L.headSrcReserve) - counterH - arrival;
    const unitFloor=L.labelPx+2*c.layout.stackUnitBorderPx;
    const unitHRaw=Math.floor(stackBudget/fig.stackCount)-unitGap;
    const unitH=Math.min(c.layout.stackUnitMaxH, Math.max(unitFloor, unitHRaw));
    const stackH=fig.stackCount*(unitH+unitGap)+arrival;
    return {chartW,chartH,unitH,unitHRaw,unitFloor,unitHOver:unitHRaw<unitFloor,stackH,counterH,stackBudget,arrivalReserve:arrival, blockH:L.phone?counterH+stackH+gap+chartH:Math.max(stackH+counterH,chartH), maxW:L.phone?cw:(c.layout.stackColWidthPx+gap+chartW)};}
  if(fig.type==="orgChart"){ if(!Array.isArray(fig.nodes)||!fig.nodes.length) return {nodeW:0,perRow:1,rows:0,ceoH:0,nodesH:0,badgeH:0,badgeReserveH:0, blockH:0, maxW:0};
    const longest=Math.max.apply(null,fig.nodes.map(n=>n.length));
    const nodeW=Math.max.apply(null, fig.nodes.map(n=>textWidth(c.formatting.nodeSymbol+" "+n, L.labelPx, c, "label")))+c.layout.nodePadX; let perRow=Math.max(1,Math.floor(cw/nodeW)); while(perRow>1 && perRow*nodeW+(perRow-1)*c.layout.orgRowGapPx>cw) perRow--; const rows=Math.ceil(fig.nodes.length/perRow);
    const ceoH=L.bodyPx*LH.body+cardPad2(c), nodesH=rows*(L.bodyPx*LH.body+10);
    const bm=badgeMetrics(fig,L,c);
    const badgeH=bm.h, badgeReserveH=bm.h*badgeCap(fig,L,c);
    return {nodeW,perRow,rows,ceoH,nodesH,badgeH,badgeReserveH, blockH:ceoH+rows*(c.layout.connectorMarginTopPx+c.layout.connectorHeightPx+c.layout.orgRowMarginTop)+nodesH+c.layout.badgeMarginTop+badgeReserveH, maxW:cw};}
  if(fig.type==="counterGraph"){const chartW=L.phone?cw:Math.min(c.layout.chartMaxW,cw*c.layout.chartWidthFrac.counter), chartH=L.phone?c.layout.chartHeights.counterPhone:c.layout.chartHeights.counterDesktop;
    const cardW=L.phone?cw:cw-chartW-c.layout.counterGapPx;
    const cardH=textLines(fig.counterLabel,cardInnerW(cardW,c),L.labelPx,c)*L.labelPx*LH.label + L.bigPx*c.layout.bigLineH + textLines(fig.counterSub,cardInnerW(cardW,c),L.bodyPx,c)*L.bodyPx*LH.body + cardPad2(c);
    return {chartW,chartH,cardW,cardH, blockH:L.phone?chartH+c.layout.counterGapPx+cardH:Math.max(chartH,cardH), maxW:L.phone?cw:chartW+c.layout.counterGapPx+cardW};}
  if(fig.type==="correction"){ if(!Array.isArray(fig.notes)||typeof fig.stamp!=="string") return {corrW:0,corrH:0,stampH:0,stampCap:0,stampAabbW:0,stampOffsetX:0,stampAabbLeftCap:0,stampAabbRightCap:0,notesH:0,stampUpReserve:0, blockH:0, maxW:0};
    const corrW=Math.min(c.layout.corrMaxW,cw);
    const corrH=textLines(fig.correction,cardInnerW(corrW,c),L.bodyPx,c)*L.bodyPx*LH.body+cardPad2(c);
    const cap=stampCap(fig,L,c), m=stampMetrics(fig,L,c);
    const upReserve=Math.max(0, (m.aabbH1 - m.h)*cap/2);
    const stampH=m.aabbH1*cap, notesH=fig.notes.length*(L.handPx*LH.label+4);
    return {corrW,corrH,stampH,stampCap:cap,stampAabbW:m.aabbW1*cap,stampOffsetX:stampAabbAt(fig,L,cap,c).offsetX,stampAabbLeftCap:stampAabbAt(fig,L,cap,c).left,stampAabbRightCap:stampAabbAt(fig,L,cap,c).right,notesH, stampUpReserve:upReserve, blockH:corrH+c.layout.corrStampMarginTop+stampH+upReserve+c.layout.corrNotesMarginTop+notesH, maxW:corrW};}
  return {blockH:0,maxW:0};
}
function figureRequired(fig,L,cfg){const c=cfg||CONFIG; const LH=c.layout.lineHeights;
  const g=figureGeom(fig,L,c);
  const headH=L.labelPx*LH.label + headingLines(fig.title,L,c)*L.titlePx*LH.title + (c.layout.headingMarginTopPx+c.layout.headingMarginBottomPx);
  const srcH=textLines(fig.source,L.contentW,L.labelPx,c,"label")*L.labelPx*LH.label+c.layout.srcMarginTopPx;
  return {id:fig.id, required:c.layout.figPadTopPx+headH+g.blockH+srcH, blockH:g.blockH, maxW:g.maxW, geom:g, stampUpReserve:g.stampUpReserve||0};
}
function fitReport(w,figures,cfg){const c=cfg||CONFIG;
  const L=computeLayout(w,c); const allErrs=validateConfig(c); const badIds={}; allErrs.forEach(e=>{ const mm=/^(fig\d+):/.exec(e); if(mm) badIds[mm[1]]=1; });
  const figs=figures.map(f=>{ if(!f||typeof f!=="object") return {id:"(absent)", rejected:true, errors:["figure record absent"]};
    if(badIds[f.id]) return {id:f.id, rejected:true, errors:allErrs.filter(e=>e.indexOf(f.id+":")===0)}; return figureRequired(f,L,c); });
  const validFigs=figs.filter(f=>!f.rejected);
  const maxReq=validFigs.length?Math.max.apply(null,validFigs.map(f=>f.required)):0;
  const allocated=L.phone?maxReq:L.contentH;
  const stageH=L.phone?L.headerH+maxReq+c.layout.bottomClearancePx:L.stageH;
  const budgetOk=!L.phone || stageH<=c.layout.phoneStageBudgetPx+0.001;
  const fontsOk=L.mastPx>=c.typeFloors.masthead && L.titlePx>=c.typeFloors.figureTitle &&
                L.bodyPx>=c.typeFloors.body && L.labelPx>=c.typeFloors.label && L.bigPx>=c.typeFloors.bigCounter;
  const rows=figs.map(f=>{ if(f.rejected) return {id:f.id, rejected:true, errors:f.errors, required:null, allocated:Math.round(allocated*10)/10, fits:false, widthOk:false};
    return {id:f.id, required:Math.round(f.required*10)/10, allocated:Math.round(allocated*10)/10,
    fits:f.required<=allocated+0.001, widthOk:f.maxW<=L.contentW+0.001,
    headingReserve:Math.round(headingLines(figures.find(x=>x&&x.id===f.id).title,L,c)*L.titlePx*c.layout.lineHeights.title*10)/10}; });
  return {width:w, phone:L.phone, clearancePx:L.phone?c.layout.bottomClearancePx:0, stageH:Math.round(stageH*10)/10, headerH:Math.round(L.headerH*10)/10, metaRows:L.metaRows, budgetOk:budgetOk,
    contentW:L.contentW, fonts:{masthead:L.mastPx, figureTitle:L.titlePx, body:L.bodyPx, label:L.labelPx, bigCounter:L.bigPx, hand:L.handPx},
    fontsOk, figures:rows, ok:fontsOk && budgetOk && rows.every(r=>r.fits&&r.widthOk)};
}
const SCHEMA_GROUPS=["palette","fonts","timing","layout","bounds","formatting","typeFloors"];
function validateConfig(cfg){
  const errs=[]; if(!cfg||typeof cfg!=="object"){errs.push("config absent"); return errs;}
  SCHEMA_GROUPS.forEach(g=>{ if(!cfg[g]||typeof cfg[g]!=="object") errs.push(g+" group absent"); });
  if(errs.length) return errs;
  const B=cfg.bounds, T=cfg.timing;
  if(typeof cfg.duration!=="number") errs.push("duration must be a number (not string-typed)");
  else if(!isFinite(cfg.duration)||cfg.duration<=0||cfg.duration>3600) errs.push("duration must be a positive finite number within 3600");
  ["masthead","dateLine","clockStart","price"].forEach(k=>{ if(typeof cfg[k]!=="string"||!cfg[k].length) errs.push(k+" must be a non-empty string"); });
  Object.keys(T).forEach(k=>{ if(typeof T[k]!=="number"||!isFinite(T[k])) errs.push("timing."+k+" must be a finite number"); });
  if(typeof T.sampleStep==="number"&&!(T.sampleStep>0&&T.sampleStep<=0.5)) errs.push("timing.sampleStep must lie in (0, 0.5]");
  ["minHoldSec","readMsPerChar","typeMsPerChar","fillSec","counterHoldSec"].forEach(k=>{ if(typeof T[k]==="number"&&!(T[k]>0)) errs.push("timing."+k+" must be positive"); });
  ["serif","sans","hand"].forEach(k=>{ if(typeof cfg.fonts[k]!=="string"||!cfg.fonts[k].length) errs.push("fonts."+k+" must be a non-empty string"); });
  Object.keys(cfg.palette).forEach(k=>{ if(typeof cfg.palette[k]!=="string") errs.push("palette."+k+" must be a string"); });
  (function sweep(o,path){ Object.keys(o).forEach(k=>{ const v=o[k]; if(typeof v==="number"&&!isFinite(v)) errs.push(path+"."+k+" must be finite"); else if(v&&typeof v==="object"&&!Array.isArray(v)) sweep(v,path+"."+k); }); })(cfg.layout,"layout");
  if(!T||!isFinite(T.minHoldSec)||!isFinite(T.readMsPerChar)||T.readMsPerChar<=0) errs.push("timing values must be finite with a positive reading rate");
  if(T&&(T.fillSec<T.minFillSec)) errs.push("timing: fill duration below its named minimum");
  if(!Array.isArray(cfg.figures)||cfg.figures.length<1||cfg.figures.length>B.maxFigures) errs.push("figures count outside supported boundary 1.."+B.maxFigures);
  (cfg.figures||[]).forEach(f=>{
    if(!f||typeof f!=="object"){errs.push("figure record absent"); return;}
    if(!isFinite(f.start)||!isFinite(f.dur)) errs.push(f.id+": figure window values must be finite");
    if(typeof f.start!=="number"||typeof f.dur!=="number") errs.push(f.id+": figure window values must be numbers (not string-typed)");
    if(!(f.dur>cfg.timing.minHoldSec)) errs.push(f.id+": window must exceed the minimum hold");
    if(f.type==="waffle"){ if(!Number.isInteger(f.waffleTotal)||f.waffleTotal<=0) errs.push(f.id+": waffleTotal must be a positive integer");
      if(!Number.isInteger(f.waffleCols)||f.waffleCols<=0) errs.push(f.id+": waffleCols must be a positive integer");
      if(f.waffleTotal>B.maxWaffle) errs.push(f.id+": waffleTotal "+f.waffleTotal+" exceeds supported boundary "+B.maxWaffle);
      if(typeof f.summaryText!=="string"||!f.summaryText.length) errs.push(f.id+": summary text absent");
      else { const st=summaryTiming(f,cfg); if(st.typeDurSec<T.minTypeSec) errs.push(f.id+": summary typing duration below its named minimum"); if(st.completionSec+st.holdSec>f.dur+1e-9) errs.push(f.id+": summary of "+f.summaryText.length+" chars cannot type and still hold for reading inside its window (needs "+(st.completionSec+st.holdSec).toFixed(2)+"s of "+f.dur+"s)"); } }
    if(f.type==="stackBars"){ if(!Array.isArray(f.bars)||!Array.isArray(f.barLabels)) errs.push(f.id+": bars records absent");
      if(!Number.isInteger(f.stackCount)||f.stackCount<=0) errs.push(f.id+": stackCount must be a positive integer");

  if(schedFigs.length){ const lastF=schedFigs[schedFigs.length-1]; if(Math.abs(lastF.start+lastF.dur-cfg.duration)>1e-9) errs.push("schedule: last figure end "+(lastF.start+lastF.dur)+" does not equal total duration "+cfg.duration); }
  (cfg.figures||[]).forEach(f=>{ if(!f||typeof f!=="object") return; const L6=computeLayout(cfg.layout.phoneProbeWidth,cfg); const g6=figureGeom(f,L6,cfg);

    strFields.forEach(([k,lim])=>{ if(typeof f[k]==="string" && f[k].length>lim) errs.push(f.id+": "+k+" length "+f[k].length+" exceeds supported boundary "+lim); });
    ["xLabels","barLabels"].forEach(k=>{ if(Array.isArray(f[k])) f[k].forEach((s2,i2)=>{ if(typeof s2==="string" && s2.length>B6.maxLabelChars) errs.push(f.id+": "+k+"["+i2+"] length "+s2.length+" exceeds supported boundary "+B6.maxLabelChars); }); });
    ["yLabels"].forEach(k=>{ if(Array.isArray(f[k])) f[k].forEach((o2,i2)=>{ if(o2 && typeof o2.text==="string" && o2.text.length>B6.maxLabelChars) errs.push(f.id+": "+k+"["+i2+"] length exceeds supported boundary "+B6.maxLabelChars); }); });
    if(Array.isArray(f.nodes)) f.nodes.forEach((s2,i2)=>{ if(typeof s2==="string" && s2.length>B6.maxNodeChars) errs.push(f.id+": nodes["+i2+"] length "+s2.length+" exceeds supported boundary "+B6.maxNodeChars); });
    if(Array.isArray(f.notes)) f.notes.forEach((s2,i2)=>{ if(typeof s2==="string" && s2.length>B6.maxNoteChars) errs.push(f.id+": notes["+i2+"] length "+s2.length+" exceeds supported boundary "+B6.maxNoteChars); });
  });
  return errs;
}

      for(let u=T.sampleStep;u<=upto;u+=T.sampleStep) d+=' L'+(geom.x0+(geom.x1-geom.x0)*u)+' '+graphYAt(u*af,fig,geom,cfg);
      d+=' L'+(geom.x0+(geom.x1-geom.x0)*upto)+' '+graphYAt(upto*af,fig,geom,cfg);
      path.setAttribute('d',d); dash.style.opacity=aq>T.cliffMarkerAt?1:0;
      return d; };
  },
  correction(mount,fig,deps){
    const P=deps.palette, L=deps.layout, dom=deps.dom, cfg=deps.config, el=dom.el;
    const g=figureGeom(fig,L,cfg);
    const corr=el('div','card',fig.correction); corr.style.cssText='border-color:'+P.red+';color:'+P.red+';font-family:'+deps.fonts.sans+';font-size:'+L.bodyPx+'px;line-height:'+cfg.layout.lineHeights.body+';max-width:'+g.corrW+'px;opacity:0'; mount.body.appendChild(corr);
    const stampWrap=el('div'); stampWrap.style.cssText='margin-top:'+cfg.layout.corrStampMarginTop+'px;min-height:'+(g.stampH+g.stampUpReserve)+'px'; stampWrap.style.paddingTop=g.stampUpReserve+'px';
    const stamp=el('span','stamp',fig.stamp); stamp.style.marginLeft=g.stampOffsetX+'px'; stamp.style.color=P.red; stamp.style.opacity=0; stamp.style.fontSize=L.bodyPx+'px'; stampWrap.appendChild(stamp);
    mount.body.appendChild(stampWrap);
    const notes=el('div'); notes.style.marginTop=cfg.layout.corrNotesMarginTop+'px'; const noteEls=fig.notes.map(n=>{const s=el('div','hand',n);s.style.cssText='font-size:'+L.handPx+'px;color:'+P.blue+';opacity:0';notes.appendChild(s);return s;}); mount.body.appendChild(notes);
    return function(p){ const st=correctionState(fig,animQ(p,fig,cfg),cfg,g.stampCap);
      corr.style.opacity=st.corr?1:0;
      stamp.style.opacity=st.stampVisible?1:0;
      const scale=st.stampScale;
      stamp.style.transform='rotate('+cfg.timing.stampRotationDeg+'deg) scale('+scale+')';
      noteEls.forEach((s,i)=>s.style.opacity=st.notes[i]); };
  }
};

  });

function render(t){
  lastT=Math.max(0,Math.min(CONFIG.duration,t)); window.__lastT=lastT;
  document.getElementById('clock').textContent=lastT.toFixed(2)+'s / '+CONFIG.duration+'s';
  const tr=transitionState(lastT,CONFIG.figures,CONFIG);
  mounted.forEach((m,i)=>{ let op=0, ty=0;
    if(i===tr.index){ op=tr.inOpacity; ty=tr.inTy; if(m.draw) m.draw(tr.inP); }
    else if(i===tr.prevIndex){ op=tr.outOpacity; ty=tr.outTy; if(m.draw) m.draw(tr.outP); }
    m.root.classList.toggle('on', op>0);
    m.root.style.opacity=op;
    m.root.style.transform=ty?('translateY('+ty+'px)'):'';
  });
}
