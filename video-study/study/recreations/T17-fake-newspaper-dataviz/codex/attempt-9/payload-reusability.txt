# T17 fake-newspaper dataviz — construction facts (neutral)

A single self-contained HTML file recreates a fake-newspaper animated dataviz: a broadsheet masthead, then six figures arriving in sequence on one stage — a line chart with a side card, a waffle chart with a typed summary card, a stacked-bars figure with a bar chart and donut, an org chart with a badge, a counter-and-graph collapse figure, and a correction figure with a stamp and margin notes. All construction is procedural from one configuration object; no specimen media, no external assets. The piece starts playing under its own startup path.

Construction facts a reviewer can verify in the source:

- Validation enumerates its required members by name: every required timing member, every required layout member (including the members of each nested layout group), every required palette and formatting member, the bounds and the type floors. A missing, null, wrongly typed, or non-finite named member is a named error carrying the member's path. Label and point arrays are shape-checked per entry — every entry of the label arrays, the point pairs, the bar values and labels, the node list, and the notes list is checked with the figure and index named in the error. The fit entry point validates before it computes any layout and returns a failure result instead of throwing when a required group or member is absent.
- One sample-count rule is shared by validation and rendering. The counter renderer's emitted points over a full draw are defined as: the initial point, one point per step sample while k times the step is below 1, and the endpoint sample. A single function computes that count from the step; the validator refuses a step whose count exceeds the named maximum in the timing configuration, and the renderer's loop is bounded by the same maximum. At a step of 1/512 a full draw emits 513 points.
- The public rebuild entry is transactional. It deep-copies the candidate, validates it, computes its fit report, and builds the complete candidate scene in a detached root that is never attached to the live stage. The live scene is replaced in one commit step only after validation, fit, and construction all succeed; the sizing for the detached build is derived from the candidate configuration's own stage width when no live viewport width exists. A candidate whose fit report fails is refused: the call returns the failure together with the fit report and the live scene stands. A candidate that fails validation returns the named errors. A candidate that throws during construction is discarded; the failure path runs no builder against the live configuration and never reconstructs the live scene. A successful swap renders the current playback position — if a preview frame stands ahead of the playback offset, the preview stands; if the piece is playing, the next animation frame continues from the pre-swap position. The no-argument rebuild form and the window-resize path run the same transactional construction.
- A complete alternate preset in the same configuration shape — masthead, date line, clock, price, every palette entry, figure titles and sources, chart titles, axis labels, point data, card text, waffle counts and callout text, stack items and labels, org nodes and badge text, counter start and end values with their labels, and the correction, stamp, and notes text — mounts through the public rebuild path, and every one of those fields' rendered effects changes with it.
- The waffle callout leader is a stepped route: it leaves the label horizontally inside its lane, runs vertically in the gap channel between two dot columns, runs horizontally in the gap channel between two dot rows, and enters its target dot at the dot's bottom edge. Its stroke width and its clearance floor are named configuration keys; the mounted census measures stroke-edge clearance for every leader segment against every non-target dot. The label is plain text with no arrow glyph.
- The fit report's rows carry, per figure, the required height, the allocated height, the occupied width, the available width, and the figure title's occupied width. Occupied width is modelled per contained element: an unbreakable text token occupies its full modelled width even when its container is narrower, wrapped text occupies its widest line, and the chart title's modelled width consumes the same letter-spacing key the renderer applies.

  const u=(sq-T.stampReboundEnd)/(1-T.stampReboundEnd); return lerp(T.stampCompress,1,1-Math.pow(1-u,T.stampEasePowOut));

function stateAt(t,figures,duration){
  const tt=Math.max(0,Math.min(duration,t));
  for(let i=0;i<figures.length;i++){const f=figures[i]; const last=i===figures.length-1;
    if(tt>=f.start && (tt<f.start+f.dur || (last && tt<=duration))) return {index:i, p:clamp01((tt-f.start)/f.dur)};}
  let idx=0; figures.forEach((f,i)=>{if(tt>=f.start) idx=i;});
  return {index:idx, p:clamp01((tt-figures[idx].start)/figures[idx].dur)};
}
function smoothstep01(v){const x=clamp01(v); return x*x*(3-2*x);}
function transitionState(t,figures,cfg){const c=cfg||CONFIG; const T=c.timing.figureTransitionSec, rise=c.layout.transitionRisePx;
  const st=stateAt(t,figures,c.duration);
  if(T>0){ for(let i=1;i<figures.length;i++){ const b=figures[i].start;
    if(t>=b-T/2 && t<=b+T/2){ const s=(t-(b-T/2))/T; const outP=smoothstep01(s/c.timing.transitionOutSpan); const inP=smoothstep01((s-c.timing.transitionInStart)/c.timing.transitionInSpan);
      const rawOut=1-outP; const rel=c.timing.transitionFloorRelease; const ramp=1-smoothstep01((s-rel)/(1-rel)); const outOpacity=Math.max(rawOut, c.timing.transitionFloorOpacity*ramp);
      return {index:i, p:clamp01((t-b)/figures[i].dur), prevIndex:i-1, outOpacity:outOpacity, outTy:-rise*outP, outP:(t>=b?1:clamp01((t-figures[i-1].start)/figures[i-1].dur)), inOpacity:inP, inTy:rise*(1-inP), inP:clamp01((t-b)/figures[i].dur), s:s}; } } }
  return {index:st.index, p:st.p, prevIndex:-1, outOpacity:0, outTy:0, outP:st.p, inOpacity:1, inTy:0, inP:st.p, s:-1};
}

function charFactor(ch,cfg,kind){const c=cfg||CONFIG; const W=c.layout.charWidths;
  if(ch===" ") return W.space;
  if("iljtf.,:;'!|()[]".indexOf(ch)>=0) return W.narrow;
  if("mwMW@#%".indexOf(ch)>=0) return W.wide;
  if(ch>="A"&&ch<="Z") return W.upper;

  if(schedFigs.length){ const f0=schedFigs[0]; if(!f0||typeof f0!=="object") errs.push("schedule: first figure record absent"); else if(f0.start!==0) errs.push("schedule: first figure "+f0.id+" start "+f0.start+" must be 0"); }
  for(let si=1;si<schedFigs.length;si++){ const prevF=schedFigs[si-1], curF=schedFigs[si]; if(!prevF||!curF||typeof prevF!=="object"||typeof curF!=="object") continue; if(Math.abs(curF.start-(prevF.start+prevF.dur))>1e-9) errs.push("schedule: figure "+curF.id+" start "+curF.start+" does not equal previous figure end "+(prevF.start+prevF.dur)+" (overlap or gap)"); }
  if(schedFigs.length){ const lastF=schedFigs[schedFigs.length-1]; if(!lastF||typeof lastF!=="object") errs.push("schedule: last figure record absent"); else if(!isFinite(lastF.start)||!isFinite(lastF.dur)) errs.push("schedule: last figure window values must be finite"); else if(Math.abs(lastF.start+lastF.dur-cfg.duration)>1e-9) errs.push("schedule: last figure end "+(lastF.start+lastF.dur)+" does not equal total duration "+cfg.duration); }
  figList.forEach(f=>{ if(!f||typeof f!=="object"||layoutBad) return; const L6=computeLayout(cfg.layout.phoneProbeWidth,cfg); const g6=figureGeom(f,L6,cfg);
    // width overflow is priced by the fit surface (fitReport widthOk + occupied), not by validation: a figure whose content exceeds the phone width is a fit failure with its occupied width on the row, not a shape error.
    const B6=cfg.bounds;
    const strFields=[["title",B6.maxTitleChars],["source",B6.maxLabelChars],["cardText",B6.maxCardChars],["summaryText",B6.maxSummaryChars],["counterLabel",B6.maxLabelChars],["counterSub",B6.maxLabelChars],["correction",B6.maxCardChars],["stamp",B6.maxStampChars],["ceo",B6.maxCeoChars],["ceoSub",B6.maxCeoChars],["badge",B6.maxLabelChars],["badgeName",B6.maxLabelChars],["onlyNote",B6.maxNoteChars],["stackLabel",B6.maxLabelChars],["donutLabel",B6.maxLabelChars],["waffleLabel",B6.maxLabelChars],["youLabel",B6.maxLabelChars],["chartTitle",B6.maxLabelChars]];
    strFields.forEach(([k,lim])=>{ if(typeof f[k]==="string" && f[k].length>lim) errs.push(f.id+": "+k+" length "+f[k].length+" exceeds supported boundary "+lim); });
    ["xLabels","barLabels"].forEach(k=>{ if(Array.isArray(f[k])) f[k].forEach((s2,i2)=>{ if(typeof s2==="string" && s2.length>B6.maxLabelChars) errs.push(f.id+": "+k+"["+i2+"] length "+s2.length+" exceeds supported boundary "+B6.maxLabelChars); }); });
    ["yLabels"].forEach(k=>{ if(Array.isArray(f[k])) f[k].forEach((o2,i2)=>{ if(o2 && typeof o2.text==="string" && o2.text.length>B6.maxLabelChars) errs.push(f.id+": "+k+"["+i2+"] length exceeds supported boundary "+B6.maxLabelChars); }); });
    if(Array.isArray(f.nodes)) f.nodes.forEach((s2,i2)=>{ if(typeof s2==="string" && s2.length>B6.maxNodeChars) errs.push(f.id+": nodes["+i2+"] length "+s2.length+" exceeds supported boundary "+B6.maxNodeChars); });
    if(Array.isArray(f.notes)) f.notes.forEach((s2,i2)=>{ if(typeof s2==="string" && s2.length>B6.maxNoteChars) errs.push(f.id+": notes["+i2+"] length "+s2.length+" exceeds supported boundary "+B6.maxNoteChars); });
  });
  return errs;
}

  if(globalProblems.length){ const gw=el('div','warn globalwarn','Validation: '+globalProblems.join('; ')); stageEl.appendChild(gw); }
  const mounted=[]; const appended=[];
  try{ cfg.figures.forEach(f=>{
    if(!f||typeof f!=="object") return;
    const root=el('section','fig'); root.id=f.id;
    const num=el('div','fignum',f.num); num.style.color=P.red; root.appendChild(num);
    const h=el('h2'); wordWrapLines(f.title,L.contentW,L.titlePx,cfg).forEach((ln,li,arr)=>{ h.appendChild(el('span','titleline',ln)); if(li<arr.length-1) h.appendChild(el('br')); });
    h.style.minHeight=(headingLines(f.title,L,cfg)*L.titlePx*cfg.layout.lineHeights.title)+'px'; root.appendChild(h);
    const body=el('div'); root.appendChild(body);
    const srcEl=el('div','src'); wordWrapLines(f.source,L.contentW,L.labelPx,cfg,"label").forEach((ln,li,arr)=>{ srcEl.appendChild(el('span','srcline',ln)); if(li<arr.length-1) srcEl.appendChild(el('br')); }); root.appendChild(srcEl);
    stageEl.appendChild(root); appended.push(root);
    const figProblems=problems.filter(x=>x.indexOf(f.id+':')===0);
    let draw=null;
    if(figProblems.length){ const warn=el('div','warn','Unsupported data — '+figProblems.join('; ')); body.appendChild(warn); if(window.console) console.warn('T17 validation: '+figProblems.join('; ')); }
    else if(BUILDERS[f.type]) draw=BUILDERS[f.type]({root:root,body:body},f,deps);
    mounted.push({fig:f,root:root,draw:draw});
  });
  } catch(e){ appended.forEach(r=>r.remove()); throw e; }
  return mounted;
}

function commitBuilt(built,cfg){
  while(stage.children.length) stage.children[0].remove();
  applyStagePresentation(stage,cfg,built.L);
  stage.style.background=built.holder.style.background; stage.style.color=built.holder.style.color;
  STAGE_VARS.forEach(k=>{ const v=built.holder.style.getPropertyValue(k); if(v) stage.style.setProperty(k,v); });

  document.body.style.background=cfg.palette.pageBg; document.getElementById('clock').style.color=cfg.palette.controlInk;
  mounted=built.mounted;
}

  const W=stage.clientWidth||CONFIG.layout.stageWidthPx;

}
function render(t){
  lastT=Math.max(0,Math.min(CONFIG.duration,t)); window.__lastT=lastT;
  document.getElementById('clock').textContent=lastT.toFixed(2)+'s / '+CONFIG.duration+'s';
  const tr=transitionState(lastT,CONFIG.figures,CONFIG);
  mounted.forEach((m,i)=>{ let op=0, ty=0;
    if(i===tr.index){ op=tr.inOpacity; ty=tr.inTy; if(m.draw) m.draw(tr.inP); }
    else if(i===tr.prevIndex){ op=tr.outOpacity; ty=tr.outTy; if(m.draw) m.draw(tr.outP); }
    m.root.classList.toggle('on', op>0);
    m.root.style.opacity=op;
    m.root.style.transform=ty?('translateY('+ty+'px)'):'';
  });
}
