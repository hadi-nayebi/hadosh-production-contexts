function scaledSchedule(cfg) {
  const s = cfg.timingScale;
  return {
    bursts: cfg.bursts.map(function (b) { return { startMs: b.startMs * s, durMs: b.durMs * s }; }),
    exit: { startMs: cfg.exit.startMs * s, durMs: cfg.exit.durMs * s },
    cycleMs: cfg.cycleMs * s
  };
}
function stateAt(ms) {
  const cfg = CONFIG;
  const sch = scaledSchedule(cfg);
  let t = ms % sch.cycleMs;
  if (t < 0) t += sch.cycleMs;
  const exitEnd = sch.exit.startMs + sch.exit.durMs;
  if (t >= exitEnd) {
    return { currentName: null, phase: "empty", visibleNames: [], outgoing: null, incoming: null, progress: 0 };
  }
  if (t >= sch.exit.startMs) {
    const p = (t - sch.exit.startMs) / sch.exit.durMs;
    const last = cfg.names[cfg.names.length - 1];
    return { currentName: last, phase: "exit", visibleNames: [last], outgoing: last, incoming: null, progress: p };
  }
  for (let i = 0; i < sch.bursts.length; i++) {
    const b = sch.bursts[i];
    const end = b.startMs + b.durMs;
    if (t >= b.startMs && t < end) {

}
