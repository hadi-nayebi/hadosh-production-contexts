/* ===== Text measurement — advance widths from the configured glyph table =====
   Widths are advance fractions of the font size for the configured font stack.
   Fit decisions multiply widths by fit.glyphErrorFactor (the upper error bound),
   so content near a boundary is rejected rather than fitted on an optimistic width. */
function tokenizeWords(text){ return String(text == null ? "" : text).split(" ").filter(Boolean); }
function isNum(v){ return typeof v === "number" && isFinite(v); }
function clampPx(minPx, vw, maxPx, widthPx){ return Math.min(Math.max(minPx, vw * widthPx / 100), maxPx); }
function glyphAdvance(ch, cfg){
  const w = cfg.glyphWidths ? cfg.glyphWidths[ch] : undefined;
  return isNum(w) ? w : cfg.fit.fallbackGlyphWidth;
}
function textWidthPx(text, fontPx, cfg, factor){
  let em = 0;
  const s = String(text == null ? "" : text);
  for (let i = 0; i < s.length; i++) em += glyphAdvance(s[i], cfg);
  return em * fontPx * (factor || 1);
}
/* Sweep thresholds on the padded advance: the bar traverses the word's padded
   box, so each character's threshold fraction is its right edge — leading
   padding, glyph advance and the letter spacing that follows the character —
   over the full padded span (both paddings, every glyph, every gap). */
function wordThresholdsPadded(word, cfg){
  const s = String(word == null ? "" : word);
  const pad = cfg.detail.wordPaddingEm;
  const spacing = parseFloat(cfg.detail.titleLetterSpacing) || 0;
  let total = pad * 2 + spacing * s.length;
  const cum = [];
  for (let i = 0; i < s.length; i++){ total += glyphAdvance(s[i], cfg); }
  if (total <= 0) return s.split("").map(function(){ return 1; });
  let run = pad;
  for (let i = 0; i < s.length; i++){ run += glyphAdvance(s[i], cfg) + spacing; cum.push(run / total); }
  return cum;
}
function wordThresholds(word, cfg){
  const s = String(word == null ? "" : word);
  let total = 0;
  const cum = [];
  for (let i = 0; i < s.length; i++){ total += glyphAdvance(s[i], cfg); cum.push(total); }
  if (total <= 0) return cum.map(function(){ return 1; });
  return cum.map(function(v){ return v / total; });
}
function wrapCount(text, fontPx, innerWidthPx, cfg, factor){
  const words = tokenizeWords(text);
  if (!words.length) return {lines:1, overflow:false};
  const spaceW = glyphAdvance(" ", cfg) * fontPx * (factor || 1);
  let lines = 1, cur = 0, overflow = false;
  words.forEach(function(w){
    const ww = textWidthPx(w, fontPx, cfg, factor);
    if (ww > innerWidthPx) overflow = true;
    if (cur === 0) cur = ww;
    else if (cur + spaceW + ww <= innerWidthPx) cur += spaceW + ww;
    else { lines++; cur = ww; }
  });
  return {lines:lines, overflow:overflow};
}

}

/* Per-word reveal onsets: characters arrive sequentially across the line at a
   uniform per-character rate; the second word additionally waits out the
   line's word-hold after the first word (and its trailing space) completes. */
function wordOnsets(line, cfg){
  const words = tokenizeWords(line.text);
  const total = String(line.text == null ? "" : line.text).length;
  const per = total > 0 ? lineRevealMs(line, cfg) / total : 0;
  const hold = lineWordHoldMs(line, cfg);
  const out = [];
  let pos = 0;
  words.forEach(function(w, wi){
    out.push({word:w, startChar:pos, onsetMs: lineRevealStart(line) + pos * per + (wi === 1 ? hold : 0),
              completeMs: lineRevealStart(line) + (pos + w.length) * per + (wi >= 1 ? hold : 0)});
    pos += w.length + 1;
  });
  return out;
}
function revealInfo(line, t, cfg){
  const start = lineRevealStart(line), dur = lineRevealMs(line, cfg);
  const total = String(line.text == null ? "" : line.text).length;
  const hold = lineWordHoldMs(line, cfg);
  /* effective span: the hold stretches the line's elapsed reveal window */
  const span = dur + (tokenizeWords(line.text).length > 1 ? hold : 0);
  let revealed = 0;
  if (t >= start){
    const per = total > 0 ? dur / total : 0;
    const words = wordOnsets(line, cfg);
    revealed = total;
    for (let wi = 0; wi < words.length; wi++){
      const w = words[wi];
      if (t < w.onsetMs) { revealed = w.startChar; break; }
      const n = Math.min(w.word.length, Math.floor((t - w.onsetMs) / per));
      if (n < w.word.length) { revealed = w.startChar + n; break; }
      revealed = w.startChar + w.word.length;
      if (wi + 1 < words.length && t < words[wi + 1].onsetMs) { revealed = w.startChar + w.word.length; break; }
    }
    if (t >= start + span) revealed = total;
  }
  const prefix = String(line.text == null ? "" : line.text).slice(0, revealed);
  return {startMs:start, durationMs:dur, spanMs:span, completeMs:start + span, totalChars:total,
          revealedChars:revealed, prefix:prefix, progress: total > 0 ? revealed / total : 1,
          complete: revealed >= total, started: t >= start};
}

    chat: clampPx(cfg.chatTextMinPx, cfg.chatTextVw, cfg.chatTextMaxPx, widthPx),
    meta: clampPx(cfg.metaMinPx, cfg.metaVw, cfg.metaMaxPx, widthPx),
    statValue: clampPx(cfg.statValueMinPx, cfg.statValueVw, cfg.statValueMaxPx, widthPx),
    statLabel: clampPx(cfg.statLabelMinPx, cfg.statLabelVw, cfg.statLabelMaxPx, widthPx),
    counter: clampPx(cfg.counterMinPx, cfg.counterVw, cfg.counterMaxPx, widthPx)
  };
  const titleColPx = screenWPx * L.titleWidthPct / 100;
  const titleAvailPx = screenWPx * (100 - L.titleLeftPct) / 100;
  const wordPadPx = d.wordPaddingEm * sizes.title * f;
  const titleWraps = [], titleWordOverflow = [];
  let titleBlockPx = 0;
  cfg.titleLines.forEach(function(line){
    tokenizeWords(line.text).forEach(function(wd){
      if (textWidthPx(wd, sizes.title, cfg, 1) + 2 * d.wordPaddingEm * sizes.title > titleColPx) titleWordOverflow.push(wd);
    });
    const w = wrapCount(line.text, sizes.title, titleColPx, cfg, f);
    const words = tokenizeWords(line.text);
    let lines = w.lines;
    /* word padding widens each rendered word; a line whose padded words no longer
       fit on one wrapped row wraps further — approximate by re-wrapping with the
       padding folded into each word's width via a widened effective font size. */
    if (!w.overflow && words.length){
      const widened = sizes.title * (titleColPx / Math.max(1, titleColPx - words.length * wordPadPx * 2));
      lines = wrapCount(line.text, widened, titleColPx, cfg, f).lines;
    }
    titleWraps.push(lines);
    titleBlockPx += lines * sizes.title * d.titleLineHeight;
  });
  const chatPad = padPair(d.chatPadding);
  const chatInnerPx = screenWPx * L.chatWidthPct / 100 - chatPad.h * 2;
  const metaRowPx = Math.max(d.avatarEm * sizes.meta, sizes.meta * d.lineHeightNormal);
  const chatWraps = [], chatWordOverflow = [], metaOverflow = [];
  let chatStackPx = 0;
  cfg.chatMessages.forEach(function(m, i){
    const w = wrapCount(m.text, sizes.chat, chatInnerPx, cfg, f);
    chatWraps.push(w.lines);
    if (w.overflow) chatWordOverflow.push(m.text);
    const metaW = d.avatarEm * sizes.meta + d.metaGapPx + textWidthPx(m.sender + " " + m.time, sizes.meta, cfg, f);
    if (metaW > chatInnerPx) metaOverflow.push(m.sender);
    chatStackPx += chatPad.v * 2 + metaRowPx + d.metaMarginPx + w.lines * sizes.chat * d.chatLineHeight;
    if (i > 0) chatStackPx += L.chatGapPx;
  });
  const statPad = padPair(d.statPadding);
  let statRowPx = 0;
  cfg.statCards.forEach(function(c, i){
    statRowPx += Math.max(textWidthPx(c.label, sizes.statLabel, cfg, f), textWidthPx(c.value, sizes.statValue, cfg, f)) + statPad.h * 2;
    if (i > 0) statRowPx += screenWPx * L.statGapPct / 100;
  });
  const statHPx = statPad.v * 2 + Math.max(sizes.statLabel, sizes.statValue) * d.lineHeightNormal;
  const counterWPx = textWidthPx(cfg.slideCounter, sizes.counter, cfg, f);
  const titleBottomPct = L.titleTopPct + titleBlockPx / screenHPx * 100;
  const chatBottomPct = L.chatTopPct + chatStackPx / screenHPx * 100;
  const statTopPct = 100 - L.statBottomPct - statHPx / screenHPx * 100;
  const rectScreen = {
    left: Math.min(L.titleLeftPct, L.statLeftPct),
    right: Math.max(100 - L.chatRightPct, 100 - L.counterRightPct, L.titleLeftPct + L.titleWidthPct,
                    L.statLeftPct + statRowPx / screenWPx * 100),
    top: Math.min(L.titleTopPct, L.chatTopPct),
    bottom: Math.max(chatBottomPct, titleBottomPct, 100 - L.statBottomPct, 100 - L.counterBottomPct)
  };
  const rect = {
    left: L.screenLeftPct + rectScreen.left * (100 - L.screenLeftPct - L.screenRightPct) / 100,
    right: L.screenLeftPct + rectScreen.right * (100 - L.screenLeftPct - L.screenRightPct) / 100,
    top: L.screenTopPct + rectScreen.top * L.screenHeightPct / 100,
    bottom: L.screenTopPct + rectScreen.bottom * L.screenHeightPct / 100
  };
  return {sizes:sizes, screenWPx:screenWPx, screenHPx:screenHPx, titleWraps:titleWraps,
          titleBlockPx:titleBlockPx, titleBottomPct:titleBottomPct, chatWraps:chatWraps,
          chatStackPx:chatStackPx, chatBottomPct:chatBottomPct, statRowPx:statRowPx,
          statTopPct:statTopPct, counterWPx:counterWPx, rect:rect, rectScreen:rectScreen,
          titleWordOverflow:titleWordOverflow, chatWordOverflow:chatWordOverflow, metaOverflow:metaOverflow};
}

const LAYOUT_KEYS = ["screenLeftPct","screenRightPct","screenTopPct","screenHeightPct","titleLeftPct","titleTopPct","titleWidthPct","chatRightPct","chatTopPct","chatWidthPct","chatGapPx","statLeftPct","statBottomPct","statGapPct","counterRightPct","counterBottomPct","spotLeftPct","spotBottomPct","spotWidthPct","spotHeightPct","speakerLeftPct","speakerBottomPct","speakerWidthPct","speakerMinPx","audienceBottomPct","audienceHeightPct","audiencePadPct"];
const DETAIL_SCHEMA = [["fontFamily","string"],["titleWeight","number"],["titleLineHeight","number"],["titleLetterSpacing","string"],["hlTopPct","number"],["hlHeightPct","number"],["screenRadius","string"],["chatPadding","padding"],["chatRadiusPx","number"],["chatLineHeight","number"],["statPadding","padding"],["statRadiusPx","number"],["audSizePx","number"],["audGapPx","number"],["audOpacity","opacity"],["perspectivePx","number"],["rotateXDeg","number"],["shadowInset","string"],["shadowOuter","string"],["statBg","color"],["statBorderColor","color"],["statBorderWidthPx","number"],["wordPaddingEm","number"],["chatOrigin","string"],["metaGapPx","number"],["metaMarginPx","number"],["metaOpacity","opacity"],["statLabelOpacity","opacity"],["statLetterSpacingEm","number"],["avatarEm","number"],["avatarColor","color"],["speakerHeadWidthPct","number"],["speakerBodyHeightPx","number"],["speakerBodyRadiusPx","number"],["speakerBodyMarginPx","number"],["counterOpacity","opacity"],["stageMinHeightPx","number"],["lineHeightNormal","number"],["statValueWeight","number"],["avatarWeight","number"],["counterWeight","number"]];

function isColor(v){
  if (typeof v !== "string") return false;
  if (/^#[0-9a-fA-F]{3}$/.test(v) || /^#[0-9a-fA-F]{6}$/.test(v)) return true;
  const m = /^rgba?\(\s*(\d{1,3})\s*,\s*(\d{1,3})\s*,\s*(\d{1,3})\s*(?:,\s*(0|1|0?\.\d+)\s*)?\)$/.exec(v);
  if (!m) return false;
  return Number(m[1]) <= 255 && Number(m[2]) <= 255 && Number(m[3]) <= 255;
}
function fitErrors(cfg){
  const errs = [];
  try {
    if (!cfg.fit || !cfg.layout || !cfg.detail || !Array.isArray(cfg.titleLines) ||
        !Array.isArray(cfg.chatMessages) || !Array.isArray(cfg.statCards)) return errs;
    const W = cfg.fit.referenceWidthPx, H = cfg.fit.minHeightPx, F = cfg.fit.glyphErrorFactor;
    if (!isNum(W) || !isNum(H) || !isNum(F)) return errs;
    const m = measureContent(cfg, W, H, F);
    if (m.titleWordOverflow.length) errs.push("measured fit: a title word exceeds the title column at the upper width bound (reject)");
    if (m.chatWordOverflow.length) errs.push("measured fit: a chat word exceeds the chat column at the upper width bound (reject)");
    if (m.metaOverflow.length) errs.push("measured fit: a chat sender/time row exceeds the chat column at the upper width bound (reject)");
    m.chatWraps.forEach(function(n, i){ if (n > cfg.fit.maxChatLinesPerMessage) errs.push("measured fit: chatMessages[" + i + "] wraps past the per-message line bound (reject)"); });
    if (m.chatBottomPct > 100 - cfg.fit.regionBottomMarginPct) errs.push("measured fit: accumulated chat height exceeds the screen at the minimum supported height (reject)");
    if (m.titleBottomPct > m.statTopPct) errs.push("measured fit: title block reaches the stat row at the minimum supported height (reject)");
    const chatLeftPct = 100 - cfg.layout.chatRightPct - cfg.layout.chatWidthPct;
    if (cfg.layout.statLeftPct + m.statRowPx / m.screenWPx * 100 > chatLeftPct) errs.push("measured fit: stat row reaches the chat column (reject)");
    const counterZonePx = m.screenWPx * (100 - cfg.layout.counterRightPct - cfg.layout.titleLeftPct - cfg.layout.titleWidthPct) / 100;
    if (m.counterWPx > counterZonePx) errs.push("measured fit: slide counter exceeds its zone (reject)");
  } catch (e) { errs.push("measured fit: model failure (" + (e && e.message) + ")"); }
  return errs;
}
function validateConfig(cfg){
  const errs = [];
  if (!cfg || typeof cfg !== "object" || Array.isArray(cfg)) return ["config not an object"];
  if (!isNum(cfg.durationMs) || cfg.durationMs <= 0) errs.push("durationMs must be a positive number");
  if (!isNum(cfg.minTailMs) || cfg.minTailMs < 0) errs.push("minTailMs must be a non-negative number");
  if (!isNum(cfg.maxIdleTailMs) || cfg.maxIdleTailMs < 0) errs.push("maxIdleTailMs must be a non-negative number");
  if (!isNum(cfg.highlightMs) || cfg.highlightMs <= 0) errs.push("highlightMs must be positive");
  if (!isNum(cfg.chatPopMs) || cfg.chatPopMs <= 0) errs.push("chatPopMs must be positive");
  if (!isNum(cfg.chatPopFrom) || !isNum(cfg.chatPopTo) || cfg.chatPopFrom <= 0 || cfg.chatPopTo <= 0 || cfg.chatPopFrom === cfg.chatPopTo) errs.push("chatPop endpoints must be two distinct positive numbers");
  if (!Number.isInteger(cfg.audienceCount) || cfg.audienceCount < 0) errs.push("audienceCount must be a non-negative integer");
  [["title", cfg.titleMinPx, cfg.titleSizeVw, cfg.titleMaxPx], ["chatText", cfg.chatTextMinPx, cfg.chatTextVw, cfg.chatTextMaxPx],
   ["meta", cfg.metaMinPx, cfg.metaVw, cfg.metaMaxPx], ["statValue", cfg.statValueMinPx, cfg.statValueVw, cfg.statValueMaxPx],
   ["statLabel", cfg.statLabelMinPx, cfg.statLabelVw, cfg.statLabelMaxPx], ["counter", cfg.counterMinPx, cfg.counterVw, cfg.counterMaxPx]
  ].forEach(function(r){
    if (!isNum(r[1]) || !isNum(r[2]) || !isNum(r[3]) || r[1] <= 0 || r[2] <= 0 || r[3] < r[1]) errs.push(r[0] + " font range must be positive with min <= max");
  });
  ["highlightColor","sweepTextColor","titleColor","defaultChatColor","bg","screenBg","chatBg","chatColor","speakerColor","spotColor","audienceColor"].forEach(function(k){
    if (!isColor(cfg[k])) errs.push(k + " must be a valid CSS colour");
  });
  if (!cfg.capacity || !Number.isInteger(cfg.capacity.maxTitleLines) || !Number.isInteger(cfg.capacity.maxMessages) ||
      !Number.isInteger(cfg.capacity.maxCuts) || !Number.isInteger(cfg.capacity.maxStatCards) ||
      cfg.capacity.maxTitleLines < 1 || cfg.capacity.maxMessages < 0 || cfg.capacity.maxCuts < 1 || cfg.capacity.maxStatCards < 0)
    errs.push("capacity limits must be integers in range");
  if (!cfg.layout || typeof cfg.layout !== "object") errs.push("layout missing");
  else LAYOUT_KEYS.forEach(function(k){
    if (!isNum(cfg.layout[k])) errs.push("layout." + k + " must be a number");
    else if (k !== "chatGapPx" && (cfg.layout[k] < 0 || cfg.layout[k] > 100)) errs.push("layout." + k + " out of percentage range");
  });
  if (!cfg.detail || typeof cfg.detail !== "object") errs.push("detail missing");
  else {
    DETAIL_SCHEMA.forEach(function(e){
      const k = e[0], v = cfg.detail[k];
      if (v === undefined || v === null) { errs.push("detail." + k + " missing"); return; }
      if (e[1] === "number" || e[1] === "opacity") {
        if (!isNum(v)) errs.push("detail." + k + " must be a number");
        else if (e[1] === "opacity" && (v < 0 || v > 1)) errs.push("detail." + k + " out of range");
        else if (DETAIL_NONNEG.indexOf(k) >= 0 && v < 0) errs.push("detail." + k + " must be non-negative");
      }
      else if (e[1] === "color") { if (!isColor(v)) errs.push("detail." + k + " must be a valid CSS colour"); }
      else if (e[1] === "padding") { if (!/^\d+(\.\d+)?px(\s+\d+(\.\d+)?px)?$/.test(String(v).trim())) errs.push("detail." + k + " must be one or two px values"); }
      else if (typeof v !== "string" || !v) errs.push("detail." + k + " must be a non-empty string");
    });
  }
  if (!cfg.fit || typeof cfg.fit !== "object") errs.push("fit missing");
  else {
    if (!isNum(cfg.fit.referenceWidthPx) || cfg.fit.referenceWidthPx <= 0) errs.push("fit.referenceWidthPx must be positive");
    if (!isNum(cfg.fit.minHeightPx) || cfg.fit.minHeightPx <= 0) errs.push("fit.minHeightPx must be positive");
    if (!Array.isArray(cfg.fit.supportedHeightsPx) || !cfg.fit.supportedHeightsPx.length || !cfg.fit.supportedHeightsPx.every(isNum)) errs.push("fit.supportedHeightsPx must be a non-empty number array");
    if (!isNum(cfg.fit.glyphErrorFactor) || cfg.fit.glyphErrorFactor < 1) errs.push("fit.glyphErrorFactor must be >= 1");
    if (!isNum(cfg.fit.perspectiveFactor) || cfg.fit.perspectiveFactor <= 0 || cfg.fit.perspectiveFactor > 1) errs.push("fit.perspectiveFactor out of range");
    if (!isNum(cfg.fit.fallbackGlyphWidth) || cfg.fit.fallbackGlyphWidth <= 0) errs.push("fit.fallbackGlyphWidth must be positive");
    if (!Number.isInteger(cfg.fit.maxChatLinesPerMessage) || cfg.fit.maxChatLinesPerMessage < 1) errs.push("fit.maxChatLinesPerMessage must be a positive integer");
    if (!isNum(cfg.fit.regionBottomMarginPct) || cfg.fit.regionBottomMarginPct < 0) errs.push("fit.regionBottomMarginPct must be non-negative");
  }
  if (!cfg.glyphWidths || typeof cfg.glyphWidths !== "object") errs.push("glyphWidths missing");
  else Object.keys(cfg.glyphWidths).forEach(function(k){ if (!isNum(cfg.glyphWidths[k]) || cfg.glyphWidths[k] <= 0 || cfg.glyphWidths[k] > 2) errs.push("glyphWidths." + k + " out of range"); });
  if (!Array.isArray(cfg.statCards)) errs.push("statCards must be an array");
  else {
    if (cfg.capacity && cfg.statCards.length > cfg.capacity.maxStatCards) errs.push("statCards exceed capacity (ill-fit policy: reject)");
    cfg.statCards.forEach(function(c, i){
      if (!c || typeof c.label !== "string" || !c.label || typeof c.value !== "string" || !c.value) errs.push("statCards[" + i + "] needs label+value strings");
    });
  }
  if (!Array.isArray(cfg.titleLines) || cfg.titleLines.length < 1) errs.push("titleLines must be a non-empty array");
  else {
    if (cfg.capacity && cfg.titleLines.length > cfg.capacity.maxTitleLines) errs.push("titleLines exceed capacity (ill-fit policy: reject)");
    let prev = -1;
    cfg.titleLines.forEach(function(L, i){
      if (!L || typeof L !== "object") { errs.push("titleLines[" + i + "] not a record"); return; }
      if (typeof L.text !== "string" || !L.text.trim()) errs.push("titleLines[" + i + "].text missing");
      if (L.highlightMs !== undefined && (!isNum(L.highlightMs) || L.highlightMs <= 0)) errs.push("titleLines[" + i + "].highlightMs must be positive");
      if (L.tMs !== undefined && L.revealStartMs !== undefined && L.tMs !== L.revealStartMs) errs.push("titleLines[" + i + "].tMs must equal revealStartMs when both are present");
      const rs = (L && typeof L === "object") ? lineRevealStart(L) : NaN;
      if (!isNum(rs)) errs.push("titleLines[" + i + "].revealStartMs not a number");
      else {
        if (rs < 0 || rs >= cfg.durationMs) errs.push("titleLines[" + i + "].revealStartMs out of range");
        if (rs <= prev) errs.push("titleLines revealStartMs must be strictly increasing (index " + i + ")");
        prev = rs;
      }
      if (L.revealMs !== undefined && (!isNum(L.revealMs) || L.revealMs <= 0)) errs.push("titleLines[" + i + "].revealMs must be positive");
      if (L.wordHoldMs !== undefined && (!isNum(L.wordHoldMs) || L.wordHoldMs < 0)) errs.push("titleLines[" + i + "].wordHoldMs must be a non-negative number");
      if (L.retainColor !== undefined && !isColor(L.retainColor)) errs.push("titleLines[" + i + "].retainColor must be a valid CSS colour");
      if (L.sweepStartMs !== undefined && (!isNum(L.sweepStartMs) || L.sweepStartMs < 0)) errs.push("titleLines[" + i + "].sweepStartMs must be a non-negative number");
      if (L.sweepStartMs !== undefined && isNum(L.sweepStartMs) && isNum(rs) && L.sweepStartMs < lineRevealComplete(L, cfg))
        errs.push("titleLines[" + i + "] sweep begins before its reveal completes (sweep/reveal check)");
      if (isNum(rs) && isNum(cfg.durationMs)) {
        const span = lineRevealMs(L, cfg) + (tokenizeWords(L.text || "").length > 1 ? lineWordHoldMs(L, cfg) : 0);
        if (rs + span > cfg.durationMs) errs.push("titleLines[" + i + "] reveal completes past durationMs");
      }
    });
    for (let i = 0; i + 1 < cfg.titleLines.length; i++){
      const a = cfg.titleLines[i], b = cfg.titleLines[i + 1];
      if (a && b && typeof a === "object" && typeof b === "object") {
        const aComplete = lineRevealStart(a) + lineRevealMs(a, cfg) + (tokenizeWords(a.text || "").length > 1 ? lineWordHoldMs(a, cfg) : 0);
        if (isNum(aComplete) && isNum(lineRevealStart(b)) && aComplete > lineRevealStart(b))
          errs.push("reveal of line " + i + " completes after the next line reveal begins (reveal neighbour check)");
      }
    }
    if (!isNum(cfg.defaultRevealMs) || cfg.defaultRevealMs <= 0) errs.push("defaultRevealMs must be positive");
    if (!isNum(cfg.defaultWordHoldMs) || cfg.defaultWordHoldMs < 0) errs.push("defaultWordHoldMs must be a non-negative number");
  }
  if (!Array.isArray(cfg.highlightTargets)) errs.push("highlightTargets must be an array");
  else if (Array.isArray(cfg.titleLines)) {
    const seen = {};
    cfg.highlightTargets.forEach(function(h, i){
      if (!h || !Number.isInteger(h.lineIndex) || !Number.isInteger(h.wordIndex)) { errs.push("highlightTargets[" + i + "] needs integer lineIndex+wordIndex"); return; }
      if (h.lineIndex < 0 || h.lineIndex >= cfg.titleLines.length) { errs.push("highlightTargets[" + i + "].lineIndex out of range"); return; }
      if (seen[h.lineIndex]) errs.push("duplicate highlight target for line " + h.lineIndex);
      seen[h.lineIndex] = true;
      const line = cfg.titleLines[h.lineIndex];
      const words = tokenizeWords(line && typeof line === "object" ? line.text : "");
      if (h.wordIndex < 0 || h.wordIndex >= words.length) errs.push("highlightTargets[" + i + "].wordIndex out of range for line " + h.lineIndex);
    });
    cfg.titleLines.forEach(function(_, i){ if (!seen[i]) errs.push("line " + i + " has no highlight target"); });
  }
  if (!Array.isArray(cfg.chatMessages)) errs.push("chatMessages must be an array");
  else {
    if (cfg.capacity && cfg.chatMessages.length > cfg.capacity.maxMessages) errs.push("chatMessages exceed capacity (ill-fit policy: reject)");
    let prev = -1;
    cfg.chatMessages.forEach(function(m, i){
      if (!m || typeof m !== "object") { errs.push("chatMessages[" + i + "] not a record"); return; }
      if (typeof m.text !== "string" || !m.text) errs.push("chatMessages[" + i + "].text missing");
      if (typeof m.sender !== "string" || !m.sender) errs.push("chatMessages[" + i + "].sender missing");
      if (typeof m.time !== "string" || !m.time) errs.push("chatMessages[" + i + "].time missing");
      if (!isColor(m.color)) errs.push("chatMessages[" + i + "].color must be a valid CSS colour");
      if (m.popMs !== undefined && (!isNum(m.popMs) || m.popMs <= 0)) errs.push("chatMessages[" + i + "].popMs must be positive");
      if (!isNum(m.tMs)) errs.push("chatMessages[" + i + "].tMs not a number");
      else {
        if (m.tMs < 0 || m.tMs >= cfg.durationMs) errs.push("chatMessages[" + i + "].tMs out of range");
        if (m.tMs <= prev) errs.push("chatMessages tMs must be strictly increasing (index " + i + ")");
        prev = m.tMs;
      }
    });
    for (let i = 0; i + 1 < cfg.chatMessages.length; i++){
      const a = cfg.chatMessages[i], b = cfg.chatMessages[i + 1];
      if (a && b && isNum(a.tMs) && isNum(b.tMs) && a.tMs + msgPopMs(a, cfg) > b.tMs)
        errs.push("pop of message " + i + " runs into the next message landing (neighbour check)");
    }
  }
  if (typeof cfg.slideCounter !== "string" || !cfg.slideCounter) errs.push("slideCounter must be a non-empty string");
  const framingFields = ["scale","x","y","speakerXPct","spotScale","audienceShiftPct"];
  function checkFraming(f, where){
    if (!f || typeof f !== "object") { errs.push(where + " missing"); return; }
    framingFields.forEach(function(k){ if (!isNum(f[k])) errs.push(where + "." + k + " not a number"); });
    if (isNum(f.scale) && f.scale <= 0) errs.push(where + ".scale must be positive");
    if (isNum(f.spotScale) && f.spotScale <= 0) errs.push(where + ".spotScale must be positive");
  }
  checkFraming(cfg.initialFraming, "initialFraming");
  if (!Array.isArray(cfg.cuts) || cfg.cuts.length < 1) errs.push("cuts must be a non-empty array");
  else {
    if (cfg.capacity && cfg.cuts.length > cfg.capacity.maxCuts) errs.push("cuts exceed capacity (ill-fit policy: reject)");
    let prev = 0;
    cfg.cuts.forEach(function(c, i){
      if (!c || typeof c !== "object") { errs.push("cuts[" + i + "] not a record"); return; }
      if (!isNum(c.tMs)) errs.push("cuts[" + i + "].tMs not a number");
      else {
        if (c.tMs <= 0 || c.tMs >= cfg.durationMs) errs.push("cuts[" + i + "].tMs out of range (cuts are events after the initial framing)");
        if (c.tMs <= prev) errs.push("cuts tMs must be strictly increasing (index " + i + ")");
        prev = c.tMs;
      }
      checkFraming(c, "cuts[" + i + "]");
    });
  }
  if (isNum(cfg.framingMinDelta) && cfg.framingMinDelta >= 0 && cfg.initialFraming && Array.isArray(cfg.cuts) && cfg.cuts.length &&
      framingFields.every(function(k){ return isNum(cfg.initialFraming[k]); })) {
    const seqF = [cfg.initialFraming].concat(cfg.cuts);
    for (let i = 1; i <= seqF.length; i++){
      const a = seqF[i - 1], b = seqF[i % seqF.length];
      if (a && b && typeof a === "object" && typeof b === "object" &&
          framingFields.every(function(k){ return isNum(a[k]) && isNum(b[k]); }) && framingDelta(a, b) < cfg.framingMinDelta)
        errs.push("framing distinction below framingMinDelta at transition " + (i - 1) + "->" + (i % seqF.length));
    }
  } else if (!isNum(cfg.framingMinDelta) || cfg.framingMinDelta < 0) errs.push("framingMinDelta must be a non-negative number");
  if (isNum(cfg.durationMs)) {
    let lastEnd = 0, lastMark = 0;
    if (Array.isArray(cfg.titleLines)) cfg.titleLines.forEach(function(L){ if (L && typeof L === "object" && isNum(lineRevealStart(L))) { lastEnd = Math.max(lastEnd, lineRevealComplete(L, cfg), lineSweepStart(L, cfg) + lineHighlightMs(L, cfg)); lastMark = Math.max(lastMark, lineRevealStart(L)); } });
    if (Array.isArray(cfg.chatMessages)) cfg.chatMessages.forEach(function(m){ if (m && isNum(m.tMs)) { lastEnd = Math.max(lastEnd, m.tMs + msgPopMs(m, cfg)); lastMark = Math.max(lastMark, m.tMs); } });
    if (Array.isArray(cfg.cuts)) cfg.cuts.forEach(function(c){ if (c && isNum(c.tMs)) { lastEnd = Math.max(lastEnd, c.tMs); lastMark = Math.max(lastMark, c.tMs); } });
    if (lastEnd + (isNum(cfg.minTailMs) ? cfg.minTailMs : 0) > cfg.durationMs)
      errs.push("durationMs too short for configured events (short-duration policy: reject)");
    if (isNum(cfg.maxIdleTailMs) && cfg.durationMs - lastEnd > cfg.maxIdleTailMs)
      errs.push("idle tail after the last event exceeds maxIdleTailMs (density policy: reject)");
  }
  if (!errs.length) errs.push.apply(errs, fitErrors(cfg));
  return errs;
}

/* ===== Framing clamp — the measured content bounds stay inside the stage =====
   The clamped rectangle comes from measureContent (the configured title lines,
   chat rows, stat row and counter at their configured sizes), and the displayed
   bounds follow the composed transform exactly: with the origin at the stage
   centre, a point p displays at 50 + scale * (p - 50 + translate), because the
   composed scale scales the translation with it. */
function clampFraming(f, cfg){
  const m = measureContent(cfg, cfg.fit.referenceWidthPx, cfg.fit.minHeightPx, 1);
  const b = m.rect;
  const w = b.right - b.left, h = b.bottom - b.top;
  let scale = Math.min(f.scale, 100 / w, 100 / h);
  let x = f.x, y = f.y;
  function rect(){
    return {left: 50 + scale * (b.left - 50 + x), right: 50 + scale * (b.right - 50 + x),
            top: 50 + scale * (b.top - 50 + y), bottom: 50 + scale * (b.bottom - 50 + y)};
  }
  let r = rect();
  if (r.left < 0) x += -r.left / scale;
  r = rect();
  if (r.right > 100) x -= (r.right - 100) / scale;
  r = rect();
  if (r.top < 0) y += -r.top / scale;
  r = rect();
  if (r.bottom > 100) y -= (r.bottom - 100) / scale;
  return {scale:scale, x:x, y:y, speakerXPct:f.speakerXPct, spotScale:f.spotScale, audienceShiftPct:f.audienceShiftPct, rect:rect(), contentRect:b};
}
function framingDelta(a, b){
  return Math.abs(a.scale - b.scale) + (Math.abs(a.x - b.x) + Math.abs(a.y - b.y)) / 100 +
         Math.abs(a.speakerXPct - b.speakerXPct) / 100 + Math.abs(a.spotScale - b.spotScale);
}

/* Per-character sweep colour: a character turns dark only once the bar has covered
   its own right edge, measured on proportional glyph widths — uncovered letters
   stay in the title colour against the dark screen throughout the sweep. */
function charColor(progress, phase, thresholds, charIndex, cfg, retainColor){
  const retain = typeof retainColor === "string" ? retainColor : cfg.titleColor;
  if (phase !== "sweeping") return retain;
  return (progress >= thresholds[charIndex]) ? cfg.sweepTextColor : retain;
}

/* Pure state from time — checkpoint tables decide everything. */
function computeState(tMs, cfg){
  const errs = validateConfig(cfg);
  if (errs.length) throw new Error("invalid CONFIG: " + errs.join("; "));
  const t = ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs;
  const titleVisible = cfg.titleLines.map(function(L){ return t >= lineRevealStart(L); });
  const titleLines = cfg.titleLines.map(function(L){
    const info = revealInfo(L, t, cfg);
    return {text:L.text, visible:info.started, revealStartMs:info.startMs, revealCompleteMs:info.completeMs,
            revealProgress:info.progress, revealedChars:info.revealedChars, totalChars:info.totalChars,
            prefix:info.prefix, complete:info.complete, retainColor:lineRetainColor(L, cfg),
            words: wordOnsets(L, cfg).map(function(w){
              const n = Math.max(0, Math.min(w.word.length, info.revealedChars - w.startChar));
              return {text:w.word, onsetMs:w.onsetMs, completeMs:w.completeMs, revealedChars:n, visibleText:w.word.slice(0, n)};
            })};
  });
  const hlByLine = {};
  cfg.highlightTargets.forEach(function(h){ hlByLine[h.lineIndex] = h; });
  const highlights = cfg.highlightTargets.map(function(h){
    const line = cfg.titleLines[h.lineIndex];
    const start = lineSweepStart(line, cfg);
    const dur = lineHighlightMs(line, cfg);
    let progress = 0, phase = "pending";
    if (t >= start + dur) { progress = 1; phase = "swept"; }
    else if (t >= start) { progress = (t - start) / dur; phase = "sweeping"; }
    const words = tokenizeWords(line.text);
    const word = words[h.wordIndex] || "";
    return {lineIndex:h.lineIndex, wordIndex:h.wordIndex, progress:progress, phase:phase, durationMs:dur, startMs:start,
            barWidthPct: phase === "sweeping" ? progress * 100 : 0,
            charThresholds: wordThresholdsPadded(word, cfg),
            textColor: phase === "sweeping" ? cfg.sweepTextColor : (phase === "swept" ? lineRetainColor(line, cfg) : lineRetainColor(line, cfg))};
  });
  const chatVisible = cfg.chatMessages.map(function(m){ return t >= m.tMs; });
  const chatScales = cfg.chatMessages.map(function(m){
    if (t < m.tMs) return cfg.chatPopFrom;
    const age = t - m.tMs;
    const dur = msgPopMs(m, cfg);
    return age >= dur ? cfg.chatPopTo : cfg.chatPopFrom + (cfg.chatPopTo - cfg.chatPopFrom) * (age / dur);
  });
  let cutIndex = -1;
  for (let i = 0; i < cfg.cuts.length; i++){ if (cfg.cuts[i].tMs <= t) cutIndex = i; else break; }
  const rawFraming = cutIndex < 0 ? cfg.initialFraming : cfg.cuts[cutIndex];
  const framing = clampFraming(rawFraming, cfg);
  return {t:t, titleVisible:titleVisible, titleLines:titleLines, highlights:highlights,
          chatVisible:chatVisible, chatScales:chatScales,
          cutIndex:cutIndex, cutCount:cfg.cuts.length, framing:framing};
}

/* Split on UNESCAPED delimiters only, keeping escape pairs verbatim in the
   fields; each field is unescaped once, after every split level is done. */
function splitEscaped(str, delim){
  const out = [];
  let cur = "", esc = false;
  const s = String(str == null ? "" : str);
  for (let i = 0; i < s.length; i++){
    const ch = s[i];
    if (esc) { cur += "\\" + ch; esc = false; }
    else if (ch === "\\") esc = true;
    else if (ch === delim) { out.push(cur); cur = ""; }
    else cur += ch;
  }
  if (esc) cur += "\\";
  out.push(cur);
  return out;
}
function unescapeField(s){
  let out = "", esc = false;
  const str = String(s == null ? "" : s);
  for (let i = 0; i < str.length; i++){
    const ch = str[i];
    if (esc) { out += ch; esc = false; }
    else if (ch === "\\") esc = true;
    else out += ch;
  }
  if (esc) out += "\\";
  return out;
}

        cs.style.visibility = revealed ? "visible" : "hidden";
        cs.style.color = (hl && wi === hl.wordIndex)
          ? charColor(hl.progress, hl.phase, hl.charThresholds, ci, CONFIG, L.retainColor)
          : L.retainColor;
      });
    });
  });
  wordEls.forEach(function(w){
    const hl = s.highlights.filter(function(h){ return h.lineIndex === w.line; })[0];
    w.bar.style.width = (hl.barWidthPct).toFixed(3) + "%";
  });
  s.chatVisible.forEach(function(v, i){
    chatEls[i].style.display = v ? "block" : "none";
    if (v) chatEls[i].style.transform = "scale(" + s.chatScales[i].toFixed(3) + ")";
  });
  const f = s.framing;
  world.style.transform = "scale(" + f.scale + ") translate(" + f.x + "%, " + f.y + "%)";
  speakerEl.style.transform = "translateX(calc(-50% + " + f.speakerXPct + "%))";
  spotEl.style.transform = "translateX(-50%) scale(" + f.spotScale + ")";
  audienceEl.style.transform = "translateX(" + f.audienceShiftPct + "%)";
  return s;
}

let playing = false, startStamp = 0, pausedAt = 0, rafId = null;
function nowMs(){ return (typeof performance !== "undefined" && performance.now) ? performance.now() : Date.now(); }
function tick(){ if (!playing) return; render(nowMs() - startStamp); rafId = requestAnimationFrame(tick); }
function play(){ if (playing) return; playing = true; startStamp = nowMs() - pausedAt; tick(); }
function pause(){ if (!playing) return; pausedAt = nowMs() - startStamp; playing = false; if (rafId) cancelAnimationFrame(rafId); rafId = null; }
function restart(){ pausedAt = 0; if (playing){ startStamp = nowMs(); } else { render(0); } }
window.addEventListener("message", function(e){
  if (!e.data || typeof e.data !== "object") return;
  if (e.data.cmd === "play") play();
  if (e.data.cmd === "pause") pause();
  if (e.data.cmd === "restart") restart();
});
render(0);
play();
