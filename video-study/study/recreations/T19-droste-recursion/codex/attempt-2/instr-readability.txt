function hexToRgba(hex, alpha){
  const h = hex.replace("#", "");

/* Checkpoint coherence — EXPLICIT REJECT/REPAIR, roles never reassigned, never sorted.
   Domain: four finite numbers; window [0, durationMs]; strictly increasing.
   Policy: (1) any non-finite value -> FALLBACK to DEFAULT_CHECKPOINTS.
   (2) otherwise REPAIR IN PLACE: clamp into the window, then a forward pass
   enforces minSeparationMs preserving the supplied order and each checkpoint's
   ROLE, then a backward pass from durationMs if the forward pass overran.
   (3) if separation cannot fit the window, separation shrinks to
   floor(durationMs/4) and the repair is flagged separationReduced; if even the
   defaults cannot fit, status stays "fallback-defaults" with that flag.
   Status is "coherent" (input untouched), "repaired" (effective != supplied,
   distinguishable from untouched), or "fallback-defaults". The effective table —
   including effectiveAnchorMs — is what computeState plays and what diagnostics
   surface during playback. */
function checkpointDiagnostics(cfg){
  const names = ["pushStartMs", "anchorMs", "pushEndMs", "holdEndMs"];
  const raw = names.map(n => cfg.checkpoints[n]);
  const errors = [];
  let sep = cfg.checkpointPolicy.minSeparationMs;
  let separationReduced = false;
  if (raw.some(v => !isFinite(v))){
    errors.push("non-finite checkpoint");
    /* fallback defaults pass through the SAME duration-aware repair as any
       supplied table, so a short durationMs can never strand them outside
       the window or collapse their separation */
    const sub = { checkpoints: DEFAULT_CHECKPOINTS, durationMs: cfg.durationMs, checkpointPolicy: cfg.checkpointPolicy };
    const repaired = checkpointDiagnostics(sub);
    return { effective: repaired.effective, status: "fallback-defaults", errors: errors,
             effectiveAnchorMs: repaired.effectiveAnchorMs,
             separationReduced: repaired.separationReduced, supplied: raw.slice() };
  }
  raw.forEach((v, i) => { if (v < 0 || v > cfg.durationMs) errors.push("out-of-window:" + names[i]); });
  for (let i = 1; i < raw.length; i++) if (!(raw[i] > raw[i - 1])) errors.push("not-ordered:" + names[i]);
  if (3 * sep > cfg.durationMs){ sep = Math.floor(cfg.durationMs / 4); separationReduced = true; }
  const v = raw.map(x => Math.max(0, Math.min(cfg.durationMs, x)));
  for (let i = 1; i < v.length; i++) v[i] = Math.max(v[i], v[i - 1] + sep);
  if (v[v.length - 1] > cfg.durationMs){
    v[v.length - 1] = cfg.durationMs;
    for (let i = v.length - 2; i >= 0; i--) v[i] = Math.min(v[i], v[i + 1] - sep);
  }
  const changed = v.some((x, i) => x !== raw[i]);
  const eff = {}; names.forEach((n, i) => eff[n] = v[i]);
  return { effective: eff, status: changed ? "repaired" : "coherent", errors: errors,
           effectiveAnchorMs: eff.anchorMs, separationReduced: separationReduced, supplied: raw.slice() };
}

}

function smoothstep(x){ x = Math.max(0, Math.min(1, x)); return x * x * (3 - 2 * x); }

/* Label legibility — the nesting factor is applied EXACTLY ONCE: the recursive
   call already passes this level's true scene width (parent scene width x
   levelScale), so the font is simply sceneWidth x labelFontFactor. The floor is
   enforced at depths 0 and 1 only. Effective sizes at a 360 px stage:
   level 0 scene 360.00 px -> 18.00 px · level 1 scene 174.96 px -> raw 8.75 px,
   floored to 11 px · level 2 scene 85.03 px -> 4.25 px (glyph-band, not legible;
   the legibility claim is three SCENE levels, matching the specimen still). */
function sceneWidthAtLevel(stageWidthPx, level, cfg){ return stageWidthPx * Math.pow(cfg.levelScale, level); }
function labelFontPx(stageWidthPx, level, cfg){
  const raw = stageWidthPx * cfg.labelFontFactor;
  if (level <= 1) return Math.max(cfg.labelMinPx, raw);
  return raw;
}

/* Displayed-state geometry model (source-bound): stage is 16:9, so stage height
   = width x 9/16. Header fit = phrase width AND header height at each level.
   Phrase width model: Georgia capitals average 0.62 em per glyph plus
   letter-spacing per glyph — a conservative upper-bound model, asserted in the
   own-check suite for levels 0-1 at 360 and 390 px. */
function headerModel(stageWidthPx, level, cfg){
  const G = cfg.geometry;
  const sceneW = sceneWidthAtLevel(stageWidthPx, level, cfg);
  const sceneH = sceneW * 9 / 16;
  const frameW = sceneW * G.frameWidthPercent / 100;
  const frameH = sceneH * G.frameHeightPercent / 100;
  const headerH = frameH * G.frameHeaderPercent / 100;
  const font = labelFontPx(sceneW, level, cfg);
  const phraseW = cfg.labelText.length * font * (0.62 + cfg.typography.letterSpacingEm);
  const borderPx = Math.max(G.frameBorderMinPx, G.frameBorderPx * Math.pow(cfg.levelScale, level));
  return { sceneW: sceneW, frameW: frameW, frameH: frameH, headerH: headerH, fontPx: font,
           phraseW: phraseW, fitsWidth: phraseW <= frameW - 2 * borderPx,
           fitsHeight: font * cfg.typography.lineHeight <= headerH, borderPx: borderPx };
}
function deepestDetailModel(stageWidthPx, cfg){
  /* displayed widths at the deepest scene level, at the opening scale zoomStart */
  const sceneW = sceneWidthAtLevel(stageWidthPx, cfg.depth - 1, cfg);
  const candleGroupW = sceneW * cfg.geometry.candleWidthPercent / 100;
  return { sceneW: sceneW,
           candleBodyPx: candleGroupW * cfg.components.candle.widthPercent / 100 * cfg.zoomStart,
           staffPx: sceneW * cfg.geometry.staffWidthPercent / 100 * cfg.zoomStart };
}

/* Pure state from time — the EFFECTIVE checkpoint table decides the push.
   Phases: settle -> push -> hold -> loopHold (continued hold; at durationMs the
   loop HARD-RESTARTS to zoomStart — named, not disguised as a reset). */
function computeState(tMs, cfg){
  const t = ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs;
  const diag = checkpointDiagnostics(cfg);
  const cp = diag.effective;
  let progress = 0;
  if (t >= cp.pushEndMs) progress = 1;
  else if (t > cp.pushStartMs) progress = smoothstep((t - cp.pushStartMs) / (cp.pushEndMs - cp.pushStartMs));
  const scale = cfg.zoomStart + (cfg.zoomEnd - cfg.zoomStart) * progress;
  const phase = t < cp.pushStartMs ? "settle" : (t < cp.pushEndMs ? "push" : (t < cp.holdEndMs ? "hold" : "loopHold"));
  return { t: t, progress: progress, scale: scale, phase: phase, depth: cfg.depth,
           checkpointsCoherent: diag.status === "coherent", checkpointStatus: diag.status,
           effectiveCheckpoints: cp, effectiveAnchorMs: diag.effectiveAnchorMs,
           checkpointErrors: diag.errors, separationReduced: diag.separationReduced };
}

/* Count-derived spacing: the CONFIG step is a CAP; the effective step shrinks so

}

    btn.style.background = P.button; btn.style.borderRadius = "50%";
    figure.appendChild(btn);
  }
  scene.appendChild(figure);
  const table = el("div", "abs");
  box(table, G.tableLeftPercent, G.tableTopPercent, G.tableWidthPercent, G.tableHeightPercent);
  table.style.background = P.table; table.style.borderRadius = G.tableRadiusPercent + "% " + G.tableRadiusPercent + "% 0% 0%";
  scene.appendChild(table);
  /* The framed picture — its interior is THIS renderer, one level deeper. */
  const frameBox = el("div", "abs");
  box(frameBox, G.frameLeftPercent, G.frameTopPercent, G.frameWidthPercent, G.frameHeightPercent);
  frameBox.style.background = P.frameBg;
  const borderPx = Math.max(G.frameBorderMinPx, G.frameBorderPx * Math.pow(cfg.levelScale, level));
  frameBox.style.border = borderPx + "px solid " + P.frameBorder;
  const header = el("div", "abs"); box(header, 0, 0, 100, G.frameHeaderPercent);
  header.style.background = P.frameBg;
  header.style.display = "flex"; header.style.alignItems = "center"; header.style.justifyContent = "center";
  const medallion = el("span", "medallion");
  medallion.textContent = cfg.medallionText; medallion.style.color = P.label;
  medallion.style.fontSize = Math.max(cfg.typography.medallionMinPx, labelFontPx(stageWidthPx, level, cfg) * cfg.typography.medallionFactor) + "px";
  medallion.style.lineHeight = String(cfg.typography.lineHeight);
  header.appendChild(medallion);
  const label = el("div", "framelabel");
  label.textContent = cfg.labelText;
  label.style.color = P.label; label.style.letterSpacing = cfg.typography.letterSpacingEm + "em";
  label.style.whiteSpace = "nowrap"; label.style.lineHeight = String(cfg.typography.lineHeight);
  label.style.fontSize = labelFontPx(stageWidthPx, level, cfg) + "px";
  header.appendChild(label);
  frameBox.appendChild(header);
  const curtainL = el("div", "abs"); box(curtainL, 0, G.frameHeaderPercent, G.curtainWidthPercent, 100 - G.frameHeaderPercent);
  curtainL.style.background = P.curtain; frameBox.appendChild(curtainL);
  const curtainR = el("div", "abs");
  curtainR.style.right = "0%"; curtainR.style.top = G.frameHeaderPercent + "%";
  curtainR.style.width = G.curtainWidthPercent + "%"; curtainR.style.height = (100 - G.frameHeaderPercent) + "%";
  curtainR.style.background = P.curtain; frameBox.appendChild(curtainR);
  const interior = el("div", "frameInterior");
  interior.style.left = G.curtainWidthPercent + "%"; interior.style.top = G.frameHeaderPercent + "%";
  interior.style.width = (100 - 2 * G.curtainWidthPercent) + "%"; interior.style.height = (100 - G.frameHeaderPercent) + "%";
  interior.style.background = P.frameInner;
  if (level + 1 < cfg.depth){
    interior.appendChild(buildScene(cfg, level + 1, stageWidthPx * cfg.levelScale));
  }
  frameBox.appendChild(interior);
  scene.appendChild(frameBox);
  return scene;
}

let lastDiagKey = "";
function surfaceDiagnostics(s){
  const diag = { checkpointStatus: s.checkpointStatus, effectiveCheckpoints: s.effectiveCheckpoints,
                 effectiveAnchorMs: s.effectiveAnchorMs, checkpointErrors: s.checkpointErrors,
                 separationReduced: s.separationReduced, overrideDiagnostics: OVERRIDE_DIAGNOSTICS };
  window.__T19_DIAGNOSTICS = diag;
  const key = JSON.stringify(diag);
  if (key === lastDiagKey) return;
  lastDiagKey = key;
  if (s.checkpointStatus !== "coherent" || OVERRIDE_DIAGNOSTICS.length){
    diagEl.style.display = "block";
    diagEl.textContent = "checkpoints: " + s.checkpointStatus + " · effective anchor " + s.effectiveAnchorMs +
      "ms · push " + s.effectiveCheckpoints.pushStartMs + "–" + s.effectiveCheckpoints.pushEndMs +
      "ms · hold ends " + s.effectiveCheckpoints.holdEndMs + "ms" +
      (s.checkpointErrors.length ? " · " + s.checkpointErrors.join(", ") : "") +
      (OVERRIDE_DIAGNOSTICS.length ? " · " + OVERRIDE_DIAGNOSTICS.join(", ") : "");
    if (typeof console !== "undefined" && console.warn) console.warn("T19 diagnostics", diag);
  } else {
    diagEl.style.display = "none"; diagEl.textContent = "";
  }
}

function render(tMs){
  const s = computeState(tMs, CONFIG);
  zoomWrap.style.transform = "scale(" + s.scale.toFixed(4) + ")";
  surfaceDiagnostics(s);
  const flames = zoomWrap.querySelectorAll ? zoomWrap.querySelectorAll(".flame") : [];
  for (let i = 0; i < flames.length; i++){
    const phase = parseFloat(flames[i].dataset.phase || "0");
    const v = CONFIG.flicker.amp * Math.sin(s.t / CONFIG.flicker.periodDivisorMs + phase);
    flames[i].style.opacity = String(1 - Math.abs(v));
  }
  return s;
}

let playing = false, startStamp = 0, pausedAt = 0, rafId = null;
function nowMs(){ return (typeof performance !== "undefined" && performance.now) ? performance.now() : Date.now(); }
function tick(){ if (!playing) return; render(nowMs() - startStamp); rafId = requestAnimationFrame(tick); }
function play(){ if (playing) return; playing = true; startStamp = nowMs() - pausedAt; tick(); }
function pause(){ if (!playing) return; pausedAt = nowMs() - startStamp; playing = false; if (rafId) cancelAnimationFrame(rafId); rafId = null; }
function restart(){ pausedAt = 0; if (playing){ startStamp = nowMs(); } else { render(0); } }

window.addEventListener("message", function(e){
  if (!e.data || typeof e.data !== "object") return;
  if (e.data.cmd === "play") play();
  if (e.data.cmd === "pause") pause();
  if (e.data.cmd === "restart") restart();
});
render(0);
play();
