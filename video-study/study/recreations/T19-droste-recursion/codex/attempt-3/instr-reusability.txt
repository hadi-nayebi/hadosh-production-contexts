const OVERRIDE_DIAGNOSTICS = [];
(function applyOverrides(){
  /* Each URL parameter is read ONCE and parsed STRICTLY: a value must be a
     complete finite number ("4junk" is rejected and reported, never truncated
     to 4). Truncation/clamping of integer fields reports requested vs
     effective values in the diagnostics record. */

  function strictNum(k){

/* Checkpoint coherence — EXPLICIT REJECT/REPAIR, roles never reassigned, never sorted.
   Domain: four finite numbers; window [0, durationMs]; strictly increasing.
   Policy: (1) any non-finite value -> FALLBACK to DEFAULT_CHECKPOINTS.
   (2) otherwise REPAIR IN PLACE: clamp into the window, then a forward pass
   enforces minSeparationMs preserving the supplied order and each checkpoint's
   ROLE, then a backward pass from durationMs if the forward pass overran.
   (3) if separation cannot fit the window, separation shrinks to
   floor(durationMs/4) and the repair is flagged separationReduced; if even the
   defaults cannot fit, status stays "fallback-defaults" with that flag.
   Status is "coherent" (input untouched), "repaired" (effective != supplied,
   distinguishable from untouched), or "fallback-defaults". The effective table —
   including effectiveAnchorMs — is what computeState plays and what diagnostics
   surface during playback. */
function checkpointDiagnostics(cfg){
  const names = ["pushStartMs", "anchorMs", "pushEndMs", "holdEndMs"];
  const raw = names.map(n => cfg.checkpoints[n]);
  const errors = [];
  let sep = cfg.checkpointPolicy.minSeparationMs;
  let separationReduced = false;
  if (raw.some(v => !isFinite(v))){
    errors.push("non-finite checkpoint");
    /* fallback defaults pass through the SAME duration-aware repair as any
       supplied table, so a short durationMs can never strand them outside
       the window or collapse their separation */
    const sub = { checkpoints: DEFAULT_CHECKPOINTS, durationMs: cfg.durationMs, checkpointPolicy: cfg.checkpointPolicy };
    const repaired = checkpointDiagnostics(sub);
    return { effective: repaired.effective, status: "fallback-defaults", errors: errors,
             effectiveAnchorMs: repaired.effectiveAnchorMs,
             separationReduced: repaired.separationReduced, supplied: raw.slice(),
             allocationDegraded: repaired.allocationDegraded };
  }
  raw.forEach((v, i) => { if (v < 0 || v > cfg.durationMs) errors.push("out-of-window:" + names[i]); });
  for (let i = 1; i < raw.length; i++) if (!(raw[i] > raw[i - 1])) errors.push("not-ordered:" + names[i]);
  if (3 * sep > cfg.durationMs){ sep = Math.floor(cfg.durationMs / 4); separationReduced = true; }
  const v = raw.map(x => Math.max(0, Math.min(cfg.durationMs, x)));
  for (let i = 1; i < v.length; i++) v[i] = Math.max(v[i], v[i - 1] + sep);
  if (v[v.length - 1] > cfg.durationMs){
    v[v.length - 1] = cfg.durationMs;
    for (let i = v.length - 2; i >= 0; i--) v[i] = Math.min(v[i], v[i + 1] - sep);
  }
  const changed = v.some((x, i) => x !== raw[i]);
  const eff = {}; names.forEach((n, i) => eff[n] = v[i]);
  /* Temporal-suitability FLAG (no auto-replacement: supplied checkpoint roles
     are preserved; a degraded allocation is reported, never silently replaced). */
  const allocationDegraded = (eff.pushEndMs - eff.pushStartMs) < 0.25 * cfg.durationMs
    || (eff.holdEndMs - eff.pushEndMs) < 0.10 * cfg.durationMs;
  return { effective: eff, status: changed ? "repaired" : "coherent", errors: errors,
           effectiveAnchorMs: eff.anchorMs, separationReduced: separationReduced, supplied: raw.slice(),
           allocationDegraded: allocationDegraded };
}
function validateCheckpoints(cfg){
  const d = checkpointDiagnostics(cfg);
  return { coherent: d.status === "coherent", errors: d.errors, status: d.status };
}
function effectiveCheckpoints(cfg){ return checkpointDiagnostics(cfg).effective; }

/* CONFIG resolution — FIELD-SPECIFIC validation, run once before first render
   (direct CONFIG edits included, not only URL overrides). Every rejection
   falls back PER FIELD to the authored default snapshot and is recorded in
   the returned diagnostics list; one rejected field never discards unrelated
   valid edits. Field table:
     durationMs            finite, > 0            (dependent: checkpoint window)
     minSeparationMs       finite, > 0            (dependent: if 3x separation
                            exceeds durationMs the repair flags
                            separationReduced instead of rejecting)
     checkpoints.<each>    finite number          (ordering handled by repair)
     depth                 integer, clamped 1..6  (clamp reported)
     candleCount           integer, clamped 0..8  (0 is a VALID zero)
     figureButtons         integer, clamped 0..6  (0 is a VALID zero)
     zoomStart / zoomEnd   finite, > 0
     flicker.amp           finite, >= 0           (0 is a VALID zero)
     labelFontFactor       finite, > 0
     labelMinPx            finite, > 0
     playbackMode          "once" | "loop"
   A null/absent checkpoints or checkpointPolicy group falls back as a group
   and never throws. */
function resolveConfig(cfg, defaults){
  const diags = [];
  const bad = (v) => typeof v !== "number" || !isFinite(v);
  function numField(obj, key, path, rule, fallback){
    const v = obj[key];
    if (bad(v) || !rule(v)){ diags.push(path + " rejected: supplied " + String(v) + " -> fallback " + fallback + " (authored default)"); obj[key] = fallback; }
  }
  function intField(obj, key, path, min, max, fallback){
    const v = obj[key];
    if (bad(v)){ diags.push(path + " rejected: supplied " + String(v) + " -> fallback " + fallback + " (authored default)"); obj[key] = fallback; return; }
    const eff = Math.max(min, Math.min(max, Math.floor(v)));
    if (eff !== v) diags.push(path + " clamped: requested " + v + " -> effective " + eff);
    obj[key] = eff;
  }
  if (!cfg.checkpoints || typeof cfg.checkpoints !== "object"){
    diags.push("checkpoints rejected: group absent -> fallback authored table");
    cfg.checkpoints = JSON.parse(JSON.stringify(defaults.checkpoints));
  }
  if (!cfg.checkpointPolicy || typeof cfg.checkpointPolicy !== "object"){
    diags.push("checkpointPolicy rejected: group absent -> fallback authored group");
    cfg.checkpointPolicy = JSON.parse(JSON.stringify(defaults.checkpointPolicy));
  }
  numField(cfg, "durationMs", "durationMs", v => v > 0, defaults.durationMs);
  numField(cfg.checkpointPolicy, "minSeparationMs", "minSeparationMs", v => v > 0, defaults.checkpointPolicy.minSeparationMs);
  ["pushStartMs", "anchorMs", "pushEndMs", "holdEndMs"].forEach(n =>
    numField(cfg.checkpoints, n, "checkpoints." + n, () => true, defaults.checkpoints[n]));
  intField(cfg, "depth", "depth", 1, 6, defaults.depth);
  intField(cfg, "candleCount", "candleCount", 0, 8, defaults.candleCount);
  intField(cfg, "figureButtons", "figureButtons", 0, 6, defaults.figureButtons);
  numField(cfg, "zoomStart", "zoomStart", v => v > 0, defaults.zoomStart);
  numField(cfg, "zoomEnd", "zoomEnd", v => v > 0, defaults.zoomEnd);
  numField(cfg.flicker, "amp", "flicker.amp", v => v >= 0, defaults.flicker.amp);
  numField(cfg, "labelFontFactor", "labelFontFactor", v => v > 0, defaults.labelFontFactor);
  numField(cfg, "labelMinPx", "labelMinPx", v => v > 0, defaults.labelMinPx);
  if (cfg.playbackMode !== "once" && cfg.playbackMode !== "loop"){
    diags.push("playbackMode rejected: supplied " + String(cfg.playbackMode) + " -> fallback " + defaults.playbackMode + " (authored default)");
    cfg.playbackMode = defaults.playbackMode;
  }
  return diags;
}

function smoothstep(x){ x = Math.max(0, Math.min(1, x)); return x * x * (3 - 2 * x); }

/* Label legibility — the nesting factor is applied EXACTLY ONCE: the recursive
   call already passes this level's true scene width (parent scene width x
   levelScale), so the font is simply sceneWidth x labelFontFactor. The floor is
   enforced at depths 0 and 1 only. Effective sizes at a 360 px stage:
   level 0 scene 360.00 px -> 18.00 px · level 1 scene 174.96 px -> raw 8.75 px,
   floored to 11 px · level 2 scene 85.03 px -> 4.25 px (glyph-band, not legible;
   the legibility claim is three SCENE levels, matching the specimen still). */
function sceneWidthAtLevel(stageWidthPx, level, cfg){ return stageWidthPx * Math.pow(cfg.levelScale, level); }
function labelFontPx(stageWidthPx, level, cfg){
  const raw = stageWidthPx * cfg.labelFontFactor;
  if (level <= 1) return Math.max(cfg.labelMinPx, raw);
  return raw;
}
function effectiveLabelSizes(stageWidthPx, cfg){
  const out = [];
  for (let l = 0; l < cfg.depth; l++) out.push(labelFontPx(sceneWidthAtLevel(stageWidthPx, l, cfg), l, cfg));
  return out;
}
function labelLegibilityStopLevel(cfg){
  let l = 0;
  while (labelFontPx(sceneWidthAtLevel(360, l, cfg), l, cfg) >= cfg.labelMinPx && l < 12) l++;
  return l;
}

/* Displayed-state geometry model (source-bound). Nesting is ANISOTROPIC: the
   width factor per level is frame width x interior width; the height factor
   is frame height x interior height (the interior starts below the header).
   Both factors are DERIVED from CONFIG geometry here, never re-entered as
   literals; the suite asserts they equal the priced anchors. Stage height =
   stage width x CONFIG stageAspectH/stageAspectW. Header fit = the complete
   medallion-plus-label row width AND the header height at each level. Glyph
   advance model: CONFIG typography glyphAdvanceEm per glyph plus
   letter-spacing per glyph — a conservative upper-bound model, asserted in
   the own-check suite for levels 0-1 at 360 and 390 px. */
function widthFactor(cfg){ const G = cfg.geometry; return (G.frameWidthPercent / 100) * ((100 - 2 * G.curtainWidthPercent) / 100); }
function heightFactor(cfg){ const G = cfg.geometry; return (G.frameHeightPercent / 100) * ((100 - G.frameHeaderPercent) / 100); }
function stageHeightPx(stageWidthPx, cfg){ return stageWidthPx * cfg.geometry.stageAspectH / cfg.geometry.stageAspectW; }
function sceneHeightAtLevel(stageWidthPx, level, cfg){ return stageHeightPx(stageWidthPx, cfg) * Math.pow(heightFactor(cfg), level); }
function headerModel(stageWidthPx, level, cfg){
  const G = cfg.geometry, T = cfg.typography;
  const sceneW = sceneWidthAtLevel(stageWidthPx, level, cfg);
  const sceneH = sceneHeightAtLevel(stageWidthPx, level, cfg);
  const frameW = sceneW * G.frameWidthPercent / 100;
  const frameH = sceneH * G.frameHeightPercent / 100;
  const headerH = frameH * G.frameHeaderPercent / 100;
  const font = labelFontPx(sceneW, level, cfg);
  const phraseW = cfg.labelText.length * font * (T.glyphAdvanceEm + T.letterSpacingEm);
  const medallionFontPx = Math.max(T.medallionMinPx, font * T.medallionFactor);
  const medallionW = medallionFontPx * T.medallionAdvanceEm;
  const rowW = phraseW + medallionW;
  const borderPx = Math.max(G.frameBorderMinPx, G.frameBorderPx * Math.pow(cfg.levelScale, level));
  return { sceneW: sceneW, sceneH: sceneH, frameW: frameW, frameH: frameH, headerH: headerH, fontPx: font,
           phraseW: phraseW, medallionFontPx: medallionFontPx, medallionW: medallionW, rowW: rowW,
           fitsWidth: rowW <= frameW - 2 * borderPx,
           fitsHeight: font * T.lineHeight <= headerH, borderPx: borderPx };
}
/* Occlusion model: the framed picture (drawn after the hall elements, inside
   its footprint) occludes whatever hall element overlaps its rect. Visible
   fraction per element = 1 - overlap area / element area, computed on the
   element rects the renderer actually places (percent of scene width for x,
   percent of scene height for y). Recognition dimensions are derived from
   the VISIBLE REMAINDER (the span above the frame's top edge for elements
   crossing it), never from the full element span. */
function rectOverlapArea(a, b){
  const w = Math.min(a.x + a.w, b.x + b.w) - Math.max(a.x, b.x);
  const h = Math.min(a.y + a.h, b.y + b.h) - Math.max(a.y, b.y);
  return (w > 0 && h > 0) ? w * h : 0;
}
function occlusionModel(stageWidthPx, cfg){
  const G = cfg.geometry, C = cfg.components;
  const level = cfg.depth - 1;
  const sceneW = sceneWidthAtLevel(stageWidthPx, level, cfg);
  const sceneH = sceneHeightAtLevel(stageWidthPx, level, cfg);
  const frame = { x: G.frameLeftPercent, y: G.frameTopPercent, w: G.frameWidthPercent, h: G.frameHeightPercent };
  const groupBottom = G.candleTopPercent + G.candleHeightPercent;
  const candleStep = derivedStep(cfg.candleCount, C.candle.widthPercent, C.candle.stepCapPercent);
  const out = { sceneW: sceneW, sceneH: sceneH, level: level, elements: {} };
  for (let i = 0; i < cfg.candleCount; i++){
    const bodyH = (C.candle.baseHeightPercent + (i % 2) * C.candle.altHeightDeltaPercent) / 100 * G.candleHeightPercent;
    const bodyTop = groupBottom - bodyH;
    const flameTop = bodyTop + (C.candle.flameTopPercent / 100) * bodyH;
    const rect = { x: G.candleLeftPercent + (i * candleStep / 100) * G.candleWidthPercent,
                   y: flameTop, w: (C.candle.widthPercent / 100) * G.candleWidthPercent, h: groupBottom - flameTop };
    const visFrac = 1 - rectOverlapArea(rect, frame) / (rect.w * rect.h);
    const visSpanPct = Math.max(0, G.frameTopPercent - rect.y);
    out.elements["candle" + i] = { rect: rect, visibleFraction: visFrac, visibleSpanPct: visSpanPct,
      visiblePxOpening: visSpanPct / 100 * sceneH * cfg.zoomStart,
      visiblePxHold: visSpanPct / 100 * sceneH * cfg.zoomEnd,
      bodyWidthPxOpening: rect.w / 100 * sceneW * cfg.zoomStart,
      pitchPx: (candleStep / 100) * (G.candleWidthPercent / 100) * sceneW };
  }
  const staffRect = { x: G.staffLeftPercent, y: G.staffTopPercent, w: G.staffWidthPercent, h: G.staffHeightPercent };
  const staffVisSpan = Math.max(0, G.frameTopPercent - staffRect.y);
  out.elements.staff = { rect: staffRect, visibleFraction: 1 - rectOverlapArea(staffRect, frame) / (staffRect.w * staffRect.h),
    visibleSpanPct: staffVisSpan, visiblePxOpening: staffVisSpan / 100 * sceneH * cfg.zoomStart,
    visiblePxHold: staffVisSpan / 100 * sceneH * cfg.zoomEnd,
    widthPxOpening: staffRect.w / 100 * sceneW * cfg.zoomStart };
  const faceRect = { x: (100 - G.figureRightPercent - G.figureWidthPercent) + (C.face.leftPercent / 100) * G.figureWidthPercent,
                     y: G.figureTopPercent + (C.face.topPercent / 100) * G.figureHeightPercent,
                     w: (C.face.widthPercent / 100) * G.figureWidthPercent, h: (C.face.heightPercent / 100) * G.figureHeightPercent };
  out.elements.face = { rect: faceRect, visibleFraction: 1 - rectOverlapArea(faceRect, frame) / (faceRect.w * faceRect.h),
    diameterPxOpening: Math.min(faceRect.w / 100 * sceneW, faceRect.h / 100 * sceneH) * cfg.zoomStart };
  const tableRect = { x: G.tableLeftPercent, y: G.tableTopPercent, w: G.tableWidthPercent, h: G.tableHeightPercent };
  out.elements.table = { rect: tableRect, visibleFraction: 1 - rectOverlapArea(tableRect, frame) / (tableRect.w * tableRect.h) };
  out.elements.frame = { rect: frame, visibleFraction: 1 };
  out.borderPx = Math.max(G.frameBorderMinPx, G.frameBorderPx * Math.pow(cfg.levelScale, level));
  return out;
}
function deepestDetailModel(stageWidthPx, cfg){
  /* displayed widths at the deepest scene level, at the opening scale zoomStart */
  const sceneW = sceneWidthAtLevel(stageWidthPx, cfg.depth - 1, cfg);
  const candleGroupW = sceneW * cfg.geometry.candleWidthPercent / 100;
  return { sceneW: sceneW,
           candleBodyPx: candleGroupW * cfg.components.candle.widthPercent / 100 * cfg.zoomStart,
           staffPx: sceneW * cfg.geometry.staffWidthPercent / 100 * cfg.zoomStart };
}

function scenePerLevelNodes(cfg){ return 19 + cfg.figureButtons + 2 * cfg.candleCount; }

/* Pure state from time — the EFFECTIVE checkpoint table decides the push.
   Phases: settle -> push -> hold -> loopHold (continued hold). In "once"
   mode time CLAMPS at durationMs and the phase at/after durationMs is
   terminalHold (progress 1, scale zoomEnd, retained); in "loop" mode time
   wraps at durationMs as before. */
function computeState(tMs, cfg){
  const t = cfg.playbackMode === "loop"
    ? ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs
    : Math.max(0, Math.min(cfg.durationMs, tMs));
  const diag = checkpointDiagnostics(cfg);
  const cp = diag.effective;
  let progress = 0;
  if (t >= cp.pushEndMs) progress = 1;
  else if (t > cp.pushStartMs) progress = smoothstep((t - cp.pushStartMs) / (cp.pushEndMs - cp.pushStartMs));
  const scale = cfg.zoomStart + (cfg.zoomEnd - cfg.zoomStart) * progress;
  const phase = (cfg.playbackMode !== "loop" && t >= cfg.durationMs) ? "terminalHold"
    : t < cp.pushStartMs ? "settle" : (t < cp.pushEndMs ? "push" : (t < cp.holdEndMs ? "hold" : "loopHold"));
  return { t: t, progress: progress, scale: scale, phase: phase, depth: cfg.depth,
           checkpointsCoherent: diag.status === "coherent", checkpointStatus: diag.status,
           effectiveCheckpoints: cp, effectiveAnchorMs: diag.effectiveAnchorMs,
           checkpointErrors: diag.errors, separationReduced: diag.separationReduced,
           allocationDegraded: diag.allocationDegraded };
}

/* Count-derived spacing: the CONFIG step is a CAP; the effective step shrinks so
   count-many items always end inside their group's bounds. */
function derivedStep(count, widthPercent, stepCapPercent){
  if (count <= 1) return 0;
  return Math.min(stepCapPercent, (100 - widthPercent) / (count - 1));
}

let lastDiagKey = "";
function surfaceDiagnostics(s){
  const diag = { checkpointStatus: s.checkpointStatus, effectiveCheckpoints: s.effectiveCheckpoints,
                 effectiveAnchorMs: s.effectiveAnchorMs, checkpointErrors: s.checkpointErrors,
                 separationReduced: s.separationReduced, overrideDiagnostics: OVERRIDE_DIAGNOSTICS,
                 resolveDiagnostics: RESOLVE_DIAGNOSTICS, allocationDegraded: s.allocationDegraded };
  window.__T19_DIAGNOSTICS = diag;
  const key = JSON.stringify(diag);
  if (key === lastDiagKey) return;
  lastDiagKey = key;
  if (s.checkpointStatus !== "coherent" || OVERRIDE_DIAGNOSTICS.length || RESOLVE_DIAGNOSTICS.length || s.allocationDegraded){
    diagEl.style.display = "block";
    diagEl.textContent = "checkpoints: " + s.checkpointStatus + " · effective anchor " + s.effectiveAnchorMs +
      "ms · push " + s.effectiveCheckpoints.pushStartMs + "–" + s.effectiveCheckpoints.pushEndMs +
      "ms · hold ends " + s.effectiveCheckpoints.holdEndMs + "ms" +
      (s.checkpointErrors.length ? " · " + s.checkpointErrors.join(", ") : "") +
      (s.allocationDegraded ? " · allocationDegraded: push " + (s.effectiveCheckpoints.pushEndMs - s.effectiveCheckpoints.pushStartMs) + "ms / hold " + (s.effectiveCheckpoints.holdEndMs - s.effectiveCheckpoints.pushEndMs) + "ms" : "") +
      (RESOLVE_DIAGNOSTICS.length ? " · " + RESOLVE_DIAGNOSTICS.join(", ") : "") +
      (OVERRIDE_DIAGNOSTICS.length ? " · " + OVERRIDE_DIAGNOSTICS.join(", ") : "");
    if (typeof console !== "undefined" && console.warn) console.warn("T19 diagnostics", diag);
  } else {
    diagEl.style.display = "none"; diagEl.textContent = "";
  }
}

function render(tMs){
  const s = computeState(tMs, CONFIG);
  zoomWrap.style.transform = "scale(" + s.scale.toFixed(4) + ")";
  surfaceDiagnostics(s);
  const flames = zoomWrap.querySelectorAll ? zoomWrap.querySelectorAll(".flame") : [];
  for (let i = 0; i < flames.length; i++){
    const phase = parseFloat(flames[i].dataset.phase || "0");
    const v = CONFIG.flicker.amp * Math.sin(s.t / CONFIG.flicker.periodDivisorMs + phase);
    flames[i].style.opacity = String(1 - Math.abs(v));
  }
  return s;
}

let playing = false, startStamp = 0, pausedAt = 0, rafId = null;
function nowMs(){ return (typeof performance !== "undefined" && performance.now) ? performance.now() : Date.now(); }
function tick(){
  if (!playing) return;
  const s = render(nowMs() - startStamp);
  if (s.phase === "terminalHold"){ /* once mode: STOP and RETAIN the terminal frame */
    playing = false; pausedAt = CONFIG.durationMs; rafId = null; return; }
  rafId = requestAnimationFrame(tick);
}
function play(){ if (playing) return; playing = true; startStamp = nowMs() - pausedAt; tick(); }
function pause(){ if (!playing) return; pausedAt = nowMs() - startStamp; playing = false; if (rafId) cancelAnimationFrame(rafId); rafId = null; }
function restart(){ /* the ONLY return to t=0 in once mode: an explicit action */
  pausedAt = 0; if (rafId) cancelAnimationFrame(rafId); rafId = null;
  if (playing){ startStamp = nowMs(); tick(); } else { playing = false; play(); } }

window.addEventListener("message", function(e){
  if (!e.data || typeof e.data !== "object") return;
  if (e.data.cmd === "play") play();
  if (e.data.cmd === "pause") pause();
  if (e.data.cmd === "restart") restart();
});
render(0);
play();
