/* Deterministic seeded RNG (mulberry32). */
function makeRng(seed){
  let a = seed >>> 0;
  return function(){
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

/* Letters in reading order: every non-space char of every line, in order. */
function letterList(cfg){
  const out = [];
  cfg.facadeLines.forEach(function(line, li){
    for (let i = 0; i < line.length; i++){
      if (line[i] !== " ") out.push({ ch: line[i], line: li, indexInLine: i });
    }
  });
  return out;
}

/* Checkpoint normalization with stated domains and postconditions.
   Domain: durationMs must be a finite number > 0 — anything else is a
   "duration-invalid" diagnostic and no checkpoint state is derived from it.
   For a valid duration, EVERY input is normalized so the result satisfies
   0 <= start < complete <= durationMs: non-finite endpoints are replaced
   (start -> 0, complete -> duration), both endpoints are clamped into
   [0, durationMs] FIRST, then ordered, then a positive span of at least
   min(CONFIG.checkpointMinSpanMs, durationMs) is enforced — complete extends
   toward the duration first, and only when pinned there does start pull back.
   Every correction is reported in the returned errors list; nothing is
   normalized silently. */
function validateCheckpoints(cfg){
  const dur = cfg.durationMs;
  if (!(typeof dur === "number" && isFinite(dur) && dur > 0)) {
    return { ok: false, errors: ["duration-invalid"], start: 0, complete: 0, durationValid: false };
  }
  const errors = [];
  let start = cfg.checkpoints.lightStart, complete = cfg.checkpoints.lightComplete;
  if (!(typeof start === "number" && isFinite(start))) { errors.push("lightStart-nonfinite->0"); start = 0; }
  if (!(typeof complete === "number" && isFinite(complete))) { errors.push("lightComplete-nonfinite->duration"); complete = dur; }
  if (start < 0) { errors.push("lightStart-clamped"); start = 0; }
  if (start > dur) { errors.push("lightStart-clamped"); start = dur; }
  if (complete < 0) { errors.push("lightComplete-clamped"); complete = 0; }
  if (complete > dur) { errors.push("lightComplete-clamped"); complete = dur; }
  if (complete < start) { errors.push("checkpoints reordered"); const tmp = start; start = complete; complete = tmp; }
  const minSpan = Math.min(cfg.checkpointMinSpanMs, dur);
  if (complete - start < minSpan) {
    errors.push("span-extended");
    complete = Math.min(dur, start + minSpan);
    if (complete - start < minSpan) { start = complete - minSpan; errors.push("span-start-pulled-back"); }
  }
  return { ok: errors.length === 0, errors: errors, start: start, complete: complete, durationValid: true };
}

/* Per-letter spacing derived from the checkpoint span under first/last-onset
   semantics: glyph 1 lights AT start, glyph n AT complete, so the spacing is
   span/(n-1). n = 1 has no spacing (its one glyph lights at start); n = 0 none. */
function perLetterMs(cfg){
  const v = validateCheckpoints(cfg);
  const n = letterList(cfg).length;
  if (!v.durationValid || n < 2) return 0;
  return (v.complete - v.start) / (n - 1);
}

/* Runtime sampling: draw the CONFIG facade text to an offscreen canvas, read its
   alpha mask, and map mask coverage onto the facade grid. A cell is owned by the
   letter whose band covers the largest share of the cell's sample rect, when that
   share reaches CONFIG.sampleCoverageMin — coverage ownership, not a single
   centre-pixel probe, so thin stems and enclosed counters survive sampling.
   The fit is TWO-AXIS: the sample font steps down until the widest line's measured
   width fits the sample width AND every line's measured ink height fits its line
   band; at the font floor an unfittable text returns a capacity diagnostic
   (fitStatus "over-capacity" + the binding axis) instead of silently overflowing.
   Per-letter bounds are MEASURED from the fitted mask and the owned cells (ink
   rows/cols, cell rows/cols) and returned with the sample.
   rasterDoc is injectable so the logic is node-testable with a stub canvas. */
function sampleFacade(cfg, rasterDoc){

  const letters0 = letterList(cfg);
  if (!doc || cfg.facadeLines.length === 0) return { cells: [], letterCount: letters0.length, cols: cfg.gridCols, rows: cfg.gridRows, fittedFontPx: cfg.sampleFontPx, fitStatus: "empty", bindingAxis: null, lineInk: [], letters: [] };
  const cv = doc.createElement("canvas");
  cv.width = cfg.sampleW; cv.height = cfg.sampleH;
  const ctx = cv.getContext("2d");
  const lineH = cfg.sampleH / cfg.facadeLines.length;
  function drawAndMeasure(fontPx){
    ctx.clearRect(0, 0, cfg.sampleW, cfg.sampleH);
    ctx.fillStyle = cfg.sampleMask;
    ctx.font = cfg.fontFamily.replace("{px}", String(fontPx));
    ctx.textBaseline = "middle";
    const bands = [];
    let globalLetter = 0;
    const lineInfo = [];
    cfg.facadeLines.forEach(function(line, li){
      const w = ctx.measureText(line).width;
      const x0 = (cfg.sampleW - w) / 2;
      ctx.fillText(line, x0, lineH * (li + 0.5));
      let acc = x0;
      for (let i = 0; i < line.length; i++){
        const cw = ctx.measureText(line[i]).width;
        const owner = line[i] === " " ? -1 : globalLetter++;
        bands.push({ ch: line[i], line: li, x0: acc, x1: acc + cw, y0: lineH * li, y1: lineH * (li + 1), owner: owner });
        acc += cw;
      }
      lineInfo.push({ widthPx: w, bandHeightPx: lineH, inkHeightPx: 0 });
    });
    const img = ctx.getImageData(0, 0, cfg.sampleW, cfg.sampleH).data;
    cfg.facadeLines.forEach(function(line, li){
      let minR = -1, maxR = -1;
      const y0 = Math.max(0, Math.floor(lineH * li)), y1 = Math.min(cfg.sampleH, Math.ceil(lineH * (li + 1)));
      for (let y = y0; y < y1; y++){
        let any = false;
        for (let x = 0; x < cfg.sampleW; x++){ if (img[(y * cfg.sampleW + x) * 4 + 3] >= cfg.sampleAlphaMin) { any = true; break; } }
        if (any) { if (minR < 0) minR = y; maxR = y; }
      }
      lineInfo[li].inkHeightPx = maxR < 0 ? 0 : (maxR - minR + 1);
    });
    return { bands: bands, img: img, lineInfo: lineInfo };
  }
  let fontPx = cfg.sampleFontPx, drawn = drawAndMeasure(fontPx);
  let fitStatus = "ok", bindingAxis = null;
  while (true) {
    const maxW = drawn.lineInfo.reduce(function(a, l){ return Math.max(a, l.widthPx); }, 0);
    const widthOk = maxW <= cfg.sampleW * cfg.sampleFitRatio;
    const heightOk = drawn.lineInfo.every(function(l){ return l.inkHeightPx <= l.bandHeightPx; });
    if (widthOk && heightOk) break;
    bindingAxis = !widthOk ? "width" : "height";
    if (fontPx - cfg.sampleFitStepPx < cfg.sampleMinFontPx) {
      if (fontPx !== cfg.sampleMinFontPx) { fontPx = cfg.sampleMinFontPx; drawn = drawAndMeasure(fontPx); }
      fitStatus = "over-capacity";
      break;
    }
    fontPx -= cfg.sampleFitStepPx;
    drawn = drawAndMeasure(fontPx);
  }
  /* coverage ownership: accumulate ink pixels per (cell, band), assign each cell
     to its best-covered letter band at/above the coverage floor. */
  const cwPx = cfg.sampleW / cfg.gridCols, chPx = cfg.sampleH / cfg.gridRows;
  const cellBandCounts = new Map();
  const maskRows = new Map(), maskCols = new Map();
  for (let y = 0; y < cfg.sampleH; y++){
    for (let x = 0; x < cfg.sampleW; x++){
      if (drawn.img[(y * cfg.sampleW + x) * 4 + 3] < cfg.sampleAlphaMin) continue;
      let band = null;
      for (let b = 0; b < drawn.bands.length; b++){
        const bd = drawn.bands[b];
        if (bd.owner >= 0 && x >= bd.x0 && x < bd.x1 && y >= bd.y0 && y < bd.y1) { band = bd; break; }
      }
      if (!band) continue;
      if (!maskRows.has(band.owner)) { maskRows.set(band.owner, new Set()); maskCols.set(band.owner, new Set()); }
      maskRows.get(band.owner).add(y); maskCols.get(band.owner).add(x);
      const cellKey = Math.floor(y / chPx) * cfg.gridCols + Math.floor(x / cwPx);
      const k = cellKey + ":" + band.owner;
      cellBandCounts.set(k, (cellBandCounts.get(k) || 0) + 1);
    }
  }
  const cellBest = new Map();
  cellBandCounts.forEach(function(count, k){
    const parts = k.split(":");
    const cellKey = Number(parts[0]), owner = Number(parts[1]);
    const coverage = count / (cwPx * chPx);
    if (coverage < cfg.sampleCoverageMin) return;
    if (!cellBest.has(cellKey) || cellBest.get(cellKey).coverage < coverage) cellBest.set(cellKey, { owner: owner, coverage: coverage });
  });
  const cells = [];
  const cellRows = new Map(), cellCols = new Map();
  cellBest.forEach(function(v, cellKey){
    const col = cellKey % cfg.gridCols, row = Math.floor(cellKey / cfg.gridCols);
    cells.push({ col: col, row: row, letter: v.owner });
    if (!cellRows.has(v.owner)) { cellRows.set(v.owner, new Set()); cellCols.set(v.owner, new Set()); }
    cellRows.get(v.owner).add(row); cellCols.get(v.owner).add(col);
  });
  cells.sort(function(a, b){ return (a.row - b.row) || (a.col - b.col); });
  const letters = letters0.map(function(l, i){
    return { ch: l.ch, line: l.line,
             maskInkRows: maskRows.has(i) ? maskRows.get(i).size : 0,
             maskInkCols: maskCols.has(i) ? maskCols.get(i).size : 0,
             inkRows: cellRows.has(i) ? cellRows.get(i).size : 0,
             inkCols: cellCols.has(i) ? cellCols.get(i).size : 0,
             cellCount: cells.reduce(function(a, c){ return a + (c.letter === i ? 1 : 0); }, 0) };
  });
  return { cells: cells, letterCount: letters0.length, cols: cfg.gridCols, rows: cfg.gridRows,
           fittedFontPx: fontPx, fitStatus: fitStatus, bindingAxis: fitStatus === "ok" ? null : bindingAxis,
           lineInk: drawn.lineInfo, letters: letters };
}

/* Facade glyph cell size at a given stage width — the legibility price. */
function cellPxAt(stageW, cfg){
  return (stageW * cfg.hero.wPct / 100) / cfg.gridCols;
}

/* Intensity hierarchy helpers: colors are scaled toward black by a CONFIG factor
   so the facade letters are the brightest group BY CONSTRUCTION (a scale, not a
   hue difference). effectiveLuminance prices a color at a scale with the BT.709
   luma coefficients — the ordering the own-check suite asserts. */
function scaleHexColor(hex, scale){
  const n = parseInt(hex.slice(1), 16);
  const r = Math.round(((n >> 16) & 255) * scale), g = Math.round(((n >> 8) & 255) * scale), b = Math.round((n & 255) * scale);
  return "rgb(" + r + "," + g + "," + b + ")";
}
function effectiveLuminance(hex, scale){
  const n = parseInt(hex.slice(1), 16);
  const r = (n >> 16) & 255, g = (n >> 8) & 255, b = n & 255;
  return (0.2126 * r + 0.7152 * g + 0.0722 * b) / 255 * scale;
}
function marqueeFontPx(stageW, cfg){
  return Math.max(cfg.marquee.fontMinPx, Math.min(cfg.marquee.fontMaxPx, stageW * cfg.marquee.fontFactor));
}

/* Marquee measured fit: the full marquee string is MEASURED at the candidate font
   and the font steps down until the measured width fits the hero strip (the hero
   width at this stage width); at the font floor an unfittable string returns the
   capacity diagnostic instead of overflowing. measure(text, fontPx) is injected
   so the fit is testable outside a browser. */
function marqueeFit(stageW, cfg, measure){
  const stripPx = stageW * cfg.hero.wPct / 100;
  let fontPx = marqueeFontPx(stageW, cfg);
  while (true) {
    const w = measure(cfg.marqueeText, fontPx);
    if (w <= stripPx) return { fontPx: fontPx, widthPx: w, stripPx: stripPx, fitStatus: "ok" };
    if (fontPx - cfg.sampleFitStepPx < cfg.marquee.fontMinPx) {
      const fw = measure(cfg.marqueeText, cfg.marquee.fontMinPx);
      return { fontPx: cfg.marquee.fontMinPx, widthPx: fw, stripPx: stripPx, fitStatus: fw <= stripPx ? "ok" : "over-capacity" };
    }
    fontPx -= cfg.sampleFitStepPx;
  }
}

/* Pure state from time — the checkpoint table decides the lighting.
   First/last-onset semantics: glyph k lights at start + (k-1) x spacing with
   inclusive-at-onset boundaries, so the facade is incomplete immediately before
   lightComplete and complete exactly at it; the hold is duration - complete.
   n = 1: the single glyph lights at start and is complete from that onset.
   n = 0: no letters, the phase never reads "hold" on letters. */
function computeState(tMs, cfg){
  const v = validateCheckpoints(cfg);
  const n = letterList(cfg).length;
  if (!v.durationValid) {
    return { t: 0, lettersLit: 0, letterCount: n, perLetterMs: 0, marqueeChars: 0,
             phase: "idle", checkpointOk: false, checkpointErrors: v.errors };
  }
  const t = ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs;
  let spacing = 0, lettersLit = 0;
  if (n === 1) {
    lettersLit = t >= v.start ? 1 : 0;
  } else if (n >= 2) {
    spacing = (v.complete - v.start) / (n - 1);
    if (t >= v.start) lettersLit = Math.min(n, Math.floor((t - v.start) / spacing) + 1);
  }
  const marqueeChars = Math.min(cfg.marqueeText.length, Math.floor(t / cfg.marqueeCharMs) + (t > 0 ? 1 : 0));
  const phase = n > 0 && lettersLit >= n ? "hold" : (lettersLit > 0 ? "lighting" : "idle");
  return { t: t, lettersLit: lettersLit, letterCount: n, perLetterMs: spacing,
           marqueeChars: t === 0 ? 0 : marqueeChars, phase: phase,
           checkpointOk: v.ok, checkpointErrors: v.errors };
}

/* Orbit passage: a directionally consistent parallax drift. The passage value is
   the elapsed fraction of the loop, so every layer's offset moves in ONE
   direction at a constant velocity sign for the whole loop — through the reveal
   and the hold — and resets at the loop wrap together with the lighting reset
   (the selected wrap behaviour: one discontinuity, shared with the reveal).
   The facade grid itself is never transformed; layers translate rigidly. */
function orbitPassage(tMs, cfg){
  const v = validateCheckpoints(cfg);
  if (!v.durationValid) return 0;
  const t = ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs;
  return t / cfg.durationMs;
}
function orbitOffsetPx(tMs, cfg, layerPx, stageW){
  return -orbitPassage(tMs, cfg) * layerPx * (stageW / cfg.orbit.referenceWidthPx);
}
