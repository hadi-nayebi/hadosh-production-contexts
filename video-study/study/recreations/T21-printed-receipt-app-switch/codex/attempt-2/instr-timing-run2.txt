/* ===== Timeline validation (execution order is part of the construction) =====
   1. duration override scales checkpoints AND subordinate durations by one ratio.
   2. individual checkpoint overrides are collected.
   3. subordinate durations are derived from neighbouring checkpoint spans.
   4. ordering + minimum spans + completion/hold space are validated.
   5. an override that fails validation is rejected ATOMICALLY (never applied)
      and recorded in CONFIG.rejectedOverrides with its reason. */
function validateTimeline(cfg){
  const cp = cfg.checkpoints, errs = [];
  const order = ["printStart","receiptFull","cardsAt","phoneAt","typedAt","replyAt","bridgeAt"];
  const minSpan = Math.max(cfg.minSpanMs, cfg.durationMs * cfg.minSpanFraction);
  for (let i = 1; i < order.length; i++)
    if (!(cp[order[i]] - cp[order[i-1]] >= minSpan)) errs.push(order[i] + " too close after " + order[i-1]);
  if (!(cfg.durationMs - cp.bridgeAt >= minSpan)) errs.push("bridgeAt leaves no hold before loop end");
  const eff = {};
  eff.cardsFadeMs = Math.min(cfg.cardsFadeMs, 0.8 * (cp.phoneAt - cp.cardsAt));
  eff.phoneRiseMs = Math.min(cfg.phoneRiseMs, 0.8 * (cp.typedAt - cp.phoneAt));
  eff.typeDelayMs = Math.min(cfg.typeDelayMs, 0.25 * (cp.typedAt - cp.phoneAt));
  eff.replyPopMs = Math.min(cfg.replyPopMs, 0.8 * (cp.bridgeAt - cp.replyAt));
  eff.bridgeMs = Math.min(cfg.bridgeMs, cfg.durationMs - cp.bridgeAt);
  const typeStart = cp.phoneAt + eff.typeDelayMs, typeWindow = cp.typedAt - typeStart;
  if (!(typeWindow >= 0.5 * (cp.typedAt - cp.phoneAt))) errs.push("typing window under half the phone-to-typed span");
  if (!(cp.replyAt + eff.replyPopMs <= cp.bridgeAt)) errs.push("reply does not complete before the bridge");
  if (!(cp.phoneAt + eff.phoneRiseMs <= cp.typedAt)) errs.push("phone rise does not complete before typing completes");
  eff.typeStart = typeStart; eff.typeWindow = typeWindow;
  cfg.effective = eff;
  return errs;
}

function clamp01(x){ return x < 0 ? 0 : (x > 1 ? 1 : x); }

/* Receipt paper geometry: wrapped line counts, overflow tier and padding all enter
   the SAME calculation that prices row exposure — a row is exposed when the fed
   length reaches its bottom offset on the moving paper. */
function paperModel(cfg){
  const scale = cfg.receiptLines.length > cfg.receiptMaxLines ? cfg.receiptOverflowScale : 1;
  const lineH = cfg.receiptLineHPx * scale;
  const rows = [{ top: cfg.receiptPadTopPx, bottom: cfg.receiptPadTopPx + cfg.receiptHeaderHPx }];
  let y = cfg.receiptPadTopPx + cfg.receiptHeaderHPx;
  for (const line of cfg.receiptLines){
    const wrapped = Math.max(1, Math.ceil(line.length / cfg.modelCharsPerLine));
    rows.push({ top: y, bottom: y + wrapped * lineH, wrapped: wrapped });
    y += wrapped * lineH;
  }
  const totalPx = y + cfg.receiptPadBottomPx;
  const windowCapPx = cfg.modelColumnHPx * (cfg.receiptLines.length > cfg.receiptMaxLines ? cfg.receiptGrowCapHPct : cfg.receiptMaxHPct) / 100;
  return { rows: rows, totalPx: totalPx, scale: scale, windowCapPx: windowCapPx, overflow: totalPx > windowCapPx };
}

function computeState(tMs, cfg){
  const t = ((tMs % cfg.durationMs) + cfg.durationMs) % cfg.durationMs;
  const cp = cfg.checkpoints;
  if (!cfg.effective) validateTimeline(cfg);
  const eff = cfg.effective;
  const printSpan = cp.receiptFull - cp.printStart;
  const printProgress = clamp01((t - cp.printStart) / printSpan);
  const ledChars = Math.floor(clamp01((t - cp.printStart) / (printSpan * cfg.ledFraction)) * cfg.ledText.length);
  const pm = paperModel(cfg);
  const fedPx = printProgress * pm.totalPx;
  /* the sheet translates down through the slot: its bottom edge sits at fedPx and
     a row is exposed once its own bottom edge has crossed the slot aperture
     (fedPx past totalPx - row.bottom on the shared geometry model) */
  const exposedRows = pm.rows.map(r => fedPx > pm.totalPx - r.bottom);
  let visibleRows = 0;
  for (const e of exposedRows) if (e) visibleRows++;
  const cardsProgress = clamp01((t - cp.cardsAt) / eff.cardsFadeMs);
  const phoneProgress = clamp01((t - cp.phoneAt) / eff.phoneRiseMs);
  const typedChars = eff.typeWindow > 0 ? Math.floor(clamp01((t - eff.typeStart) / eff.typeWindow) * cfg.phonePrompt.length) : 0;
  const replyProgress = clamp01((t - cp.replyAt) / eff.replyPopMs);
  const bridgeProgress = clamp01((t - cp.bridgeAt) / eff.bridgeMs);
  const phase = t < cp.cardsAt ? 1 : (t < cp.phoneAt ? 2 : 3);
  return { t: t, phase: phase, printProgress: printProgress, ledChars: ledChars,
           visibleRows: visibleRows, exposedRows: exposedRows, fedPx: fedPx, paperTotalPx: pm.totalPx, receiptOverflow: pm.overflow,
           cardsProgress: cardsProgress, phoneProgress: phoneProgress, typedChars: typedChars,
           replyProgress: replyProgress, bridgeProgress: bridgeProgress,
           replyOverflow: cfg.phoneReply.length > cfg.replyMaxChars,
           replyTier: cfg.phoneReply.length > cfg.replySoftChars ? cfg.replyLongScale : 1,
           valid: cfg.validationErrors.length === 0 };
}

function render(tMs){
  const s = computeState(tMs, CONFIG);
  led.textContent = CONFIG.ledText.slice(0, s.ledChars);
  /* the paper itself translates through the slot aperture: the wrap is anchored
     under the slot, its height IS the fed length in pixels, and the sheet slides
     down so its bottom edge sits exactly at the fed length — window, translation
     and row exposure all read the same fedPx on the same geometry model */
  receiptWrap.style.height = s.fedPx + "px";
  receipt.style.transform = "translateY(" + (s.fedPx - s.paperTotalPx) + "px)";
  receiptHead.style.visibility = s.exposedRows[0] ? "visible" : "hidden";
  for (let i = 0; i < lineEls.length; i++)
    lineEls[i].style.visibility = s.exposedRows[i + 1] ? "visible" : "hidden";
  cardsCol.style.opacity = String(s.cardsProgress);
  cardsCol.style.transform = "translateY(" + ((1 - s.cardsProgress) * CONFIG.cardTravelPx) + "px)";
  const dip = s.bridgeProgress * CONFIG.bridgeDipPct;
  phone.style.opacity = String(s.phoneProgress);
  phone.style.transform = "translate(-50%, " + ((1 - s.phoneProgress) * CONFIG.phoneTravelPct + dip) + "%)";
  const typed = CONFIG.phonePrompt.slice(0, s.typedChars);
  promptBubble.textContent = s.typedChars > 0 ? CONFIG.promptPrefix + typed : "";
  promptBubble.style.visibility = s.typedChars > 0 ? "visible" : "hidden";
  inputBar.textContent = typed;
  replyBubble.style.opacity = String(s.replyProgress);
  replyBubble.style.transform = "scale(" + (CONFIG.replyScaleFrom + (1 - CONFIG.replyScaleFrom) * s.replyProgress) + ")";
  replyBubble.style.visibility = s.replyProgress > 0 ? "visible" : "hidden";
  bridgeBadge.style.opacity = String(s.bridgeProgress);
  bridgeBadge.style.visibility = s.bridgeProgress > 0 ? "visible" : "hidden";
  return s;
}
let playing = false, startStamp = 0, pausedAt = 0, rafId = null;
function nowMs(){ return (typeof performance !== "undefined" && performance.now) ? performance.now() : Date.now(); }
function tick(){ if (!playing) return; render(nowMs() - startStamp); rafId = requestAnimationFrame(tick); }
function play(){ if (playing) return; playing = true; startStamp = nowMs() - pausedAt; tick(); }
function pause(){ if (!playing) return; pausedAt = nowMs() - startStamp; playing = false; if (rafId) cancelAnimationFrame(rafId); rafId = null; }
function restart(){ pausedAt = 0; if (playing){ startStamp = nowMs(); } else { render(0); } }

window.addEventListener("message", function(e){
  if (!e.data || typeof e.data !== "object") return;
  if (e.data.cmd === "play") play();
  if (e.data.cmd === "pause") pause();
  if (e.data.cmd === "restart") restart();
});
render(0);
