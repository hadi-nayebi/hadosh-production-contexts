/* ================= validation / normalization ========================= */
function strictInt(v) { return /^-?\d+$/.test(String(v).trim()) ? parseInt(v, 10) : null; }
function strictNum(v) { return /^-?\d+(\.\d+)?$/.test(String(v).trim()) ? parseFloat(v) : null; }
function isColor(v) { return /^#[0-9a-fA-F]{6}$/.test(String(v).trim()) || /^#[0-9a-fA-F]{3}$/.test(String(v).trim()); }
function validateInto(cfg, warnings) {
  const W = function (m) { warnings.push(m); };
  const clampInt = function (obj, key, lo, hi) {
    const v = strictInt(obj[key]);
    if (v === null) { W(key + ": non-integer value discarded"); return false; }
    const c = Math.max(lo, Math.min(hi, v));
    if (c !== v) W(key + ": clamped " + v + " -> " + c);
    obj[key] = c; return true;
  };
  clampInt(cfg, "dwellMs", 0, 60000); clampInt(cfg, "diveMs", 0, 10000);
  clampInt(cfg, "holdZoomMs", 0, 60000); clampInt(cfg, "loopPauseMs", 0, 60000);
  clampInt(cfg, "colsWide", 1, 6); clampInt(cfg, "colsNarrow", 1, 6);
  clampInt(cfg, "narrowBreakpointPx", 200, 2000);
  clampInt(cfg, "gapPx", 0, 40); clampInt(cfg, "borderPx", 0, 12);
  const ds = strictNum(cfg.diveScale);
  if (ds === null || ds < 1 || ds > 12) { W("diveScale: invalid, reset to default"); cfg.diveScale = CONFIG.diveScale; }
  const pa = strictNum(cfg.panelAspect);
  if (pa === null || pa < 0.5 || pa > 3) { W("panelAspect: invalid, reset to default"); cfg.panelAspect = CONFIG.panelAspect; }
  if (typeof cfg.headerText !== "string" || !cfg.headerText.trim()) { W("headerText: empty, reset to default"); cfg.headerText = CONFIG.headerText; }
  cfg.headerText = cfg.headerText.trim().slice(0, 120);
  Object.keys(cfg.palette).forEach(function (k) {
    if (!isColor(cfg.palette[k])) { W("palette." + k + ": invalid color, reset to default"); cfg.palette[k] = CONFIG.palette[k]; }
  });
  if (!Array.isArray(cfg.panels) || cfg.panels.length === 0) { W("panels: empty, reset to default"); cfg.panels = JSON.parse(JSON.stringify(CONFIG.panels)); }
  if (cfg.panels.length > 24) { W("panels: truncated to 24"); cfg.panels = cfg.panels.slice(0, 24); }
  cfg.panels.forEach(function (p, i) {
    if (!p || typeof p.title !== "string" || !p.title.trim()) { W("panels[" + i + "].title: empty, placeholder used"); p.title = "CHAPTER " + (i + 1); }
    p.title = p.title.trim().toUpperCase().slice(0, 60);
    if (!p.n) p.n = String(i + 1).padStart(2, "0");
    if (!cfg.artRegistry[p.art]) { W("panels[" + i + "].art: unknown kind '" + p.art + "', fallback 'tiles'"); p.art = "tiles"; }
  });
  if (!Array.isArray(cfg.appearances) || cfg.appearances.length === 0) { W("appearances: empty, reset to default"); cfg.appearances = JSON.parse(JSON.stringify(CONFIG.appearances)); }
  cfg.appearances = cfg.appearances.filter(function (a, i) {
    const f = strictInt(a.filled), d = strictInt(a.dive);
    if (f === null || d === null) { W("appearances[" + i + "]: non-integer filled/dive, entry dropped"); return false; }
    a.filled = Math.max(0, Math.min(cfg.panels.length, f));
    if (a.filled !== f) W("appearances[" + i + "].filled: clamped " + f + " -> " + a.filled);
    a.dive = Math.max(0, Math.min(cfg.panels.length - 1, d));
    if (a.dive !== d) W("appearances[" + i + "].dive: clamped " + d + " -> " + a.dive);
    return true;
  });
  if (cfg.appearances.length === 0) { W("appearances: all invalid, reset to default"); cfg.appearances = JSON.parse(JSON.stringify(CONFIG.appearances)); }
  if (cfg.dwellMs + cfg.diveMs + cfg.holdZoomMs + cfg.loopPauseMs <= 0) {
    W("schedule: zero total duration, timing defaults restored");
    cfg.dwellMs = CONFIG.dwellMs; cfg.diveMs = CONFIG.diveMs;
    cfg.holdZoomMs = CONFIG.holdZoomMs; cfg.loopPauseMs = CONFIG.loopPauseMs;
  }
  return cfg;
}
function resolveConfig(search) {
  const cfg = JSON.parse(JSON.stringify(CONFIG));
  const warnings = [];
  if (search) {
    const q = new URLSearchParams(search);
    const numKeys = ["dwellMs","diveMs","holdZoomMs","loopPauseMs","colsWide","colsNarrow","narrowBreakpointPx","gapPx","borderPx"];
    numKeys.forEach(function (k) {
      if (q.get(k) !== null) {
        const v = strictInt(q.get(k));
        if (v === null) warnings.push(k + ": malformed override '" + q.get(k) + "' ignored");
        else cfg[k] = v;
      }
    });
    ["diveScale","panelAspect"].forEach(function (k) {
      if (q.get(k) !== null) {
        const v = strictNum(q.get(k));
        if (v === null) warnings.push(k + ": malformed override '" + q.get(k) + "' ignored");
        else cfg[k] = v;
      }
    });
    if (q.get("panelCount") !== null) {
      const n = strictInt(q.get("panelCount"));
      if (n === null || n < 1 || n > 24) warnings.push("panelCount: malformed/out-of-range override ignored");
      else if (n < cfg.panels.length) cfg.panels = cfg.panels.slice(0, n);
      else while (cfg.panels.length < n) {
        const i = cfg.panels.length;
        cfg.panels.push({ n: String(i + 1).padStart(2, "0"), title: "CHAPTER " + (i + 1), art: "tiles" });
      }
    }
    if (q.get("panels")) {
      q.get("panels").split("|").forEach(function (t, i) {
        if (!cfg.panels[i]) return;
        const parts = t.split("@");
        if (parts[0].trim()) cfg.panels[i].title = parts[0].trim();
        if (parts[1] && parts[1].trim()) cfg.panels[i].art = parts[1].trim();
      });
    }
    if (q.get("header") !== null) cfg.headerText = q.get("header");
    if (q.get("filled") !== null) {
      const f = strictInt(q.get("filled"));
      if (f === null) warnings.push("filled: malformed override '" + q.get("filled") + "' ignored");
      else {
        const fc = Math.max(0, Math.min(cfg.panels.length, f));
        cfg.appearances = [{ filled: fc, dive: Math.min(cfg.panels.length - 1, fc) }];
      }
    }
    if (q.get("appearances")) {
      const list = [];
      q.get("appearances").split(",").forEach(function (pair) {
        const kv = pair.split(":");
        const f = strictInt(kv[0]), d = strictInt(kv[1]);
        if (f === null || d === null) warnings.push("appearances: malformed entry '" + pair + "' dropped");
        else list.push({ filled: f, dive: d });
      });
      if (list.length) cfg.appearances = list;
    }
    Object.keys(cfg.palette).forEach(function (k) {
      if (q.get(k) !== null) {
        if (isColor(q.get(k))) cfg.palette[k] = q.get(k);
        else warnings.push("palette." + k + ": malformed color '" + q.get(k) + "' ignored");
      }
    });
  }
  validateInto(cfg, warnings);
  cfg._warnings = warnings;
  return cfg;
}

function computeLayout(cfg, stageW) {
  const cols = stageW < cfg.narrowBreakpointPx ? cfg.colsNarrow : cfg.colsWide;
  const rows = Math.ceil(cfg.panels.length / cols);
  const headerH = Math.max(cfg.layout.headerMinPx, stageW * cfg.layout.headerHeightFactor);
  const panelW = (stageW - cfg.gapPx * (cols + 1)) / cols;
  const panelH = panelW / cfg.panelAspect;
  const gridH = panelH * rows + cfg.gapPx * (rows + 1);
  const stageH = headerH + gridH + cfg.layout.stageBottomPadPx;
  return { stageW: stageW, stageH: stageH, cols: cols, rows: rows, headerH: headerH, panelW: panelW, panelH: panelH, gridH: gridH };
}
function panelRect(cfg, lay, idx) {
  const col = idx % lay.cols, row = Math.floor(idx / lay.cols);
  return { x: cfg.gapPx + col * (lay.panelW + cfg.gapPx),
           y: lay.headerH + cfg.gapPx + row * (lay.panelH + cfg.gapPx),
           w: lay.panelW, h: lay.panelH };
}
/* Label policy: single-line first; two-line is the primary readable
   fallback; number-only engages only when two lines cannot fit at the
   minimum readable size. Base size derives from the actual panel width. */
function labelBasePx(cfg, panelInnerW) {
  return Math.max(cfg.labels.minPx, Math.min(cfg.labels.maxPx, panelInnerW * cfg.labels.baseFactor));
}
function estWidth(cfg, text, px) { return text.length * cfg.labels.glyphEm * px + cfg.labels.chromePx; }
function splitTwoLines(cfg, text) {
  const sep = text.indexOf(" · ");
  const num = sep >= 0 ? text.slice(0, sep) : text;
  const title = sep >= 0 ? text.slice(sep + 3) : "";
  const words = title.split(" ").filter(Boolean);
  if (!words.length) return [text];
  if (words.length === 1) return [num + " ·", words[0]];
  let best = null;
  for (let i = 1; i < words.length; i++) {
    const l1 = num + " · " + words.slice(0, i).join(" ");
    const l2 = words.slice(i).join(" ");
    const m = Math.max(l1.length, l2.length);
    if (!best || m < best.m) best = { m: m, lines: [l1, l2] };
  }
  return best.lines;
}
function labelPolicy(cfg, text, availPx, basePx) {
  const minPx = cfg.labels.minPx;
  let px = basePx;
  while (px > minPx && estWidth(cfg, text, px) > availPx) px -= 0.5;
  if (estWidth(cfg, text, px) <= availPx) return { mode: "single", px: px, lines: [text], fits: true };
  const lines = splitTwoLines(cfg, text);
  const longest = lines.reduce(function (m, l) { return Math.max(m, l.length); }, 0);
  px = basePx;
  while (px > minPx && longest * cfg.labels.glyphEm * px + cfg.labels.chromePx > availPx) px -= 0.5;
  if (longest * cfg.labels.glyphEm * px + cfg.labels.chromePx <= availPx)
    return { mode: "two-line", px: px, lines: lines, fits: true };
  const num = text.split(" ")[0];
  return { mode: "number-only", px: minPx, lines: [num], fits: estWidth(cfg, num, minPx) <= availPx };
}
function headerFitPx(cfg, text, availPx, basePx) {
  const minPx = cfg.layout.headerMinFontPx;
  let px = basePx;
  const est = function (p) { return text.length * cfg.labels.glyphEm * p + cfg.layout.headerPaddingPx * 2; };
  while (px > minPx && est(px) > availPx) px -= 0.5;
  return { px: px, fits: est(px) <= availPx };
}
/* Dive transform: translate the target panel to the stage centre and
   scale it until its transformed bounds cover the whole viewport. */
function diveTransform(cfg, lay, rect) {
  const need = Math.max(lay.stageW / rect.w, lay.stageH / rect.h);
  const s = Math.max(cfg.diveScale, need) * cfg.layout.diveCoverMargin;
  const cx = rect.x + rect.w / 2, cy = rect.y + rect.h / 2;
  const tx = lay.stageW / 2 - s * cx, ty = lay.stageH / 2 - s * cy;
  return { scale: s, tx: tx, ty: ty,
           bounds: { left: s * rect.x + tx, top: s * rect.y + ty,
                     right: s * (rect.x + rect.w) + tx, bottom: s * (rect.y + rect.h) + ty } };
}
function coversViewport(bounds, lay) {
  return bounds.left <= 0.01 && bounds.top <= 0.01 && bounds.right >= lay.stageW - 0.01 && bounds.bottom >= lay.stageH - 0.01;

function buildTimeline(cfg) {
  const steps = [];
  let start = 0;
  cfg.appearances.forEach(function (a, i) {
    [["dwell", cfg.dwellMs], ["dive", cfg.diveMs], ["hold", cfg.holdZoomMs], ["pause", cfg.loopPauseMs]].forEach(function (ph) {
      steps.push({ phase: ph[0], appearance: i, dur: ph[1], startMs: start, dive: a.dive });
      start += ph[1];
    });
  });
  return { steps: steps, totalMs: start };

function stateAt(cfg, tMs) {
  const tl = buildTimeline(cfg);
  if (tl.totalMs <= 0) return tl.steps[0];
  let t = ((tMs % tl.totalMs) + tl.totalMs) % tl.totalMs;
  for (let i = 0; i < tl.steps.length; i++) {
    const s = tl.steps[i];
    if (t >= s.startMs && t < s.startMs + s.dur) return s;
  }
  return tl.steps[tl.steps.length - 1];
}
function phaseAt(cfg, elapsedMs) {
  const tl = buildTimeline(cfg);
  const step = stateAt(cfg, elapsedMs);
  const cyclePos = tl.totalMs > 0 ? ((elapsedMs % tl.totalMs) + tl.totalMs) % tl.totalMs : 0;
  return { step: step, cyclePos: cyclePos, inPhaseMs: cyclePos - step.startMs,
           remainingMs: step.startMs + step.dur - cyclePos,
           cycleIndex: tl.totalMs > 0 ? Math.floor(elapsedMs / tl.totalMs) : 0,
           key: (tl.totalMs > 0 ? Math.floor(elapsedMs / tl.totalMs) : 0) + ":" + step.appearance + ":" + step.phase };
}

## Construction

One self-contained HTML file. A DOM grid page (cream comic page, header strip, numbered yellow-tabbed panels with code-drawn SVG miniatures) plays a looping sequence of appearances: each appearance fills more panels, holds, then the camera dives into the next chapter's panel, which takes over the frame. All variables live in one CONFIG object at the top of the script; URL parameters override them after strict validation. ### Named behaviors and constants
- **Label sizing (N1).** The tab label base size derives from the actual panel inner width: `base = clamp(panelInnerW × 0.14, 11, 18)` px. Tabs are `box-sizing: border-box` with `max-width: 100%`, so padding and border live inside the measured box. Overflow policy, in order: (1) single line, shrunk from the base toward the floor; (2) **two-line mode (primary readable policy)** — the number prefix stays on line 1 and the title balances across two lines, shrunk from the base toward the floor; (3) **number-only mode (fallback)** — engages only when two lines cannot fit at the floor. Named floor: **minimum readable label size 11 px**. Named fallback threshold: number-only engages when the tab-available width is below the width the balanced two-line form needs at 11 px (for the longest default title, whose balanced longest line is 12 glyphs: 12 × 0.52 × 11 + 14 ≈ **83 px** of tab width). A rendered-DOM refinement pass re-checks each tab's real scroll size and drops to number-only if a tab still overflows at the floor. Each tab records its mode, size, and base in `data-` attributes.
- **Header fitting (N5).** The header strip fits its text by shrinking from its width-derived base toward a named 10 px floor using the same glyph model.
- **Configuration, art registry, validation (N2).** Every layout constant lives in CONFIG: stage cap 960 px, header height factor 0.034 / minimum 18 px, header font factor 0.021, tab border 3 px, tab padding 1/5/6 px, panel background, ghost glyph factor 0.30 and opacity 0.16, dive easing `cubic-bezier(.55,0,.85,.4)`, dive cover margin 1.03, typography (weights, letter-spacing, line height 1.25), and the label model (base factor 0.14, max 18 px, floor 11 px, 0.52 em/glyph, 14 px chrome). Miniature art is a CONFIG **art registry**: each kind is data — a list of drawing primitives (rect / circle / line / polygon / path / text / grid / graph / circlesRow) with palette references (`$ink`, `$accent`); a generic renderer serializes them, and all inscriptions are registry data fields. **All URL overrides are strictly parsed and normalized**: integers must match `^-?\d+$` and are range-clamped, numbers likewise, colors must be hex, a blank header falls back, unknown art kinds fall back to the registry's tile kind — every rejection or clamp appends a named warning to `cfg._warnings` (also printed via `console.warn`). Panel-count changes are guarded (1–24; appearances are re-validated against the panel list; dive targets clamp into range with warnings).
- **Dive behavior (N3).** The dive translates the target panel to the stage centre and scales it by `max(configured scale, stageW/panelW, stageH/panelH) × 1.03`, so the transformed target bounds cover the entire viewport (every viewport edge inside the target's bounds; the header and all neighboring panels are occluded at dive end). Panels are `box-sizing: border-box`, so the rendered panel box (border included) is exactly the geometry model's panel rectangle — the camera model and the rendered layout cannot drift apart. The dive target panel is rendered **filled** (its miniature shown) throughout its appearance — during the hold and at dive start, for every appearance; non-target panels keep their filled/ghost state. Visible filled totals per appearance are therefore **5 → 9 → 12** (configured counts 4 → 8 → 12 plus the target panel, which is the chapter about to play). Every phase entry establishes its own appearance and visibility independently (dwell, dive, and hold each render the due appearance and restore full opacity), so entering any phase directly — after a stall — shows the correct, visible state.
- **Schedule (N4).** Steps carry absolute start offsets; the cycle length is **derived** as the sum of the resolved schedule (default 5,850 ms). The engine derives the current phase from elapsed time on every tick (`phaseAt`: step, position inside the phase, remaining time, cycle index), so a stalled or backgrounded tab resumes at the phase actually due — expired phases are never replayed and cycle boundaries are crossed by the same derivation. A dive that starts late animates over the time **remaining** in its phase, arriving at the phase deadline. Zero-duration steps are never selected by the phase derivation (they occupy no time interval), and a schedule whose four durations total zero is rejected at validation: the timing defaults are restored with a named warning. Pause freezes the active zoom transform where it stands and records the elapsed time; resume shifts the deadline origin by the paused duration, restores the frozen transform, and animates from it to the target over the remaining phase time.
- **Phase durations (defaults, with the tracker-reading reason).** dwell 900 ms · dive 650 ms · zoom hold 150 ms · dark pause 250 ms. Reason: with labels glance-readable in two-line mode at the 11 px floor and filled/ghost contrast carrying progress at a glance, the grid hold is priced as a flash-with-readability beat (the specimen's appearances are sub-second flashes) rather than a reading pause; the zoom hold is a brief settle only, because the dive target is already filled when the camera arrives; the dark pause is a short punctuation beat between appearances.
