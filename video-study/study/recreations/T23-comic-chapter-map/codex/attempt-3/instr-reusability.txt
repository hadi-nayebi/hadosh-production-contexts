function strictInt(v) { return /^-?\d+$/.test(String(v).trim()) ? parseInt(v, 10) : null; }
function strictNum(v) { return /^-?\d+(\.\d+)?$/.test(String(v).trim()) ? parseFloat(v) : null; }
function isColor(v) { return /^#[0-9a-fA-F]{6}$/.test(String(v).trim()) || /^#[0-9a-fA-F]{3}$/.test(String(v).trim()); }
function validateInto(cfg, warnings) {
  const W = function (m) { warnings.push(m); };
  const clampInt = function (obj, key, lo, hi) {
    const v = strictInt(obj[key]);
    if (v === null) { W(key + ": non-integer value discarded"); return false; }
    const c = Math.max(lo, Math.min(hi, v));
    if (c !== v) W(key + ": clamped " + v + " -> " + c);
    obj[key] = c; return true;
  };
  clampInt(cfg, "dwellMs", 0, 60000); clampInt(cfg, "diveMs", 0, 10000);
  clampInt(cfg, "holdZoomMs", 0, 60000); clampInt(cfg, "loopPauseMs", 0, 60000);
  clampInt(cfg, "colsWide", 1, 6); clampInt(cfg, "colsNarrow", 1, 6);
  clampInt(cfg, "narrowBreakpointPx", 200, 2000);
  clampInt(cfg, "gapPx", 0, 40); clampInt(cfg, "borderPx", 0, 12);
  const ds = strictNum(cfg.diveScale);
  if (ds === null || ds < 1 || ds > 12) { W("diveScale: invalid, reset to default"); cfg.diveScale = CONFIG.diveScale; }
  const pa = strictNum(cfg.panelAspect);
  if (pa === null || pa < 0.5 || pa > 3) { W("panelAspect: invalid, reset to default"); cfg.panelAspect = CONFIG.panelAspect; }
  if (typeof cfg.headerText !== "string" || !cfg.headerText.trim()) { W("headerText: empty, reset to default"); cfg.headerText = CONFIG.headerText; }

  Object.keys(cfg.palette).forEach(function (k) {

    else cfg.palette[k] = String(cfg.palette[k]).trim();

  /* registry primitive-field schema validation (runs BEFORE panel art
     references and the fallback kind are resolved, so both resolve

  const isNum = function (v) { return typeof v === "number" && isFinite(v); };

  const validPrim = function (pr) {
    if (!pr || primKinds.indexOf(pr.t) === -1) return false;
    if (pr.t === "rect") return numFields(pr, ["x","y","w","h"]) && pr.w >= 0 && pr.h >= 0;
    if (pr.t === "circle") return numFields(pr, ["cx","cy","r"]) && pr.r >= 0;
    if (pr.t === "line") return numFields(pr, ["x1","y1","x2","y2"]);
    if (pr.t === "polygon") return typeof pr.points === "string" && pr.points.trim().length > 0;
    if (pr.t === "path") return typeof pr.d === "string" && pr.d.trim().length > 0;
    if (pr.t === "text") {
      if (!numFields(pr, ["x","y","size"]) || pr.size <= 0) return false;

  numGuard(cfg.labels, "shrinkStepPx", 0.05, 5, CONFIG.labels.shrinkStepPx);
  if (!Number.isInteger(cfg.labels.numberPadWidth) || cfg.labels.numberPadWidth < 1 || cfg.labels.numberPadWidth > 4) {
    W("labels.numberPadWidth: out of range, reset to default"); cfg.labels.numberPadWidth = CONFIG.labels.numberPadWidth; }
  if (typeof cfg.labels.separator !== "string" || !cfg.labels.separator) { W("labels.separator: invalid, reset"); cfg.labels.separator = CONFIG.labels.separator; }
  if (typeof cfg.labels.generatedTitlePrefix !== "string" || !cfg.labels.generatedTitlePrefix) { W("labels.generatedTitlePrefix: invalid, reset"); cfg.labels.generatedTitlePrefix = CONFIG.labels.generatedTitlePrefix; }
  numGuard(cfg.typography, "headerWeight", 100, 900, CONFIG.typography.headerWeight);
  numGuard(cfg.typography, "headerLetterSpacingEm", 0, 1, CONFIG.typography.headerLetterSpacingEm);
  numGuard(cfg.typography, "tabWeight", 100, 900, CONFIG.typography.tabWeight);
  numGuard(cfg.typography, "tabLetterSpacingEm", 0, 1, CONFIG.typography.tabLetterSpacingEm);
  numGuard(cfg.typography, "tabLineHeight", 1, 2, CONFIG.typography.tabLineHeight);
  numGuard(cfg.typography, "ghostWeight", 100, 900, CONFIG.typography.ghostWeight);
  numGuard(cfg.artViewBox, "w", 20, 1000, CONFIG.artViewBox.w);
  numGuard(cfg.artViewBox, "h", 20, 1000, CONFIG.artViewBox.h);
  if (cfg.labels.maxPx < cfg.labels.minPx) { W("labels.maxPx: below minPx, raised to fit"); cfg.labels.maxPx = Math.max(CONFIG.labels.maxPx, cfg.labels.minPx); }
  const easingStr = typeof cfg.layout.diveEasing === "string" ? cfg.layout.diveEasing.trim() : "";
  const bez = /^cubic-bezier\(\s*(-?(?:\d+(?:\.\d+)?|\.\d+))\s*,\s*(-?(?:\d+(?:\.\d+)?|\.\d+))\s*,\s*(-?(?:\d+(?:\.\d+)?|\.\d+))\s*,\s*(-?(?:\d+(?:\.\d+)?|\.\d+))\s*\)$/.exec(easingStr);
  /* a cubic-bezier is only CSS-valid when its 1st and 3rd coordinates
     lie in [0,1] — an out-of-range pair is rejected by the browser
     silently, so it is rejected here with a warning instead */
  const easingOk = /^(ease|linear|ease-in|ease-out|ease-in-out|step-start|step-end)$/.test(easingStr) ||
    (bez !== null && +bez[1] >= 0 && +bez[1] <= 1 && +bez[3] >= 0 && +bez[3] <= 1);
  if (!easingOk) { W("diveEasing: invalid, reset"); cfg.layout.diveEasing = CONFIG.layout.diveEasing; }
  else cfg.layout.diveEasing = easingStr;
  if (typeof cfg.fontFamily !== "string" || !cfg.fontFamily.trim()) { W("fontFamily: invalid, reset"); cfg.fontFamily = CONFIG.fontFamily; }
  /* appearances: validate, and RE-validate restored defaults against
     the CURRENT panel count (a restore is not exempt from the range) */
  const fixAppearances = function () {
    cfg.appearances = cfg.appearances.filter(function (a, i) {
      if (!a || typeof a !== "object") { W("appearances[" + i + "]: malformed entry dropped"); return false; }
      const f = strictInt(a.filled), d = strictInt(a.dive);
      if (f === null || d === null) { W("appearances[" + i + "]: non-integer filled/dive, entry dropped"); return false; }
      a.filled = Math.max(0, Math.min(cfg.panels.length, f));
      if (a.filled !== f) W("appearances[" + i + "].filled: clamped " + f + " -> " + a.filled);
      a.dive = Math.max(0, Math.min(cfg.panels.length - 1, d));
      if (a.dive !== d) W("appearances[" + i + "].dive: clamped " + d + " -> " + a.dive);
      return true;
    });
  };
  if (!Array.isArray(cfg.appearances) || cfg.appearances.length === 0) { W("appearances: empty, reset to default"); cfg.appearances = JSON.parse(JSON.stringify(CONFIG.appearances)); }
  fixAppearances();
  if (cfg.appearances.length === 0) { W("appearances: all invalid, reset to default"); cfg.appearances = JSON.parse(JSON.stringify(CONFIG.appearances)); fixAppearances(); }
  /* relational geometry safe fallback: individually valid columns /
     gap / border can combine into non-positive inner panel dimensions
     at phone width — CLAMP the combination to a feasible one (never
     warning-only), priced at the fallback stage width */
  (function () {
    /* both breakpoint regimes are priced: the narrow regime at the
       fallback stage width, and the wide regime at its SMALLEST width
       (the breakpoint itself) — a combination feasible in one regime
       can be infeasible in the other */
    const regimes = [];
    const refKey = cfg.layout.fallbackStagePx < cfg.narrowBreakpointPx ? "colsNarrow" : "colsWide";
    regimes.push({ refW: cfg.layout.fallbackStagePx, key: refKey });
    if (refKey !== "colsWide" || cfg.narrowBreakpointPx !== cfg.layout.fallbackStagePx)
      regimes.push({ refW: cfg.narrowBreakpointPx, key: "colsWide" });
    const before = { colsNarrow: cfg.colsNarrow, colsWide: cfg.colsWide, gapPx: cfg.gapPx, borderPx: cfg.borderPx };
    const feasibleFor = function (r) {
      const cols = cfg[r.key];
      const pw = (r.refW - cfg.gapPx * (cols + 1)) / cols;
      return pw - cfg.borderPx * 2 >= 24 && pw / cfg.panelAspect - cfg.borderPx * 2 >= 12;
    };
    let guard = 0;
    while (regimes.some(function (r) { return !feasibleFor(r); }) && guard++ < 300) {
      if (cfg.borderPx > 0) cfg.borderPx--;
      else if (cfg.gapPx > 0) cfg.gapPx = Math.max(0, cfg.gapPx - 5);
      else {
        const bad = regimes.filter(function (r) { return !feasibleFor(r) && cfg[r.key] > 1; })[0];
        if (!bad) break;
        cfg[bad.key]--;
      }
    }
    regimes.forEach(function (r) {
      if (cfg[r.key] !== before[r.key]) W("geometry: " + r.key + " clamped " + before[r.key] + " -> " + cfg[r.key] + " for positive inner panel dimensions at " + r.refW + "px");
    });
    if (cfg.gapPx !== before.gapPx) W("geometry: gapPx clamped " + before.gapPx + " -> " + cfg.gapPx + " for positive inner panel dimensions");
    if (cfg.borderPx !== before.borderPx) W("geometry: borderPx clamped " + before.borderPx + " -> " + cfg.borderPx + " for positive inner panel dimensions");
  })();
  if (cfg.dwellMs + cfg.diveMs + cfg.holdZoomMs + cfg.loopPauseMs <= 0) {
    W("schedule: zero total duration, timing defaults restored");
    cfg.dwellMs = CONFIG.dwellMs; cfg.diveMs = CONFIG.diveMs;
    cfg.holdZoomMs = CONFIG.holdZoomMs; cfg.loopPauseMs = CONFIG.loopPauseMs;
  }
  return cfg;
}
function resolveConfig(search) {
  const cfg = JSON.parse(JSON.stringify(CONFIG));
  const warnings = [];
  if (search) {
    const q = new URLSearchParams(search);
    const numKeys = ["dwellMs","diveMs","holdZoomMs","loopPauseMs","colsWide","colsNarrow","narrowBreakpointPx","gapPx","borderPx"];
    numKeys.forEach(function (k) {
      if (q.get(k) !== null) {
        const v = strictInt(q.get(k));
        if (v === null) warnings.push(k + ": malformed override '" + q.get(k) + "' ignored");
        else cfg[k] = v;
      }
    });
    ["diveScale","panelAspect"].forEach(function (k) {
      if (q.get(k) !== null) {
        const v = strictNum(q.get(k));
        if (v === null) warnings.push(k + ": malformed override '" + q.get(k) + "' ignored");
        else cfg[k] = v;
      }
    });
    if (q.get("panelCount") !== null) {
      const n = strictInt(q.get("panelCount"));
      if (n === null || n < 1 || n > 24) warnings.push("panelCount: malformed/out-of-range override ignored");
      else if (n < cfg.panels.length) cfg.panels = cfg.panels.slice(0, n);
      else while (cfg.panels.length < n) {
        const i = cfg.panels.length;
        cfg.panels.push({ n: String(i + 1).padStart(cfg.labels.numberPadWidth, "0"), title: cfg.labels.generatedTitlePrefix + (i + 1), art: cfg.fallbackArtKind });
      }
    }
    if (q.get("panels") !== null) {
      if (!q.get("panels")) warnings.push("panels: empty override ignored");
      else q.get("panels").split("|").forEach(function (t, i) {
        if (!cfg.panels[i]) { warnings.push("panels: entry " + i + " beyond panel count ignored"); return; }
        const parts = t.split("@");
        if (parts.length > 2) warnings.push("panels[" + i + "]: extra '@' components ignored");
        if (parts[0].trim()) cfg.panels[i].title = parts[0].trim();
        else warnings.push("panels[" + i + "]: blank title slot ignored");
        if (parts[1] && parts[1].trim()) cfg.panels[i].art = parts[1].trim();
      });
    }
    if (q.get("numbers") !== null) {
      if (!q.get("numbers").trim()) warnings.push("numbers: empty override ignored");
      else q.get("numbers").split(",").forEach(function (n, i) {
        if (!cfg.panels[i]) { warnings.push("numbers: entry " + i + " beyond panel count ignored"); return; }
        if (n.trim()) cfg.panels[i].n = n.trim();
        else warnings.push("numbers[" + i + "]: blank number slot ignored");
      });
    }
    if (q.get("artViewBox") !== null) {
      const m = /^(\d+)x(\d+)$/.test(q.get("artViewBox").trim()) ? q.get("artViewBox").trim().split("x") : null;
      if (!m) warnings.push("artViewBox: malformed override '" + q.get("artViewBox") + "' ignored");
      else { cfg.artViewBox.w = parseInt(m[0], 10); cfg.artViewBox.h = parseInt(m[1], 10); }
    }
    if (q.get("fallbackArtKind") !== null) {
      if (q.get("fallbackArtKind").trim()) cfg.fallbackArtKind = q.get("fallbackArtKind").trim();
      else warnings.push("fallbackArtKind: blank override ignored");
    }
    if (q.get("header") !== null) cfg.headerText = q.get("header");
    if (q.get("filled") !== null) {
      const f = strictInt(q.get("filled"));
      if (f === null) warnings.push("filled: malformed override '" + q.get("filled") + "' ignored");
      else {
        const fc = Math.max(0, Math.min(cfg.panels.length, f));
        if (fc !== f) warnings.push("filled: clamped " + f + " -> " + fc);
        cfg.appearances = [{ filled: fc, dive: Math.min(cfg.panels.length - 1, fc) }];
      }
    }
    if (q.get("appearances") !== null && !q.get("appearances").trim()) warnings.push("appearances: empty override ignored");
    if (q.get("appearances")) {
      const list = [];
      q.get("appearances").split(",").forEach(function (pair) {
        const kv = pair.split(":");
        if (kv.length > 2) warnings.push("appearances: extra fields in '" + pair + "' ignored");
        const f = strictInt(kv[0]), d = strictInt(kv[1]);
        if (f === null || d === null) warnings.push("appearances: malformed entry '" + pair + "' dropped");
        else list.push({ filled: f, dive: d });
      });
      if (list.length) cfg.appearances = list;
    }
    Object.keys(cfg.palette).forEach(function (k) {
      if (q.get(k) !== null) {
        if (isColor(q.get(k))) cfg.palette[k] = q.get(k).trim();
        else warnings.push("palette." + k + ": malformed color '" + q.get(k) + "' ignored");
      }
    });
    if (q.get("fontFamily") !== null) {
      if (q.get("fontFamily").trim()) cfg.fontFamily = q.get("fontFamily");
      else warnings.push("fontFamily: blank override ignored");
    }
    /* generic nested overrides: ?cfg:layout.stageMaxPx=800&cfg:labels.minPx=12 */
    q.forEach(function (val, key) {
      if (key.indexOf("cfg:") !== 0) return;
      const path = key.slice(4).split(".");
      if (path.length !== 2 || ["layout", "labels", "typography", "artViewBox"].indexOf(path[0]) === -1) {
        warnings.push(key + ": unknown configuration path ignored"); return;
      }
      const group = cfg[path[0]], field = path[1];
      if (!Object.prototype.hasOwnProperty.call(group, field)) { warnings.push(key + ": unknown field ignored"); return; }
      if (typeof group[field] === "number") {
        const v = strictNum(val);
        if (v === null) warnings.push(key + ": malformed number '" + val + "' ignored");
        else group[field] = v;
      } else if (typeof group[field] === "string") {
        if (val.trim()) group[field] = val; else warnings.push(key + ": blank value ignored");
      }
    });
    /* unknown query keys are named, never silently ignored */
    const knownKeys = {};
    ["dwellMs","diveMs","holdZoomMs","loopPauseMs","colsWide","colsNarrow","narrowBreakpointPx","gapPx","borderPx",
     "diveScale","panelAspect","panelCount","panels","numbers","artViewBox","fallbackArtKind","header","filled",
     "appearances","fontFamily"].concat(Object.keys(cfg.palette)).forEach(function (k) { knownKeys[k] = true; });
    q.forEach(function (val, key) {
      if (key.indexOf("cfg:") === 0) return;
      if (!knownKeys[key]) warnings.push(key + ": unknown query key ignored");
    });
  }
  validateInto(cfg, warnings);
  cfg._warnings = warnings;
  return cfg;
}

/* ================= geometry (pure, source-bound model) ================ */
function computeLayout(cfg, stageW) {
  const cols = stageW < cfg.narrowBreakpointPx ? cfg.colsNarrow : cfg.colsWide;
  const rows = Math.ceil(cfg.panels.length / cols);
  const headerH = Math.max(cfg.layout.headerMinPx, stageW * cfg.layout.headerHeightFactor);
  const panelW = (stageW - cfg.gapPx * (cols + 1)) / cols;
  const panelH = panelW / cfg.panelAspect;
  const gridH = panelH * rows + cfg.gapPx * (rows + 1);
  const stageH = headerH + gridH + cfg.layout.stageBottomPadPx;
  return { stageW: stageW, stageH: stageH, cols: cols, rows: rows, headerH: headerH, panelW: panelW, panelH: panelH, gridH: gridH };
}
function panelRect(cfg, lay, idx) {
  const col = idx % lay.cols, row = Math.floor(idx / lay.cols);
  return { x: cfg.gapPx + col * (lay.panelW + cfg.gapPx),
           y: lay.headerH + cfg.gapPx + row * (lay.panelH + cfg.gapPx),
           w: lay.panelW, h: lay.panelH };
}
/* Label policy: single-line first; two-line is the primary readable
   fallback; number-only engages only when two lines cannot fit at the
   minimum readable size. Base size derives from the actual panel width. */
function labelBasePx(cfg, panelInnerW) {
  return Math.max(cfg.labels.minPx, Math.min(cfg.labels.maxPx, panelInnerW * cfg.labels.baseFactor));
}
function estWidth(cfg, text, px) { return text.length * cfg.labels.glyphEm * px + cfg.labels.chromePx; }
function splitTwoLines(cfg, text) {
  const sepStr = cfg.labels.separator;
  const sep = text.indexOf(sepStr);
  const num = sep >= 0 ? text.slice(0, sep) : text;
  const title = sep >= 0 ? text.slice(sep + sepStr.length) : "";
  const words = title.split(" ").filter(Boolean);
  if (!words.length) return [text];
  if (words.length === 1) return [num + sepStr.trimEnd(), words[0]];
  let best = null;
  for (let i = 1; i < words.length; i++) {
    const l1 = num + sepStr + words.slice(0, i).join(" ");
    const l2 = words.slice(i).join(" ");
    const m = Math.max(l1.length, l2.length);
    if (!best || m < best.m) best = { m: m, lines: [l1, l2] };
  }
  return best.lines;
}
function labelPolicy(cfg, text, availPx, basePx) {
  const minPx = cfg.labels.minPx;
  const step = cfg.labels.shrinkStepPx;
  /* every decrement is CLAMPED at the floor: the size can land exactly
     on minPx but never undershoot it (attempt 2 reached 10.81 < 11) */
  let px = Math.max(minPx, basePx);
  while (px > minPx && estWidth(cfg, text, px) > availPx) px = Math.max(minPx, px - step);
  if (estWidth(cfg, text, px) <= availPx) return { mode: "single", px: px, lines: [text], fits: true };
  const lines = splitTwoLines(cfg, text);
  const longest = lines.reduce(function (m, l) { return Math.max(m, l.length); }, 0);
  px = Math.max(minPx, basePx);
  while (px > minPx && longest * cfg.labels.glyphEm * px + cfg.labels.chromePx > availPx) px = Math.max(minPx, px - step);
  if (longest * cfg.labels.glyphEm * px + cfg.labels.chromePx <= availPx)
    return { mode: "two-line", px: px, lines: lines, fits: true };
  const sepAt = text.indexOf(cfg.labels.separator);
  const num = sepAt >= 0 ? text.slice(0, sepAt) : text.split(" ")[0];
  return { mode: "number-only", px: minPx, lines: [num], fits: estWidth(cfg, num, minPx) <= availPx };
}
/* Header policy (honored by the DOM layer): shrink toward the named
   floor; if the text still does not fit, shrink further toward the
   absolute floor (mode "reduced-floor"); only if it still does not fit
   may the header wrap (mode "wrap", fits:false) — the fits result is
   never ignored. The available width carries a named reserve. */
function headerFitPx(cfg, text, availPx, basePx) {
  const minPx = cfg.layout.headerMinFontPx;
  const absMinPx = cfg.layout.headerAbsoluteMinPx;
  const step = cfg.labels.shrinkStepPx;
  const avail = Math.max(0, availPx - cfg.layout.headerReservePx);
  const est = function (p) { return text.length * cfg.labels.glyphEm * p + cfg.layout.headerPaddingPx * 2; };
  let px = Math.max(minPx, basePx);
  while (px > minPx && est(px) > avail) px = Math.max(minPx, px - step);
  if (est(px) <= avail) return { px: px, fits: true, mode: "floor" };
  while (px > absMinPx && est(px) > avail) px = Math.max(absMinPx, px - step);
  if (est(px) <= avail) return { px: px, fits: true, mode: "reduced-floor" };
  return { px: absMinPx, fits: false, mode: "wrap" };
}
/* Dive transform: translate the target panel to the stage centre and
   scale it until its transformed bounds cover the whole viewport. */
function diveTransform(cfg, lay, rect) {
  /* coverage is computed from the panel's INNER art rectangle (inside
     its border), so the miniature itself — not just its border box —
     covers the viewport at dive end */
  const inner = { x: rect.x + cfg.borderPx, y: rect.y + cfg.borderPx,
                  w: Math.max(1, rect.w - cfg.borderPx * 2), h: Math.max(1, rect.h - cfg.borderPx * 2) };
  const need = Math.max(lay.stageW / inner.w, lay.stageH / inner.h);
  const s = Math.max(cfg.diveScale, need) * cfg.layout.diveCoverMargin;
  const cx = inner.x + inner.w / 2, cy = inner.y + inner.h / 2;
  const tx = lay.stageW / 2 - s * cx, ty = lay.stageH / 2 - s * cy;
  return { scale: s, tx: tx, ty: ty,
           bounds: { left: s * inner.x + tx, top: s * inner.y + ty,
                     right: s * (inner.x + inner.w) + tx, bottom: s * (inner.y + inner.h) + ty } };
}
function coversViewport(bounds, lay) {
  return bounds.left <= 0.01 && bounds.top <= 0.01 && bounds.right >= lay.stageW - 0.01 && bounds.bottom >= lay.stageH - 0.01;
}

/* ================= timeline (deadline-based, derived cycle) =========== */
function panelState(cfg, appearanceIdx, panelIdx, forceFilledIdx) {
  const a = cfg.appearances[appearanceIdx % cfg.appearances.length];
  return { filled: panelIdx < a.filled || panelIdx === forceFilledIdx,
           isDive: panelIdx === a.dive, filledCount: a.filled };
}
function buildTimeline(cfg) {
  const steps = [];
  let start = 0;
  cfg.appearances.forEach(function (a, i) {
    [["dwell", cfg.dwellMs], ["dive", cfg.diveMs], ["hold", cfg.holdZoomMs], ["pause", cfg.loopPauseMs]].forEach(function (ph) {
      steps.push({ phase: ph[0], appearance: i, dur: ph[1], startMs: start, dive: a.dive });
      start += ph[1];
    });
  });
  return { steps: steps, totalMs: start };
}
function stateAt(cfg, tMs) {
  const tl = buildTimeline(cfg);
  if (tl.totalMs <= 0) return tl.steps[0];
  let t = ((tMs % tl.totalMs) + tl.totalMs) % tl.totalMs;
  for (let i = 0; i < tl.steps.length; i++) {
    const s = tl.steps[i];
    if (t >= s.startMs && t < s.startMs + s.dur) return s;
  }
  return tl.steps[tl.steps.length - 1];
}
function phaseAt(cfg, elapsedMs) {
  const tl = buildTimeline(cfg);
  const step = stateAt(cfg, elapsedMs);
  const cyclePos = tl.totalMs > 0 ? ((elapsedMs % tl.totalMs) + tl.totalMs) % tl.totalMs : 0;
  return { step: step, cyclePos: cyclePos, inPhaseMs: cyclePos - step.startMs,
           remainingMs: step.startMs + step.dur - cyclePos,
           cycleIndex: tl.totalMs > 0 ? Math.floor(elapsedMs / tl.totalMs) : 0,
           key: (tl.totalMs > 0 ? Math.floor(elapsedMs / tl.totalMs) : 0) + ":" + step.appearance + ":" + step.phase };
}
function schedulerDelay(stepEndMs, elapsedMs, minStepMs) {
  return Math.max(minStepMs, stepEndMs - elapsedMs);
}

/* ================= art renderer (generic, registry-driven) ============ */
function esc(s) { return String(s).replace(/&/g, "&amp;").replace(/</g, "&lt;").replace(/>/g, "&gt;"); }
function fillOf(palette, v) {
  if (typeof v === "string" && v.charAt(0) === "$") {
    const k = v.slice(1);
    return Object.prototype.hasOwnProperty.call(palette, k) ? palette[k] : v;
  }
  return v;
}
function primSVG(p, palette) {
  const F = function (v) { return fillOf(palette, v); };
  const SW = function (q) { return q.strokeWidth !== undefined ? q.strokeWidth : 1; };
  if (p.t === "rect") return '<rect x="' + p.x + '" y="' + p.y + '" width="' + p.w + '" height="' + p.h + '"' +
    (p.rx ? ' rx="' + p.rx + '"' : "") + ' fill="' + F(p.fill) + '"' +
    (p.stroke ? ' stroke="' + F(p.stroke) + '" stroke-width="' + SW(p) + '"' : "") +
    (p.opacity !== undefined ? ' opacity="' + p.opacity + '"' : "") +
    (p.transform ? ' transform="' + p.transform + '"' : "") + "/>";
  if (p.t === "circle") return '<circle cx="' + p.cx + '" cy="' + p.cy + '" r="' + p.r + '" fill="' + F(p.fill) + '"' +
    (p.stroke ? ' stroke="' + F(p.stroke) + '" stroke-width="' + SW(p) + '"' : "") + "/>";
  if (p.t === "line") return '<line x1="' + p.x1 + '" y1="' + p.y1 + '" x2="' + p.x2 + '" y2="' + p.y2 +
    '" stroke="' + F(p.stroke) + '" stroke-width="' + SW(p) + '"/>';
  if (p.t === "polygon") return '<polygon points="' + p.points + '" fill="' + F(p.fill) + '"' +
    (p.opacity !== undefined ? ' opacity="' + p.opacity + '"' : "") + "/>";
  if (p.t === "path") return '<path d="' + p.d + '" fill="' + F(p.fill || "none") + '"' +
    (p.stroke ? ' stroke="' + F(p.stroke) + '" stroke-width="' + SW(p) + '"' : "") + "/>";
  if (p.t === "text") {
    const inner = p.segments ? p.segments.map(function (sg) {
      return '<tspan fill="' + F(sg.fill) + '">' + esc(sg.t) + "</tspan>"; }).join("") : esc(p.content || "");
    return '<text x="' + p.x + '" y="' + p.y + '" font-size="' + p.size + '"' +
      (p.anchor ? ' text-anchor="' + p.anchor + '"' : "") +
      ' fill="' + F(p.fill) + '" font-weight="' + (p.weight || "normal") + '">' + inner + "</text>";
  }
  if (p.t === "grid") { let s = "";
    for (let r = 0; r < p.rows; r++) for (let c = 0; c < p.cols; c++)
      s += '<rect x="' + (p.x + c * (p.cellW + p.gapX)) + '" y="' + (p.y + r * (p.cellH + p.gapY)) +
           '" width="' + p.cellW + '" height="' + p.cellH + '" fill="' + F(p.fills[(r + c) % p.fills.length]) + '"/>';
    return s; }

## Construction (what changed in this attempt, stated neutrally)

One self-contained HTML file: a DOM grid page (cream comic page, header strip, numbered yellow-tabbed panels with code-drawn SVG miniatures) playing a looping sequence of appearances; each appearance fills more panels, holds, then the camera dives into the next chapter's panel, which takes over the frame. - **V-audit (validation termination + range audit).** Every numeric field of the nested groups (`layout`, `labels`, `typography`, `artViewBox`) now carries an explicit terminating range with default restoration on violation — including `labels.shrinkStepPx`, which must be strictly positive (range 0.05–5) because both shrink helpers loop on it. `labels.maxPx` is raised to at least the resolved `minPx` when the pair inverts. `labels.numberPadWidth` is range-checked as an integer; `separator` / `generatedTitlePrefix` must be non-empty strings. `layout.diveEasing` must be a CSS keyword or a well-formed `cubic-bezier(...)` (four numeric components, leading-dot decimals allowed).
