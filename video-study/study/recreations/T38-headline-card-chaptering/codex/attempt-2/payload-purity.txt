## Technique facts (neutral)
Every section of the specimen opens on a designed title card in one visual system:
a headline, often with a date/provenance chip, plus the section's key object
(a computed pie; a formula with a legend). Cards double as the video's chapter
map. Anchor frames: card 1 (headline "Dark energy", provenance line, pie with
segment labels, takeaway pill) fully formed at segment t≈4s; card 2 (headline
"Albert Einstein", date pill chip "1905", formula hero with legend, takeaway
pill) is a specimen from elsewhere in the same video. The digest grid-change
census (2s bins) at the two anchors reads 0.022 (t=302) and 0.030 (t=718),
0.21 (t=722): no blank interval at a card anchor.

## Construction facts
- One self-contained HTML file; all content, timing, palette, geometry and
  style values live in CONFIG; construction reads CONFIG only.
- Timeline is derived, not hand-set: each card carries durationMs (formation
  start → chapter boundary) and relative entrance offsets; deriveTimeline()
  computes absolute checkpoints, the handover overlap (incoming formation
  starts handoverMs before the boundary; the outgoing card stays fully opaque
  until its boundary, then fades over fadeOutMs), formed times and holds.
  validateConfig() rejects non-increasing offsets, formation ending after the
  handover start, invalid pie data, and a timeline total ≠ totalDurationMs.
- Per-card information load is computed by loadUnits() (headline + chip +
  takeaway + hero words, plus the demonstration tag where present); the hold
  pool is shared in proportion to load.
- Derived timeline (ms): card 1 — start 0, boundary 7000, end 7300, formed
  3420, hold 2980, load 26 units, load-proportional hold target 3440
  (residual −460). Card 2 — start 6400, boundary/end 14000, formed 9570,
  hold 4430, load 30 units, target 3970 (residual +460). Both residuals are
  inside the model's stated ±600ms uncertainty.
- Card 2 carries role "demonstration" and a rendered tag
  "SYSTEM DEMONSTRATION — SECOND CHAPTER FORM"; the timing model states the
  segment's second half is a labelled demonstration, not a source claim.
- Pie segment label text is generated from segment values; a segment spanning
  ≥359.99° renders a full-ellipse case; a zero/empty total is a validation
  error and renders no paths.
- Every CSS variable used in the stylesheet or SVG is bound from CONFIG

function wordCount(s){ return s ? s.trim().split(/\s+/).filter(Boolean).length : 0; }
function loadUnits(card){
  let u = wordCount(card.headline) + wordCount(card.chip.text) + wordCount(card.takeaway);
  if (card.demoTag) u += wordCount(card.demoTag);

function pieSpans(segments){
  const total = segments.reduce((a,s)=>a+s.value,0);
  if (!(total > 0)) return { total, spans: [], error: "zero-total" };
  let a = CONFIG.pieGeom.startDeg;
  const spans = segments.map(s=>{
    const span = (s.value/total)*360, out = { name: s.name, value: s.value, start: a, span, pct: (s.value/total)*100, full: span >= 359.99 };
    a += span; return out;
  });
  return { total, spans, error: null };
}
function segLabelText(seg, total){

function deriveTimeline(cfg){
  /* durationMs = formation start -> chapter boundary. The outgoing card stays
     fully opaque until its boundary, then fades over fadeOutMs (its end).
     The next card's formation starts handoverMs BEFORE the boundary. */
  const H = cfg.anim.handoverMs, F = cfg.anim.fadeOutMs;
  let prevBoundary = 0;
  const tl = cfg.cards.map((card, i)=>{
    const start = i === 0 ? 0 : prevBoundary - H;
    const boundary = start + card.durationMs;
    const end = i + 1 < cfg.cards.length ? boundary + F : boundary;
    const cp = {};
    for (const k of Object.keys(card.relOffsets)) cp[k] = start + card.relOffsets[k];
    const formed = cp.pillMs + cfg.anim.pillMs;
    const holdUntil = i + 1 < cfg.cards.length ? boundary - H : end;
    const out = { index: i, start, boundary, end, checkpoints: cp, formedMs: formed,
      holdMs: holdUntil - formed, loadUnits: loadUnits(card), holdTargetMs: 0 };
    prevBoundary = boundary;
    return out;
  });
  const pool = tl.reduce((a,c)=>a+c.holdMs,0), units = tl.reduce((a,c)=>a+c.loadUnits,0);
  for (const c of tl) c.holdTargetMs = Math.round(pool * c.loadUnits / units);
  return { cards: tl, totalMs: prevBoundary, handoverMs: H, fadeOutMs: F };
}
function validateConfig(cfg){
  const errs = [];
  const tl = deriveTimeline(cfg);
  cfg.cards.forEach((card, i)=>{
    const offs = Object.values(card.relOffsets);
    for (let j = 1; j < offs.length; j++) if (!(offs[j] > offs[j-1])) errs.push("card"+i+" offsets not increasing");
    if (i + 1 < cfg.cards.length && tl.cards[i].formedMs > tl.cards[i].boundary - cfg.anim.handoverMs) errs.push("card"+i+" forms after handover start");
    if (card.hero.type === "pie"){
      const r = pieSpans(card.hero.segments);
      if (r.error) errs.push("card"+i+" pie " + r.error);
      for (const s of card.hero.segments) if (!(s.value >= 0)) errs.push("card"+i+" negative segment");
    }
  });
  if (tl.totalMs !== cfg.totalDurationMs) errs.push("timeline total " + tl.totalMs + " != totalDurationMs " + cfg.totalDurationMs);
  return { errors: errs, timeline: tl };
}

    if (!visible) continue;
    const ph = phase(t, cp.headlineMs, A.fadeMs);
    o.h.style.opacity = ph; o.h.style.transform = `translateY(${(1-ph)*A.risePx}px)`;
    o.chip.style.opacity = phase(t, cp.chipMs, A.fadeMs);
    if (o.demo) o.demo.style.opacity = phase(t, cp.chipMs, A.fadeMs);
    const hh = phase(t, cp.heroMs, A.heroMs);
    o.hero.style.opacity = hh; o.hero.style.transform = `scale(${CONFIG.pieGeom.heroScaleFrom + (1-CONFIG.pieGeom.heroScaleFrom)*hh})`;
    o.labels.forEach((l,i)=>{ l.style.opacity = phase(t, cp.labelsMs + i*A.labelStaggerMs, A.fadeMs); });
    o.pill.style.opacity = phase(t, cp.pillMs, A.pillMs);
  }
}
/* clock: standalone rAF + external slave API for comparison.html */
let playing = false, startStamp = 0, baseMs = 0, rafId = 0;
function nowMs(){ return baseMs + (playing ? performance.now() - startStamp : 0); }
function tick(){ applyAt(nowMs() % CONFIG.totalDurationMs); if (playing) rafId = requestAnimationFrame(tick); }
window.__rec = {
  play(){ if (!playing){ playing = true; startStamp = performance.now(); tick(); } },
  pause(){ if (playing){ baseMs = nowMs(); playing = false; cancelAnimationFrame(rafId); } },
  seek(ms){ baseMs = ms; if (playing) startStamp = performance.now(); applyAt(ms); },
  durationMs: CONFIG.totalDurationMs,
  timeline: TIMELINE
};
applyAt(0);
window.__rec.play();
