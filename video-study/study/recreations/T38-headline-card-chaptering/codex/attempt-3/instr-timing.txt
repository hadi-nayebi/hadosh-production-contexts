function pieSpans(segments){
  const total = segments.reduce((a,s)=>a+s.value,0);
  if (!(total > 0)) return { total, spans: [], error: "zero-total" };
  let a = CONFIG.pieGeom.startDeg;
  const spans = segments.map(s=>{
    const span = (s.value/total)*360, out = { name: s.name, value: s.value, start: a, span, pct: (s.value/total)*100, full: span >= 359.99 };
    a += span; return out;
  });
  return { total, spans, error: null };
}

function deriveTimeline(cfg){
  /* SEQUENTIAL handover: durationMs = formation start -> chapter boundary.
     A non-last card fades out over fadeOutMs ENDING at its boundary
     (fade-out start = boundary − fadeOutMs); the next card's formation
     starts AT that boundary — no incoming checkpoint precedes it.
     The last card has no fade-out; its end is the derived total.
     formedMs = the LATEST completion across ALL elements of the card,
     staggered labels included. The derived total is the sole duration
     authority: there is no separate total-duration config value. */
  const F = cfg.anim.fadeOutMs;
  let prevBoundary = 0;
  const tl = cfg.cards.map((card, i)=>{
    const start = prevBoundary;
    const boundary = start + card.durationMs;
    const end = boundary;
    const cp = {};
    for (const k of Object.keys(card.relOffsets)) cp[k] = start + card.relOffsets[k];
    const completions = [
      cp.headlineMs + cfg.anim.fadeMs,
      cp.chipMs + cfg.anim.fadeMs,
      cp.heroMs + cfg.anim.heroMs,
      cp.pillMs + cfg.anim.pillMs
    ];
    const nSeg = card.hero && card.hero.type === "pie" ? card.hero.segments.length : 1;
    for (let li = 0; li < nSeg; li++) completions.push(cp.labelsMs + li * cfg.anim.labelStaggerMs + cfg.anim.fadeMs);
    const formed = Math.max.apply(null, completions);
    const isLast = i + 1 >= cfg.cards.length;
    const deadline = isLast ? end : boundary - F;
    const out = { index: i, start, boundary, end, fadeOutStart: isLast ? null : boundary - F,
      checkpoints: cp, formedMs: formed, deadlineMs: deadline,
      holdMs: deadline - formed, loadUnits: loadUnits(card), holdTargetMs: 0 };
    prevBoundary = boundary;
    return out;
  });
  const pool = tl.reduce((a,c)=>a+c.holdMs,0), units = tl.reduce((a,c)=>a+c.loadUnits,0);
  for (const c of tl) c.holdTargetMs = units > 0 ? Math.round(pool * c.loadUnits / units) : 0;
  return { cards: tl, totalMs: prevBoundary, fadeOutMs: F };
}
function validateConfig(cfg){
  const errs = [];
  const tl = deriveTimeline(cfg);
  cfg.cards.forEach((card, i)=>{
    const offs = Object.values(card.relOffsets);
    for (let j = 1; j < offs.length; j++) if (!(offs[j] > offs[j-1])) errs.push("card"+i+" offsets not increasing");
    if (tl.cards[i].formedMs > tl.cards[i].deadlineMs) errs.push("card"+i+" forms after its deadline (formed " + tl.cards[i].formedMs + " > deadline " + tl.cards[i].deadlineMs + ")");
    if (card.hero.type === "pie"){
      const r = pieSpans(card.hero.segments);
      if (r.error) errs.push("card"+i+" pie " + r.error);
      for (const s of card.hero.segments) if (!(s.value >= 0)) errs.push("card"+i+" negative segment");
    }
  });
  return { errors: errs, timeline: tl };
}

function pieSVG(card){
  const { total, spans, error } = pieSpans(card.hero.segments);
  const NS = "http://www.w3.org/2000/svg", PG = CONFIG.pieGeom;
  const svg = document.createElementNS(NS, "svg");
  svg.setAttribute("viewBox", `0 0 ${PG.vbW} ${PG.vbH}`); svg.style.width = "var(--pieSize)"; svg.style.overflow = "visible";
  if (error) { svg.setAttribute("data-pie-error", error); return svg; }
  const {cx, cy, rx, ry, depth} = PG;
  const pt = (a,r)=>{ const r2 = a*Math.PI/180; return [cx + r*Math.cos(r2), cy + (r*Math.sin(r2))*ry/rx]; };
  for (const sp of spans){
    const s = card.hero.segments[spans.indexOf(sp)];
    if (sp.full){
      const side = document.createElementNS(NS,"ellipse");
      side.setAttribute("cx", cx); side.setAttribute("cy", cy+depth); side.setAttribute("rx", rx); side.setAttribute("ry", ry);
      side.setAttribute("fill", s.color); side.setAttribute("opacity", String(PG.sideOpacity)); svg.appendChild(side);
      const top = document.createElementNS(NS,"ellipse");
      top.setAttribute("cx", cx); top.setAttribute("cy", cy); top.setAttribute("rx", rx); top.setAttribute("ry", ry);
      top.setAttribute("fill", s.color); top.setAttribute("stroke", P.bg); top.setAttribute("stroke-width", String(PG.strokeW));
      svg.appendChild(top); continue;
    }
    const a0 = sp.start, a1 = sp.start + sp.span, [x0,y0] = pt(a0,rx), [x1,y1] = pt(a1,rx);
    const large = sp.span > 180 ? 1 : 0;
    const side = document.createElementNS(NS,"path");
    side.setAttribute("d", `M ${x0} ${y0+depth} L ${x0} ${y0} A ${rx} ${ry} 0 ${large} 1 ${x1} ${y1} L ${x1} ${y1+depth} A ${rx} ${ry} 0 ${large} 0 ${x0} ${y0+depth} Z`);
    side.setAttribute("fill", s.color); side.setAttribute("opacity", String(PG.sideOpacity)); svg.appendChild(side);
    const top = document.createElementNS(NS,"path");
    top.setAttribute("d", `M ${cx} ${cy} L ${x0} ${y0} A ${rx} ${ry} 0 ${large} 1 ${x1} ${y1} Z`);
    top.setAttribute("fill", s.color); top.setAttribute("stroke", P.bg); top.setAttribute("stroke-width", String(PG.strokeW));
    svg.appendChild(top);
  }
  return svg;
}

function clamp01(v){ return v < 0 ? 0 : v > 1 ? 1 : v; }
function phase(t, at, dur){ return clamp01((t - at) / dur); }
function applyAt(t){
  for (const o of cardEls){
    const cp = o.tl.checkpoints, A = CONFIG.anim;
    const isLast = o.tl.index === cardEls.length - 1;
    const visible = t >= o.tl.start && t <= o.tl.end;
    let cardOpacity = visible ? 1 : 0;
    if (visible && !isLast) cardOpacity = 1 - phase(t, o.tl.boundary - A.fadeOutMs, A.fadeOutMs);
    o.c.style.opacity = cardOpacity;
    if (!visible) continue;
    const ph = phase(t, cp.headlineMs, A.fadeMs);
    o.h.style.opacity = ph; o.h.style.transform = `translateY(${(1-ph)*A.risePx}px)`;
    o.chip.style.opacity = phase(t, cp.chipMs, A.fadeMs);
    if (o.demo) o.demo.style.opacity = phase(t, cp.chipMs, A.fadeMs);
    const hh = phase(t, cp.heroMs, A.heroMs);
    o.hero.style.opacity = hh; o.hero.style.transform = `scale(${CONFIG.pieGeom.heroScaleFrom + (1-CONFIG.pieGeom.heroScaleFrom)*hh})`;
    o.labels.forEach((l,i)=>{ l.style.opacity = phase(t, cp.labelsMs + i*A.labelStaggerMs, A.fadeMs); });
    o.pill.style.opacity = phase(t, cp.pillMs, A.pillMs);
  }
}

let playing = false, startStamp = 0, baseMs = 0, rafId = 0;
function nowMs(){ return baseMs + (playing ? performance.now() - startStamp : 0); }
function tick(){ applyAt(nowMs() % TIMELINE.totalMs); if (playing) rafId = requestAnimationFrame(tick); }
window.__rec = {
  play(){ if (!playing){ playing = true; startStamp = performance.now(); tick(); } },
  pause(){ if (playing){ baseMs = nowMs(); playing = false; cancelAnimationFrame(rafId); } },
  seek(ms){ baseMs = ms; if (playing) startStamp = performance.now(); applyAt(ms); },
  durationMs: TIMELINE.totalMs,
  timeline: TIMELINE
};
