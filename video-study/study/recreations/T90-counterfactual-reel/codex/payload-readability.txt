/* ================= Render (pure function of t) ================= */
function branchPath(upto){
  const g = CFG.geom, steps = 240;
  const n = Math.round(steps * upto);
  const p0 = cubic(g.branchP0, g.branchP1, g.branchP2, g.branchP3, 0);
  if (n <= 0) return { d: 'M ' + p0[0].toFixed(1) + ' ' + p0[1].toFixed(1), end: p0 };
  let d = '';
  for (let i = 0; i <= n; i++){
    const p = cubic(g.branchP0, g.branchP1, g.branchP2, g.branchP3, (i / steps));
    d += (i === 0 ? 'M ' : ' L ') + p[0].toFixed(1) + ' ' + p[1].toFixed(1);
    if (i === n) return { d: d, end: p };
  }
  return { d: d, end: cubic(g.branchP0, g.branchP1, g.branchP2, g.branchP3, upto) };
}
function beatCompleteAt(b){
  return (typeof b.completeMs === 'number') ? b.completeMs : CFG.timing.reelEndMs;
}
let lastBranchKey = '', lastBeatKey = '';
function render(tMs){
  const total = CFG.timing.totalMs;
  const t = ((tMs % total) + total) % total;
  const P = CFG.palette, g = CFG.geom;
  /* branch draws during intro, then holds complete */
  const prog = Math.min(1, t / CFG.introMs);
  const bk = prog.toFixed(4);
  if (bk !== lastBranchKey){
    lastBranchKey = bk;
    const bp = branchPath(prog);
    branchG.innerHTML = '<path d="'+bp.d+'" fill="none" stroke="'+P.branch+'" stroke-width="'+g.branchWidthPx+'" stroke-linecap="round"/>'
      + '<circle cx="'+bp.end[0]+'" cy="'+bp.end[1]+'" r="'+g.branchDotPx+'" fill="'+P.accent+'"/>';
  }
  /* beats: discrete visibility, accumulate, never change once landed */
  const key = CFG.beats.map(function(b){ return t >= b.atMs ? '1' : '0'; }).join('')
    + '|' + CFG.beats.map(function(b){ return (b.icon === 'lamp-counter' && t >= beatCompleteAt(b)) ? '1' : '0'; }).join('');
  if (key !== lastBeatKey){
    lastBeatKey = key;
    let out = '';
    CFG.beats.forEach(function(b, i){
      if (t < b.atMs) { lineEls[i].style.display = 'none'; return; }
      const np = nodePoint(i);
      out += glyph(b.icon, np[0], np[1], g.iconPx, P.ink, P.accent);
      if (b.icon === 'lamp-counter' && t >= beatCompleteAt(b)) out += glyph('counter-part', np[0], np[1], g.iconPx, P.ink, P.accent);
      lineEls[i].style.display = 'block';
    });
    beatG.innerHTML = out;
  }
}
