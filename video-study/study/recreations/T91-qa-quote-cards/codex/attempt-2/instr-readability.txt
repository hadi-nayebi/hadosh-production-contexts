function deepMerge(base, over){
  if (Array.isArray(over)){
    if (!Array.isArray(base)) return over.slice();
    const merged = base.map(function(b, i){ return i < over.length ? deepMerge(b, over[i]) : b; });
    for (let i = base.length; i < over.length; i++) merged.push(over[i]);
    return merged;
  }
  if (over && typeof over === "object"){
    const out = Array.isArray(base) ? base.slice() : Object.assign({}, base);
    for (const k of Object.keys(over)) out[k] = deepMerge(base ? base[k] : undefined, over[k]);
    return out;
  }
  return over;
}

    return {mode: mode, slots: slots, freeMs: s.seqFreeMs, freeStart: start, cycleMs: start + s.seqFreeMs};
  }
  let start = 0;
  const slots = cfg.cards.map(function(c, i){
    const slot = {idx: i, start: start, slotMs: cfg.cardMs, enterMs: cfg.enterMs, exitMs: 0, revealSpanMs: 0};
    start += cfg.cardMs; return slot;
  });
  return {mode: mode, slots: slots, freeMs: cfg.endHoldMs, freeStart: start, cycleMs: start + cfg.endHoldMs};
}
function cycleMs(cfg){ return resolveSchedule(cfg).cycleMs; }

/* ============ line-growth reveal — stage times for a wrapped line count.
   1 line: [0]. 2 lines: [0, span] (the second line is the final line; the final
   anchor governs). 3+ lines: first at 0, second at min(2000, span/4), last at
   span, interior lines evenly between the second and last anchors. ============ */
function revealTimes(cfg, spanMs, n){
  if (n <= 1) return [0];
  if (n === 2) return [0, spanMs];
  const t2 = Math.min(cfg.schedule.secondLineMs, spanMs * cfg.schedule.secondLineFrac);
  const out = [0, t2];
  for (let k = 3; k < n; k++) out.push(t2 + (spanMs - t2) * (k - 2) / (n - 2));
  out.push(spanMs);
  return out;
}
function visibleLineCount(cfg, spanMs, n, local){
  if (!spanMs || spanMs <= 0) return n;
  const times = revealTimes(cfg, spanMs, n);
  let v = 0;
  for (let i = 0; i < times.length; i++) if (local >= times[i]) v = i + 1;
  return Math.max(1, v);
}

/* ============ phase machine — exactly one card, or the named card-free phase,
   owns every instant. Reveal clock starts at the slot's own start. ============ */
function stateAt(cfg, t, lineCounts){
  const sch = resolveSchedule(cfg);
  const total = sch.cycleMs;
  const u = ((t % total) + total) % total;
  if (u >= sch.freeStart){
    if (sch.mode === "uniform") return {idx: cfg.cards.length - 1, phase: "endhold", p: 1, visibleLines: lineCounts ? lineCounts[cfg.cards.length - 1] : 0, slotStart: sch.slots[sch.slots.length - 1].start};
    return {idx: -1, phase: "cardfree", p: 1, visibleLines: 0, slotStart: sch.freeStart};
  }
  for (let i = 0; i < sch.slots.length; i++){
    const sl = sch.slots[i];
    if (u >= sl.start && u < sl.start + sl.slotMs){
      const local = u - sl.start;
      const n = lineCounts ? lineCounts[sl.idx] : 0;
      if (sl.exitMs > 0 && local >= sl.slotMs - sl.exitMs)
        return {idx: sl.idx, phase: "exit", p: (local - (sl.slotMs - sl.exitMs)) / sl.exitMs, visibleLines: n, slotStart: sl.start};
      if (sl.enterMs > 0 && local < sl.enterMs)
        return {idx: sl.idx, phase: "enter", p: local / sl.enterMs, visibleLines: visibleLineCount(cfg, sl.revealSpanMs, n, local), slotStart: sl.start};
      return {idx: sl.idx, phase: "hold", p: 1, visibleLines: visibleLineCount(cfg, sl.revealSpanMs, n, local), slotStart: sl.start};
    }
  }
  return {idx: -1, phase: "cardfree", p: 1, visibleLines: 0, slotStart: sch.freeStart};
}

/* ============ viewport-derived geometry — effective stage width is the smaller
   of the configured stage width and the viewport width; one scale factor scales
   every horizontal geometry value; fonts never scale. ============ */
const H_KEYS = ["hudX","cardX","cardW","cardPad","tagX","tagPadX","markX","quoteX","quoteW","attrX","attrTextX"];
function effectiveGeom(cfg, viewportW){
  const effW = Math.min(cfg.stage.width, viewportW || cfg.stage.width);
  const s = effW / cfg.stage.width;
  const g = Object.assign({}, cfg.geom);
  for (const k of H_KEYS) g[k] = cfg.geom[k] * s;
  return {geom: g, scale: s, effW: effW};
}
function quoteZoneDepth(g){ return g.cardH - g.attrBottom - g.attrH - g.quoteY; }
function attrTextWidth(g){ return g.cardW - g.attrX - g.cardPad - g.attrTextX; }

/* ============ text wrapping on an injected metric basis — the browser supplies
   canvas measurements in the rendered font; other callers supply their own
   measurer in the same font. measure(text, px, kind) returns width in px. ============ */
function wrapText(text, maxW, px, measure, kind){
  const words = String(text).split(/\s+/).filter(Boolean);
  const lines = []; let cur = "";
  for (const w of words){
    const trial = cur ? cur + " " + w : w;
    if (cur && measure(trial, px, kind) > maxW){ lines.push(cur); cur = w; }
    else cur = trial;
  }
  if (cur) lines.push(cur);
  return lines.length ? lines : [""];
}

/* ============ fit contract — deterministic and ordered: shrink the quote size
   in fixed steps down to the floor until the quote fits its zone; if it still
   does not fit at the floor, reject naming the card and the overflow in px. ============ */
function fitCard(cfg, g, card, measure){
  const ty = cfg.type;
  const zone = quoteZoneDepth(g);
  let px = ty.quotePx;
  let lines = wrapText(card.quote, g.quoteW, px, measure, "quote");
  let blockH = lines.length * px * ty.quoteLineHeight;
  while (blockH > zone && px > cfg.fit.floorPx){
    px -= cfg.fit.stepPx;
    lines = wrapText(card.quote, g.quoteW, px, measure, "quote");
    blockH = lines.length * px * ty.quoteLineHeight;
  }
  if (blockH > zone) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: Math.round((blockH - zone) * 10) / 10};
  const spkLines = wrapText(card.speaker, attrTextWidth(g), ty.speakerPx, measure, "speaker");
  const roleText = card.role + cfg.labels.sep + card.year;
  const roleLines = wrapText(roleText, attrTextWidth(g), ty.rolePx, measure, "role");
  const attrH = ty.speakerPx * ty.speakerLineHeight + g.attrGap + roleLines.length * ty.rolePx * ty.roleLineHeight;
  if (spkLines.length !== 1) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: 0, attrFail: "speaker wraps to " + spkLines.length + " lines (expected exactly 1)"};
  if (roleLines.length > 4) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: 0, attrFail: "role wraps to " + roleLines.length + " lines (maximum 4)"};
  if (attrH > g.attrH) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: Math.round((attrH - g.attrH) * 10) / 10, attrFail: "attribution block " + attrH.toFixed(1) + "px exceeds reserved " + g.attrH + "px"};
  return {ok: true, px: px, lines: lines.length, blockH: blockH, zone: zone, spkLines: spkLines.length, roleLines: roleLines.length, attrH: attrH};
}
