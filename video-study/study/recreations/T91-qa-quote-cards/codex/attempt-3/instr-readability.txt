   span, interior lines evenly between the second and last anchors. ============ */
function revealTimes(cfg, spanMs, n){
  if (n <= 1) return [0];
  if (n === 2) return [0, spanMs];
  const t2 = Math.min(cfg.schedule.secondLineMs, spanMs * cfg.schedule.secondLineFrac);
  const out = [0, t2];
  for (let k = 3; k < n; k++) out.push(t2 + (spanMs - t2) * (k - 2) / (n - 2));
  out.push(spanMs);
  return out;
}
function visibleLineCount(cfg, spanMs, n, local){
  if (!spanMs || spanMs <= 0) return n;
  const times = revealTimes(cfg, spanMs, n);
  let v = 0;
  for (let i = 0; i < times.length; i++) if (local >= times[i]) v = i + 1;
  return Math.max(1, v);
}

/* ============ phase machine — exactly one card, or the named card-free phase,
   owns every instant. Reveal clock starts at the slot's own start. ============ */
function stateAt(cfg, t, lineCounts){
  const sch = resolveSchedule(cfg);
  const total = sch.cycleMs;
  const u = ((t % total) + total) % total;
  if (u >= sch.freeStart){
    return {idx: -1, phase: "cardfree", p: 1, visibleLines: 0, slotStart: sch.freeStart};
  }
  for (let i = 0; i < sch.slots.length; i++){
    const sl = sch.slots[i];
    if (u >= sl.start && u < sl.start + sl.slotMs){
      const local = u - sl.start;
      const n = lineCounts ? lineCounts[sl.idx] : 0;
      if (sl.exitMs > 0 && local >= sl.slotMs - sl.exitMs)
        return {idx: sl.idx, phase: "exit", p: (local - (sl.slotMs - sl.exitMs)) / sl.exitMs, visibleLines: n, slotStart: sl.start};
      if (sl.enterMs > 0 && local < sl.enterMs)
        return {idx: sl.idx, phase: "enter", p: local / sl.enterMs, visibleLines: visibleLineCount(cfg, sl.revealSpanMs, n, local), slotStart: sl.start};
      return {idx: sl.idx, phase: "hold", p: 1, visibleLines: visibleLineCount(cfg, sl.revealSpanMs, n, local), slotStart: sl.start};
    }
  }
  return {idx: -1, phase: "cardfree", p: 1, visibleLines: 0, slotStart: sch.freeStart};
}

/* ============ viewport-derived geometry — effective stage width is the smaller
   of the configured stage width and the viewport width; one scale factor scales
   the layout keys listed in H_KEYS; the two stroke widths (borderW, attrBarW)
   are applied at face value; fonts never scale. ============ */
const H_KEYS = ["hudX","cardX","cardW","cardPad","tagX","tagPadX","markX","quoteX","quoteW","attrX","attrTextX"];
function effectiveGeom(cfg, viewportW){
  const effW = Math.min(cfg.stage.width, viewportW || cfg.stage.width);
  const s = effW / cfg.stage.width;
  const g = Object.assign({}, cfg.geom);
  for (const k of H_KEYS) g[k] = cfg.geom[k] * s;
  return {geom: g, scale: s, effW: effW};
}
function quoteZoneDepth(g){ return g.cardH - g.attrBottom - g.attrH - g.quoteY; }
function attrTextWidth(g){ return g.cardW - g.attrX - g.cardPad - g.attrTextX; }
function recomputeAt(cfg, viewportW, measure){
  if (typeof viewportW === "number" && isFinite(viewportW) && viewportW < cfg.stage.minWidth)
    return {ok: false, error: "viewport width " + viewportW + "px is below the supported minimum width " + cfg.stage.minWidth + "px"};
  const eff = effectiveGeom(cfg, viewportW);
  const fits = cfg.cards.map(function(c){ return fitCard(cfg, eff.geom, c, measure); });
  return {ok: true, eff: eff, fits: fits, lineCounts: fits.map(function(f){ return f.lines; })};
}

/* ============ text wrapping on an injected metric basis — the browser supplies
   canvas measurements in the rendered font; other callers supply their own
   measurer in the same font. measure(text, px, kind) returns width in px. ============ */
function wrapText(text, maxW, px, measure, kind){
  const words = String(text).split(/\s+/).filter(Boolean);
  const lines = []; let cur = "";
  for (const w of words){
    const trial = cur ? cur + " " + w : w;
    if (cur && measure(trial, px, kind) > maxW){ lines.push(cur); cur = w; }
    else cur = trial;
  }
  if (cur) lines.push(cur);
  return lines.length ? lines : [""];
}

/* ============ fit contract — deterministic and ordered: shrink the quote size
   in fixed steps down to the floor until the quote fits its zone in height and
   in width; if it still does not fit at the floor, reject naming the card and
   the overflow in px. ============ */
function wideWord(lines, maxW, px, measure, kind){
  for (const ln of lines){
    const w = measure(ln, px, kind);
    if (w > maxW) return {word: ln, overflowPx: Math.round((w - maxW) * 10) / 10};
  }
  return null;
}
function fitCard(cfg, g, card, measure){
  const ty = cfg.type;
  const zone = quoteZoneDepth(g);
  let px = ty.quotePx;
  let lines = wrapText(card.quote, g.quoteW, px, measure, "quote");
  let blockH = lines.length * px * ty.quoteLineHeight;
  let wide = wideWord(lines, g.quoteW, px, measure, "quote");
  let steps = 0;
  while ((blockH > zone || wide) && px > cfg.fit.floorPx){
    if (steps >= cfg.fit.maxSteps) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: 0, attrFail: "shrink step bound (" + cfg.fit.maxSteps + ") exceeded"};
    px = Math.max(cfg.fit.floorPx, px - cfg.fit.stepPx);
    steps++;
    lines = wrapText(card.quote, g.quoteW, px, measure, "quote");
    blockH = lines.length * px * ty.quoteLineHeight;
    wide = wideWord(lines, g.quoteW, px, measure, "quote");
  }
  if (blockH > zone) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: Math.round((blockH - zone) * 10) / 10};
  if (wide) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: wide.overflowPx, hFail: "single word \"" + wide.word + "\" exceeds the quote width by " + wide.overflowPx + "px at the floor size"};
  const spkLines = wrapText(card.speaker, attrTextWidth(g), ty.speakerPx, measure, "speaker");
  const roleText = card.role + cfg.labels.sep + card.year;
  const roleLines = wrapText(roleText, attrTextWidth(g), ty.rolePx, measure, "role");
  const attrH = ty.speakerPx * ty.speakerLineHeight + g.attrGap + roleLines.length * ty.rolePx * ty.roleLineHeight;
  const spkWide = wideWord(spkLines, attrTextWidth(g), ty.speakerPx, measure, "speaker");
  if (spkWide) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: spkWide.overflowPx, attrFail: "speaker token \"" + spkWide.word + "\" exceeds the attribution width by " + spkWide.overflowPx + "px"};
  const roleWide = wideWord(roleLines, attrTextWidth(g), ty.rolePx, measure, "role");
  if (roleWide) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: roleWide.overflowPx, attrFail: "role token \"" + roleWide.word + "\" exceeds the attribution width by " + roleWide.overflowPx + "px"};
  if (spkLines.length !== 1) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: 0, attrFail: "speaker wraps to " + spkLines.length + " lines (expected exactly 1)"};
  if (roleLines.length > cfg.fit.roleMaxLines) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: 0, attrFail: "role wraps to " + roleLines.length + " lines (maximum " + cfg.fit.roleMaxLines + ")"};
  if (attrH > g.attrH) return {ok: false, cardQuote: card.quote, px: px, lines: lines.length, blockH: blockH, zone: zone, overflowPx: Math.round((attrH - g.attrH) * 10) / 10, attrFail: "attribution block " + attrH.toFixed(1) + "px exceeds reserved " + g.attrH + "px"};
  return {ok: true, px: px, lines: lines.length, blockH: blockH, zone: zone, spkLines: spkLines.length, roleLines: roleLines.length, attrH: attrH};
}

/* ============ validation — runs before first paint. Any violation renders the
   error state instead of the schedule. ============ */
function isHex(v){ return typeof v === "string" && /^#[0-9a-fA-F]{6}$/.test(v); }
const MAX_MS = 3600000, MAX_GEOM = 10000, MAX_TYPE = 500, MS_PER_MIN = 60000;
function validateConfig(cfg){
  const errors = [];
  if (!cfg || typeof cfg !== "object" || Array.isArray(cfg)) return ["config: expected an object"];
  const GROUPS = ["hud","palette","labels","schedule","fit","type","stage","geom"];
  for (const g of GROUPS){
    if (!cfg[g] || typeof cfg[g] !== "object" || Array.isArray(cfg[g])) errors.push(g + ": expected a non-null object group");
  }
  if (!Array.isArray(cfg.cards) || cfg.cards.length === 0) errors.push("cards: expected a non-empty array of cards");
  if (errors.length) return errors;
  function num(path, v, lo, hi, integer, loOpen){
    const okType = typeof v === "number" && isFinite(v);
    if (!okType){ errors.push(path + ": expected a finite number"); return false; }
    if (integer && !Number.isInteger(v)){ errors.push(path + ": expected an integer"); return false; }
    if (loOpen ? !(v > lo) : !(v >= lo)){ errors.push(path + ": expected a number " + (loOpen ? "> " : ">= ") + lo); return false; }
    if (hi !== null && !(v <= hi)){ errors.push(path + ": expected a number <= " + hi); return false; }
    return true;
  }
  function str(path, v){ if (typeof v !== "string" || !v) errors.push(path + ": expected a non-empty string"); }
  str("hud.dayLine", cfg.hud.dayLine); str("hud.clock", cfg.hud.clock);
  if (!Number.isInteger(cfg.hud.year)) errors.push("hud.year: expected an integer");
  str("labels.tag", cfg.labels.tag); str("labels.mark", cfg.labels.mark);
  if (typeof cfg.labels.sep !== "string") errors.push("labels.sep: expected a string");
  for (const k of Object.keys(cfg.palette)) if (!isHex(cfg.palette[k])) errors.push("palette." + k + ": expected a #rrggbb hex color");
  for (const k of Object.keys(cfg.geom)) num("geom." + k, cfg.geom[k], 0, MAX_GEOM, false, true);
  num("enterMs", cfg.enterMs, 0, MAX_MS, true, true);
  const ty = cfg.type;
  for (const k of Object.keys(ty)){
    if (/Px$/.test(k)) num("type." + k, ty[k], 0, MAX_TYPE, false, true);
    else if (/LineHeight$/.test(k)) { if (!(typeof ty[k] === "number" && isFinite(ty[k]) && ty[k] >= 1 && ty[k] <= 5)) errors.push("type." + k + ": expected a finite number in [1, 5]"); }
    else if (/SpacingEm$/.test(k)) { if (!(typeof ty[k] === "number" && isFinite(ty[k]) && ty[k] >= 0 && ty[k] < 2)) errors.push("type." + k + ": expected a finite number in [0, 2)"); }
    else if (typeof ty[k] === "string") str("type." + k, ty[k]);
  }
  num("stage.width", cfg.stage.width, 0, MAX_GEOM, false, true);
  num("stage.minWidth", cfg.stage.minWidth, 0, MAX_GEOM, false, true);
  if (typeof cfg.stage.minWidth === "number" && typeof cfg.stage.width === "number" && cfg.stage.minWidth > cfg.stage.width) errors.push("stage.minWidth (" + cfg.stage.minWidth + ") must be <= stage.width (" + cfg.stage.width + ")");
  const s = cfg.schedule;
  if (s.mode === "uniform") errors.push("schedule.mode: the uniform mode is retired — expected one of specimen, sequence");
  else if (["specimen","sequence"].indexOf(s.mode) < 0) errors.push("schedule.mode: expected one of specimen, sequence");
  const SMS = [["specimenRevealMs",0],["specimenHoldMs",0],["specimenExitMs",0],["specimenFreeMs",0],["exitMs",0],["seqFreeMs",0],["minHoldMs",0],["minHoldFloorMs",0],["secondLineMs",0],["defaultRevealSpanMs",0]];
  for (const pair of SMS) num("schedule." + pair[0], s[pair[0]], pair[1], MAX_MS, true, pair[0] === "specimenHoldMs" || pair[0] === "specimenExitMs" || pair[0] === "exitMs" || pair[0] === "minHoldMs" || pair[0] === "minHoldFloorMs");
  if (!(typeof s.secondLineFrac === "number" && isFinite(s.secondLineFrac) && s.secondLineFrac > 0 && s.secondLineFrac <= 1)) errors.push("schedule.secondLineFrac: expected a finite number in (0, 1]");
  num("schedule.readingWpm", s.readingWpm, 0, 1000, false, true);
  if (!Number.isInteger(s.specimenCard)) errors.push("schedule.specimenCard: expected an integer index into cards");
  cfg.cards.forEach(function(c, i){
    if (!c || typeof c !== "object" || Array.isArray(c)){ errors.push("cards[" + i + "]: expected an object"); return; }
    str("cards[" + i + "].quote", c.quote); str("cards[" + i + "].speaker", c.speaker); str("cards[" + i + "].role", c.role);
    if (!Number.isInteger(c.year)) errors.push("cards[" + i + "].year: expected an integer");
    num("cards[" + i + "].slotMs", c.slotMs, 0, MAX_MS, true, true);
    num("cards[" + i + "].revealSpanMs", c.revealSpanMs, 0, MAX_MS, true, false);
  });
  const ft = cfg.fit;
  num("fit.floorPx", ft.floorPx, 0, MAX_TYPE, false, true);
  num("fit.stepPx", ft.stepPx, 0, MAX_TYPE, false, true);
  num("fit.minLegiblePx", ft.minLegiblePx, 0, MAX_TYPE, false, true);
  if (!Number.isInteger(ft.maxSteps) || ft.maxSteps <= 0) errors.push("fit.maxSteps: expected a positive integer");
  if (!Number.isInteger(ft.roleMaxLines) || ft.roleMaxLines < 1 || ft.roleMaxLines > 20) errors.push("fit.roleMaxLines: expected an integer in [1, 20]");
  if (typeof ft.floorPx === "number" && typeof ft.minLegiblePx === "number" && ft.floorPx < ft.minLegiblePx) errors.push("fit.floorPx (" + ft.floorPx + ") is below fit.minLegiblePx (" + ft.minLegiblePx + ")");
  if (typeof ft.floorPx === "number" && typeof ty.quotePx === "number" && ft.floorPx >= ty.quotePx) errors.push("fit.floorPx (" + ft.floorPx + ") must be < type.quotePx (" + ty.quotePx + ")");
  if (typeof ft.stepPx === "number" && typeof ft.floorPx === "number" && typeof ty.quotePx === "number" && ft.stepPx > ty.quotePx - ft.floorPx) errors.push("fit.stepPx (" + ft.stepPx + ") must be <= type.quotePx - fit.floorPx (" + (ty.quotePx - ft.floorPx) + ")");
  if (!errors.length){
    for (const bw of [cfg.stage.width, cfg.stage.minWidth]){
      const eg = effectiveGeom(cfg, bw).geom, effW = Math.min(cfg.stage.width, bw);
      if (eg.cardX + eg.cardW > effW + 0.001) errors.push("geometry: card extends past the stage width at viewport " + bw + "px");
      if (eg.quoteX + eg.quoteW > eg.cardW + 0.001) errors.push("geometry: quote zone extends past the card at viewport " + bw + "px");
      if (eg.markX >= eg.cardW || eg.tagX >= eg.cardW || eg.hudX >= effW) errors.push("geometry: a positioned element starts outside its container at viewport " + bw + "px");
    }
    const gv = cfg.geom;
    if (gv.cardY + gv.cardH > gv.stageH) errors.push("geometry: card extends past the stage height");
    if (gv.attrBottom + gv.attrH > gv.cardH) errors.push("geometry: attribution block extends past the card");
    if (gv.quoteY >= gv.cardH || gv.markY >= gv.cardH || gv.tagY >= gv.cardH) errors.push("geometry: a positioned element starts outside the card vertically");
  }
  if (errors.length) return errors;
  if (s.mode === "specimen"){
    if (!(s.specimenCard >= 0 && s.specimenCard < cfg.cards.length)) errors.push("schedule.specimenCard: index " + s.specimenCard + " is outside the cards list of length " + cfg.cards.length);
    else {
      const maxTok = Math.max.apply(null, cfg.cards.map(tokenCount));
      const floor = tokenCount(cfg.cards[s.specimenCard]) === maxTok ? s.minHoldFloorMs : s.minHoldMs;
      if (s.specimenHoldMs < floor) errors.push("schedule: specimen hold " + s.specimenHoldMs + "ms is below the minimum full hold " + floor + "ms for the selected card");
    }
    return errors;
  }
  const maxTok = Math.max.apply(null, cfg.cards.map(tokenCount));
  cfg.cards.forEach(function(c, i){
    if (cfg.enterMs >= c.slotMs) errors.push("enterMs (" + cfg.enterMs + ") must be < cards[" + i + "].slotMs (" + c.slotMs + ")");
    if (s.exitMs >= c.slotMs) errors.push("schedule.exitMs (" + s.exitMs + ") must be < cards[" + i + "].slotMs (" + c.slotMs + ")");
    const minHold = tokenCount(c) === maxTok ? s.minHoldFloorMs : s.minHoldMs;
    if (c.revealSpanMs + minHold + s.exitMs > c.slotMs)
      errors.push("cards[" + i + "]: reveal " + c.revealSpanMs + " + minimum hold " + minHold + " + exit " + s.exitMs + " exceeds slotMs " + c.slotMs);
    const reading = Math.ceil(tokenCount(c) * MS_PER_MIN / s.readingWpm);
    if (c.slotMs < reading) errors.push("cards[" + i + "]: slotMs " + c.slotMs + " is below the reading interval " + reading + "ms for its " + tokenCount(c) + " tokens at " + s.readingWpm + " wpm");
  });
  return errors;
}
