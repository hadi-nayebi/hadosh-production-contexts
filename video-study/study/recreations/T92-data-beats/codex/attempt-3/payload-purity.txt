    const v = over.$replace;
    return Array.isArray(v) ? v.slice() : (v && typeof v === "object" ? Object.assign({}, v) : v);
  }
  if (Array.isArray(over)){
    if (!Array.isArray(base)) return over.slice();
    const merged = base.map(function(b, i){ return i < over.length ? deepMerge(b, over[i]) : b; });
    for (let i = base.length; i < over.length; i++) merged.push(over[i]);
    return merged;
  }

    const out = Array.isArray(base) ? base.slice() : Object.assign({}, base);

    cfg.beats.forEach(function(b, i){
      if (!b || typeof b !== "object") return;
      if (b.kind === "chart" && Array.isArray(b.bars) && b.bars.length && b.bars.every(function(bar){ return bar && fin(bar.h) && bar.h >= 0; })){
        const nb = b.bars.length;
        const seriesRight = g.chartInsetX + g.barInsetX + (nb - 1) * (g.barW + g.barGap) + g.barW;
        if (seriesRight > g.chartW - g.chartInsetX) problems.push("beats["+i+"] series right edge exceeds the chart width bound");
        const stepStart = g.chartInsetX + g.barInsetX + nb * (g.barW + g.barGap) + g.stepStartGap;
        const stepRight = stepStart + (g.stepCount - 1) * g.stepDx + g.stepW;
        if (stepRight > g.chartW) problems.push("beats["+i+"] steps right edge exceeds the chart width");
        const inkTop = g.beamY - (g.stepCount - 1) * g.stepRise - g.stepMarkGap - g.markPx;
        if (inkTop < 0) problems.push("beats["+i+"] step mark ink top is above the chart top");
        b.bars.forEach(function(bar, j){ if (bar.h > g.beamY) problems.push("beats["+i+"].bars["+j+"] height exceeds beamY"); });
        const pm = /^\s*(-?\d+(?:\.\d+)?)\s*%\s*$/.exec(String(b.value));
        if (pm){
          const hangH = parseFloat(pm[1]) * g.pxPerPercent;
          if (hangH >= 0 && hangH > g.beamY) problems.push("beats["+i+"] positive hang exceeds beamY");
          if (hangH < 0 && -hangH > g.chartH - (g.beamY + g.beamH + g.hangGap)) problems.push("beats["+i+"] negative hang exceeds the chart space below the beam");
          if (isObj(cfg.type) && fin(cfg.type.valuePx)){
            const base = g.beamY + g.beamH + Math.max(0, -hangH) + g.numeralGap + cfg.type.valuePx * g.numeralBaseRatio;
            if (base > g.chartH) problems.push("beats["+i+"] numeral baseline exceeds the chart height");
          }
        }
      }
      if (/(\d+)\s*in\s*(\d+)/.test(String(b.value)) || b.units !== undefined){
        let du = null;
        try { du = deriveUnits(b); } catch(e) { du = null; }
        if (du && Number.isInteger(du.total) && du.total >= 1){
          const rowW = du.total * g.unitSize + (du.total - 1) * g.unitGap;
          if (rowW > cfg.stage.width) problems.push("beats["+i+"] unit row width exceeds the stage width");
        }
      }
    });
    if (g.chartX + g.chartW > cfg.stage.width) problems.push("chart right edge exceeds the stage width");
    if (g.chartY + g.chartH > g.stageH) problems.push("chart bottom exceeds the stage height");
    if (g.cardX + g.cardW > cfg.stage.width) problems.push("card right edge exceeds the stage width");
  }
  if (!fin(cfg.endHoldMs) || cfg.endHoldMs < 0) problems.push("endHoldMs must be a finite number >= 0");
  return problems;
}

/* ============ phase machine — one beat owns the stage (phase exclusivity) ============ */
function cycleMs(cfg){ return cfg.beats.reduce(function(s,b){ return s + b.slotMs; }, 0) + cfg.endHoldMs; }
function stateAt(cfg, t){
  const total = cycleMs(cfg);
  const tt = ((t % total) + total) % total;
  let acc = 0;
  for (let i = 0; i < cfg.beats.length; i++){
    const b = cfg.beats[i], slot = b.slotMs;
    if (tt < acc + slot){
      const local = tt - acc;
      const lead = b.dialogue.graphicEntrance, eb = b.enterBorderMs || 0, ef = (b.dialogue.followUp - b.dialogue.graphicEntrance) - eb;
      const clear = b.clearMs || 0, restore = b.restoreMs || 0;
      if (local < lead) return {idx: i, phase:"lead", p: local / Math.max(lead, 1)};
      if (local < lead + eb + ef) return {idx: i, phase:"enter", p: (local - lead) / Math.max(eb + ef, 1),
        bp: eb > 0 ? Math.min(1, (local - lead) / eb) : 1,
        tp: ef > 0 ? Math.min(1, Math.max(0, (local - lead - eb) / ef)) : 1};
      if (clear + restore > 0 && local >= slot - clear - restore){
        if (local < slot - restore) return {idx: i, phase:"clear", p: (local - (slot - clear - restore)) / Math.max(clear, 1)};
        return {idx: i, phase:"restore", p: (local - (slot - restore)) / Math.max(restore, 1)};
      }
      return {idx: i, phase:"hold", p: 1};
    }
    acc += slot;
  }
  return {idx: cfg.beats.length - 1, phase:"endhold", p: 1};
}
function chartStagesFor(cfg, beat, p){
  const cr = cfg.chartReveal, g = cfg.geom;
  const seg = function(a, b){ return Math.min(1, Math.max(0, (p - a) / Math.max(b - a, 1e-9))); };
  const nb = (beat.bars || []).length, ns = g.stepCount;
  const win = function(start, span, i, count){ const w = span / Math.max(count, 1); return seg(start + i * w, start + (i + 1) * w); };
  return { beam: seg(cr.beamStart, cr.beamStart + cr.beamSpan),
    bars: Array.from({length: nb}, function(_, i){ return win(cr.barsStart, cr.barsSpan, i, nb); }),
    hang: seg(cr.hangStart, cr.hangStart + cr.hangSpan),
    steps: Array.from({length: ns}, function(_, i){ return win(cr.stepsStart, cr.stepsSpan, i, ns); }),
    numeral: seg(cr.numeralStart, cr.numeralStart + cr.numeralSpan) };
}
function hudYearAt(cfg, t){
  const st = stateAt(cfg, t);
  if (st.phase === "lead" || st.phase === "restore") return cfg.hud.sceneYear;
  return cfg.beats[st.idx].year;
}
