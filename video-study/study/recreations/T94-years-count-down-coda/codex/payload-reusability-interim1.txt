  /* ruler draw progress */
  s.rulerProg = t < T.rulerStartMs ? 0 : Math.min(1, (t - T.rulerStartMs) / T.rulerDrawMs);
  /* ticks lit, in order, after the line is drawn */
  const step = tickStepMs();
  let ticks = 0;
  if (s.rulerProg >= 1 && t >= T.tickStartMs) ticks = Math.min(CFG.geom.tickCount, 1 + Math.floor((t - T.tickStartMs) / step));
  s.ticks = ticks;
  s.figure = s.rulerProg >= 1 && t >= T.figureMs;
  return s;
}
function rulerSVG(s){
  const g = CFG.geom, P = CFG.palette;
  if (s.rulerProg <= 0) return '';
  const x0 = g.rulerX0, x1 = g.rulerX0 + (g.rulerX1 - g.rulerX0) * s.rulerProg, y = g.rulerY;
  let out = '<line x1="'+x0+'" y1="'+y+'" x2="'+x1.toFixed(1)+'" y2="'+y+'" stroke="'+P.ink+'" stroke-width="'+g.rulerWidthPx+'"/>';
  if (s.rulerProg >= 1) out += '<circle cx="'+g.rulerX1+'" cy="'+y+'" r="'+g.endDotPx+'" fill="'+P.ink+'"/>';
  /* ticks: small lit windows above the line, spaced over the year domain */
  for (let i = 0; i < s.ticks; i++){
    const tx = g.rulerX0 + (i + 1) * ((g.rulerX1 - g.rulerX0) / (g.tickCount + 1));
    out += '<rect x="'+(tx - g.tickSizePx / 2).toFixed(1)+'" y="'+(y - g.tickGapAbovePx - g.tickSizePx)+'" width="'+g.tickSizePx+'" height="'+g.tickSizePx+'" fill="'+P.window+'"/>';
  }
  /* figure at the ruler's left end (fromYear position on the domain) */
  if (s.figure){
    const fx = g.rulerX0 + g.figureGapPx, fh = g.figureH;
    out += '<g fill="'+P.accent+'"><circle cx="'+fx+'" cy="'+(y - fh + 4)+'" r="3.4"/><path d="M '+(fx - 4)+' '+y+' L '+(fx - 2.6)+' '+(y - fh + 9)+' L '+(fx + 2.6)+' '+(y - fh + 9)+' L '+(fx + 4)+' '+y+' Z"/></g>';
    out += '<text x="'+fx+'" y="'+(y - fh - g.markGapPx)+'" font-size="'+CFG.typeSize.markPx+'" fill="'+P.ink+'" text-anchor="middle">'+CFG.ruler.mark+'</text>';
  }
  return out;
}
let lastKey = '';
function render(tMs){
  const total = CFG.timing.totalMs;
  const t = ((tMs % total) + total) % total;
  const s = stateAt(t);
  const key = [s.hud, s.year, s.rulerProg.toFixed(3), s.ticks, s.figure, s.entryProg.toFixed(3)].join('|');
  hudEl.style.display = s.hud ? 'block' : 'none';
  cardEl.style.display = s.card ? 'block' : 'none';
  if (key === lastKey) return;
  lastKey = key;
  if (s.card){
    const dy = (1 - s.entryProg) * -CFG.geom.entryDropPx;
    const sc = CFG.geom.entryScale + (1 - CFG.geom.entryScale) * s.entryProg;
    cardEl.style.transform = 'translateY(' + dy.toFixed(1) + 'px) scale(' + sc.toFixed(3) + ')';
    cardYearEl.textContent = String(s.year);
  }
  rulerSvg.innerHTML = rulerSVG(s);
  rulerLabel.style.display = s.rulerProg >= 1 ? 'block' : 'none';
}
