function deepMerge(base, over){
  if (Array.isArray(over)) return over.slice();
  if (over && typeof over === 'object'){
    const out = {};
    for (const k of Object.keys(base)) out[k] = deepMerge(base[k], over[k]);
    for (const k of Object.keys(over)) if (!(k in out)) out[k] = over[k];
    return out;
  }
  return over === undefined ? base : over;
}

/* ONE phase resolver — half-open intervals, exactly one state per t. */
function phaseAt(t){
  const tt = ((t % cycleMs) + cycleMs) % cycleMs;
  const scenesMs = CFG.scenes.length * CFG.sceneMs;
  if (tt >= scenesMs) return { sceneIndex: CFG.scenes.length - 1, paletteMode: "blue", localTime: tt - scenesMs, endHold: true };
  const idx = Math.floor(tt / CFG.sceneMs), local = tt - idx * CFG.sceneMs;
  const inBreak = idx === CFG.break.atScene && local >= CFG.break.offsetMs && local < CFG.break.offsetMs + CFG.breakMs;
  return { sceneIndex: idx, paletteMode: inBreak ? "dusk" : "blue", localTime: local, endHold: false };
}
function skyline(g, P, seedOff){
